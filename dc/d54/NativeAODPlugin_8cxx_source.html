41// A fragment which holds a natively read tree, and the read operations
42// planned for it.
43class NativeAODFileFragment : public arrow::dataset::FileFragment
44{
45 public:
46 NativeAODFileFragment(arrow::dataset::FileSource source,
47 std::shared_ptr<arrow::dataset::FileFormat> format,
48 arrow::compute::Expression partition_expression,
49 std::shared_ptr<arrow::Schema> physical_schema)
50 : FileFragment(source, format, std::move(partition_expression), physical_schema)
51 {
52 auto rootFS = std::dynamic_pointer_cast<VirtualRootFileSystemBase>(this->source().filesystem());
53 if (rootFS.get() == nullptr) {
54 throw runtime_error_f("Unknown filesystem %s when reading %s.",
55 source.filesystem()->type_name().c_str(), source.path().c_str());
56 }
57 auto objectHandler = rootFS->GetObjectHandler(source);
58 if (!objectHandler->format->Equals(*format)) {
59 throw runtime_error_f("Cannot read source %s with format %s to populate a NativeAODFileFragment.",
60 source.path().c_str(), objectHandler->format->type_name().c_str());
61 }
62 // The handle is owned by the filesystem, which the source keeps alive.
63 mHandle = objectHandler->GetObjectAsOwner<NativeTreeHandle>().release();
64 // Unless the file is mapped, the baskets come from the read plan of the DF.
65 if (auto nativeFS = std::dynamic_pointer_cast<NativeAODFileSystem>(rootFS)) {
66 if (auto plan = nativeFS->planFor(source.path())) {
67 mSource = std::make_unique<o2::aod::native::PlannedByteSource>(mHandle->source, std::move(plan));
68 }
69 }
70 mDecoder = std::make_unique<BranchDecoder>(mSource ? *mSource : *mHandle->source, mHandle->options);
71 }
72
73 NativeTreeHandle* handle() { return mHandle; }
74 BranchDecoder& decoder() { return *mDecoder; }
75 std::vector<ColumnOp>& ops() { return mOps; }
76
77 private:
78 NativeTreeHandle* mHandle = nullptr;
79 std::unique_ptr<o2::aod::native::ByteSource> mSource;
80 std::unique_ptr<BranchDecoder> mDecoder;
81 std::vector<ColumnOp> mOps;
82};

88class NativeAODDeferredReadOutputStream : public arrow::io::OutputStream
89{
90 public:
91 NativeAODDeferredReadOutputStream(std::shared_ptr<NativeAODFileFragment> fragment,
92 const std::shared_ptr<arrow::ResizableBuffer>& buffer)
93 : mFragment(std::move(fragment)),
94 buffer_(buffer),
95 is_open_(true),
96 capacity_(buffer->size()),
97 position_(0),
98 mutable_data_(buffer->mutable_data())
99 {
100 }
101
102 // By the time we call the destructor, the contents
103 // of the buffer are already moved to fairmq
104 // for being sent.
105 ~NativeAODDeferredReadOutputStream() override = default;
106
107 arrow::Status Close() override
108 {
109 if (is_open_) {
110 is_open_ = false;
111 if (position_ < capacity_) {
112 RETURN_NOT_OK(buffer_->Resize(position_, false));
113 }
114 }
115 return arrow::Status::OK();
116 }
117
118 [[nodiscard]] bool closed() const override { return !is_open_; }
119 [[nodiscard]] arrow::Result<int64_t> Tell() const override { return position_; }
120
121 arrow::Status Write(const void* data, int64_t nbytes) override
122 {
123 if (ARROW_PREDICT_FALSE(!is_open_)) {
124 return arrow::Status::IOError("OutputStream is closed");
125 }
126 if (ARROW_PREDICT_TRUE(nbytes == 0)) {
127 return arrow::Status::OK();
128 }
129 if (ARROW_PREDICT_FALSE(position_ + nbytes >= capacity_)) {
130 RETURN_NOT_OK(Reserve(nbytes));
131 }
132 auto& ops = mFragment->ops();
133 // This is a real address which needs to be copied. Do it!
134 auto ref = (int64_t)data;
135 if (ref < 0 || ref >= (int64_t)ops.size()) {
136 memcpy(mutable_data_ + position_, data, nbytes);
137 position_ += nbytes;
138 return arrow::Status::OK();
139 }
140 auto& op = ops[ref];
141 if ((int64_t)op.byteSize != nbytes) {
142 return arrow::Status::Invalid("Deferred read of ", nbytes, " bytes for an operation of ", op.byteSize);
143 }
144 // Decoding happens here, straight into the message buffer.
145 o2::aod::native::executeColumnOp(mFragment->decoder(), op, mutable_data_ + position_);
146 position_ += nbytes;
147 return arrow::Status::OK();
148 }
149
150 using OutputStream::Write;
151
152 private:
153 arrow::Status Reserve(int64_t nbytes)
154 {
155 int64_t new_capacity = position_ + nbytes;
156 if (new_capacity > capacity_) {
157 RETURN_NOT_OK(buffer_->Resize(new_capacity));
158 capacity_ = new_capacity;
159 mutable_data_ = buffer_->mutable_data();
160 }
161 return arrow::Status::OK();
162 }
163
164 std::shared_ptr<NativeAODFileFragment> mFragment;
165 std::shared_ptr<arrow::ResizableBuffer> buffer_;
166 bool is_open_;
167 int64_t capacity_;
168 int64_t position_;
169 uint8_t* mutable_data_;
170};

245 arrow::Result<arrow::RecordBatchGenerator> ScanBatchesAsync(
246 const std::shared_ptr<arrow::dataset::ScanOptions>& options,
247 const std::shared_ptr<arrow::dataset::FileFragment>& fragment) const override
248 {
249 assert(options->dataset_schema != nullptr);
250 auto datasetSchema = options->dataset_schema;
251 auto nativeFragment = std::dynamic_pointer_cast<NativeAODFileFragment>(fragment);
252 if (nativeFragment.get() == nullptr) {
253 return {arrow::Status::NotImplemented("Not a native AO2D fragment")};
254 }
255
256 auto generator = [pool = options->pool, nativeFragment, datasetSchema, &totalCompressedSize = mTotCompressedSize,
257 &totalUncompressedSize = mTotUncompressedSize]() -> arrow::Future<std::shared_ptr<arrow::RecordBatch>> {
258 auto* handle = nativeFragment->handle();
259 auto const& tree = handle->tree;
260 O2_SIGNPOST_ID_FROM_POINTER(tid, native_aod, handle);
261 O2_SIGNPOST_START(native_aod, tid, "Generator", "Creating batch for tree %{public}s", tree.name.c_str());
262 if (datasetSchema->num_fields() > (int)tree.branches.size()) {
263 throw runtime_error_f("One TTree must have all the fields requested in a table");
264 }
265 // Plan the reads. VLA offsets are decoded now, because they size the
266 // values; everything else is decoded later, in the deferred Write().
267 int64_t rows = 0;
268 auto& ops = nativeFragment->ops();
269 ops = o2::aod::native::planColumns(tree, datasetSchema, nativeFragment->decoder(), pool, rows);
270 // The pointer to each buffer is an incremental number, indexing ops.
271 std::vector<std::shared_ptr<arrow::Buffer>> placeholders;
272 for (size_t i = 0; i < ops.size(); ++i) {
273 placeholders.push_back(std::make_shared<arrow::Buffer>((uint8_t*)i, ops[i].byteSize));
274 O2_SIGNPOST_EVENT_EMIT(native_aod, tid, "Op", "Created op for branch %{public}s with %lli entries, size of the buffer %zu.",
275 ops[i].branch->name.c_str(), (long long)ops[i].branch->entries, ops[i].byteSize);
276 }
277 auto batch = o2::aod::native::makeBatch(datasetSchema, ops, placeholders, rows);
278 totalCompressedSize += tree.zipBytes;
279 totalUncompressedSize += tree.totBytes;
280 O2_SIGNPOST_END(native_aod, tid, "Generator", "Done creating batch compressed:%zu uncompressed:%zu", totalCompressedSize, totalUncompressedSize);
281 return batch;
282 };
283 return generator;
284 }

293RootArrowFactory* createNativeAODFactory()
294{
295 auto context = new NativeAODPluginContext;
296 context->format = std::make_shared<NativeAODFileFormat>(context->totalCompressedSize, context->totalUncompressedSize);
297 return new RootArrowFactory{
298 .options = [context]() { return context->format->DefaultWriteOptions(); },
299 .format = [context]() { return context->format; },
300 .deferredOutputStreamer = [](std::shared_ptr<arrow::dataset::FileFragment> fragment, const std::shared_ptr<arrow::ResizableBuffer>& buffer) -> std::shared_ptr<arrow::io::OutputStream> {
301 auto nativeFragment = std::dynamic_pointer_cast<NativeAODFileFragment>(fragment);
302 return std::make_shared<NativeAODDeferredReadOutputStream>(nativeFragment, buffer);
303 }};
304}

310// The capability lives here rather than in Framework/Core, because recognising
311// a native handle needs NativeAODFileSystem. Being first in the capability
312// list, it wins for files opened natively; for any other filesystem it returns
313// no handle, and DPL falls through to the RNTuple and TTree capabilities.
314struct NativeAODObjectReadingCapability : o2::framework::RootObjectReadingCapabilityPlugin {
315 RootObjectReadingCapability* create() override
316 {
317 return new RootObjectReadingCapability{
318 .name = "native-ttree",
319 .lfn2objectPath = [](std::string s) { return s; },
320 .getHandle = [](std::shared_ptr<arrow::fs::FileSystem> fs, std::string const& path) -> void* {
321 auto nativeFS = std::dynamic_pointer_cast<NativeAODFileSystem>(fs);
322 if (!nativeFS) {
323 return nullptr;
324 }
325 try {
326 return nativeFS->getTreeHandle(path);
327 } catch (AODReadError const&) {
328 throw;
329 } catch (std::exception const& e) {
330 // A DF which cannot be parsed: --aod-skip-invalid-reads skips it.
331 throw AODReadError(AODReadError::Kind::InvalidTimeframe, e.what());
332 }
333 },
334 .checkSupport = [](char const* className) -> bool { return std::string_view(className) == "TTree"; },
335 .accountBytes = [](void* handle, size_t& compressed, size_t& uncompressed) {
336 auto* h = (NativeTreeHandle*)handle;
337 compressed += h->tree.zipBytes;
338 uncompressed += h->tree.totBytes; },
339 .factory = []() -> RootArrowFactory& {
340 static RootArrowFactory* factory = createNativeAODFactory();
341 return *factory;
342 }};
343 }
344};
