38struct GPUDisplayGUIUpdate { // GUI → display
39 uint64_t id = 0;
40 std::vector<int32_t> keys; // Keys pressed in the GUI and not handled by it, display key codes
41 std::vector<std::pair<uint32_t, GPUDisplayGUIValue>> values; // Elements changed in the GUI: element id, new value
42 std::optional<float> guiScale; // GUI scale changed with Ctrl + mouse wheel / Ctrl + 0
43 std::optional<int8_t> guiDarkMode; // System color scheme, reported when the GUI starts if not yet set
44 bool empty() const { return keys.empty() && values.empty() && !guiScale && !guiDarkMode; }
45 void setValue(uint32_t element, GPUDisplayGUIValue value) // Replaces an earlier value of the same element
46 {
47 auto it = std::find_if(values.begin(), values.end(), [element](const auto& v) { return v.first == element; });
48 if (it != values.end()) {
49 it->second = std::move(value);
50 } else {
51 values.emplace_back(element, std::move(value));
52 }
53 }
54 void merge(const GPUDisplayGUIUpdate& other) // Newer changes on top of this update, id not changed
55 {
56 keys.insert(keys.end(), other.keys.begin(), other.keys.end());
57 for (const auto& [element, value] : other.values) {
58 setValue(element, value);
59 }
60 guiScale = other.guiScale ? other.guiScale : guiScale;
61 guiDarkMode = other.guiDarkMode ? other.guiDarkMode : guiDarkMode;
62 }
63};

65// Communication between the Qt GUI thread (GPUDisplayGUI) and the display thread (GPUDisplayFrontend::HandleExternalGUISignals), Qt-free.
66// The GUI sends at most one update per timer tick, the display processes it once per frame and publishes the complete settings with the id of the last processed update.
67// If the GUI went to sleep with an update ready and the display picks it up, the GUI waits for the reply when it wakes up, so it continues with the settings including its update.

73 GPUDisplayOptions mCfgO;
74 GPUDisplayControl mCfgC;
75 };
