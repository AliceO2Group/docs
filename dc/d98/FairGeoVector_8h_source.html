247inline void FairGeoVector::round(Double_t d, Int_t n)
248{
249 // rounds d to a precision with n digits
250 if (d > 0) {
251 d = floor(d * pow(10., n) + 0.5) / pow(10., n);
252 } else {
253 d = -floor((-d) * pow(10., n) + 0.5) / pow(10., n);
254 }
255}
