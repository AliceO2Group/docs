123TEST_CASE("LabeledArrayBoostOptionsRetrieverTest")
124{
125 auto specs = std::vector<ConfigParamSpec>{
126 {"someLabeledArray", VariantType::LabeledArrayInt, defaultLabeledArray(), {"some labeled array option"}}};
127
128 // full JSON on the command line, its labels are kept
129 auto registry = registryFromCommandLine(specs, {"--someLabeledArray", fullLabeledJSON});
130 auto la = registry->get<LabeledArray<int>>("someLabeledArray");
131 REQUIRE(la.getLabelsRows() == std::vector<std::string>{"a", "b"});
132 REQUIRE(la.getLabelsCols() == std::vector<std::string>{"x", "y"});
133 REQUIRE(la.get("b", "y") == 8);
134
135 // absent from the command line: the default value
136 registry = registryFromCommandLine(specs, {});
137 la = registry->get<LabeledArray<int>>("someLabeledArray");
138 REQUIRE(la.get("r2", "c2") == 4);
139
140 // only values on the command line: the default labels are used
141 registry = registryFromCommandLine(specs, {"--someLabeledArray", R"({"values":[[5,6],[7,8]]})"});
142 la = registry->get<LabeledArray<int>>("someLabeledArray");
143 REQUIRE(la.getLabelsRows() == std::vector<std::string>{"r1", "r2"});
144 REQUIRE(la.getLabelsCols() == std::vector<std::string>{"c1", "c2"});
145 REQUIRE(la.get("r2", "c1") == 7);
146
147 // values which do not match the default labels
148 REQUIRE_THROWS(registryFromCommandLine(specs, {"--someLabeledArray", R"({"values":[[5,6,7]]})"}));
149}

151TEST_CASE("DefaultValueForHelpTest")
152{
153 auto const max = ConfigParamsHelper::MaxHelpDefaultLength;
154 REQUIRE(ConfigParamsHelper::defaultValueForHelp("") == "");
155 REQUIRE(ConfigParamsHelper::defaultValueForHelp("short") == "short");
156 std::string atLimit(max, 'x');
157 REQUIRE(ConfigParamsHelper::defaultValueForHelp(atLimit) == atLimit);
158
159 std::string longValue = std::string("BEGIN") + std::string(100, '-') + "END";
160 auto shown = ConfigParamsHelper::defaultValueForHelp(longValue);
161 REQUIRE(shown.size() == max);
162 REQUIRE(shown.starts_with("BEGIN"));
163 REQUIRE(shown.ends_with("END"));
164 REQUIRE(shown.find("...") != std::string::npos);
165 REQUIRE(ConfigParamsHelper::defaultValueForHelp(std::string(max + 1, 'x')).size() == max);
166}

168TEST_CASE("LongDefaultInHelpTest")
169{
170 namespace bpo = boost::program_options;
171 std::vector<int> values(30, 12345);
172 std::vector<std::string> rows{"row"};
173 std::vector<std::string> cols;
174 for (auto i = 0U; i < values.size(); ++i) {
175 cols.push_back("column" + std::to_string(i));
176 }
177 LabeledArray<int> la{values.data(), 1, static_cast<uint32_t>(values.size()), rows, cols};
178 auto workflowOptions = std::vector<ConfigParamSpec>{
179 {"longLabeledArray", VariantType::LabeledArrayInt, la, {"a labeled array option with a long default"}}};
180 auto full = workflowOptions[0].defaultValue.asString();
181 REQUIRE(full.size() > ConfigParamsHelper::MaxHelpDefaultLength);
182
183 // the help shows the abbreviated default only
184 auto description = ConfigParamsHelper::prepareOptionDescriptions(std::vector<DataProcessorSpec>{}, workflowOptions);
185 std::stringstream help;
186 help << description;
187 REQUIRE(help.str().find(ConfigParamsHelper::defaultValueForHelp(full)) != std::string::npos);
188 REQUIRE(help.str().find(full) == std::string::npos);
189
190 // the default value used when parsing is still the full one
191 bpo::options_description forwarded;
192 ConfigParamsHelper::dpl2BoostOptions(workflowOptions, forwarded);
193 const char* noArgs[] = {"test"};
194 bpo::variables_map vm;
195 bpo::store(parse_command_line(1, noArgs, forwarded), vm);
196 REQUIRE(vm["longLabeledArray"].as<std::string>() == full);
197
198 // also for the options registered for the devices
199 bpo::options_description deviceOptions;
200 ConfigParamsHelper::populateBoostProgramOptions(deviceOptions, workflowOptions);
201 std::stringstream deviceHelp;
202 deviceHelp << deviceOptions;
203 REQUIRE(deviceHelp.str().find(full) == std::string::npos);
204 bpo::variables_map deviceVm;
205 bpo::store(parse_command_line(1, noArgs, deviceOptions), deviceVm);
206 REQUIRE(deviceVm["longLabeledArray"].as<std::string>() == full);
207}
