28struct DigitKey {
29 ULong64_t high;
30 UInt_t low;
31
32 bool operator<(const DigitKey& other) const
33 {
34 if (high != other.high) {
35 return high < other.high;
36 }
37 return low < other.low;
38 }
39};

56 static DigitKey getOrderingKey(ULong64_t bc, UInt_t tdc, uint16_t row, uint16_t col)
57 {
58 DigitKey key;
59 uint32_t orbit = bc / o2::constants::lhc::LHCMaxBunches;
60 uint16_t bunch = bc % o2::constants::lhc::LHCMaxBunches;
61 key.high = (static_cast<ULong64_t>(orbit) << 32) | (static_cast<UInt_t>(bunch) << 16) | (static_cast<UInt_t>(tdc) & 0xFFFF);
62 key.low = (static_cast<UInt_t>(row) << 16) | (static_cast<UInt_t>(col) & 0xFFFF);
63 return key;
64 }
65
66 private:
67 double mTime = 0.;
68 ULong64_t mBc = 0;
69 Int_t mTdc = 0;
70 ClassDefNV(Digit, 1);
71};
