34void* extractFromTFile(TFile& file, TClass const* cl, const char* what)
35{
36 if (!cl) {
37 return nullptr;
38 }
39#if defined(_LIBCPP_VERSION)
40 DataRefUtils::registerLibstdcxxStdArray(file, cl);
41#endif
42 auto object = file.GetObjectChecked(what, cl);
43 if (!object) {
44 // it could be that object was stored with previous convention
45 // where the classname was taken as key
46 std::string objectName(cl->GetName());
47 objectName.erase(std::find_if(objectName.rbegin(), objectName.rend(), [](unsigned char ch) {
48 return !std::isspace(ch);
49 }).base(),
50 objectName.end());
51 objectName.erase(objectName.begin(), std::find_if(objectName.begin(), objectName.end(), [](unsigned char ch) {
52 return !std::isspace(ch);
53 }));
54
55 object = file.GetObjectChecked(objectName.c_str(), cl);
56 LOG(warn) << "Did not find object under expected name " << what;
57 if (!object) {
58 return nullptr;
59 }
60 LOG(warn) << "Found object under deprecated name " << cl->GetName();
61 }
62 auto result = object;
63 // We need to handle some specific cases as ROOT ties them deeply
64 // to the file they are contained in
65 if (cl->IsTObject()) {
66 // make a clone
67 // detach from the file
68 auto tree = dynamic_cast<TTree*>((TObject*)object);
69 if (tree) {
70 tree->LoadBaskets(0x1L << 32); // make tree memory based
71 tree->SetDirectory(nullptr);
72 result = tree;
73 } else {
74 auto h = dynamic_cast<TH1*>((TObject*)object);
75 if (h) {
76 h->SetDirectory(nullptr);
77 result = h;
78 }
79 }
80 }
81 return result;
82}

86// The only member of std::array<T,N> is T _M_elems[N] in libstdc++, T __elems_[N] in libc++, so
87// objects written on Linux carry a StreamerInfo with a checksum the libc++ class does not have.
88// When opening the file, ROOT drops that StreamerInfo as an alias of the in-memory one
89// (TStreamerInfo::BuildCheck), and the read fails with "Could not find the StreamerInfo with a
90// checksum", leaving the array zero-filled. Register the on-file layout as a version of its own,
91// under the in-memory member name, so that the payload, T[N] for both, is read as usual.
92void DataRefUtils::registerLibstdcxxStdArray(TFile& file, TClass const* cl)
93{
94 auto* infos = std::strncmp(cl->GetName(), "array<", 6) == 0 ? file.GetStreamerInfoCache() : nullptr;
95 auto* onfile = infos ? dynamic_cast<TStreamerInfo*>(infos->FindObject(cl->GetName())) : nullptr;
96 if (!onfile || onfile->GetNelement() != 1 || std::strcmp(onfile->GetElement(0)->GetName(), "_M_elems") != 0) {
97 return;
98 }
99 // Exists before BuildCheck, so that the on-file layout gets a version of its own
100 auto* current = static_cast<TStreamerInfo*>(const_cast<TClass*>(cl)->GetStreamerInfo());
101 auto* info = static_cast<TStreamerInfo*>(cl->FindStreamerInfo(onfile->GetCheckSum()));
102 if (!info) {
103 info = static_cast<TStreamerInfo*>(onfile->Clone());
104 info->BuildCheck(&file);
105 }
106 // Also when ROOT registered it itself: it then reads nothing, without any error
107 if (std::strcmp(info->GetElement(0)->GetName(), current->GetElement(0)->GetName()) != 0) {
108 info->GetElement(0)->SetName(current->GetElement(0)->GetName());
109 info->Clear("build");
110 info->BuildOld();
111 }
112}

118void* DataRefUtils::decodeCCDB(DataRef const& ref, std::type_info const& tinfo)
119{
120 void* result = nullptr;
121 Int_t previousErrorLevel = gErrorIgnoreLevel;
122 gErrorIgnoreLevel = kFatal;
123 auto* dh = o2::header::get<o2::header::DataHeader*>(ref.header);
124 const char* buff = const_cast<char*>(ref.payload);
125 size_t flSize = dh->payloadSize;
126 // does it have a flattened headers map attached in the end?
127 constexpr char FlatHeaderAnnot[] = "$HEADER$";
128 constexpr int Offset = sizeof(int) + sizeof(FlatHeaderAnnot);
129 int headerSize = 0;
130 LOGP(debug, "DHPayloadSize={}>{} Ref:{}/{} Cmp {}:{}", dh->payloadSize, Offset, dh->dataOrigin.as<std::string>(), dh->dataDescription.as<std::string>(), std::string{buff + dh->payloadSize - sizeof(FlatHeaderAnnot)}, std::string{FlatHeaderAnnot});
131
132 if (dh->payloadSize >= Offset &&
133 !std::strncmp(buff + dh->payloadSize - sizeof(FlatHeaderAnnot), FlatHeaderAnnot, sizeof(FlatHeaderAnnot))) {
134 headerSize = *reinterpret_cast<const int*>(buff + dh->payloadSize - Offset);
135 }
136 if (headerSize < 0) {
137 LOGP(fatal, "Anomalous flattened header size {} extracted for CCDB object {}/{}", headerSize, dh->dataOrigin.as<std::string>(), dh->dataDescription.as<std::string>());
138 }
139 TMemFile memFile("name", const_cast<char*>(ref.payload), dh->payloadSize - headerSize, "READ");
140 gErrorIgnoreLevel = previousErrorLevel;
141 if (memFile.IsZombie()) {
142 return nullptr;
143 }
144
145 TClass* tcl = TClass::GetClass(tinfo);
146 result = extractFromTFile(memFile, tcl, "ccdb_object");
147 if (!result) {
148 throw runtime_error_f("Couldn't retrieve object corresponding to %s from TFile", tcl->GetName());
149 }
150 memFile.Close();
151 return result;
152}

154std::map<std::string, std::string> DataRefUtils::extractCCDBHeaders(DataRef const& ref)
155{
156 auto* dh = o2::header::get<o2::header::DataHeader*>(ref.header);
157 const char* buff = const_cast<char*>(ref.payload);
158 // does it have a flattened headers map attached in the end?
159 constexpr char FlatHeaderAnnot[] = "$HEADER$";
160 constexpr int Offset = sizeof(int) + sizeof(FlatHeaderAnnot);
161 int headerSize = 0, ss0 = 0;
162 std::map<std::string, std::string> res;
163 if (dh->payloadSize >= Offset && !std::strncmp(buff + dh->payloadSize - sizeof(FlatHeaderAnnot), FlatHeaderAnnot, sizeof(FlatHeaderAnnot))) {
164 headerSize = *reinterpret_cast<const int*>(buff + dh->payloadSize - Offset);
165 } else { // header was not added
166 LOGP(warn, "CCDB headers were not added to condition object blob, returning dummy header map");
167 return res;
168 }
169
170 if (headerSize < 0) {
171 LOGP(fatal, "Anomalous flattened header size {} extracted for CCDB object {}/{}", headerSize, dh->dataOrigin.as<std::string>(), dh->dataDescription.as<std::string>());
172 }
173
174 buff += dh->payloadSize - headerSize; // jump to the start of flattened header
175 headerSize -= Offset;
176 const char* str0 = &buff[ss0++];
177 while (ss0 < headerSize) {
178 if (buff[ss0++] == 0) {
179 if (!str0) {
180 str0 = &buff[ss0]; // new key string is found
181 } else {
182 res.emplace(std::string(str0), std::string(&buff[ss0])); // new value string found, add key value to the map
183 LOGP(debug, "Header{} {}:{}", res.size(), std::string(str0), std::string(&buff[ss0]));
184 str0 = nullptr;
185 }
186 }
187 }
188 return res;
189}

192{
193 auto* dh = o2::header::get<o2::header::DataHeader*>(ref.header);
194 const char* buff = ref.payload;
195 size_t payloadSize = dh->payloadSize;
196 constexpr char FlatHeaderAnnot[] = "$HEADER$";
197 constexpr size_t Offset = sizeof(int) + sizeof(FlatHeaderAnnot);
198 int headerSize = 0;
199 if (payloadSize >= Offset &&
200 !std::strncmp(buff + payloadSize - sizeof(FlatHeaderAnnot), FlatHeaderAnnot, sizeof(FlatHeaderAnnot))) {
201 headerSize = *reinterpret_cast<const int*>(buff + payloadSize - Offset);
202 }
203 return {buff, payloadSize - headerSize};
204}
