36// Event of the CPUDevice backend. Recording a marker increments nRecorded when the marker is enqueued, the command queue sets nCompleted when the marker is processed. A wait captures the value of nRecorded at the time the wait is enqueued, so it behaves like cudaStreamWaitEvent.
37struct GPUReconstructionCPUDeviceEvent {
38 struct waitingQueue {
39 uint64_t value;
40 GPUReconstructionCPUDeviceInternals* internals;
41 int32_t stream;
42 };
43 std::mutex mutex;
44 std::condition_variable cv; // For host threads waiting for the event
45 std::vector<waitingQueue> waiting; // Suspended command queues, which are resumed when the event reaches the value
46 uint64_t nRecorded = 0;
47 uint64_t nCompleted = 0;
48};
49
50// Command in a command queue: wait for events, run the work, mark an event as completed
51struct GPUReconstructionCPUDeviceTask {
52 std::vector<std::pair<GPUReconstructionCPUDeviceEvent*, uint64_t>> waits;
53 uint32_t nextWait = 0;
54 std::function<void()> work;
55 GPUReconstructionCPUDeviceEvent* record = nullptr;
56 uint64_t recordValue = 0;
57};

59// In-order command queue. It does not own a thread: When commands are enqueued, a TBB task is started which processes them, and which returns once the queue is empty, or when the next command must wait for an event. In the latter case, the queue is registered at the event,
60// and a new TBB task is started when the event completes. At most one TBB task processes a queue at a time.
61struct GPUReconstructionCPUDeviceQueue {
62 std::mutex mutex;
63 std::condition_variable cv; // For host threads synchronizing the queue
64 std::deque<GPUReconstructionCPUDeviceTask> tasks;
65 bool active = false; // Queue has pending commands, and is either being processed by a TBB task or suspended waiting for an event
66};
