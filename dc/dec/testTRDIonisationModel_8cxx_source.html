32 TRandom3 rnd(4711);

34 IonisationModel model(gas);

47BOOST_AUTO_TEST_CASE(TRDIonisationModel_zeroCollisions)
48{
49 TRandom3 rnd(815);
50 IonisationModel model(SimParam::GasMixture::Xenon);
51 const double bg = 4., length = 0.02;
52 for (double z2 : {1., 4.}) {
53 const int n = 100000;
54 int zero = 0;
55 for (int i = 0; i < n; ++i) {

64BOOST_AUTO_TEST_CASE(TRDIonisationModel_minimum)
65{
66 IonisationModel model(SimParam::GasMixture::Xenon);
67 double bgMin = 0., nMin = 1e9;
68 for (double lbg = 0.; lbg <= 1.; lbg += 0.01) {
69 const double n = model.collisionsPerCm(std::pow(10., lbg));
70 if (n < nMin) {
71 nMin = n;
72 bgMin = std::pow(10., lbg);
73 }
74 }
75 BOOST_TEST(nMin > 40.);
76 BOOST_TEST(nMin < 47.);
77 BOOST_TEST(bgMin > 2.5);
78 BOOST_TEST(bgMin < 5.);
79 BOOST_TEST(model.collisionsPerCm(1e5) == model.collisionsPerCm(1e6));
80}

83BOOST_AUTO_TEST_CASE(TRDIonisationModel_collisionEnergy)
84{
85 TRandom3 rnd(42);
86 IonisationModel model(SimParam::GasMixture::Xenon);
87 // steps short enough that almost every step with energy has exactly one collision
88 const double length = 1e-4, bg = 4.;
89 int n = 0;
90 double sum = 0.;
91 for (int i = 0; i < 2000000; ++i) {
92 const double e = model.stepEnergy(length, bg, 1., rnd);
93 if (e > 0.) {
94 BOOST_TEST(e >= 12.);
95 BOOST_TEST(e <= IonisationModel::deltaRayCut());
96 sum += e;
97 ++n;
98 }
99 }
100 BOOST_TEST(n > 5000);
101 BOOST_TEST(sum / n == model.restrictedDEdx(bg) / model.collisionsPerCm(bg), boost::test_tools::tolerance(0.2));
102}
