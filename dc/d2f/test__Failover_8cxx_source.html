23// A replica whose byte i is i % 251. Its behaviour is shared between the
24// instances opened for the same URL, so that a reopen sees the same state.
25struct Behaviour {
26 size_t size = 1000;
27 bool opens = true;
28 std::atomic<int> failingReads = 0; // the next reads which fail with IoError
29 std::atomic<int> opened = 0;
30 std::atomic<int> reads = 0;
31};
32
33struct FakeSource : ByteSource {
34 FakeSource(std::string u, Behaviour& b) : name(std::move(u)), behaviour(b) {}
35 uint64_t size() const override { return behaviour.size; }
36 void readInto(uint64_t off, size_t len, void* dst) override
37 {
38 ReadRequest r{off, len, dst};
39 readvInto({&r, 1});
40 }
41 void readvInto(std::span<ReadRequest> requests) override
42 {
43 ++behaviour.reads;
44 int failing = behaviour.failingReads.load();
45 while (failing > 0 && !behaviour.failingReads.compare_exchange_weak(failing, failing - 1)) {
46 }
47 if (failing > 0) {
48 throw IoError(name, "Operation expired");
49 }
50 for (auto& r : requests) {
51 for (size_t i = 0; i < r.length; ++i) {
52 ((uint8_t*)r.dst)[i] = (r.offset + i) % 251;
53 }
54 bytes += r.length;
55 }
56 ++calls;
57 }
58 bool isRemote() const override { return true; }
59 uint64_t bytesRead() const override { return bytes; }
60 uint64_t readCalls() const override { return calls; }
61 std::string url() const override { return name; }
62 std::string name;
63 Behaviour& behaviour;
64 std::atomic<uint64_t> bytes = 0;
65 std::atomic<uint64_t> calls = 0;
66};

68struct Setup {
69 std::map<std::string, Behaviour> replicas;
70 std::vector<std::string> log;
71 Options options;
72 Setup() { options.retryDelay = 0; }
73
74 std::unique_ptr<FailoverByteSource> open()
75 {
76 std::vector<Replica> list;
77 for (auto& [url, _] : replicas) {
78 list.push_back({url, "SE-" + url});
79 }
80 auto opener = [this](std::string const& url, Options const&) -> std::unique_ptr<ByteSource> {
81 auto& b = replicas.at(url);
82 ++b.opened;
83 if (!b.opens) {
84 throw IoError(url, "no such file");
85 }
86 return std::make_unique<FakeSource>(url, b);
87 };
88 return std::make_unique<FailoverByteSource>(list, options, opener, [this](std::string const& m) { log.push_back(m); });
89 }
90};

92bool readsCorrectly(ByteSource& s, uint64_t off = 100, size_t len = 50)
93{
94 std::vector<uint8_t> buf(len);
95 s.readInto(off, len, buf.data());
96 for (size_t i = 0; i < len; ++i) {
97 if (buf[i] != (off + i) % 251) {
98 return false;
99 }
100 }
101 return true;
102}
103} // namespace

209{

216 for (int t = 0; t < 4; ++t) {
217 threads.emplace_back([&, t] {
218 for (int i = 0; i < 20; ++i) {
219 good += readsCorrectly(*f, 10 * t + i, 40);
