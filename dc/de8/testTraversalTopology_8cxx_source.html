29std::vector<SurfaceDescriptor> catalog(uint16_t count)
30{
31 std::vector<SurfaceDescriptor> result;
32 result.reserve(count);

73const Edge* findEdge(const TraversalTopology& topology, LayerId from, LayerId to)
74{
75 const auto edge = std::find_if(topology.edges.begin(), topology.edges.end(), [&](const auto& candidate) {
76 return candidate.from == from && candidate.to == to;
77 });
78 return edge == topology.edges.end() ? nullptr : &*edge;
79}
