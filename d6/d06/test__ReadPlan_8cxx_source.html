21// File contents where byte i is i % 251, recording every read.
22struct RecordingSource : ByteSource {
23 explicit RecordingSource(size_t n) : bytes(n)
24 {
25 for (size_t i = 0; i < n; ++i) {
26 bytes[i] = i % 251;
27 }
28 }
29 uint64_t size() const override { return bytes.size(); }
30 void readInto(uint64_t off, size_t len, void* dst) override
31 {
32 reads.emplace_back(off, len);
33 std::memcpy(dst, bytes.data() + off, len);
34 }
35 void readvInto(std::span<ReadRequest> requests) override
36 {
37 for (auto& r : requests) {
38 readInto(r.offset, r.length, r.dst);
39 }
40 }
41 uint64_t bytesRead() const override { return 0; }
42 uint64_t readCalls() const override { return reads.size(); }
43 std::string url() const override { return "memory"; }
44
45 std::vector<uint8_t> bytes;
46 std::vector<std::pair<uint64_t, size_t>> reads;
47};
48
49bool holds(std::span<const uint8_t> span, uint64_t offset)
50{
51 for (size_t i = 0; i < span.size(); ++i) {
52 if (span[i] != (offset + i) % 251) {
53 return false;
54 }
55 }
56 return true;
57}

59Options withGap(size_t gap)
60{
61 Options options;
62 options.coalesceGap = gap;
63 return options;
64}
65} // namespace

67TEST_CASE("ranges closer than the gap are fetched in one read, on first use")
68{
69 auto source = std::make_shared<RecordingSource>(1 << 20);
70 ReadPlan plan{source, withGap(100)};
71 plan.add(5000, 200);
72 plan.add(1000, 100);
73 plan.add(1150, 50); // 50 bytes after the first: merged
74 plan.add(1300, 10); // 100 bytes after the second: merged
75 REQUIRE(source->reads.empty());
76
77 auto bytes = plan.find(1150, 50);
78 REQUIRE(holds(bytes, 1150));
79 REQUIRE(source->reads == std::vector<std::pair<uint64_t, size_t>>{{1000, 310}, {5000, 200}});
80 REQUIRE(plan.bytesHeld() == 510);
81
82 REQUIRE(holds(plan.find(5100, 100), 5100));
83 REQUIRE(holds(plan.find(1000, 310), 1000));
84 REQUIRE(source->reads.size() == 2);
85}

98TEST_CASE("ranges added after a fetch are fetched on their first use")
99{
100 auto source = std::make_shared<RecordingSource>(1 << 20);
101 ReadPlan plan{source, withGap(1000)};
102 plan.add(1000, 100);
103 REQUIRE(holds(plan.find(1000, 100), 1000));
104 auto first = plan.find(1000, 100);
105
106 // Within the gap of the fetched chunk on both sides, but not merged across it.
107 plan.add(500, 100);
108 plan.add(1500, 100);
109 plan.add(1000, 100); // already in memory: not read again
110 REQUIRE(holds(plan.find(1500, 100), 1500));

114 REQUIRE(plan.find(1000, 100).data() == first.data());

146TEST_CASE("concurrent lookups fetch the pending ranges once")
147{
148 auto source = std::make_shared<RecordingSource>(1 << 20);
149 ReadPlan plan{source, withGap(0)};
150 for (int i = 0; i < 64; ++i) {
151 plan.add(i * 10000, 1000);
152 }
153 std::vector<std::thread> threads;
154 std::atomic<int> good = 0;
155 for (int t = 0; t < 8; ++t) {
156 threads.emplace_back([&, t] {
157 for (int i = t; i < 64; i += 8) {
158 good += holds(plan.find(i * 10000 + 10, 900), i * 10000 + 10);
159 }
160 });
161 }
162 for (auto& t : threads) {
163 t.join();
164 }
165 REQUIRE(good == 64);
166 REQUIRE(source->reads.size() == 64);
167}
