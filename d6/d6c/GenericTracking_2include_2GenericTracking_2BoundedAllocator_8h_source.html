113template <typename T>
114inline void deepVectorClear(std::vector<T>& vec)
115{
116 std::vector<T>().swap(vec);
117}
118
119template <typename T>
120inline void deepVectorClear(bounded_vector<T>& vec, std::pmr::memory_resource* mr = nullptr)
121{
122 std::pmr::memory_resource* tmr = (mr != nullptr) ? mr : vec.get_allocator().resource();
123 std::destroy_at(&vec);
124 new (&vec) bounded_vector<T>(std::pmr::polymorphic_allocator<T>{tmr});
125}
126
127template <typename T>
128inline void deepVectorClear(std::vector<bounded_vector<T>>& vec, std::pmr::memory_resource* mr = nullptr)
129{
130 for (auto& v : vec) {
131 deepVectorClear(v, mr);
132 }
133}
134
135template <typename T, size_t S>
136inline void deepVectorClear(std::array<bounded_vector<T>, S>& arr, std::pmr::memory_resource* mr = nullptr)
137{
138 for (size_t i{0}; i < S; ++i) {
139 deepVectorClear(arr[i], mr);
140 }
141}

143template <typename T>
144inline void clearResizeBoundedVector(bounded_vector<T>& vec, size_t sz, std::pmr::memory_resource* mr = nullptr, T def = T())
145{
146 // Rebuild a valid empty vector before allocation or element construction can
147 // throw. The old allocation is released first, as required by bounded pools.
148 deepVectorClear(vec, mr);
149 vec.assign(sz, def);
150}
151
152// GTRK's capacity-reusing resize: retained storage is intended for arrays
153// rewritten by the caller. A non-null different resource requires a rebuild.
154template <typename T>
155inline void resizeBoundedVector(BoundedVector<T>& vec, size_t sz, std::pmr::memory_resource* mr = nullptr)
156{
157 if (sz > vec.capacity() || (mr != nullptr && mr != vec.get_allocator().resource())) {
158 clearResizeBoundedVector(vec, sz, mr);
159 return;
160 }
161 vec.resize(sz);
162}
163
164// Release several bounded vectors onto one resource.
165template <typename... T>
166inline void deepVectorClear(std::pmr::memory_resource* mr, BoundedVector<T>&... vecs)
167{
168 (deepVectorClear(vecs, mr), ...);
169}
170
171template <typename T>
172inline void clearResizeBoundedVector(std::vector<bounded_vector<T>>& vec, size_t size, std::pmr::memory_resource* mr)
173{
174 vec.clear();
175 vec.reserve(size);
176 for (size_t i = 0; i < size; ++i) {
177 vec.emplace_back(std::pmr::polymorphic_allocator<bounded_vector<T>>{mr});
178 }
179}
180
181template <typename T, size_t S>
182inline void clearResizeBoundedArray(std::array<bounded_vector<T>, S>& arr, size_t size, std::pmr::memory_resource* mr = nullptr, T def = T())
183{
184 for (size_t i{0}; i < S; ++i) {
185 clearResizeBoundedVector(arr[i], size, mr, def);
186 }
187}

191{
192 std::vector<T> t(b.size());
193 std::copy(b.cbegin(), b.cend(), t.begin());
