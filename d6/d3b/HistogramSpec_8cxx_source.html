70void AxisSpec::makeLogarithmic()
71{
72 if (binEdges.size() > 2) {
73 LOG(fatal) << "Cannot make a variabled bin width axis logaritmic";
74 }
75
76 const double min = binEdges[0];
77 if (min <= 0.) {
78 LOG(fatal) << "Cannot have the first bin limit of the log. axis below 0: " << min;
79 }
80 const double max = binEdges[1];
81 binEdges.clear();
82 const double logmin = std::log10(min);
83 const double logmax = std::log10(max);
84 const int nbins = nBins.value();
85 const double logdelta = (logmax - logmin) / (static_cast<double>(nbins));
86 LOG(debug) << "Making a logaritmic binning from " << min << " to " << max << " with " << nbins << " bins";
87 for (int i = 0; i < nbins + 1; i++) {
88 const auto nextEdge = std::pow(10, logmin + i * logdelta);
89 LOG(debug) << i << "/" << nbins - 1 << ": " << nextEdge;
90 binEdges.push_back(nextEdge);
91 }
92 nBins = std::nullopt;
93}

104// helper function to generate the actual histograms
105template <typename T>
106T* generateHist(const std::string& name, const std::string& title, const std::size_t nDim,
107 const int nBins[], const double lowerBounds[], const double upperBounds[], const int nSteps)
108{
109 if constexpr (std::is_base_of_v<StepTHn, T>) {
110 return new T(name.data(), title.data(), nSteps, nDim, nBins, lowerBounds, upperBounds);
111 } else if constexpr (std::is_base_of_v<THnBase, T>) {
112 return new T(name.data(), title.data(), nDim, nBins, lowerBounds, upperBounds);
113 } else if constexpr (std::is_base_of_v<TH3, T>) {
114 return (nDim != 3) ? nullptr
115 : new T(name.data(), title.data(), nBins[0], lowerBounds[0],
116 upperBounds[0], nBins[1], lowerBounds[1], upperBounds[1],
117 nBins[2], lowerBounds[2], upperBounds[2]);
118 } else if constexpr (std::is_base_of_v<TH2, T>) {
119 return (nDim != 2) ? nullptr
120 : new T(name.data(), title.data(), nBins[0], lowerBounds[0],
121 upperBounds[0], nBins[1], lowerBounds[1], upperBounds[1]);
122 } else if constexpr (std::is_base_of_v<TH1, T>) {
123 return (nDim != 1)
124 ? nullptr
125 : new T(name.data(), title.data(), nBins[0], lowerBounds[0], upperBounds[0]);
126 }
127 return nullptr;
128}

130// main function for creating arbitrary histograms
131template <typename T, typename BASE>
132static std::unique_ptr<BASE> createHistFull(const HistogramSpec& histSpec)
133{
134 constexpr std::size_t MAX_DIM{20};
135 const std::size_t nAxes{histSpec.config.axes.size()};
136 if (nAxes == 0 || nAxes > MAX_DIM) {
137 LOGF(fatal, "The histogram specification contains no (or too many) axes.");
138 }
139
140 int nBins[MAX_DIM]{0};
141 double lowerBounds[MAX_DIM]{0.};
142 double upperBounds[MAX_DIM]{0.};
143
144 // first figure out number of bins and dimensions
145 for (std::size_t i = 0; i < nAxes; i++) {
146 nBins[i] = (histSpec.config.axes[i].nBins) ? *histSpec.config.axes[i].nBins : histSpec.config.axes[i].binEdges.size() - 1;
147 lowerBounds[i] = histSpec.config.axes[i].binEdges.front();
148 upperBounds[i] = histSpec.config.axes[i].binEdges.back();
149 }
150
151 // create histogram
152 std::unique_ptr<BASE> hist{generateHist<T>(histSpec.name, histSpec.title, nAxes, nBins, lowerBounds, upperBounds, histSpec.config.nSteps)};
153 if (!hist) {
154 LOGF(fatal, "The number of dimensions specified for histogram %s does not match the type.", histSpec.name);
155 }
156
157 // set axis properties
158 for (std::size_t i = 0; i < nAxes; i++) {
159 TAxis* axis{HistFactory::getAxis(i, hist.get())};
160 if (axis) {
161 if (histSpec.config.axes[i].title) {
162 axis->SetTitle((*histSpec.config.axes[i].title).data());
163 }
164
165 // this helps to have axes not only called 0,1,2... in ndim histos
166 if constexpr (std::is_base_of_v<THnBase, T>) {
167 if (histSpec.config.axes[i].name) {
168 axis->SetName((std::string(axis->GetName()) + "-" + *histSpec.config.axes[i].name).data());
169 }
170 }
171
172 // move the bin edges in case a variable binning was requested
173 if (!histSpec.config.axes[i].nBins) {
174 if (!std::is_sorted(std::begin(histSpec.config.axes[i].binEdges), std::end(histSpec.config.axes[i].binEdges))) {
175 LOGF(fatal, "The bin edges in histogram %s are not in increasing order!", histSpec.name);
176 }
177 axis->Set(nBins[i], histSpec.config.axes[i].binEdges.data());
178 }
179 }
180 }
181 if (histSpec.callSumw2) {
182 hist->Sumw2();
183 }
184 return hist;
185}

248// helper function to get the axis via index for any type of root histogram
249template <typename T>
250TAxis* HistFactory::getAxis(const int i, T* hist)
251{
252 if constexpr (std::is_base_of_v<THnBase, T> || std::is_base_of_v<StepTHn, T>) {
253 return hist->GetAxis(i);
254 } else {
255 if (i == 0) {
256 return hist->GetXaxis();
257 } else if (i == 1) {
258 return hist->GetYaxis();
259 } else if (i == 2) {
260 return hist->GetZaxis();
261 } else {
262 return nullptr;
263 }
264 }
265}
