21ReadBufferPool::Buffer ReadBufferPool::get(size_t size)
22{
23 std::lock_guard lock{mMutex};
24 auto best = mFree.end();
25 for (auto it = mFree.begin(); it != mFree.end(); ++it) {
26 if (it->capacity >= size && (best == mFree.end() || it->capacity < best->capacity)) {
27 best = it;
28 }
29 }
30 if (best != mFree.end()) {
31 auto buffer = std::move(*best);
32 mFree.erase(best);
33 return buffer;
34 }
35 // All the free buffers are too small: give them back rather than keep them
36 // resident next to the new one. Some headroom, so that the next DF, a bit
37 // larger, still fits.
38 mFree.clear();
39 auto capacity = std::max<size_t>(size + size / 8, 1);
40 void* p = mmap(nullptr, capacity, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
41 if (p == MAP_FAILED) {
42 throw std::bad_alloc();
43 }
44 return {std::unique_ptr<uint8_t[], Unmap>((uint8_t*)p, Unmap{capacity}), capacity};
45}

52void ReadBufferPool::put(Buffer buffer)
53{
54 std::lock_guard lock{mMutex};
55 mFree.push_back(std::move(buffer));
56 if (mMaxFreeBytes == 0 && mFree.size() > kMaxFree) {
57 mFree.erase(std::ranges::min_element(mFree, {}, &Buffer::capacity));
58 }
59 auto freeBytes = [this] {
60 size_t total = 0;
61 for (auto const& b : mFree) {
62 total += b.capacity;
63 }
64 return total;
65 };
66 // The same headroom as get() allocates, so that a buffer for a group just
67 // under the budget can still be kept.
68 while (mMaxFreeBytes > 0 && !mFree.empty() && freeBytes() > mMaxFreeBytes + mMaxFreeBytes / 8) {
69 mFree.erase(std::ranges::max_element(mFree, {}, &Buffer::capacity));
70 }
71}

85void ReadPlan::add(uint64_t offset, size_t length)
86{
87 if (length == 0 || offset > mSource->size() || length > mSource->size() - offset) {
88 // Left to the decoder, which reports it with the branch it belongs to.
89 return;
90 }
91 std::lock_guard lock{mMutex};
92 mPending.emplace_back(offset, length);

119std::span<const uint8_t> ReadPlan::lookup(uint64_t offset, size_t length) const
120{
121 auto it = std::upper_bound(mChunks.begin(), mChunks.end(), offset, [](uint64_t off, Chunk const& c) { return off < c.offset; });
122 if (it == mChunks.begin()) {
123 return {};
124 }
125 --it;
126 if (offset - it->offset + length > it->size) {
127 return {};
128 }
129 return {it->data + (offset - it->offset), length};
130}

132void ReadPlan::fetchPending()
133{
134 std::ranges::sort(mPending);
135 std::vector<Chunk> fetched;
136 for (auto [offset, length] : mPending) {
137 if (!lookup(offset, length).empty()) {
138 continue;
139 }
140 auto end = offset + length;
141 if (!fetched.empty()) {
142 auto& last = fetched.back();
143 auto lastEnd = last.offset + last.size;
144 auto next = std::lower_bound(mChunks.begin(), mChunks.end(), lastEnd, [](Chunk const& c, uint64_t off) { return c.offset < off; });
145 bool bridgesFetched = next != mChunks.end() && next->offset < offset;
146 if (offset <= lastEnd + mCoalesceGap && !bridgesFetched) {
147 if (end > lastEnd) {
148 last.size = end - last.offset;
149 }
150 continue;
151 }
152 }
153 fetched.push_back({offset, nullptr, length});
154 }
155 mPending.clear();
156 if (fetched.empty()) {
157 return;
158 }
159 // One buffer for all the chunks of this fetch. Earlier buffers stay where
160 // they are, so spans handed out before remain valid.
161 size_t total = 0;
162 for (auto& chunk : fetched) {
163 total += chunk.size;
164 }
165 auto buffer = mPool->get(total);
166 std::vector<ReadRequest> requests;
167 auto* at = buffer.data.get();
168 for (auto& chunk : fetched) {
169 chunk.data = at;
170 at += chunk.size;
171 requests.push_back({chunk.offset, chunk.size, chunk.data});
172 }
173 mBuffers.push_back(std::move(buffer));
174 mSource->readvInto(requests);
175 mChunks.insert(mChunks.end(), fetched.begin(), fetched.end());
176 std::ranges::sort(mChunks, {}, &Chunk::offset);
177}

179std::span<const uint8_t> ReadPlan::find(uint64_t offset, size_t length)
180{
181 std::lock_guard lock{mMutex};
182 auto bytes = lookup(offset, length);
183 if (bytes.empty() && !mPending.empty()) {
184 fetchPending();
185 bytes = lookup(offset, length);
186 }
187 return bytes;
188}
