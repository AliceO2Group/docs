436template<typename T>
437void FairRootManager::AddMemoryBranchAny(const char* brname, T** obj)
438{
439 if (fAnyBranchMap.find(brname) == fAnyBranchMap.end()) {
440 auto& ot = typeid(T*);
441 auto& pt = typeid(T);
442 fAnyBranchMap[brname] = std::unique_ptr<TypeAddressPair const>(new TypeAddressPair(ot, pt, (void*)obj));
443 }
444}

446// try to retrieve an object address from the registered branches/names
447template<typename T>
448T FairRootManager::GetMemoryBranchAny(const char* brname) const
449{
450 static_assert(std::is_pointer<T>::value, "Return type of GetMemoryBranchAny has to be a pointer");
451 using P = typename std::remove_pointer<T>::type;
452 auto iter = fAnyBranchMap.find(brname);
453 if (iter != fAnyBranchMap.end()) {
454 // verify type consistency
455 if (typeid(P).hash_code() != iter->second->origtypeinfo.hash_code()) {
456 EmitMemoryBranchWrongTypeWarning(brname, typeid(P).name(), iter->second->origtypeinfo.name());
457 return nullptr;
458 }
459 return static_cast<T>(iter->second->ptraddr);
460 }
461 return nullptr;
462}

464template<typename T>
465void FairRootManager::RegisterAny(const char* brname, T*& obj, bool persistence)
466{
467 AddBranchToList(brname);
468 // we are taking the address of the passed pointer
469 AddMemoryBranchAny<T>(brname, &obj);
470 if (persistence) {
471 auto& ot = typeid(T*);
472 auto& pt = typeid(T);
473 if (fSink)
474 fSink->RegisterAny(brname, ot, pt, &obj);
475 else
476 LOG(fatal) << "The sink does not exist to store persistent branches.";
477 }
478}

480// this function serves as a factory (or lookup) for memory managed
481// instances associated to branches
482// it returns a pointer to unmodifiable instance of T
483template<typename TPtr>
484TPtr FairRootManager::InitObjectAs(const char* brname)
485{
486 static_assert(std::is_pointer<TPtr>::value, "Return type of GetObjectAs has to be a pointer");
487 using X = typename std::remove_pointer<TPtr>::type;
488 static_assert(std::is_const<X>::value, "Return type of GetObjectAs has to be pointer to const class");
489 using T = typename std::remove_const<X>::type;
490
491 // is there already an object associated to the branch in memory??
492 // then just return
493 T** obj = GetMemoryBranchAny<T**>(brname);
494 // obj is some address/instance holding TPtr instances
495 if (obj != nullptr)
496 return *obj;
497
498 if (!fSource) {
499 return nullptr;
500 }
501
502 // it does not seem to be the case, let us create the pointer which will be initialized
503 // with the data (pointer to T)
504 T** addr = new T*;
505 // init the pointee to a default obj which we can return
506 (*addr) = new T;
507 // try to find and activate in the source
508 auto succeeded = fSource->ActivateObjectAny((void**)addr, typeid(T), brname);
509
510 if (!succeeded) {
511 delete (*addr);
512 delete addr;
513 return nullptr;
514 }
515 // add into branch list
516 AddMemoryBranchAny<T>(brname, addr);
517 // register as a **requested** branch
518 // (duplications are explicitely allowed)
519 fReqBrNames.emplace_back(brname);
520
521 // NOTE: ideally we would do proper resource management for addr and *addr
522 // since the FairRootManager becomes owner of these pointers/instances; Unfortunately this
523 // is quite a difficult task since we would have to store something like std::unique_ptr<T> in a member
524 // container which we cannot know a priori; Some solutions we could think of in the future are
525 // a) use the Destructor mechanism of ROOT::TClass since we still have the type info.
526 // b) investigate if boost::any could be of help here
527 // In any case, this problem is not very critical in the sense that FairRootManager is a singleton and hence
528 // cannot really leak memory (Assuming that the destructors of T are not doing something non-trivial).
529 return *addr;
530}
