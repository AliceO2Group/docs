29// A sector whose last time bins lie beyond the timeframe limit must not pass their labels to the next sector
30BOOST_AUTO_TEST_CASE(DigitContainer_noLabelCarryOver)
31{
32 auto& cdb = CDBInterface::instance();
33 cdb.setUseDefaults();
34 // limit the flush to one orbit (time bin 447) and propagate the ADC values unchanged
35 o2::conf::ConfigurableParam::updateFromString(fmt::format("DigiParams.maxOrbitsToDigitize=1;TPCEleParam.DigiMode={}", (int)DigitzationMode::PropagateADC));
36 const Mapper& mapper = Mapper::instance();
37 SAMPAProcessing::instance();
38
39 const GlobalPadNumber globalPad = mapper.getPadNumberInROC(PadROCPos(CRU(0).roc(), PadPos(10, 10)));
40 const MCCompLabel labelFirst(1, 1, 0, false);
41 const MCCompLabel labelSecond(2, 2, 0, false);
42
43 DigitContainer digitContainer;
44 digitContainer.reset();
45
46 // first sector: a signal beyond the timeframe limit, which the final flush does not reach
47 digitContainer.addDigit(labelFirst, CRU(0), 600, globalPad, 100);
48 std::vector<Digit> digits;
49 dataformats::MCTruthContainer<MCCompLabel> labels;
50 std::vector<CommonMode> commonMode;
51 digitContainer.fillOutputContainer(digits, labels, commonMode, Sector(0), 0, true, true);
52 BOOST_CHECK(digits.empty());
53
54 // second sector, as after Digitizer::setSector(): same pad, early time bin
55 digitContainer.reset();
56 digits.clear();
57 labels.clear();
58 commonMode.clear();
59 digitContainer.addDigit(labelSecond, CRU(10), 153, globalPad, 100);
60 digitContainer.fillOutputContainer(digits, labels, commonMode, Sector(1), 0, true, true);
61
62 BOOST_REQUIRE(digits.size() == 1);
63 const auto digitLabels = labels.getLabels(0);
64 BOOST_CHECK_EQUAL(digitLabels.size(), 1);
65 for (const auto& label : digitLabels) {
66 BOOST_CHECK(label == labelSecond);
67 }
68}
