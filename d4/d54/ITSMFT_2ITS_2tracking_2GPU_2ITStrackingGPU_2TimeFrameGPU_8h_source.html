364template <int NLayers>
365inline std::vector<unsigned int> TimeFrameGPU<NLayers>::getClusterSizes()
366{
367 std::vector<unsigned int> sizes(this->mUnsortedClusters.size());
368 std::transform(this->mUnsortedClusters.begin(), this->mUnsortedClusters.end(), sizes.begin(),
369 [](const auto& v) { return static_cast<unsigned int>(v.size()); });
370 return sizes;
371}
