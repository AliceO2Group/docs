55ServiceSpec actsGeometryServiceSpec()
56{
57 return ServiceSpec{
58 .name = "acts-geometry",
59 .init = [](ServiceRegistryRef, DeviceState&, fair::mq::ProgOptions&) -> ServiceHandle {
60 // Nothing is built here: the TGeo geometry is not available yet at service
61 // init time. The first getGeometry() call, from run(), triggers the build.
62 return ServiceHandle{TypeIdHelpers::uniqueId<ActsGeometryService>(), new ActsGeometryService(),
63 ServiceKind::DeviceGlobal, "acts-geometry"};
64 },
65 .configure = CommonServices::noConfiguration(),
66 .exit = [](ServiceRegistryRef, void* service) {
67 TrackingGeometryManager::instance().clear();
68 delete reinterpret_cast<ActsGeometryService*>(service); },
69 .kind = ServiceKind::DeviceGlobal};
70}
