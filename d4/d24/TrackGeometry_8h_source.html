146inline float dSector(float phi, int nSectors = 18, float sectorPhaseRad = 0.f)
147{
148 if (!std::isfinite(phi) || !std::isfinite(sectorPhaseRad) || nSectors <= 0) {
149 return std::numeric_limits<float>::quiet_NaN();
150 }
151 const float x = 0.5f * nSectors * (phi - sectorPhaseRad) / o2::constants::math::PI;
152 const float phase = x - std::floor(x);
153 return phase < 1.f ? phase : 0.f;
154}

229inline TrackParametersAtRadius getParametersAtRadiusH(const o2::track::TrackPar& track, float radius, float bzKG,
230 o2::track::DirType dir = o2::track::DirType::DirOutward)
231{
232 TrackParametersAtRadius out;
233 if (!(radius > 0.f) || !std::isfinite(radius) || !std::isfinite(bzKG) || !track.isValid()) {
234 out.status = TrackParametersStatus::InvalidInput;
235 return out;
236 }
237
238 o2::track::TrackPar::dim3_t xyz{}, pxyz{};
239 if (!extractGlobalState(track, xyz, pxyz)) {
240 out.status = TrackParametersStatus::MomentumInvalid;
241 return out;
242 }
243
244 // Use double precision for the analytic construction even though TrackPar and
245 // the public result are float-based. For high-pT tracks the circle centre is
246 // O(1/curvature) away while the requested detector radius is O(10^2 cm).
247 // Performing the circle/circle intersection in float loses several microrad
248 // through cancellation. K0 is therefore the numerical oracle for the final
249 // float state, while H keeps the fast analytic transport.
250 const double x0 = static_cast<double>(xyz[0]);
251 const double y0 = static_cast<double>(xyz[1]);
252 const double z0 = static_cast<double>(xyz[2]);
253 const double px0 = static_cast<double>(pxyz[0]);
254 const double py0 = static_cast<double>(pxyz[1]);
255 const double pz0 = static_cast<double>(pxyz[2]);
256 const double targetR = static_cast<double>(radius);
257 const double pt = std::hypot(px0, py0);
258 if (!(pt > 0.) || !std::isfinite(pt)) {
259 out.status = TrackParametersStatus::MomentumInvalid;
260 return out;
261 }
262
263 const double phi0 = std::atan2(py0, px0);
264 const double tgl = pz0 / pt;
265 const double curv = static_cast<double>(track.getQ2Pt()) *
266 static_cast<double>(bzKG) *
267 static_cast<double>(o2::constants::math::B2C);
268 constexpr double eps = 1.e-10;
269 constexpr double curvEps = 1.e-12;
270 const double twoPi = 2.0 * static_cast<double>(o2::constants::math::PI);
271
272 bool foundAny = false;
273 bool foundDirection = false;
274 double bestSxy = std::numeric_limits<double>::infinity();
275 double bestX = 0., bestY = 0., bestPhiMom = 0.;
276
277 auto directionMatchesD = [&](double drds) {
278 constexpr double dirEps = 1.e-10;
279 if (dir == o2::track::DirType::DirOutward) {
280 return drds >= -dirEps;
281 }
282 if (dir == o2::track::DirType::DirInward) {
283 return drds <= dirEps;
284 }
285 return true;
286 };
287
288 auto consider = [&](double x, double y, double phiMom, double sxy) {
289 if (!std::isfinite(sxy) || sxy < -eps) {
290 return;
291 }
292 foundAny = true;
293 sxy = std::max(0.0, sxy);
294 const double drds = (x * std::cos(phiMom) + y * std::sin(phiMom)) / targetR;
295 if (!directionMatchesD(drds)) {
296 return;
297 }
298 foundDirection = true;
299 if (sxy < bestSxy) {
300 bestSxy = sxy;
301 bestX = x;
302 bestY = y;
303 bestPhiMom = phiMom;
304 }
305 };
306
307 if (std::abs(curv) < curvEps) {
308 const double ux = px0 / pt, uy = py0 / pt;
309 const double b = x0 * ux + y0 * uy;
310 const double c = x0 * x0 + y0 * y0 - targetR * targetR;
311 double disc = b * b - c;
312 if (disc < -eps) {
313 out.status = TrackParametersStatus::UnreachableRadius;
314 return out;
315 }
316 disc = std::max(0.0, disc);
317 const double root = std::sqrt(disc);
318 for (const double sxy : {-b - root, -b + root}) {
319 consider(x0 + sxy * ux, y0 + sxy * uy, phi0, sxy);
320 }
321 } else {
322 const double xc = x0 - std::sin(phi0) / curv;
323 const double yc = y0 + std::cos(phi0) / curv;
324 const double rho = 1.0 / std::abs(curv);
325 const double d = std::hypot(xc, yc);
326 if (!(d > eps) || d > targetR + rho + eps || d < std::abs(targetR - rho) - eps) {
327 out.status = TrackParametersStatus::UnreachableRadius;
328 return out;
329 }
330
331 const double a = (targetR * targetR - rho * rho + d * d) / (2.0 * d);
332 double h2 = targetR * targetR - a * a;
333 if (h2 < -eps) {
334 out.status = TrackParametersStatus::UnreachableRadius;
335 return out;
336 }
337 h2 = std::max(0.0, h2);
338 const double h = std::sqrt(h2);
339 const double bx = a * xc / d, by = a * yc / d;
340 const double ox = -yc * h / d, oy = xc * h / d;
341
342 const std::array<std::array<double, 2>, 2> intersections{{{{bx + ox, by + oy}},
343 {{bx - ox, by - oy}}}};
344 for (const auto& xy : intersections) {
345 const double x = xy[0], y = xy[1];
346 double sinPhi = curv * (x - xc);
347 double cosPhi = -curv * (y - yc);
348 const double norm = std::hypot(sinPhi, cosPhi);
349 if (!(norm > 0.) || !std::isfinite(norm)) {
350 continue;
351 }
352 sinPhi /= norm;
353 cosPhi /= norm;
354 const double phiMom = std::atan2(sinPhi, cosPhi);
355 double delta = phiMom - phi0;
356 if (curv > 0.) {
357 while (delta < -eps) {
358 delta += twoPi;
359 }
360 } else {
361 while (delta > eps) {
362 delta -= twoPi;
363 }
364 }
365 consider(x, y, phiMom, delta / curv);
366 }
367 }
368
369 if (!foundDirection || !std::isfinite(bestSxy)) {
370 out.status = foundAny ? TrackParametersStatus::WrongDirection : TrackParametersStatus::UnreachableRadius;
371 return out;
372 }
373
374 out.gx = static_cast<float>(bestX);
375 out.gy = static_cast<float>(bestY);
376 out.gz = static_cast<float>(z0 + bestSxy * tgl);
377 out.px = static_cast<float>(pt * std::cos(bestPhiMom));
378 out.py = static_cast<float>(pt * std::sin(bestPhiMom));
379 out.pz = static_cast<float>(pz0);
380 out.status = TrackParametersStatus::OK;
381 return out;
382}

397inline TrackParametersAtRadius getParametersAtRadiusK0(const o2::track::TrackPar& track, float radius, float bzKG,
398 o2::track::DirType dir = o2::track::DirType::DirOutward)
399{
400 TrackParametersAtRadius out;
401 if (!(radius > 0.f) || !std::isfinite(radius) || !std::isfinite(bzKG) || !track.isValid()) {
402 out.status = TrackParametersStatus::InvalidInput;
403 return out;
404 }
405
406 float xAtR = 0.f;
407 if (!track.getXatLabR(radius, xAtR, bzKG, dir)) {
408 out.status = TrackParametersStatus::UnreachableRadius;
409 return out;
410 }
411
412 auto propagated = track;
413 if (!propagated.propagateParamTo(xAtR, bzKG)) {
414 out.status = TrackParametersStatus::PropagationFailed;
415 return out;
416 }
417
418 o2::track::TrackPar::dim3_t xyz{}, pxyz{};
419 if (!extractGlobalState(propagated, xyz, pxyz)) {
420 out.status = TrackParametersStatus::MomentumInvalid;
421 return out;
422 }
423
424 // A successful internal TrackPar update is not sufficient: the returned
425 // global state must actually lie on the requested cylinder (Architect A5).
426 constexpr float radiusToleranceCm = 2.e-3f;
427 const float resultRadius = std::hypot(xyz[0], xyz[1]);
428 if (!std::isfinite(resultRadius) || std::abs(resultRadius - radius) > radiusToleranceCm) {
429 out.status = TrackParametersStatus::PropagationFailed;
430 return out;
431 }
432
433 out.gx = xyz[0];
434 out.gy = xyz[1];
435 out.gz = xyz[2];
436 out.px = pxyz[0];
437 out.py = pxyz[1];
438 out.pz = pxyz[2];
439 out.status = TrackParametersStatus::OK;
440 return out;
441}

515template <typename Evaluator>
516inline TrackParametersAtRadiusBatchResult primaryTrackParametersAtRadiusBatch(
517 const float* phi, const float* tgl, const float* qpt,
518 const float* vx, const float* vy, const float* vz,
519 std::size_t n, float radius, float bzKG, int nSectors, float sectorPhaseRad, Evaluator&& evaluate)
520{
521 TrackParametersAtRadiusBatchResult out;
522 out.resize(n);
523 for (std::size_t i = 0; i < n; ++i) {
524 if (!phi || !tgl || !qpt || !std::isfinite(qpt[i]) || std::abs(qpt[i]) < 1.e-12f) {
525 TrackParametersAtRadius invalid;
526 invalid.status = TrackParametersStatus::InvalidInput;
527 out.set(i, invalid, nSectors, sectorPhaseRad);
528 continue;
529 }
530 const auto track = makeTrackParFromPrimaryApproximation(phi[i], tgl[i], qpt[i],
531 vx ? vx[i] : 0.f,
532 vy ? vy[i] : 0.f,
533 vz ? vz[i] : 0.f);
534 out.set(i, evaluate(track, radius, bzKG), nSectors, sectorPhaseRad);
535 }
536 return out;
537}

838inline float rowRadius(const o2::tpc::Mapper& mapper, int row)
839{

900inline bool trackOutwardReference(const o2::track::TrackPar& track, float& radius)
901{
902 o2::track::TrackPar::dim3_t xyz{}, pxyz{};
903 if (!track.isValid() || !o2::recohelpers::extractGlobalState(track, xyz, pxyz)) {
904 return false;
905 }
906 radius = std::hypot(xyz[0], xyz[1]);
907 if (!std::isfinite(radius) || !std::isfinite(track.getQ2Pt()) ||
908 std::abs(track.getQ2Pt()) < 1.e-12f) {
909 return false;
910 }
911 // Negative radial motion at the supplied reference is not the first outward branch.
912 return radius < 1.e-5f || xyz[0] * pxyz[0] + xyz[1] * pxyz[1] >= 0.f;
913}

916inline TrajectoryPoint helixAtRadius(const o2::track::TrackPar& track, float radius, float bzKG)
917{
918 TrajectoryPoint p;
919 if (!track.isValid() || !std::isfinite(bzKG) || !(radius > 0.f)) {
920 return p;
921 }
922 const auto state = o2::recohelpers::getParametersAtRadiusH(
923 track, radius, bzKG, o2::track::DirType::DirOutward);
924 if (!state.valid()) {
925 return p;
926 }
927 p.x = state.gx;
928 p.y = state.gy;
929 p.z = state.gz;
930 p.phiPos = o2::recohelpers::statePhiPos(state);
931 p.phiMom = o2::recohelpers::statePhiMom(state);
932 const float c = std::abs(std::cos(p.phiMom - p.phiPos));
933 const float tgl = o2::recohelpers::stateTgl(state);
934 p.dsDr = std::sqrt(1.f + tgl * tgl) / std::max(c, 1.e-6f);
935 p.valid = std::isfinite(p.dsDr);
936 return p;
937}

959inline void accumulateRow(const o2::tpc::Mapper& mapper, int row, const TrajectoryPoint& p, float rowLength,
960 const TPCSectorEdgeConfig& cfg, TPCSectorEdgeResult& out)
961{
962 if (!rowWithinAcceptance(p, cfg) || !std::isfinite(rowLength) || rowLength <= 0.f) {
963 return;
964 }
965
966 const float phi = std::atan2(p.y, p.x);
967 const int sector = o2::math_utils::angle2Sector(phi);
968 const float alpha = o2::math_utils::sector2Angle(sector);
969 const float localY = -p.x * std::sin(alpha) + p.y * std::cos(alpha);
970 const float halfWidth = rowActiveHalfWidth(mapper, row);
971 const float edgeDistance = halfWidth - std::abs(localY);
972 const float guard = localY < 0.f ? cfg.guardLowerCm : cfg.guardUpperCm;
973
974 ++out.nRowsCrossed;
975 out.maxRadiusCm = rowRadius(mapper, row);
976 out.lengthCrossedCm += rowLength;
977 setMaskBit(out.crossedMask, row);
978
979 if (edgeDistance < 0.f) {
980 ++out.nRowsDead;
981 out.lengthDeadCm += rowLength;
982 setMaskBit(out.deadMask, row);
983 if (out.firstDeadRow < 0) {
984 out.firstDeadRow = row;
985 }
986 out.lastDeadRow = row;
987 } else if (edgeDistance < guard) {
988 ++out.nRowsExcluded;
989 out.lengthExcludedCm += rowLength;
990 setMaskBit(out.excludedMask, row);
991 } else {
992 ++out.nRowsAvailable;
993 out.lengthAvailableCm += rowLength;
994 setMaskBit(out.availableMask, row);
995 }
996}

1003inline TPCSectorEdgeResult sectorEdgeHelixRows(const o2::track::TrackPar& track, float bzKG, const TPCSectorEdgeConfig& cfg = {},
1004 std::vector<o2::recohelpers::TrackParametersAtRadius>* trajectory = nullptr)
1005{
1006 TPCSectorEdgeResult out;
1007 if (trajectory) {
1008 trajectory->clear();
1009 }
1010 float startRadius = 0.f;
1011 if (!std::isfinite(bzKG) || !trackOutwardReference(track, startRadius)) {
1012 out.status = SectorEdgeStatus::InvalidInput;
1013 return out;
1014 }
1015 const auto& mapper = o2::tpc::Mapper::instance();
1016
1017 for (int row = 0; row < mapper.getNumberOfRows(); ++row) {
1018 const float r = rowRadius(mapper, row);
1019 if (r + 1.e-4f < startRadius) {
1020 if (trajectory) {
1021 trajectory->emplace_back(); // Preserve trajectory[row] indexing for skipped rows.
1022 }
1023 continue; // A full track can start inside the TPC; do not count earlier rows.
1024 }
1025 const auto p = helixAtRadius(track, r, bzKG);
1026 if (!p.valid) {
1027 break; // a larger radius cannot re-enter the first outward primary-helix branch
1028 }
1029 if (trajectory) {
1030 appendTrajectoryState(*trajectory, p, track.getQ2Pt(), track.getTgl());
1031 }
1032 if (reachedMaxSnp(p, cfg)) {
1033 break;
1034 }
1035 accumulateRow(mapper, row, p, rowLengthRadial(mapper, row) * p.dsDr, cfg, out);
1036 }
1037
1038 if (!out.nRowsCrossed) {
1039 out.status = SectorEdgeStatus::NoRowsCrossed;
1040 }
1041 return out;
1042}

1065inline bool plausibleFirstOutwardStep(float dr, float ds, float tgl, float maxSnp)
1066{
1067 if (!(dr > 0.f) || !(ds > 0.f) || !std::isfinite(ds) || !std::isfinite(tgl) ||
1068 !(maxSnp > 0.f && maxSnp < 1.f)) {
1069 return false;
1070 }
1071 const float minRadialCos = std::sqrt(1.f - maxSnp * maxSnp);
1072 return ds <= 10.f * dr * std::sqrt(1.f + tgl * tgl) / minRadialCos;
1073}

1086inline TPCSectorEdgeResult sectorEdgeO2PropagatorRows(const o2::track::TrackPar& inputTrack, float bzKG,
1087 const TPCSectorEdgeConfig& cfg = {},
1088 std::vector<o2::recohelpers::TrackParametersAtRadius>* trajectory = nullptr)
1089{
1090 TPCSectorEdgeResult out;
1091 if (trajectory) {
1092 trajectory->clear();
1093 }
1094 const auto& mapper = o2::tpc::Mapper::instance();
1095 auto* prop = o2::base::Propagator::Instance();
1096 if (!prop->hasMagFieldSet()) {
1097 out.status = SectorEdgeStatus::FieldNotInitialized;
1098 return out;
1099 }
1100 float startRadius = 0.f;
1101 if (!std::isfinite(bzKG) || !trackOutwardReference(inputTrack, startRadius) ||
1102 !std::isfinite(cfg.maxSnp) || !(cfg.maxSnp > 0.f && cfg.maxSnp < 1.f)) {
1103 out.status = SectorEdgeStatus::InvalidInput;
1104 return out;
1105 }
1106
1107 int firstRow = 0;
1108 while (firstRow < mapper.getNumberOfRows() &&
1109 rowRadius(mapper, firstRow) + 1.e-4f < startRadius) {
1110 ++firstRow;
1111 }
1112 if (firstRow == mapper.getNumberOfRows()) {
1113 out.status = SectorEdgeStatus::NoRowsCrossed;
1114 return out;
1115 }
1116 const float r0 = rowRadius(mapper, firstRow);
1117 if (trajectory) {
1118 trajectory->resize(firstRow); // Keep diagnostic trace indexed by global TPC row.
1119 }
1120 // Preserve the reviewed K first-row seeding convention, but use the complete
1121 // original track rather than reconstructing a primary vertex. K field-map
1122 // transport from the original reference remains a separate physics-validation item.
1123 const auto p0 = helixAtRadius(inputTrack, r0, bzKG);
1124 if (!p0.valid) {
1125 out.status = SectorEdgeStatus::NoRowsCrossed;
1126 return out;
1127 }
1128
1129 const float qpt = inputTrack.getQ2Pt();
1130 const float pt = 1.f / std::abs(qpt);
1131 const int charge = qpt >= 0.f ? 1 : -1;
1132 o2::track::TrackPar::dim3_t xyz{{p0.x, p0.y, p0.z}};
1133 o2::track::TrackPar::dim3_t pxyz{{pt * std::cos(p0.phiMom), pt * std::sin(p0.phiMom), pt * inputTrack.getTgl()}};
1134 o2::track::TrackPar track(xyz, pxyz, charge, true);
1135 o2::track::TrackLTIntegral path;
1136 path.clear();
1137 path.setTimeNotNeeded();
1138 float previousPathCm = 0.f;
1139 float previousRadiusCm = r0;
1140
1141 for (int row = firstRow; row < mapper.getNumberOfRows(); ++row) {
1142 const float radius = rowRadius(mapper, row);
1143 if (row > firstRow) {
1144 // Record accumulated transported path length. A later outward *orbit* can have positive dr/ds
1145 // and lie on the requested cylinder; radialCos > 0 alone would accept that wrong branch.
1146 // With radial |snp| < maxSnp, a first-branch step should be short relative to its radial
1147 // increment. This deliberately generous guard only rejects gross discontinuities; numerical
1148 // H/K row-mask agreement is checked separately by the physics oracle.
1149 if (!prop->propagateToR(track, radius, false, o2::base::Propagator::MAX_SIN_PHI,
1150 o2::base::Propagator::MAX_STEP,
1151 o2::base::Propagator::MatCorrType::USEMatCorrNONE, &path)) {
1152 out.status = SectorEdgeStatus::PropagationFailed;
1153 return out;
1154 }
1155 const float deltaPathCm = path.getL() - previousPathCm;
1156 const float deltaRadiusCm = radius - previousRadiusCm;
1157 if (!plausibleFirstOutwardStep(deltaRadiusCm, deltaPathCm, track.getTgl(), cfg.maxSnp)) {
1158 out.status = SectorEdgeStatus::PropagationFailed;
1159 return out;
1160 }
1161 previousPathCm = path.getL();
1162 previousRadiusCm = radius;
1163 }
1164
1165 const auto xyzg = track.getXYZGlo();
1166 TrajectoryPoint p;
1167 p.x = xyzg.X();
1168 p.y = xyzg.Y();
1169 p.z = xyzg.Z();
1170 p.phiPos = track.getPhiPos();
1171 p.phiMom = track.getPhi();
1172 const float dphi = p.phiMom - p.phiPos;
1173 const float radialCos = std::cos(dphi);
1174 // propagateToR() may return a state on a different branch or slightly off the target cylinder.
1175 // Neither may contribute a crossed row. The radial-frame snp limit is checked independently below.
1176 constexpr float radiusToleranceCm = 2.e-3f;
1177 const float actualRadius = std::hypot(p.x, p.y);
1178 if (!std::isfinite(actualRadius) || std::abs(actualRadius - radius) > radiusToleranceCm ||
1179 !std::isfinite(radialCos)) {
1180 out.status = SectorEdgeStatus::PropagationFailed;
1181 return out;
1182 }
1183 if (radialCos <= 0.f) {
1184 break; // reached the transverse turning point or a returning arc
1185 }
1186 p.dsDr = std::sqrt(1.f + track.getTgl() * track.getTgl()) / radialCos;
1187 p.valid = true;
1188 if (trajectory) {
1189 appendTrajectoryState(*trajectory, p, track.getQ2Pt(), track.getTgl());
1190 }
1191 if (reachedMaxSnp(p, cfg)) {
1192 break;
1193 }
1194 accumulateRow(mapper, row, p, rowLengthRadial(mapper, row) * p.dsDr, cfg, out);
1195 }
1196
1197 if (!out.nRowsCrossed && out.status == SectorEdgeStatus::OK) {
1198 out.status = SectorEdgeStatus::NoRowsCrossed;
1199 }
1200 return out;
1201}

1226inline TPCSectorEdgeResult sectorEdgeStackLinear(const o2::track::TrackPar& track, float bzKG, const TPCSectorEdgeConfig& cfg = {},
1227 std::vector<o2::recohelpers::TrackParametersAtRadius>* trajectory = nullptr)
1228{
1229 TPCSectorEdgeResult out;
1230 if (trajectory) {
1231 trajectory->clear();
1232 }
1233 float startRadius = 0.f;
1234 if (!std::isfinite(bzKG) || !trackOutwardReference(track, startRadius)) {
1235 out.status = SectorEdgeStatus::InvalidInput;
1236 return out;
1237 }
1238 const auto& mapper = o2::tpc::Mapper::instance();
1239
1240 int first = 0;
1241 while (first < mapper.getNumberOfRows()) {
1242 const float h = rowLengthRadial(mapper, first);
1243 int last = first;
1244 while (last + 1 < mapper.getNumberOfRows() && std::abs(rowLengthRadial(mapper, last + 1) - h) < 1.e-6f) {
1245 ++last;
1246 }
1247
1248 if (rowRadius(mapper, first) + 1.e-4f < startRadius) {
1249 if (trajectory) {
1250 trajectory->emplace_back(); // Preserve global row indexing.
1251 }
1252 ++first; // Ignore rows inside the complete track's reference radius.
1253 continue;
1254 }
1255 const float r0 = rowRadius(mapper, first);
1256 const auto p0 = helixAtRadius(track, r0, bzKG);
1257 if (!p0.valid || reachedMaxSnp(p0, cfg)) {
1258 break;
1259 }
1260
1261 // The frozen Linear contract keeps a reachable partial stack. If the nominal
1262 // outer endpoint is unreachable or already beyond maxSnp, walk inward to the
1263 // last reachable row below maxSnp and use that row as the interpolation endpoint.
1264 int endpoint = last;
1265 auto p1 = helixAtRadius(track, rowRadius(mapper, endpoint), bzKG);
1266 bool stopAfterStack = false;
1267 while (endpoint > first && (!p1.valid || reachedMaxSnp(p1, cfg))) {
1268 --endpoint;
1269 p1 = helixAtRadius(track, rowRadius(mapper, endpoint), bzKG);
1270 stopAfterStack = true;
1271 }
1272 if (!p1.valid || reachedMaxSnp(p1, cfg)) {
1273 break;
1274 }
1275 stopAfterStack = stopAfterStack || endpoint < last;
1276
1277 const float r1 = rowRadius(mapper, endpoint);
1278 float dphi = p1.phiPos - p0.phiPos;
1279 o2::math_utils::detail::bringToPMPi(dphi);
1280 const float dr = std::max(r1 - r0, 1.e-6f);
1281 const float dphiDr = dphi / dr, dzDr = (p1.z - p0.z) / dr;
1282
1283 for (int row = first; row <= endpoint; ++row) {
1284 const float r = rowRadius(mapper, row), f = (r - r0) / dr;
1285 const float phi = p0.phiPos + f * dphi;
1286 TrajectoryPoint p;
1287 p.x = r * std::cos(phi);
1288 p.y = r * std::sin(phi);
1289 p.z = p0.z + f * (p1.z - p0.z);
1290 p.phiPos = phi;
1291 p.phiMom = phi + std::atan(r * dphiDr);
1292 p.dsDr = std::sqrt(1.f + (r * dphiDr) * (r * dphiDr) + dzDr * dzDr);
1293 // When the limit leaves a single reachable row, finite-difference derivatives are undefined.
1294 // Reuse the actual helix tangent and path length at that row rather than setting the slopes to zero.
1295 if (endpoint == first) {
1296 p.phiMom = p0.phiMom;
1297 p.dsDr = p0.dsDr;
1298 }
1299 p.valid = true;
1300 if (trajectory) {
1301 appendTrajectoryState(*trajectory, p, track.getQ2Pt(), track.getTgl());
1302 }
1303 if (reachedMaxSnp(p, cfg)) {
1304 stopAfterStack = true;
1305 break;
1306 }
1307 accumulateRow(mapper, row, p, rowLengthRadial(mapper, row) * p.dsDr, cfg, out);
1308 }
1309
1310 if (stopAfterStack) {
1311 break;
1312 }
1313 first = last + 1;
1314 }
1315
1316 if (!out.nRowsCrossed) {
1317 out.status = SectorEdgeStatus::NoRowsCrossed;
1318 }
1319 return out;
1320}
