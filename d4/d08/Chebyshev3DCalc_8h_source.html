181 EvalScratch(int nrows, int ncols)
182 {
183 if (nrows <= kStackSize && ncols <= kStackSize) {
184 mRows = mStackRows;
185 mCols = mStackCols;
186 } else {
187 thread_local std::vector<Float_t> rows, cols;
188 rows.resize(std::max<size_t>(nrows, rows.size()));
189 cols.resize(std::max<size_t>(ncols, cols.size()));
190 mRows = rows.data();
191 mCols = cols.data();
192 }
193 }
194 EvalScratch(const EvalScratch&) = delete;
195 EvalScratch& operator=(const EvalScratch&) = delete;
196 Float_t* rows() { return mRows; }
197 Float_t* cols() { return mCols; }

232inline Float_t Chebyshev3DCalc::chebyshevEvaluation1D(Float_t x, const Float_t* array, int ncf)
233{
234 if (ncf <= 0) {
235 return 0;
236 }
237
238 Float_t b0, b1, b2, x2 = x + x;
239 b0 = array[--ncf];
240 b1 = b2 = 0;
241
242 for (int i = ncf; i--;) {
243 b2 = b1;
244 b1 = b0;
245 // Clenshaw recurrence, grouped as fma(x2, b1, array[i] - b2). Mathematically
246 // identical to `array[i] + x2 * b1 - b2`, but `array[i] - b2` does not depend
247 // on the just-updated b1, so the loop-carried chain collapses to a single FMA
248 // latency instead of a dependent multiply+add+subtract. This kernel dominates
249 // magnetic-field evaluation in (e.g.) muon track extrapolation.
250 b0 = std::fma(x2, b1, array[i] - b2);
251 }
252 return std::fma(-x, b1, b0);
253}

257inline Float_t Chebyshev3DCalc::Eval(const Float_t* par) const
258{
259 EvalScratch scratch(mNumberOfRows, mNumberOfColumns);
260 Float_t* coefs2D = scratch.cols();
261 Float_t* coefs1D = scratch.rows();
262 for (int id0 = mNumberOfRows; id0--;) {
263 int nCLoc = mNumberOfColumnsAtRow[id0]; // number of significant coefs on this row
264 int col0 = mColumnAtRowBeginning[id0]; // beginning of local column in the 2D boundary matrix
265 for (int id1 = nCLoc; id1--;) {
266 int id = id1 + col0;
267 coefs2D[id1] = chebyshevEvaluation1D(par[2], mCoefficients + mCoefficientBound2D1[id], mCoefficientBound2D0[id]);
268 }
269 coefs1D[id0] = chebyshevEvaluation1D(par[1], coefs2D, nCLoc);
270 }
271 return chebyshevEvaluation1D(par[0], coefs1D, mNumberOfRows);
272}
