23// Serializes values big-endian, as ROOT stores them.
24template <typename T>
25std::vector<uint8_t> bigEndian(std::vector<T> const& values)
26{
27 BigEndianWriter w;
28 for (auto v : values) {
29 w.put(std::bit_cast<std::conditional_t<sizeof(T) == 4, uint32_t, std::conditional_t<sizeof(T) == 8, uint64_t, std::conditional_t<sizeof(T) == 2, uint16_t, uint8_t>>>>(v));
30 }
31 return w.bytes;
32}

35struct BranchBuilder {
36 std::vector<uint8_t> file = std::vector<uint8_t>(100, 0); // some leading bytes, so seeks are non-zero
37 BranchInfo branch;
38
39 BranchBuilder(char leafClass, int fixedLen = 1)
40 {
41 branch.name = "fX";
42 branch.leafClass = leafClass;
43 branch.isUnsigned = false;
44 branch.fixedLen = fixedLen;
45 branch.entries = branch.totBytes = branch.zipBytes = 0;
46 }
47
48 void add(Algo algo, std::vector<uint8_t> const& values, int32_t nEntries, size_t tail = 0, size_t chunkSize = 0xffffff, bool ioBits = false)
49 {
50 auto basket = makeBasket(algo, values, nEntries, tail, chunkSize, ioBits);
51 branch.baskets.push_back(BasketRef{file.size(), (uint32_t)basket.size(), branch.entries, nEntries, false, {}});
52 file.insert(file.end(), basket.begin(), basket.end());
53 branch.entries += nEntries;
54 }
55
56 void addEmbedded(std::vector<uint8_t> const& values, int32_t nEntries)
57 {
58 branch.baskets.push_back(BasketRef{0, 0, branch.entries, nEntries, true, values});
59 branch.entries += nEntries;
60 }
61};

75TEST_CASE("float values across codecs, uncompressed and embedded baskets")
76{
77 BranchBuilder b{'F'};
78 std::vector<float> expected;
79 int n = 0;
80 for (auto algo : {Algo::ZSTD, Algo::None, Algo::ZLIB, Algo::LZMA, Algo::LZ4}) {
81 std::vector<float> values;
82 for (int i = 0; i < 1000; ++i) {
83 values.push_back(n++ * 0.5f);
84 }
85 // Entry-offset tail after the values must never reach the destination.
86 b.add(algo, bigEndian(values), values.size(), 4004);
87 expected.insert(expected.end(), values.begin(), values.end());
88 }
89 std::vector<float> tail{-1.f, -2.f, -3.f};
90 b.addEmbedded(bigEndian(tail), tail.size());
91 expected.insert(expected.end(), tail.begin(), tail.end());
92
93 for (bool mappable : {true, false}) {
94 for (int threads : {1, 4}) {
95 MemoryByteSource source{b.file, mappable};
96 Options options;
97 options.threads = threads;
98 BranchDecoder decoder{source, options};
99 std::vector<float> out(expected.size(), 42.f);
100 decoder.decodeValues(b.branch, (uint8_t*)out.data(), out.size() * sizeof(float));
101 CHECK(out == expected);
102 }
103 }
104}

125TEST_CASE("decoding stops exactly at the value region for every codec")
126{
127 // A poisoned entry-offset tail after the values: nothing past fLast - fKeylen
128 // may reach the destination, which is followed by guard bytes. Sizes span
129 // several zstd blocks and compression chunks.
130 for (auto algo : {Algo::ZSTD, Algo::ZLIB, Algo::LZMA, Algo::LZ4, Algo::None}) {
131 for (size_t chunkSize : {size_t(100000), size_t(0xffffff)}) {
132 INFO("algorithm " << (int)algo << ", chunk size " << chunkSize);
133 BranchBuilder b{'D'};
134 std::vector<double> values(70001);
135 for (size_t i = 0; i < values.size(); ++i) {
136 values[i] = i * 0.25 - 7;
137 }
138 b.add(algo, bigEndian(values), values.size(), 8 * values.size() + 4, chunkSize);
139 MemoryByteSource source{b.file};
140 BranchDecoder decoder{source, Options{}};
141 std::vector<double> out(values.size() + 16, 1e300);
142 decoder.decodeValues(b.branch, (uint8_t*)out.data(), values.size() * sizeof(double));

162TEST_CASE("booleans are packed across basket and window boundaries")
163{
164 // Basket sizes chosen so that boundaries fall inside bitmap bytes, and one
165 // basket is larger than the 64 KiB window.
166 BranchBuilder b{'O'};
167 std::vector<bool> expected;
168 for (int size : {3, 13, (int)BranchDecoder::kBooleanWindow + 5, 1, 7}) {
169 std::vector<uint8_t> bytes;
170 for (int i = 0; i < size; ++i) {
171 bool v = ((expected.size() * 7) % 3) == 0;
172 expected.push_back(v);
173 // Any non-zero byte is true.
174 bytes.push_back(v ? (i % 2 ? 1 : 0x80) : 0);
175 }
176 b.add(Algo::ZSTD, bytes, size, 8, 30000);
177 }
178 MemoryByteSource source{b.file};
179 BranchDecoder decoder{source, Options{}};
180 std::vector<uint8_t> bitmap((expected.size() + 7) / 8, 0xFF);
181 decoder.decodeBooleans(b.branch, bitmap.data(), bitmap.size());
182 for (size_t i = 0; i < expected.size(); ++i) {
183 INFO("value " << i);
184 CHECK(((bitmap[i / 8] >> (i % 8)) & 1) == expected[i]);
185 }
186 // Padding bits are zero.
187 for (size_t i = expected.size(); i < bitmap.size() * 8; ++i) {
188 CHECK(((bitmap[i / 8] >> (i % 8)) & 1) == 0);
189 }
190}

192TEST_CASE("variable-length arrays")
193{
194 BranchBuilder sizes{'I'};
195 sizes.branch.name = "fY_size";
196 std::vector<int32_t> s1{2, 0, 3}, s2{1};
197 sizes.add(Algo::ZLIB, bigEndian(s1), 3);
198 sizes.add(Algo::None, bigEndian(s2), 1);
199
200 // Values baskets do not need to align with size baskets.
201 BranchBuilder values{'D'};
202 values.branch.name = "fY";
203 values.branch.countBranch = "fY_size";
204 std::vector<double> v1{1.5, 2.5}, v2{3.5, 4.5, 5.5, 6.5};
205 values.add(Algo::LZ4, bigEndian(v1), 2);
206 values.add(Algo::ZSTD, bigEndian(v2), 2);
207
208 std::vector<uint8_t> file = sizes.file;
209 // Place the values baskets after the size baskets.
210 auto shift = file.size() - 100;
211 file.insert(file.end(), values.file.begin() + 100, values.file.end());
212 for (auto& basket : values.branch.baskets) {
213 basket.seek += shift;
214 }
215
216 MemoryByteSource source{file};
217 BranchDecoder decoder{source, Options{}};
218 auto offsets = decoder.decodeOffsets(sizes.branch);
219 CHECK(offsets == std::vector<int32_t>{0, 2, 2, 5, 6});
220 std::vector<double> out(6);
221 decoder.decodeVLAValues(values.branch, offsets, (uint8_t*)out.data(), out.size() * sizeof(double));
222 CHECK(out == std::vector<double>{1.5, 2.5, 3.5, 4.5, 5.5, 6.5});
223}
