44void GPUDisplayGUI::timerTick()
45{
46 std::unique_lock lock(mChannel.mutex);
47 mChannel.guiSleeping = false;
48 if (mUpdate.empty()) { // Display processes the update we went to sleep with, wait for the reply (with new changes we would not apply it anyway)
49 mChannel.stateUpdated.wait_for(lock, mChannel.replyTimeout, [this] { return !mChannel.displayProcessingCurrentUpdate || mChannel.stop; });
50 }
51 if (!mUpdate.empty() && mLastSentId - mLastAppliedId >= mChannel.maxUpdatesAhead) { // Too far ahead of the display, wait for the reply to the last update
52 mChannel.stateUpdated.wait_for(lock, mChannel.replyTimeout, [this] { return mChannel.state.appliedUpdate == mLastSentId || mChannel.stop; });
53 }
54 if (mChannel.stop) {
55 lock.unlock();
56 close();
57 return;
58 }
59
60 const bool newState = mChannel.state.counter != mState.counter;
61 if (newState) {
62 mState = mChannel.state;
63 if (mState.appliedUpdate == mLastSentId) {
64 mLastAppliedId = mLastSentId;
65 }
66 }
67 const bool apply = newState && mUpdate.empty() && mState.appliedUpdate == mLastSentId; // State contains all our updates, and we have no unsent changes
68 if (!mUpdate.empty() && mLastSentId - mLastAppliedId < mChannel.maxUpdatesAhead) {
69 if (mChannel.update) { // Previous update not yet picked up by the display, merge
70 auto& update = *mChannel.update;
71 update.keys.insert(update.keys.end(), mUpdate.keys.begin(), mUpdate.keys.end());
72 if (mUpdate.lineWidth) {
73 update.lineWidth = mUpdate.lineWidth;
74 }
75 } else {
76 mChannel.update = std::move(mUpdate);
77 }
78 mChannel.update->id = ++mLastSentId;
79 mChannel.guiSleeping = true;
80 mUpdate = {};
81 }
82 lock.unlock();
83
84 if (apply) {
85 applySettings();
86 }
87}

114int32_t GPUDisplayGUI::translateKey(const QKeyEvent* event)
115{
116 switch (event->key()) {
117 case Qt::Key_Up:
118 return GPUDisplayFrontend::KEY_UP;
119 case Qt::Key_Down:
120 return GPUDisplayFrontend::KEY_DOWN;
121 case Qt::Key_Left:
122 return GPUDisplayFrontend::KEY_LEFT;
123 case Qt::Key_Right:
124 return GPUDisplayFrontend::KEY_RIGHT;
125 case Qt::Key_PageUp:
126 return GPUDisplayFrontend::KEY_PAGEUP;
127 case Qt::Key_PageDown:
128 return GPUDisplayFrontend::KEY_PAGEDOWN;
129 case Qt::Key_Home:
130 return GPUDisplayFrontend::KEY_HOME;
131 case Qt::Key_End:
132 return GPUDisplayFrontend::KEY_END;
133 case Qt::Key_Insert:
134 return GPUDisplayFrontend::KEY_INSERT;
135 case Qt::Key_Escape:
136 return GPUDisplayFrontend::KEY_ESCAPE;
137 case Qt::Key_Return:
138 case Qt::Key_Enter:
139 return GPUDisplayFrontend::KEY_ENTER;
140 default:
141 break;
142 }
143 if (event->key() >= Qt::Key_F1 && event->key() <= Qt::Key_F12) {
144 static_assert(GPUDisplayFrontend::KEY_F12 - GPUDisplayFrontend::KEY_F1 == Qt::Key_F12 - Qt::Key_F1);
145 return GPUDisplayFrontend::KEY_F1 + (event->key() - Qt::Key_F1);
146 }
147 const QString text = event->text();
148 if (text.size() == 1 && text[0].unicode() >= 32 && text[0].unicode() < 127) { // Printable ASCII, with shift applied
149 return text[0].unicode();
150 }
151 return -1;
152}
