32{

34 const std::string_view rtcSource(_binary_GPUReconstructionCUDArtc_src_start, _binary_GPUReconstructionCUDArtc_src_len);
35 const std::string rtcparam = rtc::paramCode(proc, param());
36 rtc::tmpFilename(filename, "/tmp/o2cagpu_rtc_");
37
38 std::vector<std::string> kernels;
39 getRTCKernelCalls(kernels);
40 const std::string kernelsall = rtc::joinKernels(kernels);

42 const std::string baseCommand = rtc::baseCommand(proc, std::string_view(_binary_GPUReconstructionCUDArtc_command_start, _binary_GPUReconstructionCUDArtc_command_len)) + " " +
43 (proc.rtctech.overrideArchitecture != "" ? proc.rtctech.overrideArchitecture : std::string(_binary_GPUReconstructionCUDArtc_command_arch_start, _binary_GPUReconstructionCUDArtc_command_arch_len));
44 const std::string noFastMath(_binary_GPUReconstructionCUDArtc_command_no_fast_math_start, _binary_GPUReconstructionCUDArtc_command_no_fast_math_len);

46 rtc::loadLaunchBounds(proc, *mParDevice);
47 if constexpr (std::string_view("CUDA") == "HIP") { // Check if we are RTC-compiling for HIP
48 if (proc.hipOverrideAMDEUSperCU > 0) {
49 mParDevice->par_AMD_EUS_PER_CU = proc.hipOverrideAMDEUSperCU;
50 } else if (mParDevice->par_AMD_EUS_PER_CU <= 0) {

58 }
59 const std::string launchBounds = rtc::launchBoundsCode(proc, *mParDevice, mParDevice->par_AMD_EUS_PER_CU ? (mParDevice->par_AMD_EUS_PER_CU * mWarpSize) : 0, mWarpSize);
60 if (proc.rtctech.printLaunchBounds >= 2) {
61 return 1;
62 }
63

65 const std::string compilerVersions = getBackendVersions();

67 if (!cache.load(filename, mRtcBinExtension, nCompile)) {
68 rtc::compile(proc, nCompile, "CUDA", [&](uint32_t i) {
69 const std::string fileBase = filename + "_" + std::to_string(i);
70 if (proc.debugLevel >= 3) {
71 printf("Compiling %s\n", (fileBase + mRtcSrcExtension).c_str());
72 }
73 const bool deterministic = rtc::deterministic(proc, GetKernelName(i));
74 const std::string kernel = std::string("extern \"C\" {") + (proc.rtc.compilePerKernel ? kernels[i] : kernelsall) + "}";

76 const std::string command = baseCommand + (deterministic ? " " + noFastMath : "") + " -c " + fileBase + mRtcSrcExtension + " -o " + fileBase + mRtcBinExtension;
77 rtc::runCommand(proc, command, fileBase + mRtcSrcExtension, "Error during CUDA compilation");
78 });
79 cache.store(filename, mRtcBinExtension, nCompile);
80 }
81 return 0;
