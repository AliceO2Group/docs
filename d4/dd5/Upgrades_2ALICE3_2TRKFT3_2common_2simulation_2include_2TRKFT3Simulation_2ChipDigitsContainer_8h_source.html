71inline o2::trkft3::PreDigit* ChipDigitsContainer::findDigit(ULong64_t key)
72{
73 // finds the digit corresponding to global key
74 auto digitentry = mDigits.find(key);
75 return digitentry != mDigits.end() ? &(digitentry->second) : nullptr;
76}

87template <int DetID>
88void o2::trkft3::ChipDigitsContainer::addNoise(UInt_t rofMin, UInt_t rofMax, const o2::trkft3::DigiParams<DetID>* params, int subDetID, int layer)
89{
90 UInt_t row = 0;
91 UInt_t col = 0;
92 Int_t nhits = 0;
93 float mean = 0.f;
94 int nel = 0;
95 int maxRows = 0;
96 int maxCols = 0;
97
98 if (subDetID == 0) {
99 maxRows = o2::trk::constants::VD::petal::layer::nRows[layer];
100 maxCols = o2::trk::constants::VD::petal::layer::nCols;
101 } else {
102 maxRows = o2::trk::constants::moduleMLOT::chip::nRows;
103 maxCols = o2::trk::constants::moduleMLOT::chip::nCols;
104 }
105 mean = params->getNoisePerPixel() * maxRows * maxCols;
106 nel = static_cast<int>(params->getChargeThreshold() * 1.1);
107
108 LOG(debug) << "Adding noise for chip " << mChipIndex << " with mean " << mean << " and charge " << nel;
109
110 for (UInt_t rof = rofMin; rof <= rofMax; rof++) {
111 nhits = gRandom->Poisson(mean);
112 for (Int_t i = 0; i < nhits; ++i) {
113 row = gRandom->Integer(maxRows);
114 col = gRandom->Integer(maxCols);
115 LOG(debug) << "Generated noise hit at ROF " << rof << ", row " << row << ", col " << col;
116 if (mNoiseMap && mNoiseMap->isNoisy(mChipIndex, row, col)) {
117 continue;
118 }
119 if (mDeadChanMap && mDeadChanMap->isNoisy(mChipIndex, row, col)) {
120 continue;
121 }
122 auto key = getOrderingKey(rof, row, col);
123 if (!findDigit(key)) {
124 addDigit(key, rof, row, col, nel, o2::MCCompLabel(true), 0.f);
125 }
126 }
127 }
128}
