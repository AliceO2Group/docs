74void FairRun::StoreTaskNames(const FairTask* task)
75{
76 if (task) {
77 fFileHeader->AddTaskClassName(task->ClassName());
78 fNTasks++;
79
80 const TList* subtasks = task->GetListOfTasks();
81 if (subtasks) {
82 for (const auto subtask : *subtasks) {
83 StoreTaskNames(static_cast<FairTask*>(subtask));
84 }
85 }
86 }
87}

89void FairRun::SetTask(FairTask* task)
90{
91 delete fTask;
92 fTask = task;
93 fNTasks = 0;
94 TList* tasklist = fFileHeader->GetListOfTasks();
95 if (tasklist) {
96 tasklist->Clear();
97 StoreTaskNames(task);
98 }
99}
