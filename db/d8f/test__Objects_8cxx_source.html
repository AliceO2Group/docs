21TEST_CASE("Cursor reads big-endian integers")
22{
23 std::vector<uint8_t> buf{0x12, 0x34, 0x56, 0x78, 0x9a, 0xbc, 0xde, 0xf0, 0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07};
24 std::string file = "memory";
25 Cursor c{buf, file};
26 CHECK(c.u16() == 0x1234);
27 CHECK(c.u32() == 0x56789abc);
28 CHECK(c.u8() == 0xde);
29 CHECK(c.u64() == 0xf001020304050607ULL);
30 CHECK(c.remaining() == 0);
31 CHECK_THROWS_AS(c.u8(), CorruptData);
32}

70TEST_CASE("truncated tree payloads raise errors, never crash")
71{
72 // Arbitrary bytes, then every truncation of them: each must fail cleanly.
73 std::vector<uint8_t> junk(200);
74 for (size_t i = 0; i < junk.size(); ++i) {
75 junk[i] = (uint8_t)(i * 37 + 11);
76 }
77 // A plausible start: byte count with the flag set, TTree version 20.
78 junk[0] = 0x40;
79 junk[4] = 0x00;
80 junk[5] = 20;
81 for (size_t n = 0; n <= junk.size(); n += 7) {
82 std::span<const uint8_t> part{junk.data(), n};
83 CHECK_THROWS_AS(readTreePayload(part, 64, "memory", 0), FormatError);
84 }
85}
