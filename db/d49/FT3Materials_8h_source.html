33/*
34 * Materials of the FT3 detector.
35 *
36 * Everything the simulation needs to know about a material lives in the
37 * materials map below, keyed by its FT3-local ID: composition, density,
38 * transport parameters and the colour its volumes are drawn in.
39 * Detector::createMaterials() registers the whole map with the MaterialManager,
40 * and FT3Module/FT3Layer reach the media through getMedium() below, so no ID,
41 * density or colour is ever written out by hand.
42 */
43enum class MaterialID : unsigned {
44 Air = 1,
45 Silicon,
46 Copper,
47 Kapton,
48 CarbonFiber,
49 Epoxy,
50 Aluminum,
51 Foam,
52 Water,
53 FR4
54};
55
56// Transport parameters of a medium, in the order expected by Detector::Medium()
57struct TrackingParams {
58 float tmaxfd; // maximum field-induced angular deviation per step, degrees
59 float stemax; // maximum step length, cm
60 float deemax; // maximum fractional energy loss per step
61 float epsil; // tracking precision, cm
62 float stmin; // minimum step length, cm
63};
64
65constexpr TrackingParams sensitiveTracking = {0.1f, 0.0075f, 0.1f, 1.0e-4f, 0.0f};
66constexpr TrackingParams passiveTracking = {0.1f, 1.0f, 0.1f, 1.0e-4f, 0.0f};
67
68// Maximum number of elements any of the mixtures below is built from
69constexpr unsigned maxMaterialComponents = 4;
70using ComponentArray = std::array<float, maxMaterialComponents>;
71
72struct MaterialProperties {
73 const char* name;
74 int colour; // ROOT colour every volume made of this material is drawn in
75 float density; // g/cm3
76 // Radiation and nuclear interaction length, cm. Only single elements carry
77 // them: Mixture() derives both from the composition and takes no such
78 // arguments. A non-positive value lets the transport engine compute it.
79 float radl;
80 float absl;
81 int nComponents; // 0: single element; > 0: mixture by weight; < 0: mixture by atom count
82 ComponentArray a; // mass numbers; only a[0] is used for a single element
83 ComponentArray z; // atomic numbers; only z[0] is used for a single element
84 ComponentArray w; // weight fractions or atom counts; unused for a single element
85 TrackingParams tracking;
86};
