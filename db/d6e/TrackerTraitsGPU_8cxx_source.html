34template <int NLayers>
35void TrackerTraitsGPU<NLayers>::initialiseTimeFrame(const int iteration)
36{
37 this->mTaskArena->execute([&] {
38 mTimeFrameGPU->initialise(this->mTrkParams[iteration], this->mTrkParams[iteration].NLayers, iteration);
39 });
40 // load iteration parameters
41 mTimeFrameGPU->loadIterationParameters(this->mTrkParams[iteration]);
42
43 if (this->mTrkParams[iteration].PassFlags[IterationStep::FirstPass]) {
44 // on default stream
45 mTimeFrameGPU->loadVertices();
46 if (this->mTrkParams[iteration].PassFlags[IterationStep::LoadPersistentTables]) {
47 mTimeFrameGPU->loadROFOverlapTable(); // this can be put in constant memory actually
48 mTimeFrameGPU->loadTrackingTopologies();
49 }
50 mTimeFrameGPU->loadROFVertexLookupTable();
51 // once the tables are in persistent memory just re-upload the vertex one
52 // mTimeFrameGPU->uploadROFVertexLookupTable();
53 mTimeFrameGPU->loadIndexTableUtils();
54 // pinned on host
55 mTimeFrameGPU->createUsedClustersDeviceArray();
56 mTimeFrameGPU->createClustersDeviceArray();
57 mTimeFrameGPU->createUnsortedClustersDeviceArray();
58 mTimeFrameGPU->createClustersIndexTablesArray();
59 mTimeFrameGPU->createTrackingFrameInfoDeviceArray();
60 mTimeFrameGPU->createROFrameClustersDeviceArray();
61 // device array
62 mTimeFrameGPU->createClusterRadiiDevice();
63 mTimeFrameGPU->uploadClusterRadii();
64 mTimeFrameGPU->createTrackletsLUTDeviceArray();
65 mTimeFrameGPU->createTrackletsBuffersArray();
66 mTimeFrameGPU->createCellsBuffersArray();
67 mTimeFrameGPU->createCellsLUTDeviceArray();
68 mTimeFrameGPU->createClusterOwnersDeviceArray();
69 }
70 if (this->mTrkParams[iteration].PassFlags[IterationStep::FirstPass] || this->mTrkParams[iteration].PassFlags[IterationStep::UseUPCMask]) {
71 mTimeFrameGPU->loadROFCutMask(iteration);
72 }
73}

82template <int NLayers>
83void TrackerTraitsGPU<NLayers>::computeLayerTracklets(const int iteration, int iVertex)
84{
85 const auto topology = mTimeFrameGPU->getDeviceTrackingTopologyView();
86 const auto hostTopology = mTimeFrameGPU->getTrackingTopologyView();
87 const bool loadFirstPassData = this->mTrkParams[iteration].PassFlags[IterationStep::FirstPass] && iVertex <= 0; // load data only on first pass and first vertex
88 for (int iLayer{0}; iLayer < this->mTrkParams[iteration].NLayers; ++iLayer) {
89 if (loadFirstPassData) {
90 mTimeFrameGPU->createUsedClustersDevice(iLayer);
91 mTimeFrameGPU->loadUnsortedClustersDevice(iLayer);
92 mTimeFrameGPU->loadROFrameClustersDevice(iLayer);
93 mTimeFrameGPU->sortClustersDevice(iLayer, this->mTrkParams[iteration]);
94 }
95 mTimeFrameGPU->recordEvent(iLayer);
96 }
97
98 for (int linkId{0}; linkId < hostTopology.nLinks; ++linkId) {
99 mTimeFrameGPU->createTrackletsLUTDevice(loadFirstPassData, linkId); // on first pass allocates, then only clears memory
100 }
101
102 // Stack allocations created from trackleting through road finding are scoped to one tracker pass.
103 // With per-primary-vertex processing, the chain is called once per vertex while initialisation is only done once.
104 mTimeFrameGPU->pushMemoryStack(iteration);
105
106 const auto nClusters = mTimeFrameGPU->getClusterSizes();
107 const bool useDiamond = this->mTrkParams[iteration].UseDiamond;
108 if (useDiamond) {
109 const Vertex diamondVert = makeDiamondVertex(this->mTrkParams[iteration]);
110 mTimeFrameGPU->createDiamondDevice(diamondVert);
111 mTimeFrameGPU->recordEvent(0);
112 }
113 const Vertex* deviceVertices = useDiamond ? mTimeFrameGPU->getDeviceDiamond() : mTimeFrameGPU->getDeviceVertices();
114 const bool seedingPass = this->mTrkParams[iteration].PassFlags[IterationStep::SeedingVertexPass];
115 bounded_vector<float> vtxPhiCuts(seedingPass ? hostTopology.nLinks : 0,
116 this->mTrkParams[iteration].VtxPhiCut,
117 this->getMemoryPool().get());
118 auto& linkPhiCuts = seedingPass ? vtxPhiCuts : mTimeFrameGPU->getLinkPhiCuts();
119
120 for (int linkId{0}; linkId < hostTopology.nLinks; ++linkId) {
121 const auto link = hostTopology.getLink(linkId);
122 mTimeFrameGPU->waitEvent(linkId, link.fromLayer);
123 mTimeFrameGPU->waitEvent(linkId, link.toLayer);
124 if (useDiamond) {
125 mTimeFrameGPU->waitEvent(linkId, 0); // links not anchored on layer 0 must still wait for the diamond upload
126 }
127 const auto key = CapacityEstimator::makeKey(SlabSite::Tracklets, iteration, iVertex + 1, linkId);
128 const auto scale = static_cast<double>(nClusters[link.fromLayer]);
129 runOnSlab(mTimeFrameGPU->getCapacityEstimator(), key, scale, [&](const int capacity) {
130 mTimeFrameGPU->createTrackletsBuffers(linkId, capacity);
131 return TrackingKernels<NLayers>::computeTrackletsInROFsHandler(mTimeFrameGPU->getDeviceIndexTableUtils(),
132 mTimeFrameGPU->getDeviceROFMaskTableView(),
133 linkId,
134 link.fromLayer,
135 link.toLayer,
136 mTimeFrameGPU->getDeviceROFOverlapTableView(),
137 mTimeFrameGPU->getDeviceROFVertexLookupTableView(),
138 iVertex,
139 deviceVertices,
140 useDiamond,
141 mTimeFrameGPU->getDeviceArrayClusters(),
142 nClusters,
143 mTimeFrameGPU->getDeviceROFrameClusters(),
144 (const uint8_t**)mTimeFrameGPU->getDeviceArrayUsedClusters(),
145 mTimeFrameGPU->getDeviceArrayClustersIndexTables(),
146 mTimeFrameGPU->getDeviceArrayTracklets(),
147 mTimeFrameGPU->getDeviceTracklets(),
148 mTimeFrameGPU->getNTracklets(),
149 capacity,
150 mTimeFrameGPU->getDeviceTrackletsLUTs(),
151 this->mTrkParams[iteration].PassFlags[IterationStep::SelectUPCVertices],
152 this->mTrkParams[iteration].NSigmaCut,
153 topology,
154 linkPhiCuts,
155 this->mTrkParams[iteration].PVres,
156 mTimeFrameGPU->getDeviceMinRs(),
157 mTimeFrameGPU->getDeviceMaxRs(),
158 mTimeFrameGPU->getPositionResolutions(),
159 this->mTrkParams[iteration].LayerRadii,
160 mTimeFrameGPU->getLinkMSAngles(),
161 mTimeFrameGPU->getFrameworkAllocator(),
162 mTimeFrameGPU->getStreams());
163 });
164 mTimeFrameGPU->recordEvent(linkId);
165 }
166}

168template <int NLayers>
169void TrackerTraitsGPU<NLayers>::computeLayerCells(const int iteration)
170{
171 const auto topology = mTimeFrameGPU->getDeviceTrackingTopologyView();
172 const auto hostTopology = mTimeFrameGPU->getTrackingTopologyView();
173 for (int iLayer{0}; iLayer < this->mTrkParams[iteration].NLayers; ++iLayer) {
174 if (this->mTrkParams[iteration].PassFlags[IterationStep::FirstPass]) {
175 mTimeFrameGPU->loadUnsortedClustersDevice(iLayer); // latched: a no-op if trackleting already did it
176 mTimeFrameGPU->loadTrackingFrameInfoDevice(iLayer);
177 }
178 mTimeFrameGPU->recordEvent(iLayer);
179 }
180
181 for (int cellTopologyId{hostTopology.nCells}; cellTopologyId--;) {
182 const auto cellTopology = hostTopology.getCell(cellTopologyId);
183 const auto first = hostTopology.getLink(cellTopology.firstLink);
184 const auto second = hostTopology.getLink(cellTopology.secondLink);
185 const float cellDeltaPhiCut = this->mTrkParams[iteration].PassFlags[IterationStep::SeedingVertexPass] ? math_utils::cellDeltaPhiBound(this->mBz, this->mTrkParams[iteration].CellDeltaPhiMinPt,
186 this->mTrkParams[iteration].LayerRadii[first.fromLayer],
187 this->mTrkParams[iteration].LayerRadii[first.toLayer],
188 this->mTrkParams[iteration].LayerRadii[second.toLayer],
189 mTimeFrameGPU->getLinkMSAngle(cellTopology.firstLink))
190 : o2::constants::math::TwoPI;
191 const float cellTanLNSigma = this->mTrkParams[iteration].CellDeltaTanLambdaNSigma > 0.f
192 ? this->mTrkParams[iteration].CellDeltaTanLambdaNSigma
193 : this->mTrkParams[iteration].NSigmaCut;
194 const int currentLayerTrackletsNum{static_cast<int>(mTimeFrameGPU->getNTracklets()[cellTopology.firstLink])};
195 if (!currentLayerTrackletsNum || !mTimeFrameGPU->getNTracklets()[cellTopology.secondLink]) {
196 mTimeFrameGPU->getNCells()[cellTopologyId] = 0;
197 continue;
198 }
199
200 mTimeFrameGPU->createCellsLUTDevice(cellTopologyId);
201 mTimeFrameGPU->waitEvent(cellTopologyId, cellTopology.firstLink);
202 mTimeFrameGPU->waitEvent(cellTopologyId, cellTopology.secondLink);
203 mTimeFrameGPU->waitEvent(cellTopologyId, first.fromLayer);
204 mTimeFrameGPU->waitEvent(cellTopologyId, first.toLayer);
205 mTimeFrameGPU->waitEvent(cellTopologyId, second.toLayer);
206 const auto key = CapacityEstimator::makeKey(SlabSite::Cells, iteration, 0, cellTopologyId);
207 const auto scale = static_cast<double>(currentLayerTrackletsNum);
208 const int emitted = runOnSlab(mTimeFrameGPU->getCapacityEstimator(), key, scale, [&](const int capacity) {
209 mTimeFrameGPU->createCellsBuffers(cellTopologyId, capacity);
210 return TrackingKernels<NLayers>::computeCellsHandler(mTimeFrameGPU->getDeviceArrayClusters(),
211 mTimeFrameGPU->getDeviceArrayUnsortedClusters(),
212 mTimeFrameGPU->getDeviceArrayTrackingFrameInfo(),
213 mTimeFrameGPU->getDeviceArrayTracklets(),
214 mTimeFrameGPU->getDeviceArrayTrackletsLUT(),
215 currentLayerTrackletsNum,
216 cellTopologyId,
217 topology,
218 mTimeFrameGPU->getDeviceCells()[cellTopologyId],
219 capacity,
220 mTimeFrameGPU->getDeviceCellLUTs()[cellTopologyId],
221 this->mBz,
222 this->mTrkParams[iteration].MaxChi2ClusterAttachment,
223 this->mTrkParams[iteration].CellDeltaTanLambdaSigma,
224 cellDeltaPhiCut,
225 cellTanLNSigma,
226 mTimeFrameGPU->getDeviceLayerxX0(),
227 mTimeFrameGPU->getCapacityEstimator(),
228 iteration,
229 !this->mTrkParams[iteration].PassFlags[IterationStep::SeedingVertexPass],
230 mTimeFrameGPU->getFrameworkAllocator(),
231 mTimeFrameGPU->getStreams());
232 });
233 mTimeFrameGPU->getNCells()[cellTopologyId] = emitted;
234 mTimeFrameGPU->recordEvent(cellTopologyId);
235 }
236 mTimeFrameGPU->syncStreams(false);
237}

239template <int NLayers>
240void TrackerTraitsGPU<NLayers>::computeVertexCandidates(const int iteration)
241{
242 const int nCells = mTimeFrameGPU->getNCells()[0];
243 if (!nCells) {
244 mTimeFrameGPU->setNLinesTotal(0);
245 return;
246 }
247 mTimeFrameGPU->createClusterOwnersDevice();
248 mTimeFrameGPU->createLinesDevice(nCells);
249 mTimeFrameGPU->resetClusterOwnersDevice();
250 mTimeFrameGPU->syncStreams(false);
251
252 TrackingKernels<NLayers>::registerClusterOwnershipHandler(mTimeFrameGPU->getDeviceCells()[0],
253 nCells,
254 mTimeFrameGPU->getDeviceArrayClusterOwners(),
255 mTimeFrameGPU->getStream(0));
256
257 TrackingKernels<NLayers>::linearizeCellsToLinesHandler(nCells,
258 mTimeFrameGPU->getDeviceCells()[0],
259 mTimeFrameGPU->getDeviceArrayClusterOwners(),
260 mTimeFrameGPU->getDeviceROFramesClusters(1),
261 mTimeFrameGPU->getNrof(1),
262 this->mTrkParams[iteration].CellLineSharedClusterCut,
263 mTimeFrameGPU->getDeviceLines(),
264 mTimeFrameGPU->getDeviceLineRof(),
265 mTimeFrameGPU->getDeviceLineClusters(),
266 mTimeFrameGPU->getDeviceLineSlots(),
267 mTimeFrameGPU->getBeamX(),
268 mTimeFrameGPU->getBeamY(),
269 this->mTrkParams[iteration].VtxMaxZPositionAllowed,
270 this->mTrkParams[iteration].VtxLineMinPt,
271 mTimeFrameGPU->getDeviceLineZs(),
272 mTimeFrameGPU->getDeviceLineTimes(),
273 mTimeFrameGPU->getDeviceLineChi2(),
274 mTimeFrameGPU->getDeviceLinePt(),
275 mTimeFrameGPU->getFrameworkAllocator(),
276 mTimeFrameGPU->getStream(0));
277 const unsigned int nLines = mTimeFrameGPU->downloadLinesDevice();
278
279 TrackingKernels<NLayers>::sortLinesHandler(nLines,
280 mTimeFrameGPU->getNrof(1),
281 mTimeFrameGPU->getLineProjSoA(),
282 mTimeFrameGPU->getLineProjSortedSoA(),
283 mTimeFrameGPU->getDeviceLineRof(),
284 mTimeFrameGPU->getDeviceRofLineOffsets(),
285 mTimeFrameGPU->getFrameworkAllocator(),
286 mTimeFrameGPU->getStream(0));
287
288 const auto& tp = this->mTrkParams[iteration];
289 const float zWindow = 0.5f * tp.VtxClusterCut;
290 TrackingKernels<NLayers>::scanDensityHandler(static_cast<int>(nLines),
291 mTimeFrameGPU->getLineProjSortedSoA(),
292 mTimeFrameGPU->getDeviceRofLineOffsets(),
293 mTimeFrameGPU->getDeviceLineDensity(),
294 mTimeFrameGPU->getDeviceLineWin(),
295 zWindow,
296 mTimeFrameGPU->getStream(0));
297
298 const float fineZWindow = tp.VtxFineZWindow;
299 const bool doFine = fineZWindow > 0.f && fineZWindow < zWindow;
300 if (doFine) {
301 TrackingKernels<NLayers>::scanDensityHandler(static_cast<int>(nLines),
302 mTimeFrameGPU->getLineProjSortedSoA(),
303 mTimeFrameGPU->getDeviceRofLineOffsets(),
304 mTimeFrameGPU->getDeviceLineDensityFine(),
305 mTimeFrameGPU->getDeviceLineWinFine(),
306 fineZWindow,
307 mTimeFrameGPU->getStream(0));
308 }
309
310 TrackingKernels<NLayers>::findPeaksHandler(nLines,
311 mTimeFrameGPU->getNrof(1),
312 mTimeFrameGPU->getLineProjSortedSoA(),
313 mTimeFrameGPU->getDeviceRofLineOffsets(),
314 mTimeFrameGPU->getDeviceLineDensity(),
315 mTimeFrameGPU->getDeviceLineWin(),
316 mTimeFrameGPU->getDeviceLineIsPeak(),
317 doFine ? mTimeFrameGPU->getDeviceLineDensityFine() : nullptr,
318 doFine ? mTimeFrameGPU->getDeviceLineWinFine() : nullptr,
319 tp.VtxFineMinDensity,
320 doFine ? mTimeFrameGPU->getDeviceLineIsPeakFine() : nullptr,
321 mTimeFrameGPU->getDevicePeakScan(),
322 mTimeFrameGPU->getDevicePeakLineIdx(),
323 mTimeFrameGPU->getDevicePeakOffsets(),
324 mTimeFrameGPU->getFrameworkAllocator(),
325 mTimeFrameGPU->getStream(0));
326
327 const float goodLinePtCut = std::abs(mTimeFrameGPU->getBz()) > 0.01f ? tp.VtxGoodLinePtCut : -1.f;
328 TrackingKernels<NLayers>::fitPeaksHandler(mTimeFrameGPU->getDeviceNPeaks(),
329 mTimeFrameGPU->getDevicePeakLineIdx(),
330 mTimeFrameGPU->getDeviceLineWin(),
331 mTimeFrameGPU->getLineProjSortedSoA(),
332 mTimeFrameGPU->getDeviceLines(),
333 mTimeFrameGPU->getDeviceLineChi2(),
334 mTimeFrameGPU->getDeviceLinePt(),
335 tp.VtxGoodLineChi2Cut,
336 goodLinePtCut,
337 tp.VtxPairCut * tp.VtxPairCut,
338 tp.VtxNSigmaCut,
339 tp.VtxClusterContributorsCut,
340 mTimeFrameGPU->getBeamX(),
341 mTimeFrameGPU->getBeamY(),
342 doFine ? mTimeFrameGPU->getDeviceLineIsPeakFine() : nullptr,
343 tp.VtxFineMaxDrift,
344 mTimeFrameGPU->getDeviceVertexCands(),
345 mTimeFrameGPU->getStream(0));
346
347 const float duplicateZCut = tp.VtxDuplicateZCut > 0.f
348 ? tp.VtxDuplicateZCut
349 : std::max(4.f * tp.VtxPairCut, 0.5f * tp.VtxClusterCut);
350 TrackingKernels<NLayers>::dedupVertexCandidatesHandler(mTimeFrameGPU->getDeviceNPeaks(),
351 mTimeFrameGPU->getDevicePeakLineIdx(),
352 mTimeFrameGPU->getDevicePeakOffsets(),
353 mTimeFrameGPU->getLineProjSortedSoA(),
354 duplicateZCut,
355 tp.VtxDuplicateZScale,
356 mTimeFrameGPU->getDeviceVertexCands(),
357 mTimeFrameGPU->getStream(0));
358
359 const bool withMC = mTimeFrameGPU->hasMCinformation() && this->mTrkParams[iteration].CreateArtefactLabels;
360 mTimeFrameGPU->downloadVertexCandsDevice(); // sets nPeaks + host candidate/peak-offset mirrors
361 if (withMC) {
362 // pull back the peak windows and re-run the membership test here
363 mTimeFrameGPU->downloadPeakMembershipInputs();
364 }
365 const auto& lines = mTimeFrameGPU->getHostLines();
366 const auto& lineRof = mTimeFrameGPU->getHostLineRof();
367 const auto& lineClusters = mTimeFrameGPU->getHostLineClusters();
368 const int nRofs = mTimeFrameGPU->getNrof(1);
369 if (withMC) {
370 mTimeFrameGPU->getLineLabelFlat() = bounded_vector<o2::MCCompLabel>(nLines, o2::MCCompLabel(), mTimeFrameGPU->getMemoryPool().get());
371 }
372 auto lineLabel = [&](const int* cl) -> o2::MCCompLabel {
373 const auto l0 = mTimeFrameGPU->getClusterLabels(0, cl[0]);
374 const auto l1 = mTimeFrameGPU->getClusterLabels(1, cl[1]);
375 const auto l2 = mTimeFrameGPU->getClusterLabels(2, cl[2]);
376 for (const auto& a : l0) {
377 if (!a.isValid()) {
378 continue;
379 }
380 bool in1{false}, in2{false};
381 for (const auto& b : l1) {
382 if (b == a) {
383 in1 = true;
384 break;
385 }
386 }
387 for (const auto& c : l2) {
388 if (c == a) {
389 in2 = true;
390 break;
391 }
392 }
393 if (in1 && in2) {
394 return a;
395 }
396 }
397 return o2::MCCompLabel();
398 };
399 for (unsigned int i{0}; i < nLines; ++i) {
400 const int rof = lineRof[i];
401 if (rof < 0 || rof >= nRofs) {
402 LOGP(fatal, "ITS GPU linearizer: line {} carries out-of-range ROF {} (nRofs={}).", i, rof, nRofs);
403 }
404 const auto& l = lines[i];
405 mTimeFrameGPU->getLines(rof).emplace_back(std::array<float, 3>{l.originPoint[0], l.originPoint[1], l.originPoint[2]},
406 std::array<float, 3>{l.cosinesDirector[0], l.cosinesDirector[1], l.cosinesDirector[2]},
407 l.mTime);
408 if (withMC) {
409 const auto lbl = lineLabel(&lineClusters[3 * i]);
410 mTimeFrameGPU->getLinesLabel(rof).emplace_back(lbl);
411 mTimeFrameGPU->getLineLabelFlat()[i] = lbl; // same label, global-indexed for the vote
412 }
413 }
414 mTimeFrameGPU->setNLinesTotal(nLines);
415}

417template <int NLayers>
418void TrackerTraitsGPU<NLayers>::computeVertices(const int iteration)
419{
420 const int nRofs = mTimeFrameGPU->getNrof(1);
421 const bool withMC = mTimeFrameGPU->hasMCinformation() && this->mTrkParams[iteration].CreateArtefactLabels;
422 const int suppressLowMultDebris = this->mTrkParams[iteration].VtxSuppressLowMultDebris;
423 const bool skipHighMultRofs = this->mTrkParams[iteration].PassFlags[IterationStep::SkipROFsAboveThreshold];
424
425 const auto& cands = mTimeFrameGPU->getHostVertexCands(); // per compacted peak slot [0, nPeaks)
426 const auto& peakOffsets = mTimeFrameGPU->getHostPeakOffsets(); // nRofs+1; per-ROF peak slices
427
428 std::vector<std::vector<Vertex>> rofVertices(nRofs);
429 std::vector<std::vector<VertexLabel>> rofLabels(nRofs);
430 const float goodSig = this->mTrkParams[iteration].VtxGoodContributorsSignificance;
431
432 for (int rofId = 0; rofId < nRofs; ++rofId) {
433 if (skipHighMultRofs &&
434 static_cast<int>(mTimeFrameGPU->getROFVertexLookupTableView().getVertices(1, rofId).getEntries()) > this->mTrkParams[iteration].VertPerRofThreshold) {
435 continue;
436 }
437 // Survivors of this ROF, sorted by contributor count desc
438 std::vector<int> accepted;
439 for (int p = peakOffsets[rofId]; p < peakOffsets[rofId + 1]; ++p) {
440 if (cands[p].keep) {
441 accepted.push_back(p);
442 }
443 }
444 std::sort(accepted.begin(), accepted.end(), [&](const int a, const int b) { return cands[a].size > cands[b].size; });
445
446 double rofLoad = 0.;
447 if (goodSig > 0.f) { // compute the number of contributors in this ROF
448 for (int p = peakOffsets[rofId]; p < peakOffsets[rofId + 1]; ++p) {
449 if (cands[p].ok && !cands[p].fine) {
450 rofLoad += cands[p].size;
451 }
452 }
453 }
454 const float sigThreshold = goodSig > 0.f ? goodSig * std::sqrt(static_cast<float>(std::max(rofLoad, 1.))) : 0.f;
455
456 for (const int p : accepted) {
457 const auto& c = cands[p];
458 if (!rofVertices[rofId].empty()) {
459 if (goodSig > 0.f) {
460 if (c.nGood <= sigThreshold) {
461 continue;
462 }
463 } else if (c.size < suppressLowMultDebris) {
464 continue;
465 }
466 }
467 const float pos[3] = {c.x, c.y, c.z};
468 Vertex vertex(pos, c.rms2, static_cast<ushort>(c.size), c.avgDist2);
469 vertex.setTimeStamp(c.time);
470 if (this->mTrkParams[iteration].PassFlags[IterationStep::MarkVerticesAsUPC]) {
471 vertex.setFlags(Vertex::UPCMode);
472 }
473 rofVertices[rofId].push_back(vertex);
474 if (withMC) {
475 // Re-apply here the rule the fit used: a line contributes when it is
476 // time-compatible with the peak line and within pairCut of the candidate's seed
477 const auto& memb = mTimeFrameGPU->getHostPeakMembership();
478 const auto& lines = mTimeFrameGPU->getHostLines();
479 const auto& lineLabelFlat = mTimeFrameGPU->getLineLabelFlat(); // nLines, global-indexed
480 const int nLab = static_cast<int>(lineLabelFlat.size());
481 const float pairCut2 = this->mTrkParams[iteration].VtxPairCut * this->mTrkParams[iteration].VtxPairCut;
482 const int k = memb.peakLineIdx[p];
483 const auto tk = memb.times[k];
484 const auto wk = memb.win[k];
485 std::vector<o2::MCCompLabel> labels;
486 labels.reserve(wk.hi - wk.lo);
487 for (int j = wk.lo; j < wk.hi; ++j) {
488 const auto tj = memb.times[j];
489 if (!tk.isCompatible(tj)) {
490 continue;
491 }
492 const int gi = memb.sortedToLine[j];
493 if (gi < 0 || gi >= nLab) {
494 LOGP(error, "Seeding vertexer: member line {} out of range (nLines={}), skipping label", gi, nLab);
495 continue;
496 }
497 if (o2::its::Line::getDistance2FromPoint(lines[gi], c.seed) >= pairCut2) {
498 continue;
499 }
500 labels.push_back(lineLabelFlat[gi]);
501 }
502 rofLabels[rofId].push_back(computeMainVertexLabel(labels));
503 }
504 }
505 }
506
507 for (int rofId = 0; rofId < nRofs; ++rofId) {
508 for (auto& vertex : rofVertices[rofId]) {
509 mTimeFrameGPU->addPrimaryVertex(vertex);
510 }
511 if (withMC) {
512 for (auto& label : rofLabels[rofId]) {
513 mTimeFrameGPU->addPrimaryVertexLabel(label);
514 }
515 }
516 }
517
518 auto& pvs = mTimeFrameGPU->getPrimaryVertices();
519 std::vector<size_t> indices(pvs.size());
520 std::iota(indices.begin(), indices.end(), 0);
521 std::sort(indices.begin(), indices.end(), [&pvs](const size_t i, const size_t j) {
522 const auto aLower = pvs[i].getTimeStamp().lower();
523 const auto bLower = pvs[j].getTimeStamp().lower();
524 if (aLower != bLower) {
525 return aLower < bLower;
526 }
527 return pvs[i].getNContributors() > pvs[j].getNContributors();
528 });
529 std::decay_t<decltype(pvs)> sortedVtx(pvs.get_allocator());
530 sortedVtx.reserve(pvs.size());
531 for (const size_t idx : indices) {
532 sortedVtx.push_back(pvs[idx]);
533 }
534 pvs.swap(sortedVtx);
535 if (withMC) {
536 auto& mc = mTimeFrameGPU->getPrimaryVerticesLabels();
537 std::decay_t<decltype(mc)> sortedMC(mc.get_allocator());
538 sortedMC.reserve(mc.size());
539 for (const size_t idx : indices) {
540 sortedMC.push_back(mc[idx]);
541 }
542 mc.swap(sortedMC);
543 }
544 mTimeFrameGPU->updateROFVertexLookupTable();
545

549template <int NLayers>
550void TrackerTraitsGPU<NLayers>::findCellsNeighbours(const int iteration)
551{
552 const auto hostTopology = mTimeFrameGPU->getTrackingTopologyView();
553 bounded_vector<int> sourceTopologies(this->getMemoryPool().get());
554 sourceTopologies.reserve(hostTopology.nCells);
555 for (int outerLayer{0}; outerLayer < NLayers; ++outerLayer) {
556 for (int targetCellTopologyId{0}; targetCellTopologyId < hostTopology.nCells; ++targetCellTopologyId) {
557 const auto targetCellTopology = hostTopology.getCell(targetCellTopologyId);
558 if (targetCellTopology.hitLayerMask.last() != outerLayer) {
559 continue;
560 }
561 const int targetCellsNum{static_cast<int>(mTimeFrameGPU->getNCells()[targetCellTopologyId])};
562 sourceTopologies.clear();
563 size_t sourceCellCount{0};
564 for (int sourceCellTopologyId{0}; sourceCellTopologyId < hostTopology.nCells; ++sourceCellTopologyId) {
565 const auto sourceCellTopology = hostTopology.getCell(sourceCellTopologyId);
566 const int sourceCellsNum{static_cast<int>(mTimeFrameGPU->getNCells()[sourceCellTopologyId])};
567 if (!sourceCellsNum || sourceCellTopology.secondLink != targetCellTopology.firstLink) {
568 continue;
569 }
570 sourceTopologies.push_back(sourceCellTopologyId);
571 sourceCellCount += sourceCellsNum;
572 }
573 if (!targetCellsNum || sourceTopologies.empty()) {
574 mTimeFrameGPU->getNNeighbours()[targetCellTopologyId] = 0;
575 mTimeFrameGPU->createNeighboursDevice(targetCellTopologyId, 0);
576 mTimeFrameGPU->recordEvent(targetCellTopologyId);
577 continue;
578 }
579 mTimeFrameGPU->createNeighboursLUTDevice(targetCellTopologyId, targetCellsNum);
580 auto& stream = mTimeFrameGPU->getStream(targetCellTopologyId);
581 int* outputCounter = mTimeFrameGPU->getDeviceNeighboursLUT(targetCellTopologyId) + targetCellsNum;
582
583 const auto key = CapacityEstimator::makeKey(SlabSite::Neighbours, iteration, 0, targetCellTopologyId);
584 const auto scale = static_cast<double>(sourceCellCount);
585 const int emitted = runOnSlab(mTimeFrameGPU->getCapacityEstimator(), key, scale, [&](const int capacity) {
586 mTimeFrameGPU->createNeighboursDevice(targetCellTopologyId, capacity);
587 resetOutputCounterHandler(outputCounter, stream);
588 for (const int sourceCellTopologyId : sourceTopologies) {
589 mTimeFrameGPU->waitEvent(targetCellTopologyId, sourceCellTopologyId);
590 TrackingKernels<NLayers>::computeCellNeighboursHandler(mTimeFrameGPU->getDeviceArrayCells(),
591 mTimeFrameGPU->getDeviceArrayCellsLUT(),
592 mTimeFrameGPU->getDeviceNeighbours(targetCellTopologyId),
593 outputCounter,
594 capacity,
595 sourceCellTopologyId,
596 targetCellTopologyId,
597 this->mTrkParams[iteration].MaxChi2ClusterAttachment,
598 this->mBz,
599 mTimeFrameGPU->getNCells()[sourceCellTopologyId],
600 mTimeFrameGPU->getFrameworkAllocator(),
601 stream);
602 }
603 return finalizeCellNeighboursHandler(mTimeFrameGPU->getDeviceNeighbours(targetCellTopologyId),
604 mTimeFrameGPU->getDeviceNeighboursLUT(targetCellTopologyId),
605 targetCellsNum,
606 capacity,
607 mTimeFrameGPU->getFrameworkAllocator(),
608 stream);
609 });
610 mTimeFrameGPU->getNNeighbours()[targetCellTopologyId] = emitted;
611 mTimeFrameGPU->recordEvent(targetCellTopologyId);
612 }
613 }
614 mTimeFrameGPU->syncStreams(false);
615}

617template <int NLayers>
618void TrackerTraitsGPU<NLayers>::findRoads(const int iteration)
619{
620 bounded_vector<bounded_vector<int>> firstClusters(this->mTrkParams[iteration].NLayers, bounded_vector<int>(this->getMemoryPool().get()), this->getMemoryPool().get());
621 firstClusters.resize(this->mTrkParams[iteration].NLayers);
622 const auto hostTopology = mTimeFrameGPU->getTrackingTopologyView();
623 const bool extendTop = this->mTrkParams[iteration].PassFlags[IterationStep::TrackFollowerTop];
624 const bool extendBot = this->mTrkParams[iteration].PassFlags[IterationStep::TrackFollowerBot];
625 const bool extendTracks = extendTop || extendBot;
626 for (int startLevel{this->mTrkParams[iteration].CellsPerRoad()}; startLevel >= this->mTrkParams[iteration].CellMinimumLevel(); --startLevel) {
627 // The cells that may start a road at this level, as the scale the estimator predicts from.
628 size_t startCells{0};
629 for (int startCellTopologyId{0}; startCellTopologyId < hostTopology.nCells; ++startCellTopologyId) {
630 const int startLayer = hostTopology.getCell(startCellTopologyId).hitLayerMask.last();
631 if (this->mTrkParams[iteration].StartLayerMask.has(startLayer)) {
632 startCells += mTimeFrameGPU->getNCells()[startCellTopologyId];
633 }
634 }
635 if (!startCells) {
636 continue;
637 }
638 const auto key = CapacityEstimator::makeKey(SlabSite::TrackSeeds, iteration, startLevel, 0);
639 auto& estimator = mTimeFrameGPU->getCapacityEstimator();
640 const int nSeeds = runOnSlab(estimator, key, static_cast<double>(startCells), [&](const int capacity) {
641 mTimeFrameGPU->createTrackSeedsDevice(capacity);
642 int cursor{0};
643 for (int startCellTopologyId{0}; startCellTopologyId < hostTopology.nCells; ++startCellTopologyId) {
644 const int startLayer = hostTopology.getCell(startCellTopologyId).hitLayerMask.last();
645 if (!(this->mTrkParams[iteration].StartLayerMask.has(startLayer)) || mTimeFrameGPU->getNCells()[startCellTopologyId] == 0) {
646 continue;
647 }
648 TrackingKernels<NLayers>::processNeighboursHandler(startLevel,
649 startCellTopologyId,
650 mTimeFrameGPU->getDeviceArrayCells(),
651 mTimeFrameGPU->getDeviceCells()[startCellTopologyId],
652 nullptr,
653 nullptr,
654 mTimeFrameGPU->getArrayNCells().data(),
655 (const uint8_t**)mTimeFrameGPU->getDeviceArrayUsedClusters(),
656 mTimeFrameGPU->getDeviceArrayNeighbours(),
657 mTimeFrameGPU->getDeviceArrayNeighboursCellLUT(),
658 mTimeFrameGPU->getDeviceArrayTrackingFrameInfo(),
659 mTimeFrameGPU->getDeviceTrackSeeds(),
660 capacity,
661 cursor,
662 mTimeFrameGPU->getCapacityEstimator(),
663 iteration,
664 this->mBz,
665 this->mTrkParams[iteration].MaxChi2ClusterAttachment,
666 this->mTrkParams[iteration].MaxChi2NDF,
667 this->mTrkParams[iteration].MaxHoles,
668 this->mTrkParams[iteration].getMinSeedingClusters(),
669 this->mTrkParams[iteration].HoleLayerMask,
670 this->mTrkParams[iteration].getNonSeedingLayerMask(),
671 mTimeFrameGPU->getDeviceLayerxX0(),
672 mTimeFrameGPU->getDevicePropagator(),
673 this->mTrkParams[iteration].CorrType,
674 mTimeFrameGPU->getFrameworkAllocator());
675 }
676 return cursor; }, estimator.peakCapacity(key));
677 if (!nSeeds) {
678 LOGP(debug, "No track seeds found, skipping track finding");
679 continue;
680 }
681 if (extendTracks) { // independent of the slab size, so it must not be redone on a retry
682 mTimeFrameGPU->createTrackExtensionScratchDevice(gpu::gridThreads(gpu::ResidentBlocks.fitTrackSeedsExtended),
683 this->mTrkParams[iteration].TrackFollowerMaxHypotheses);
684 }
685 const auto trackKey = CapacityEstimator::makeKey(extendTracks ? SlabSite::TracksExtended : SlabSite::Tracks,
686 iteration, startLevel, 0);
687 const int nTracks = runOnSlab(estimator, trackKey, static_cast<double>(nSeeds), [&](const int capacity) {
688 mTimeFrameGPU->createTrackITSExtDevice(capacity);
689 return TrackingKernels<NLayers>::computeTrackSeedHandler(mTimeFrameGPU->getDeviceTrackSeeds(),
690 mTimeFrameGPU->getDeviceArrayTrackingFrameInfo(),
691 mTimeFrameGPU->getDeviceArrayUnsortedClusters(),
692 mTimeFrameGPU->getDeviceIndexTableUtils(),
693 mTimeFrameGPU->getDeviceROFMaskTableView(),
694 mTimeFrameGPU->getDeviceROFOverlapTableView(),
695 mTimeFrameGPU->getDeviceArrayClusters(),
696 (const unsigned char**)mTimeFrameGPU->getDeviceArrayUsedClusters(),
697 mTimeFrameGPU->getDeviceArrayClustersIndexTables(),
698 mTimeFrameGPU->getDeviceROFrameClusters(),
699 mTimeFrameGPU->getDeviceTrackITSExt(),
700 mTimeFrameGPU->getDeviceTrackIndices(),
701 mTimeFrameGPU->getDeviceTrackSeedIndices(),
702 mTimeFrameGPU->getDeviceTrackCounter(),
703 capacity,
704 extendTracks ? mTimeFrameGPU->getDeviceActiveTrackExtensionHypotheses() : nullptr,
705 extendTracks ? mTimeFrameGPU->getDeviceNextTrackExtensionHypotheses() : nullptr,
706 mTimeFrameGPU->getDeviceLayerRadii(),
707 mTimeFrameGPU->getDeviceMinPts(),
708 mTimeFrameGPU->getDeviceLayerxX0(),
709 static_cast<unsigned int>(nSeeds),
710 this->mBz,
711 this->mTrkParams[iteration].MaxChi2ClusterAttachment,
712 this->mTrkParams[iteration].MaxChi2NDF,
713 this->mTrkParams[iteration].ReseedIfShorter,
714 this->mTrkParams[iteration].RepeatRefitOut,
715 this->mTrkParams[iteration].ShiftRefToCluster,
716 this->mTrkParams[iteration].NLayers,
717 this->mTrkParams[iteration].PhiBins,
718 this->mTrkParams[iteration].TrackFollowerMaxHypotheses,
719 extendTop,
720 extendBot,
721 this->mTrkParams[iteration].TrackFollowerNSigmaCutPhi,
722 this->mTrkParams[iteration].TrackFollowerNSigmaCutZ,
723 mTimeFrameGPU->getDevicePropagator(),
724 this->mTrkParams[iteration].CorrType,
725 mTimeFrameGPU->getFrameworkAllocator()); }, estimator.peakCapacity(trackKey));
726 mTimeFrameGPU->createTrackITSExtHost(nTracks);
727 mTimeFrameGPU->downloadTrackITSExtDevice();
728
729 auto& tracks = mTimeFrameGPU->getTrackITSExt();
730 const auto& trackIndices = mTimeFrameGPU->getTrackIndices();
731 this->acceptTracks(iteration, tracks, trackIndices, firstClusters);
732 mTimeFrameGPU->loadUsedClustersDevice();
733 }
734 this->markTracks(iteration);
735 // wipe the artefact memory
736 mTimeFrameGPU->popMemoryStack(iteration);
737};
