34 float minPtOppositeSides = 0.f; // TPC-only legs on opposite TPC sides: reject cosmics with the pT of either leg or of the refitted cosmic below this (scaled with field; 0: no cut)
35 float timeToleranceMUS = 0.f;
36 float maxStep = 10.f;
37 float maxSnp = 0.99f;
38 float minSeedPt = 0.10f; // use only tracks above this pT (scaled with field)

40 float minSeedDCAxy = 0.f; // use only tracks with |DCA_xy| to the beam line >= this [cm] (0: no cut; rejects collision tracks in physics data)
41 float minSeedDCAxyNSigma = 0.f; // use only tracks with |DCA_xy| >= this * sigma(DCA_xy) (0: no cut; poorly measured collision tracks)
42 float minSeedDCAxyTOF = -1.f; // with tofFlightSelection: TPC-only legs failing the two cuts above but with |DCA_xy| >= this [cm] are used in TOF-confirmed pairs only (< 0: off; e.g. cosmics crossing the ITS inner barrel)
43 float minSeedDCAxyNSigmaTOF = 3.f; // the same for these legs in units of sigma(DCA_xy)
44 bool constrainTPCOnlyZ = false; // TPC-only legs: test z at a common time (same side, or a leg with known time), else require the time implied by z continuity in both brackets
45 bool vetoSameHalf = false; // reject pairs whose two legs lie on the same side of the closest approach (two pieces of one leg)
46 bool refitSameSideAtCommonTime = false; // TPC-only legs on the same side: compare them refitted at the centre of their brackets' overlap instead of at their own time0s
47 bool tofFlightSelection = false; // needs TOF clusters: accepted pairs of TPC-only legs pointing to a top / bottom TOF hit pair with the muon's flight time win the selection, refit at that time (if that fails, at their time without TOF)
48 float tofRoad = 5.f; // half-width [cm] in y and z of the road at the TOF around the outward continuation of a TPC-only leg
49 float tofFlightTolerance = 2.f; // max. deviation [ns] of the top / bottom TOF time difference from the flight time along the helix
50 float tofTimeError = 0.1f; // error [mus] of the TOF time of a confirmed cosmic, for its refit and time window (covers TPC vs TOF offsets)
51 float nSigmaTError = 4.f; // number of sigmas on track time error for matching (except for TPC which provides an interval)
52 float tpcExtraZError2 = 1.f; // extra error^2 on the TPC-only track Z coordinate
53 float fiducialRIP = 1.0f; // consider track having |Y@x=0|< this as passing DCA cut (if requested)
54 float fiducialZIP = 20.f; // consider track having |Z@x=0|< this as passing DCA cut (if requested)
55 bool allowTPCOnly = true;
56 bool discardPVContributors = true; // used only if the global option --use-pv-info is requested
57 o2::base::Propagator::MatCorrType matCorr = o2::base::Propagator::MatCorrType::USEMatCorrLUT;
58
59 O2ParamDef(MatchCosmicsParams, "cosmicsMatch");
60};
