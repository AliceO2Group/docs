28framework::WorkflowSpec getWorkflow(const ca::WorkflowOptions& options)
29{
30 using namespace ca;
31 framework::WorkflowSpec specs;
32 const auto& tracker = options.tracker;
33 const bool useGeom = tracker.geometry == GeometrySource::Full;
34 const bool writeTracks = options.output == OutputPolicy::All || options.output == OutputPolicy::TracksAndClusterROFs;
35 if (options.kind == WorkflowKind::Reconstruction) {
36 if (options.input == InputStage::DigitsFile) {
37 specs.emplace_back(o2::itsmft::getMFTDigitReaderSpec(tracker.useMC, options.staggering, false, true, "mftdigits.root"));
38 }
39 if (options.input != InputStage::UpstreamClusters) {
40 specs.emplace_back(o2::itsmft::getMFTClustererSpec(tracker.useMC, options.staggering));
41 }
42 if (options.output != OutputPolicy::None) {
43 specs.emplace_back(o2::itsmft::getMFTClusterWriterSpec(tracker.useMC, options.staggering, options.output != OutputPolicy::All));
44 }
45 }
46 if (options.runTracking) {
47 if (tracker.irFrames == IRFrameSource::File) {
48 specs.emplace_back(o2::globaltracking::getIRFrameReaderSpec("ITS", 0, "its-irframe-reader", "o2_its_irframe.root"));
49 }
50 specs.emplace_back(o2::mft::getCATrackerSpec(tracker));
51 if (writeTracks) {
52 specs.emplace_back(o2::mft::getTrackWriterSpec(tracker.useMC, true));
53 }
54 if (options.assessment) {
55 specs.emplace_back(o2::mft::getMFTAssessmentSpec(tracker.useMC, useGeom, options.processGenerated));
56 }
57 if (options.tracksToRecords) {
58 specs.emplace_back(o2::mft::getTracksToRecordsSpec());
59 }
60 }
61 return specs;
62}
