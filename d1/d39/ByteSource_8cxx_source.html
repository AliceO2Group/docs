28std::unique_ptr<ByteSource> ByteSource::open(std::string url, const Options& options)
29{
30 if (url.starts_with("file://")) {
31 return std::make_unique<LocalByteSource>(url.substr(7), options);
32 }
33 for (auto scheme : {"root://", "roots://", "xroot://", "xroots://"}) {
34 if (url.starts_with(scheme)) {
35#ifdef AODNATIVE_WITH_XROOTD
36 return std::make_unique<XRootDByteSource>(std::move(url), options);
37#else
38 throw IoError(url, "the native reader was built without XRootD");
39#endif
40 }
41 }
42 for (auto scheme : {"http://", "https://"}) {
43 if (url.starts_with(scheme)) {
44#ifdef AODNATIVE_WITH_CURL
45 return std::make_unique<HttpByteSource>(std::move(url), options);
46#else
47 throw IoError(url, "the native reader was built without curl");
48#endif
49 }
50 }
51 // alien:// names are resolved to root:// replicas by the caller.
52 for (auto scheme : {"alien://"}) {
53 if (url.starts_with(scheme)) {
54 throw IoError(url, "alien:// names must be resolved to a replica first");
55 }
56 }
57 if (url.find("://") != std::string::npos) {
58 throw IoError(url, "unknown URL scheme");
59 }
60 return std::make_unique<LocalByteSource>(std::move(url), options);
61}

63std::unique_ptr<ByteSource> ByteSource::openReplicas(std::vector<Replica> replicas, const Options& options,
64 std::function<void(std::string const&)> log)
65{
66 return std::make_unique<FailoverByteSource>(std::move(replicas), options, &ByteSource::open, std::move(log));
