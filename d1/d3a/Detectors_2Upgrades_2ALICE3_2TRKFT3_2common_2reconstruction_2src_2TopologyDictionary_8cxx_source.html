30template <int DetID>
31int TopologyDictionary<DetID>::layerOf(const ClusterType& cluster) noexcept
32{
33 if (cluster.layer < 0) {
34 return -1;
35 }
36 if constexpr (DetID == o2::detectors::DetID::TRK) {
37 if (cluster.subDetID == 0) {
38 return cluster.layer;
39 }
40 if (cluster.subDetID == 1) {
41 return o2::trk::constants::VD::petal::nLayers + cluster.layer;
42 }
43 return -1;
44 } else {
45 return cluster.layer;
46 }
47}

59template <int DetID>
60math_utils::Point3D<float> TopologyDictionary<DetID>::centreOfGravity(const ClusterType& cluster, const unsigned char* pattern) noexcept
61{
62 float cogRow = 0.f, cogCol = 0.f;
63 itsmft::ClusterPattern::getCOG(pattern[0], pattern[1], pattern + 2, cogRow, cogCol);
64 float x = 0.f, z = 0.f;
65 SegmentationChip::detectorToLocalUnchecked(cluster.row, cluster.col, x, z, cluster.subDetID, cluster.layer, cluster.layer);
66 const bool vd = cluster.subDetID == 0;
67 x -= cogRow * (vd ? SegmentationChip::PitchRowVD : SegmentationChip::PitchRowMLOT);
68 z += cogCol * (vd ? SegmentationChip::PitchColVD : SegmentationChip::PitchColMLOT);
69 return {x, 0.f, z};
70}

72template <int DetID>
73int TopologyDictionary<DetID>::findID(int layer, const unsigned char* pattern) const
74{
75 const auto& data = mLayers.at(layer);
76 const int rowSpan = pattern[0], colSpan = pattern[1];
77 // The hash leaves out a large pattern's last bytes: compare them all.
78 const auto nBytes = static_cast<std::size_t>(2 + ((rowSpan * colSpan + 7) / 8));
79 const auto [first, last] = data.hashMap.equal_range(itsmft::ClusterTopology::getCompleteHash(rowSpan, colSpan, pattern + 2));
80 for (auto it = first; it != last; ++it) {
81 const auto& entryPattern = data.entries[it->second].pattern;
82 if (entryPattern.size() == nBytes && std::memcmp(entryPattern.data(), pattern, nBytes) == 0) {
83 return it->second;
84 }
85 }
86 return static_cast<int>(data.entries.size()) - NGroups + groupOf(rowSpan, colSpan);
87}

89template <int DetID>
90math_utils::Point3D<float> TopologyDictionary<DetID>::getFlatCoordinates(const ClusterType& cluster, const unsigned char* pattern, int id) const
91{
92 const int layer = layerOf(cluster);
93 const auto& data = mLayers.at(layer);
94 const auto& entry = data.entries.at(id);
95 math_utils::Point3D<float> flat;
96 if (entry.isGroup) {
97 flat = centreOfGravity(cluster, pattern);
98 } else {
99 float x = 0.f, z = 0.f;
100 SegmentationChip::detectorToLocalUnchecked(cluster.row, cluster.col, x, z, cluster.subDetID, cluster.layer, cluster.layer);
101 const bool vd = cluster.subDetID == 0;
102 flat.SetX(x + entry.xCOG * (vd ? SegmentationChip::PitchRowVD : SegmentationChip::PitchRowMLOT));
103 flat.SetZ(z + entry.zCOG * (vd ? SegmentationChip::PitchColVD : SegmentationChip::PitchColMLOT));
104 }
105 if (mApplyOffsets) {
106 flat.SetX(flat.X() + entry.xOffset);
107 flat.SetZ(flat.Z() + entry.zOffset);
108 }
109 flat.SetY(data.depth);
110 return flat;
111}

113template <int DetID>
114math_utils::Point3D<float> TopologyDictionary<DetID>::getClusterCoordinates(const ClusterType& cluster, const unsigned char* pattern, int id) const
115{
116 auto local = getFlatCoordinates(cluster, pattern, id);
117 if (cluster.subDetID == 0) {
118 const auto curved = SegmentationChip::flatToCurved(cluster.layer, local.X(), local.Y());
119 local.SetX(curved.X());
120 local.SetY(curved.Y());
121 }
122 return local;
123}

125template <int DetID>
126void TopologyDictionary<DetID>::print() const
127{
128 LOGP(info, "{} TopologyDictionary: {} layers, offsets {}", o2::detectors::DetID(DetID).getName(), mLayers.size(), mApplyOffsets ? "applied" : "not applied");
129 for (std::size_t layer = 0; layer < mLayers.size(); ++layer) {
130 LOGP(info, " layer {:2}: depth {:6.2f} um, {} common patterns", layer, 1e4 * mLayers[layer].depth, mLayers[layer].entries.size() - NGroups);
131 }
132}

134template <int DetID>
135TopologyDictionary<DetID>* TopologyDictionary<DetID>::loadFrom(const std::string& fileName, const std::string& objName)
136{
137 LOGP(info, "Loading {} TopologyDictionary from {} with name {}", o2::detectors::DetID(DetID).getName(), fileName, objName);
138 std::unique_ptr<TFile> file{TFile::Open(fileName.c_str(), "READ")};
139 if (!file || file->IsZombie()) {
140 throw std::runtime_error(fmt::format("Failed to open {}", fileName));
141 }
142 auto* dictionary = file->Get<TopologyDictionary<DetID>>(objName.c_str());
143 if (dictionary == nullptr) {
144 throw std::runtime_error(fmt::format("Failed to load {} from {}", objName, fileName));
145 }
146 return dictionary;
147}
