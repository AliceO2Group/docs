33TGeoVolume* TF3Chip::createChip(const Double_t ychip,
34 const Double_t ysens,
35 const char* chipName,
36 const char* sensName,
37 const Bool_t dummy,
38 const TGeoManager* mgr)
39{
40
41 Double_t xchip, zchip;
42 Double_t ylen, ypos;
43
44 xchip = 0.5 * sChipSizeX;
45 zchip = 0.5 * sChipSizeZ;
46
47 // First create all needed shapes
48 ylen = ysens;
49 if (ysens > ychip) {
50 LOG(warning) << "Sensor half thickness (" << ysens
51 << ") greater than chip half thickness (" << ychip
52 << "), setting equal";
53 ylen = ychip;
54 }
55
56 // The chip
57 TGeoBBox* chip = new TGeoBBox(xchip, 0.5 * ychip, zchip);
58
59 // The sensor
60 TGeoBBox* sensor = new TGeoBBox(xchip, 0.5 * ylen, zchip);
61
62 // We have all shapes: now create the real volumes
63 TGeoMedium* medSi = mgr->GetMedium("TF3CHIP_SI$");
64 if (!medSi) {
65 Int_t fieldType;
66 Float_t maxField;
67 o2::base::Detector::initFieldTrackingParams(fieldType, maxField);
68 createMaterials(0, fieldType, maxField);
69 medSi = mgr->GetMedium("TF3CHIP_SI$");
70 }
71 TGeoMedium* medAir = mgr->GetMedium("TF3CHIP_AIR$");
72
73 TGeoMedium* medChip;
74
75 if (dummy) {
76 medChip = medAir;
77 } else {
78 medChip = medSi;
79 }
80
81 TGeoVolume* chipVol = new TGeoVolume(chipName, chip, medChip);
82 chipVol->SetVisibility(kTRUE);
83 chipVol->SetLineColor(kOrange);
84 chipVol->SetFillColor(kOrange - 9);
85 chipVol->SetLineWidth(3);
86 chipVol->SetTransparency(20);
87
88 TGeoVolume* sensVol = new TGeoVolume(sensName, sensor, medChip);
89 sensVol->SetVisibility(kTRUE);
90 sensVol->SetLineColor(kRed);
91 sensVol->SetLineWidth(3);
92 sensVol->SetFillColor(kRed - 9);
93 sensVol->SetFillStyle(4000); // 0% transparent
94
95 // Now build up the chip
96 ypos = chip->GetDY() - sensor->GetDY();
97 chipVol->AddNode(sensVol, 1, new TGeoTranslation(0., ypos, 0.));
98
99 // Done, return the chip
100 return chipVol;
101}

104void TF3Chip::createMaterials(Int_t id, Int_t ifield, Float_t fieldm)
105{
106 auto& mgr = o2::base::MaterialManager::Instance();
107
108 Float_t tmaxfd = 0.1; // Degree
109 Float_t stemax = 1.0; // cm
110 Float_t deemax = 0.1; // Fraction of particle's energy 0<deemax<=1
111 Float_t epsil = 1.0E-4; // cm
112 Float_t stmin = 0.0; // cm "Default value used"
113
114 Float_t tmaxfdSi = 0.1; // Degree
115 Float_t stemaxSi = 0.0075; // cm
116 Float_t deemaxSi = 0.1; // Fraction of particle's energy 0<deemax<=1
117 Float_t epsilSi = 1.0E-4; // cm
118 Float_t stminSi = 0.0; // cm "Default value used"
119
120 Float_t tmaxfdAir = 0.1; // Degree
121 Float_t stemaxAir = 1.0; // cm
122 Float_t deemaxAir = 0.1; // Fraction of particle's energy 0<deemax<=1
123 Float_t epsilAir = 1.0E-4; // cm
124 Float_t stminAir = 0.0; // cm "Default value used"
125
126 // AIR
127 Float_t aAir[4] = {12.0107, 14.0067, 15.9994, 39.948};
128 Float_t zAir[4] = {6., 7., 8., 18.};
129 Float_t wAir[4] = {0.000124, 0.755267, 0.231781, 0.012827};
130 Float_t dAir = 1.20479E-3;
131
132 if (mgr.getMediumID("TF3CHIP", id) < 0) {
133 mgr.Mixture("TF3CHIP", id, "AIR$", aAir, zAir, dAir, 4, wAir);
134 mgr.Medium("TF3CHIP", id, "AIR$", id, 0, ifield, fieldm, tmaxfdAir, stemaxAir, deemaxAir, epsilAir, stminAir);
135 id++;
136 }
137
138 if (mgr.getMediumID("TF3CHIP", id) < 0) {
139 mgr.Material("TF3CHIP", id, "SI$", 0.28086E+02, 0.14000E+02, 0.23300E+01, 0.93600E+01, 0.99900E+03);
140 mgr.Medium("TF3CHIP", id, "SI$", id, 0, ifield, fieldm, tmaxfdSi, stemaxSi, deemaxSi, epsilSi, stminSi);
141 }
142}
