79std::map<std::string, std::string> RootAODInputFile::readStringMap(char const* key)
80{
81 std::map<std::string, std::string> result;
82 std::unique_ptr<TMap> map{(TMap*)mFile->Get(key)};
83 if (!map) {
84 return result;
85 }
86 map->SetOwnerKeyValue(true, true);
87 std::unique_ptr<TIterator> it{map->MakeIterator()};
88 while (auto* key = it->Next()) {
89 auto* value = (TObjString*)map->GetValue(key);
90 result[key->GetName()] = value ? value->String().Data() : "";
91 }
92 return result;
93}

108std::vector<GridReplica> gridReplicas(std::string const& lfn)
109{
110 TUrl url(lfn.c_str());
111 if (url.GetAnchor() && url.GetAnchor()[0] != '\0') {
112 return {};
113 }
114 if (!gGrid) {
115 TGrid::Connect("alien://");
116 }
117 if (!gGrid) {
118 return {};
119 }
120 // As TJAlienFile::Open does for reading.
121 std::unique_ptr<TGridResult> result(gGrid->Command(fmt::format("access read {}", url.GetFile()).c_str()));
122 if (!result) {
123 return {};
124 }
125 std::vector<GridReplica> replicas;
126 for (auto* object : *result) {
127 auto* map = dynamic_cast<TMap*>(object);
128 if (!map) {
129 continue;
130 }
131 auto value = [map](char const* key) -> std::string {
132 auto* v = map->GetValue(key);
133 return v ? v->GetName() : "";
134 };
135 auto pfn = value("url");
136 auto envelope = value("envelope");
137 if (pfn.empty() || envelope.empty() || pfn.find('#') != std::string::npos) {
138 continue;
139 }
140 replicas.push_back({pfn + "?&authz=" + envelope, value("se")});
141 }
142 return replicas;
143}

145std::shared_ptr<AODInputFile> openNativeAOD(std::string const& filename, RootObjectReadingFactory& factory, o2::aod::native::Options const& options)
146{
147 if (!filename.starts_with("alien://")) {
148 return NativeAODInputFile::open(filename, factory, options);
149 }
150 auto start = std::chrono::steady_clock::now();
151 auto replicas = gridReplicas(filename);
152 if (replicas.empty()) {
153 throw o2::aod::native::IoError(filename, "no replica the native reader can read");
154 }
155 // The file is read from the first replica which opens; a read error later on
156 // is retried and then continues on the next replica (FailoverByteSource).
157 std::vector<o2::aod::native::Replica> list;
158 for (auto const& replica : replicas) {
159 list.push_back({replica.url, replica.se});
160 }
161 auto source = o2::aod::native::ByteSource::openReplicas(std::move(list), options, [filename](std::string const& message) {
162 LOGP(info, "\"{}\": {}", filename, message);
163 });
164 auto se = source->replicaLabel();
165 auto file = NativeAODInputFile::open(std::move(source), factory, options, filename);
166 file->setGridInfo(se, std::chrono::duration<double>(std::chrono::steady_clock::now() - start).count());
167 return file;
168}
