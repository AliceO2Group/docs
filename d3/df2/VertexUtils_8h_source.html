31template <typename Container>
32VertexLabel computeMainVertexLabel(const Container& elements)
33{
34 // we only care about the source&event of the tracks, not the trackId
35 auto composeVtxLabel = [](const o2::MCCompLabel& lbl) -> o2::MCCompLabel {
36 return {o2::MCCompLabel::maxTrackID(), lbl.getEventID(), lbl.getSourceID(), lbl.isFake()};
37 };
38 std::unordered_map<o2::MCCompLabel, size_t> frequency;
39 for (const auto& element : elements) {
40 ++frequency[composeVtxLabel(element)];
41 }
42 o2::MCCompLabel elem{};
43 size_t maxCount = 0;
44 for (const auto& [key, count] : frequency) {
45 if (count > maxCount) {
46 maxCount = count;
47 elem = key;
48 }
49 }
50 if (maxCount <= 1) { // need >50%
51 elem.setFakeFlag();
52 }
53 return std::make_pair(elem, static_cast<float>(maxCount) / static_cast<float>(elements.size()));
54}
