36float Line::getDCA2(const Line& firstLine, const Line& secondLine, const float precision)
37{
38 const auto& a = firstLine.cosinesDirector;
39 const auto& b = secondLine.cosinesDirector;
40 const float n[3] = {a[1] * b[2] - a[2] * b[1],
41 a[2] * b[0] - a[0] * b[2],
42 a[0] * b[1] - a[1] * b[0]};
43 const float norm2 = n[0] * n[0] + n[1] * n[1] + n[2] * n[2];
44
45 float d[3];
46 for (int i = 0; i < 3; ++i) {
47 d[i] = secondLine.originPoint[i] - firstLine.originPoint[i];
48 }
49
50 if (norm2 <= precision * precision) {
51 // lines are parallel, fall back to point-to-line distance
52 float proj = 0.f;
53 for (int i = 0; i < 3; ++i) {
54 proj += d[i] * a[i];
55 }
56 float res2 = 0.f;
57 for (int i = 0; i < 3; ++i) {
58 const float residual = d[i] - proj * a[i];
59 res2 += residual * residual;
60 }
61 return res2;
62 }
63
64 const float numerator = d[0] * n[0] + d[1] * n[1] + d[2] * n[2];
65 return (numerator * numerator) / norm2;
66}

81ClusterLines::ClusterLines(gsl::span<const int> lineIndices, gsl::span<const Line> lines)
82{
83 if (lineIndices.size() < 2) {
84 return;
85 }
86 for (const auto idx : lineIndices) {
87 add(lines[idx]);
88 }
89 computeClusterCentroid();
90 if (!mIsValid) {
91 return;
92 }
93 for (const auto idx : lineIndices) {
94 addResidual(lines[idx]);
95 }
96}
