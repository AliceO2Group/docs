122struct DetInfoResid { // detector info associated with residual
123 uint32_t word = 0; // container interpreted in a different way depending on the detector type
124 //
125 // TPC view: qTot and qMax of the cluster
126 uint16_t qTotTPC() const { return static_cast<uint16_t>(word & 0xFFFFu); }
127 uint16_t qMaxTPC() const { return static_cast<uint16_t>((word >> 16) & 0xFFFFu); }
128 void setTPC(uint16_t qTot, uint16_t qMax) { word = (static_cast<uint32_t>(qMax) << 16) | static_cast<uint32_t>(qTot); }
129 //
130 // TRD view: q0, q1, q2 + calibrated slope (truncated to in +-3.5 range)
131 static constexpr uint32_t TRDQ0NB = 7, TRDQ1NB = 7, TRDQ2NB = 6, TRDSlpNB = 12;
132 static constexpr uint32_t TRDQ0Msk = (1 << TRDQ0NB) - 1, TRDQ1Msk = (1 << TRDQ1NB) - 1, TRDQ2Msk = ((1 << TRDQ2NB) - 1), TRDSlpMsk = (1 << TRDSlpNB) - 1;
133 static constexpr float TRDMaxSlope = 3.5, TRDSlope2Int = ((1 << TRDSlpNB) - 1) / (2 * TRDMaxSlope), TRDInt2Slope = 1.f / TRDSlope2Int;
134 uint16_t q0TRD() const { return static_cast<uint16_t>(word & TRDQ0Msk); }
135 uint16_t q1TRD() const { return static_cast<uint16_t>((word >> TRDQ0NB) & TRDQ1Msk); }
136 uint16_t q2TRD() const { return static_cast<uint16_t>((word >> (TRDQ0NB + TRDQ1NB)) & TRDQ2Msk); }
137 float slopeTRD() const { return ((word >> (TRDQ0NB + TRDQ1NB + TRDQ2NB)) & TRDSlpMsk) * TRDInt2Slope - TRDMaxSlope; }
138 void setTRD(uint8_t q0, uint8_t q1, uint8_t q2, float slope)
139 {
140 float rslope = (slope + TRDMaxSlope) * TRDSlope2Int;
141 if (rslope < 0.f) {
142 rslope = 0;
143 } else if (rslope > TRDSlpMsk) {
144 rslope = TRDSlpMsk;
145 }
146 uint32_t slpI = std::round(rslope);
147 word = (static_cast<uint32_t>(slpI << (TRDQ0NB + TRDQ1NB + TRDQ2NB)) |
148 static_cast<uint32_t>((q2 & TRDQ2Msk) << (TRDQ0NB + TRDQ1NB)) |
149 static_cast<uint32_t>((q1 & TRDQ1Msk) << TRDQ0NB) |
150 static_cast<uint32_t>(q0 & TRDQ0Msk));
151 }
152 //
153 // TOF view (time difference in \mus wrt seeding ITS-TPC track)
154 float timeTOF() const { return std::bit_cast<float>(word); }
155 void setTOF(float t) { word = std::bit_cast<uint32_t>(t); }
156 //
157 // No info for ITS is stored
158 //
159 // PV view (time difference in \mus wrt contributing ITS-TPC track)
160 float timePV() const { return std::bit_cast<float>(word); }
161 void setPV(float t) { word = std::bit_cast<uint32_t>(t); }
162
163 ClassDefNV(DetInfoResid, 1);
164};
