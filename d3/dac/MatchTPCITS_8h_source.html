179struct MatchRecord {
180 float chi2 = -1.f;
181 int partnerID = MinusOne;
182 int nextRecID = MinusOne;
183 int matchedIC = MinusOne;
184 MatchRecord(int partID, float chi2match, int nxt = MinusOne, int candIC = MinusOne) : partnerID(partID), chi2(chi2match), nextRecID(nxt), matchedIC(candIC) {}
185 MatchRecord() = default;
186
187 bool isBetter(float otherChi2, int otherIC = MinusOne) const
188 { // prefer record with matched IC candidate, otherwise, better chi2
189 if (otherIC == MinusOne) {
190 return matchedIC == MinusOne ? chi2 < otherChi2 : true;
191 } else {
192 return matchedIC == MinusOne ? false : chi2 < otherChi2;
193 }
194 }
195
196 bool isBetter(const MatchRecord& other) const
197 {
198 return isBetter(other.chi2, other.matchedIC);
199 }
200};

273 bool checkLinkHasUsedClusters(int linkID, const ITSClusStatus& clStatus) const
274 {
275 // check if some clusters used by the link or its parents are forbidden (already used by validatet track)
276 while (linkID > MinusOne) {
277 const auto& link = getLink(linkID);
278 if (link.clID > MinusOne && clStatus[o2::itsmft::clusID2Layer(link.clID)][o2::itsmft::clusID2Index(link.clID)] != MinusOne) {
279 return true;
280 }
281 linkID = link.parentID;
282 }
283 return false;
284 }
285 void flagLinkUsedClusters(int linkID, ITSClusStatus& clStatus) const
286 {
287 // check if some clusters used by the link or its parents are forbidden (already used by validated track)
288 while (linkID > MinusOne) {
289 const auto& link = getLink(linkID);
290 if (link.clID > MinusOne) {
291 clStatus[o2::itsmft::clusID2Layer(link.clID)][o2::itsmft::clusID2Index(link.clID)] = MinusTen;
292 }
293 linkID = link.parentID;
294 }
295 }
296 size_t sizeInternal() const { return sizeof(ABTrackLink) * getNLinks(); }
297 size_t capInternal() const { return sizeof(ABTrackLink) * getNLinks(); }
