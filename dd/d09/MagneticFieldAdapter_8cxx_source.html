59Acts::Result<Acts::Vector3> MagneticFieldAdapter::getField(const Acts::Vector3& position,
60 Cache& /*cache*/) const
61{
62 const double xyz[3] = {position[0] / Acts::UnitConstants::cm,
63 position[1] / Acts::UnitConstants::cm,
64 position[2] / Acts::UnitConstants::cm};
65 double bxyz[3] = {0., 0., 0.};
66 mField->Field(xyz, bxyz);
67
68 return Acts::Result<Acts::Vector3>::success(
69 Acts::Vector3{bxyz[0] * kGaussToActs, bxyz[1] * kGaussToActs, bxyz[2] * kGaussToActs});
70}
