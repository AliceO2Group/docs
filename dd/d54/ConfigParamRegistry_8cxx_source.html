74// ptree::erase only removes direct children, so the parent of a nested key has to be looked up first
75void eraseImpl(boost::property_tree::ptree& tree, std::string_view key)
76{
77 auto pos = key.rfind('.');
78 if (pos == std::string_view::npos) {
79 tree.erase(std::string(key));
80 return;
81 }
82 if (auto parent = tree.get_child_optional(std::string(key.substr(0, pos)))) {
83 parent->erase(std::string(key.substr(pos + 1)));
84 }
85}

88template <ConfigValueType T>
89T ConfigParamRegistry::get(const char* key) const
90{
91 try {
92 return getImpl<T>(this->mStore->store(), key);
93 } catch (std::exception& e) {
94 throw std::invalid_argument(std::string("missing option: ") + key + " (" + e.what() + ")");
95 } catch (...) {
96 throw std::invalid_argument(std::string("error parsing option: ") + key);
97 }

100void ConfigParamRegistry::override(const char* key, ConfigValueType auto const& val) const
101{
102 using T = std::decay_t<decltype(val)>;
103 try {
104 auto& store = mStore->store();
105 if constexpr (SimpleConfigValueType<T> || StringConfigValueType<T>) {
106 store.put(key, val);
107 } else if constexpr (VectorConfigValueType<T>) {
108 // arrays are stored as subtrees, which have to be replaced as a whole
109 store.put_child(key, vectorToBranch(val.data(), val.size()));
110 } else if constexpr (Array2DLike<T>) {
111 store.put_child(key, array2DToBranch(T(val)));
112 } else if constexpr (LabeledArrayLike<T>) {
113 store.put_child(key, labeledArrayToBranch(T(val)));
114 }
115 } catch (std::exception& e) {
116 throw std::invalid_argument(std::string("failed to store an option: ") + key + " (" + e.what() + ")");
117 } catch (...) {
118 throw std::invalid_argument(std::string("failed to store an option: ") + key);
119 }
120}
