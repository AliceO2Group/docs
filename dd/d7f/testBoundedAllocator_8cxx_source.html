26{

32struct ThrowingCopy {
33 static inline int copiesBeforeFailure = -1;
34 ThrowingCopy() = default;
35 ThrowingCopy(const ThrowingCopy&)
36 {
37 if (copiesBeforeFailure == 0) {
38 throw std::runtime_error{"copy"};
39 }
40 if (copiesBeforeFailure > 0) {
41 --copiesBeforeFailure;
42 }
43 }
44 ThrowingCopy& operator=(const ThrowingCopy&) = default;
45};

48BOOST_AUTO_TEST_CASE(huge_request_cannot_wrap_past_budget)
