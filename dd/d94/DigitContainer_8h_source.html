117inline DigitContainer::DigitContainer()
118{
119 auto& detParam = ParameterDetector::Instance();
120 auto& gasParam = ParameterGas::Instance();
121 auto& eleParam = ParameterElectronics::Instance();
122 mTmaxTriggered = detParam.TmaxTriggered;
123
124 // always have 50 % contingency for the size of the container depending on the input
125 mOffset = static_cast<TimeBin>(detParam.TPCRecoWindowSim * detParam.TPClength / gasParam.DriftV / eleParam.ZbinWidth);
126 mTimeBins.resize(mOffset, nullptr);
127}
128
129inline void DigitContainer::reset()
130{
131 mFirstTimeBin = 0;
132 mEffectiveTimeBin = 0;
133 // drop all time bins, including those beyond the timeframe limit, so that no MC labels survive
134 for (auto time : mTimeBins) {
135 delete time;
136 }
137 mTimeBins.assign(mOffset, nullptr);
138 if (mPrevDigArr) {
139 std::fill(mPrevDigArr->begin(), mPrevDigArr->end(), PrevDigitInfo{});
140 }
141}
142
143inline void DigitContainer::reserve(TimeBin eventTimeBin)
144{
145 const auto space = mOffset + eventTimeBin - mFirstTimeBin;
146 if (mTimeBins.size() < space) {
147 mTimeBins.resize(space);
148 }
149}

151inline void DigitContainer::addDigit(const MCCompLabel& label, const CRU& cru, TimeBin timeBin, GlobalPadNumber globalPad,
152 float signal)
153{
154 mEffectiveTimeBin = timeBin - mFirstTimeBin;
155 if (mEffectiveTimeBin >= mTimeBins.size()) {
156 // LOG(warning) << "Out of bound access to digit container .. dropping digit";
157 return;
158 }
159
160 if (mTimeBins[mEffectiveTimeBin] == nullptr) {
161 mTimeBins[mEffectiveTimeBin] = new DigitTime();
162 }
163
164 mTimeBins[mEffectiveTimeBin]->addDigit(label, cru, globalPad, signal);
165}
