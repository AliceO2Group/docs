66template <int DetID>
67void ClusterReader<DetID>::run(ProcessingContext& pc)
68{
69 constexpr auto origin = DetID == o2::detectors::DetID::TRK ? o2::header::gDataOriginTRK : o2::header::gDataOriginFT3;
70 const auto entry = mTree->GetReadEntry() + 1;
71 // A timeframe without a collision has no entry: publish empty streams, so
72 // that the consumers still see the timeframe (as the digit reader does).
73 const bool noEntry = entry >= mTree->GetEntries();
74 if (noEntry) {
75 LOG(info) << detName<DetID>() << "ClusterReader: no entry to read, publishing empty output";
76 } else {
77 mTree->GetEntry(entry);
78 }
79 static const std::vector<o2::trkft3::Cluster<DetID>> noClusters;
80 static const std::vector<unsigned char> noPatterns;
81 static const std::vector<o2::trkft3::ROFRecord> noROFs;
82 static const o2::dataformats::MCTruthContainer<o2::MCCompLabel> noLabels;
83 for (int layer = 0; layer < NLayers; ++layer) {
84 const auto subspec = static_cast<DataAllocator::SubSpecificationType>(layer);
85 const auto& clusters = noEntry ? noClusters : *mClusters[layer];
86 const auto& rofs = noEntry ? noROFs : *mROFs[layer];
87 LOG(info) << detName<DetID>() << "ClusterReader on layer " << layer << " pushes " << clusters.size() << " clusters in " << rofs.size()
88 << " RO frames at entry " << entry;
89 pc.outputs().snapshot(Output{origin, "COMPCLUSTERS", subspec}, clusters);
90 pc.outputs().snapshot(Output{origin, "PATTERNS", subspec}, noEntry ? noPatterns : *mPatterns[layer]);
91 pc.outputs().snapshot(Output{origin, "CLUSTERSROF", subspec}, rofs);
92 if (mUseMC) {
93 pc.outputs().snapshot(Output{origin, "CLUSTERSMCTR", subspec}, noEntry ? noLabels : *mLabels[layer]);
94 }
95 }
96 if (noEntry || mTree->GetReadEntry() + 1 >= mTree->GetEntries()) {
97 pc.services().get<ControlService>().endOfStream();
98 pc.services().get<ControlService>().readyToQuit(QuitRequest::Me);
99 }
100}

102template <int DetID>
103void ClusterReader<DetID>::connectTree(const std::string& filename)
104{
105 mTree.reset();
106 mFile.reset(TFile::Open(filename.c_str()));
107 if (!mFile || mFile->IsZombie()) {
108 throw std::runtime_error(detName<DetID>() + "ClusterReader: cannot open " + filename);
109 }
110 mTree.reset(mFile->Get<TTree>("o2sim"));
111 if (!mTree) {
112 throw std::runtime_error(detName<DetID>() + "ClusterReader: no tree o2sim in " + filename);
113 }
114 for (int layer = 0; layer < NLayers; ++layer) {
115 setBranchAddress("ClusterComp", layer, mClusters[layer]);
116 setBranchAddress("ClusterPatt", layer, mPatterns[layer]);
117 setBranchAddress("ClustersROF", layer, mROFs[layer]);
118 if (mUseMC) {
119 setBranchAddress("ClusterMCTruth", layer, mLabels[layer]);
120 }
121 }
122 LOG(info) << detName<DetID>() << "ClusterReader: " << filename << " with " << mTree->GetEntries() << " entries";
123}

125template <int DetID>
126template <typename Ptr>
127void ClusterReader<DetID>::setBranchAddress(const std::string& base, int layer, Ptr& address)
128{
129 const auto name = branchName<DetID>(base, layer);
130 if (!mTree->GetBranch(name.c_str())) {
131 throw std::runtime_error(detName<DetID>() + "ClusterReader: no branch " + name + (base == "ClusterMCTruth" ? " (MC labels requested)" : ""));
132 }
133 if (const Int_t status = mTree->SetBranchAddress(name.c_str(), &address); status != 0) {
134 throw std::runtime_error(std::format("{}ClusterReader: cannot set the address of {} ({})", detName<DetID>(), name, status));
135 }
136}

140template <int DetID>
141DataProcessorSpec getClusterReaderSpecT(bool useMC, std::string defname)
142{
143 constexpr auto origin = DetID == o2::detectors::DetID::TRK ? o2::header::gDataOriginTRK : o2::header::gDataOriginFT3;
144 std::vector<OutputSpec> outputs;
145 for (int layer = 0; layer < ClusterReader<DetID>::NLayers; ++layer) {
146 outputs.emplace_back(origin, "COMPCLUSTERS", layer, Lifetime::Timeframe);
147 outputs.emplace_back(origin, "PATTERNS", layer, Lifetime::Timeframe);
148 outputs.emplace_back(origin, "CLUSTERSROF", layer, Lifetime::Timeframe);
149 if (useMC) {
150 outputs.emplace_back(origin, "CLUSTERSMCTR", layer, Lifetime::Timeframe);
151 }
152 }
153 return DataProcessorSpec{
154 lowerName<DetID>() + "-cluster-reader",
155 Inputs{},
156 outputs,
157 AlgorithmSpec{adaptFromTask<ClusterReader<DetID>>(useMC)},
158 Options{{lowerName<DetID>() + "-cluster-infile", VariantType::String, std::move(defname), {"Name of the input cluster file"}}}};
159}
160} // namespace
