30struct consoleArgumentError : public std::runtime_error { // Wrong arguments for an existing command, what(): the error, usage: the argument types
31 consoleArgumentError(const std::string& cmd, const std::string& error, const std::string& types) : std::runtime_error(error), command(cmd), usage("[")
32 {
33 static constexpr std::pair<char, const char*> names[] = {{'s', "string"}, {'i', "int"}, {'d', "double"}, {'a', "any"}, {'+', "[optional]"}};
34 for (char t : types) {
35 if (const auto* n = std::find_if(std::begin(names), std::end(names), [t](const auto& entry) { return entry.first == t; }); n != std::end(names)) {
36 usage += (usage.size() > 1 ? " " : "") + std::string(n->second);
37 }
38 }
39 usage += "]";
40 }
41 std::string command, usage;
42};

53void GPUDisplayConsole::handleConsoleKey(uint8_t key, bool shift)
54{
55 if (key == KEY_PAGEUP || key == KEY_PAGEDOWN || (shift && (key == KEY_UP || key == KEY_DOWN))) { // Half a page / one line
56 const int32_t lines = key == KEY_PAGEUP || key == KEY_PAGEDOWN ? std::max<int32_t>(1, mVisibleLines / 2) : 1;
57 scroll(key == KEY_PAGEUP || key == KEY_UP ? lines : -lines);
58 return;
59 }
60 const uint32_t nHistory = mConsoleHistory.size();
61 auto isOutput = [this](uint32_t i) { return mConsoleHistory[i].size() && mConsoleHistory[i][0] == ' '; }; // Indented output of commands, not selectable
62 if (key == KEY_UP || key == KEY_DOWN) { // Entry 0 is the title, not selectable
63 if (key == KEY_UP) {
64 uint32_t i = mConsoleTextCurrent ? mConsoleTextCurrent : nHistory;
65 while (--i >= 1 && isOutput(i)) {
66 }
67 mConsoleTextCurrent = i >= 1 ? i : mConsoleTextCurrent;
68 } else if (mConsoleTextCurrent) {
69 uint32_t i = mConsoleTextCurrent;
70 while (++i < nHistory && isOutput(i)) {
71 }
72 mConsoleTextCurrent = i < nHistory ? i : 0;
73 }
74 mConsoleTextPos = shownText().size(); // At the end, as when editing the entry starts
75 return;
76 } else if (key == KEY_ENTER) {
77 std::string command = mConsoleTextCurrent ? mConsoleHistory[mConsoleTextCurrent] : mConsoleText;
78 command.erase(0, std::min(command.size(), command.find_first_not_of(' '))); // Leading spaces would mark it as output in the history
79 mConsoleText.clear();
80 mConsoleTextCurrent = mConsoleTextPos = mScroll = 0;
81 if (!command.empty()) {
82 consoleAppend(command); // Before its output
83 handleConsoleCommand(command);
84 }
85 return;
86 } else if (key != KEY_LEFT && key != KEY_RIGHT && key != KEY_BACKSPACE && key != KEY_DELETE && (key < 32 || key >= 127)) { // Other special keys are ignored
87 return;
88 }
89 if (mConsoleTextCurrent) { // Edit a copy of the history entry, with the cursor at its end (left: on its last character), delete then removes nothing
90 mConsoleText = mConsoleHistory[mConsoleTextCurrent];
91 mConsoleTextCurrent = 0;
92 mConsoleTextPos = mConsoleText.size() - (key == KEY_LEFT && mConsoleText.size());
93 if (key == KEY_LEFT || key == KEY_RIGHT) {
94 return;
95 }
96 }
97 mConsoleTextPos = std::min<uint32_t>(mConsoleTextPos, mConsoleText.size());
98 if (key == KEY_LEFT) {
99 mConsoleTextPos -= mConsoleTextPos > 0;
100 } else if (key == KEY_RIGHT) {
101 mConsoleTextPos += mConsoleTextPos < mConsoleText.size();
102 } else if (key == KEY_BACKSPACE) {
103 if (mConsoleTextPos > 0) {
104 mConsoleText.erase(--mConsoleTextPos, 1);
105 }
106 } else if (key == KEY_DELETE) {
107 if (mConsoleTextPos < mConsoleText.size()) {
108 mConsoleText.erase(mConsoleTextPos, 1);
109 }
110 } else {
111 mConsoleText.insert(mConsoleTextPos++, 1, (char)key);
112 }
113}

115void GPUDisplayConsole::consolePrint(const char* format, ...)
116{
117 va_list args, argsCopy;
118 va_start(args, format);
119 va_copy(argsCopy, args);
120 std::string buffer(std::max(0, vsnprintf(nullptr, 0, format, args)), '\0');
121 va_end(args);
122 vsnprintf(buffer.data(), buffer.size() + 1, format, argsCopy);
123 va_end(argsCopy);
124 consoleAppend(" " + buffer); // Indented: output, not a command, skipped by up / down
125 if (mPrintToTerminal) {
126 GPUInfo("%s", buffer.c_str());
127 }

130void GPUDisplayConsole::consoleInfo(const char* format, const char* text)
131{
132 const auto now = std::chrono::steady_clock::now();
133 if (!mConsoleInfoLastMessage.empty() && mConsoleInfoLastMessage == format && now - mConsoleInfoLastTime < std::chrono::seconds(2)) {
134 mConsoleHistory.back() = std::string(" ") + text; // Same message again shortly after (animation progress, held keys): update its line
135 } else {
136 consoleAppend(std::string(" ") + text);
137 }
138 mConsoleInfoLastMessage = strcmp(format, "%s") ? format : ""; // A plain "%s" can be any message, never merged
139 mConsoleInfoLastTime = now;

148void GPUDisplayConsole::consoleLimitHistory(uint32_t historySize)
149{
150 const size_t size = std::max<uint32_t>(historySize, 2);
151 if (mConsoleHistory.size() < 2 * size) {
152 return;
153 }
154 const size_t removed = mConsoleHistory.size() - size;
155 std::move(mConsoleHistory.begin() + 1 + removed, mConsoleHistory.end(), mConsoleHistory.begin() + 1); // Keep the title (entry 0)
156 mConsoleHistory.resize(size);
157 mConsoleTextCurrent = mConsoleTextCurrent > removed ? mConsoleTextCurrent - removed : 0; // Same entry, the end if it was removed
158 mConsoleTextPos = std::min<uint32_t>(mConsoleTextPos, shownText().size());
159 mScroll = mScroll + 1 >= size ? 0 : mScroll; // Counted from the end: same lines, the end if the lowest shown one was removed
160}

162// Tokens separated by spaces: text in single / double quotes and characters after a backslash are plain text (the token is a string),
163// otherwise a token is int64_t for [+-]digits, double for other decimal numbers (also with exponent), a string otherwise. Throws std::runtime_error for an unterminated quote / trailing backslash.
164
165std::tuple<const GPUDisplayConsole::GPUDisplayConsoleCommand&, std::vector<GPUDisplayConsole::consoleArg>> GPUDisplayConsole::parseConsoleCommand(std::string_view cmd) const
166{
167 std::vector<std::pair<std::string, bool>> tokens; // Text, plain text (quoted / escaped)
168 std::string token;
169 bool inToken = false, plainText = false, escape = false;
170 char quote = 0;
171 auto finishToken = [&]() {
172 if (inToken) {
173 tokens.emplace_back(std::move(token), plainText);
174 token.clear();
175 inToken = plainText = false;
176 }
177 };
178 for (char c : cmd) {
179 if (escape) {
180 token += c;
181 escape = false;
182 } else if (quote) {
183 if (c == quote) {
184 quote = 0;
185 } else {
186 token += c;
187 }
188 } else if (c == '\\') {
189 escape = inToken = plainText = true;
190 } else if (c == '"' || c == '\'') {
191 quote = c;
192 inToken = plainText = true;
193 } else if (c == ' ') {
194 finishToken();
195 } else {
196 token += c;
197 inToken = true;
198 }
199 }
200 if (quote || escape) {
201 throw std::runtime_error(quote ? "unterminated quote" : "backslash at the end");
202 }
203 finishToken();
204 if (tokens.empty()) {
205 throw std::runtime_error("empty command");
206 }
207
208 const std::string& name = tokens[0].first;
209 const auto it = mConsoleCommands.find(name);
210 if (it == mConsoleCommands.end()) {
211 throw std::runtime_error("unknown command " + name);
212 }
213 const std::string& types = it->second.args;
214 const bool more = types.size() && types.back() == '+';
215 const size_t nFixed = types.size() - more, nArgs = tokens.size() - 1;
216 if (nArgs < nFixed || (!more && nArgs > nFixed)) {
217 throw consoleArgumentError(name, std::to_string(nFixed) + (more ? " or more" : "") + " arguments expected, got " + std::to_string(nArgs), types);
218 }
219 auto autoType = [](const std::string& text, bool plain) -> consoleArg { // int64_t for [+-]digits, double for other decimal numbers (also with exponent, as saveSettings writes them), string otherwise or if plain text
220 const char *begin = text.c_str() + (text[0] == '+' || text[0] == '-'), *end = text.c_str() + text.size();
221 const bool number = !plain && begin < end && ((*begin >= '0' && *begin <= '9') || *begin == '.') && std::all_of(begin, end, [](char c) { return (c >= '0' && c <= '9') || c == '.' || c == 'e' || c == 'E' || c == '+' || c == '-'; });
222 if (!number) {
223 return text;
224 }
225 int64_t intValue = 0;
226 double doubleValue = 0.;
227 if (const auto rInt = std::from_chars(begin, end, intValue); rInt.ec == std::errc() && rInt.ptr == end) {
228 return text[0] == '-' ? -intValue : intValue;
229 } else if (const auto rDouble = std::from_chars(begin, end, doubleValue); rDouble.ec == std::errc() && rDouble.ptr == end) {
230 return text[0] == '-' ? -doubleValue : doubleValue; // Also integers out of the int64_t range
231 }
232 return text;
233 };
234 std::vector<consoleArg> args;
235 for (size_t i = 0; i < nArgs; i++) {
236 const auto& [text, plain] = tokens[i + 1];
237 const char type = i < nFixed ? types[i] : '+';
238 consoleArg value = type == 's' ? consoleArg(text) : autoType(text, plain);
239 if (type == 'i' && !std::holds_alternative<int64_t>(value)) {
240 throw consoleArgumentError(name, "argument " + std::to_string(i + 1) + " (" + text + ") must be an integer", types);
241 } else if (type == 'd') {
242 if (std::holds_alternative<int64_t>(value)) {
243 value = (double)std::get<int64_t>(value);
244 } else if (!std::holds_alternative<double>(value)) {
245 throw consoleArgumentError(name, "argument " + std::to_string(i + 1) + " (" + text + ") must be a number", types);
246 }
247 } else if (type != 's' && type != 'i' && type != 'a' && type != '+') {
248 throw std::runtime_error(name + ": invalid argument types " + types);
249 }
250 args.emplace_back(std::move(value));
251 }
252 return {it->second, std::move(args)};
253}

255void GPUDisplayConsole::handleConsoleCommand(std::string_view cmd)
256{
257 if (cmd.find_first_not_of(' ') == std::string_view::npos) { // Empty line
258 return;
259 }
260 const GPUDisplayConsoleCommand* command = nullptr;
261 std::vector<consoleArg> args;
262 try {
263 auto parsed = parseConsoleCommand(cmd);
264 command = &std::get<0>(parsed);
265 args = std::move(std::get<1>(parsed));
266 } catch (const consoleArgumentError& e) {
267 consolePrint("Invalid arguments for %s: %s %s", e.command.c_str(), e.what(), e.usage.c_str());
268 return;
269 } catch (const std::runtime_error& e) {
270 consolePrint("Invalid command: %s", e.what());
271 return;
272 }
273 try { // A failing command must not end the display
274 command->func(args);
275 } catch (const std::exception& e) {
276 consolePrint("Error in command %.*s: %s", (int32_t)cmd.size(), cmd.data(), e.what());
277 }
278}

280void GPUDisplayConsole::runScript(std::string_view script)
281{
282 if (mScriptDepth >= 16) {
283 consolePrint("Scripts nested too deeply");
284 return;
285 }
286 std::vector<ScriptLine> lines;
287 for (size_t pos = 0; pos < script.size();) {
288 const size_t end = std::min(script.find_first_of("\r\n", pos), script.size());
289 if (end > pos) { // Empty lines (consecutive \r / \n) are ignored
290 lines.push_back({std::string(script.substr(pos, end - pos)), mScriptDepth + 1});
291 }
292 pos = end + 1;
293 }
294 mScriptQueue.insert(mScriptQueue.begin(), lines.begin(), lines.end()); // Before the remaining lines of a script running this one
295 if (mScriptDepth == 0) { // From a script, its processScripts loop continues with these lines
296 processScripts(false);
297 }
298}

300void GPUDisplayConsole::processScripts(bool newFrame)
301{
302 if (newFrame) {
303 mWaitFrames -= mWaitFrames > 0;
304 if (mBenchmarkLeft && --mBenchmarkLeft == 0) {
305 const double time = std::chrono::duration<double>(std::chrono::steady_clock::now() - mBenchmarkStart).count();
306 consolePrint("benchmark: %u frames in %.3f s, %.2f fps, %.3f ms per frame", mBenchmarkFrames, time, mBenchmarkFrames / time, time * 1000. / mBenchmarkFrames);
307 }
308 }
309 if (mWaitCondition && mWaitCondition()) {
310 mWaitCondition = nullptr;
311 }
312 const uint32_t outerDepth = mScriptDepth;
313 for (uint32_t n = 0; n < 100000 && mScriptQueue.size() && !mWaitFrames && !mBenchmarkLeft && !mWaitCondition && std::chrono::steady_clock::now() >= mWaitUntil; n++) { // Limit per call: a script must not block the display
314 ScriptLine line = std::move(mScriptQueue.front());
315 mScriptQueue.pop_front();
316 mScriptDepth = line.depth;
317 handleConsoleCommand(line.text);
318 }
319 mScriptDepth = outerDepth;
320}

341{

348 for (const auto* c : commands) {
349 print(" " + c->first + (c->second.args.empty() ? "" : " ") + c->second.args + ": " + c->second.description);
