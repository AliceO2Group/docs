163inline void Segmentation::localToDetectorUnchecked(float xRow, float zCol, int& iRow, int& iCol, const int subDetectorID) const
164{
165 // convert to row/col w/o over/underflow check
166 if (subDetectorID != 0 && subDetectorID != 1) {
167 iRow = iCol = -1;
168 return;
169 }
170 const auto& specsConfig = ChipSpecificsParam::Instance();
171 xRow = 0.5 * (specsConfig.ActiveMatrixSizeRows() - specsConfig.PassiveEdgeTop + specsConfig.PassiveEdgeReadOut) - xRow; // coordinate wrt top edge of Active matrix
172 zCol += 0.5 * specsConfig.ActiveMatrixSizeCols(); // coordinate wrt left edge of Active matrix
173 iRow = int(xRow / specsConfig.PitchRow);
174 iCol = int(zCol / specsConfig.PitchCol);
175 // check pixel passive region
176 if (std::abs(xRow - (iRow + 0.5) * specsConfig.PitchRow) > (0.5 * specsConfig.PitchRow - specsConfig.PixelPassiveEdgeX) || std::abs(zCol - (iCol + 0.5) * specsConfig.PitchCol) > (0.5 * specsConfig.PitchCol - specsConfig.PixelPassiveEdgeZ)) {
177 iRow = iCol = -1;
178 return;
179 }
180 if (xRow < 0) {
181 iRow -= 1;
182 }
183 if (zCol < 0) {
184 iCol -= 1;
185 }
186}
