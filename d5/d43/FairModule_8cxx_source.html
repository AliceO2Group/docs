198void FairModule::SetGeometryFileName(TString fname, TString)
199{
200
201 // If absolute path is given as argument, try to find it there.
202 // If the file don't exist break. Don't look anywhere else.
203 if (fname.BeginsWith("/")) {
204 if (gSystem->AccessPathName(fname.Data())) {
205 LOG(fatal) << fName << ": geometry file " << fname << " not found in absolut path!";
206 fgeoName = "";
207 } // file not found
208 fgeoName = fname;
209 LOG(debug) << fName << ": using geometry file " << fgeoName;
210 return;
211 }
212
213 // If GEOMPATH is set, try to find the file there.
214 // If there is no such file go on to look in the default location.
215 TString userPath = getenv("GEOMPATH");
216 userPath.ReplaceAll("//", "/");
217 if (!userPath.IsNull()) {
218 if (!userPath.EndsWith("/")) {
219 userPath += "/";
220 }
221 fgeoName = userPath + fname;
222 if (!gSystem->AccessPathName(fgeoName.Data())) {
223 LOG(debug) << fName << ": using geometry file " << fgeoName;
224 return;
225 }
226 LOG(debug) << fName << ": geometry file " << fname << " not found in GEOMPATH " << userPath;
227 }
228
229 // Look in the standard path
230 fgeoName = getenv("VMCWORKDIR");
231 fgeoName += "/geometry/" + fname;
232 if (!gSystem->AccessPathName(fgeoName.Data())) {
233 LOG(debug) << fName << ": using geometry file " << fgeoName;
234 return;
235 }
236
237 // File not found
238 LOG(fatal) << fName << ": geometry file " << fname << " not found in standard path ";
239 fgeoName = "";
240 return;
241}

481 // Loop over new media which are not in GeoBase and shift the ID

486 med->SetId(i + 1);

558void FairModule::SetDefaultMatrixName(TGeoMatrix* matrix)
559{
560 // Copied from root TGeoMatrix::SetDefaultName() and modified (memory leak)
561 // If no name was supplied in the ctor, the type of transformation is checked.
562 // A letter will be prepended to the name :
563 // t - translation
564 // r - rotation
565 // s - scale
566 // c - combi (translation + rotation)
567 // g - general (tr+rot+scale)
568 // The index of the transformation in gGeoManager list of transformations will
569 // be appended.
570 if (!gGeoManager) {
571 return;
572 }
573 if (strlen(matrix->GetName())) {
574 return;
575 }
576 char type = 'n';
577 if (matrix->IsTranslation()) {
578 type = 't';
579 }
580 if (matrix->IsRotation()) {
581 type = 'r';
582 }
583 if (matrix->IsScale()) {
584 type = 's';
585 }
586 if (matrix->IsCombi()) {
587 type = 'c';
588 }
589 if (matrix->IsGeneral()) {
590 type = 'g';
591 }
592 TObjArray* matrices = gGeoManager->GetListOfMatrices();
593 Int_t index = 0;
594 if (matrices) {
595 index = matrices->GetEntriesFast() - 1;
596 }
597 matrix->SetName(Form("%c%i", type, index));
598}

600void FairModule::AssignMediumAtImport(TGeoVolume* v)
601{
602
608 if (!FairGeoLoader::Instance()) {
609 LOG(fatal) << "FairModule: no FairGeoLoader, the ASCII/ROOT geometry import of FairRoot is not set up in O2";
610 }
611 FairGeoMedia* Media = FairGeoLoader::Instance()->getGeoInterface()->getMedia();
612 FairGeoBuilder* geobuild = FairGeoLoader::Instance()->getGeoBuilder();
613
614 TGeoMedium* med1 = v->GetMedium();
615
616 if (med1) {
617 // In newer ROOT version also a TGeoVolumeAssembly has a material and medium.
618 // This medium is called dummy and is automatically set when the geometry is constructed.
619 // Since this material and medium is neither in the TGeoManager (at this point) nor in our
620 // ASCII file we have to create it the same way it is done in TGeoVolume::CreateDummyMedium()
621 // In the end the new medium and material has to be added to the TGeomanager, because this is
622 // not done automatically when using the default constructor. For all other constructors the
623 // newly created medium or material is added to the TGeomanger.
624 // Create the medium and material only the first time.
625 TString medName = static_cast<TString>(med1->GetName());
626 if (medName.EqualTo("dummy") && nullptr == gGeoManager->GetMedium(medName)) {
627
628 TGeoMaterial* dummyMaterial = new TGeoMaterial();
629 dummyMaterial->SetName("dummy");
630
631 TGeoMedium* dummyMedium = new TGeoMedium();
632 dummyMedium->SetName("dummy");
633 dummyMedium->SetMaterial(dummyMaterial);
634
635 gGeoManager->GetListOfMedia()->Add(dummyMedium);
636 gGeoManager->AddMaterial(dummyMaterial);
637 }
638
639 TGeoMaterial* mat1 = v->GetMaterial();
640 TGeoMaterial* newMat = gGeoManager->GetMaterial(mat1->GetName());
641 if (newMat == 0) {
644 FairGeoMedium* FairMedium = Media->getMedium(mat1->GetName());
645 if (!FairMedium) {
646 LOG(fatal) << "Material " << mat1->GetName() << "is not defined in ASCII file nor in Root file.";
647 // FairMedium=new FairGeoMedium(mat1->GetName());
648 // Media->addMedium(FairMedium);
649 } else {
650
651 Int_t nmed = geobuild->createMedium(FairMedium);
652 v->SetMedium(gGeoManager->GetMedium(nmed));
653 gGeoManager->SetAllIndex();
654 }
655 } else {
657 TGeoMedium* med2 = gGeoManager->GetMedium(mat1->GetName());
658 v->SetMedium(med2);
659 }
660 } else {
661 if (strcmp(v->ClassName(), "TGeoVolumeAssembly") != 0) {
662 //[R.K.-3.3.08] // When there is NO material defined, set it to avoid conflicts in Geant
663 LOG(fatal) << "The volume " << v->GetName()
664 << "has no medium information and not an Assembly so we have to quit";
665 }
666 }
