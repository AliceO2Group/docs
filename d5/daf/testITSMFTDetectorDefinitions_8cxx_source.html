38float parseToken(const char* token)
39{
40 char* endptr = nullptr;
41 const float value = std::strtof(token, &endptr);
42 BOOST_REQUIRE_MESSAGE(endptr != nullptr && *endptr == '\0', "strtof left unparsed characters in \"" << token << "\"");
43 return value;

83template <std::size_t N>
84void checkAuthoredLiteralsMatchExpectedTokens(const std::array<SurfaceDescriptor, N>& surfaces, const std::vector<ExpectedSurface>& expected)
85{
86 BOOST_REQUIRE_EQUAL(surfaces.size(), expected.size());
87 for (const auto& row : expected) {
88 const auto& authored = surfaces[row.index];
89 const float fromToken = parseToken(row.referenceCoordinateToken);
90 BOOST_CHECK_MESSAGE(bitsOf(authored.referenceCoordinate) == bitsOf(fromToken),
91 "surface " << row.index << ": authored literal (bits 0x" << std::hex
92 << bitsOf(authored.referenceCoordinate) << ") does not bit-match expected token \""
93 << row.referenceCoordinateToken << "\" (bits 0x" << bitsOf(fromToken) << ")" << std::dec);
94 }
95}

97template <std::size_t N>
98void checkIdentityAndKind(const std::array<SurfaceDescriptor, N>& surfaces, const std::vector<ExpectedSurface>& expected)
99{

104 BOOST_CHECK(authored.kind == row.kind);
