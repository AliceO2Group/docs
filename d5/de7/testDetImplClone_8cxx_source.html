69BOOST_AUTO_TEST_CASE(clone_gets_own_hit_buffers)
70{
71 o2::test::CloneTestDetector master;
72 master.initializeLate();
73 master.BeginEvent();
74 BOOST_REQUIRE(master.getHits(0) != nullptr);
75
76 std::unique_ptr<o2::test::CloneTestDetector> clone(static_cast<o2::test::CloneTestDetector*>(master.CloneModule()));
77 BOOST_CHECK(!clone->initialized());
78 for (int b = 0; b < 3; ++b) {
79 BOOST_CHECK(clone->cachedBuffer(b) == nullptr);
80 BOOST_CHECK(clone->busyFlag(b) == nullptr);
81 }
82
83 clone->initializeLate();
84 clone->BeginEvent();
85 BOOST_REQUIRE(clone->getHits(0) != nullptr);
86 BOOST_CHECK(clone->getHits(0) != master.getHits(0));
87 for (int b = 0; b < 3; ++b) {
88 BOOST_CHECK(clone->cachedBuffer(b) != master.cachedBuffer(b));
89 BOOST_CHECK(clone->busyFlag(b) != master.busyFlag(b));
90 }
91
92 // hits of the clone do not reach the master, and the master survives the clone
93 clone->getHits(0)->emplace_back();
94 BOOST_CHECK(master.getHits(0)->empty());
95 clone.reset();
96 master.getHits(0)->emplace_back();
97 BOOST_CHECK_EQUAL(master.getHits(0)->size(), 1);
98}
