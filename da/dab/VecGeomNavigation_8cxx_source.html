53void installVecGeomNavigator()
54{
55 auto const& g4Params = o2::conf::G4Params::Instance();
56
57 if (TMCManager::Instance() != nullptr) {
58 LOG(fatal) << "G4.navmode=kVecGeom cannot be used with the multi-engine TMCManager: restoring a "
59 "geometry state across engines goes through the TGeo navigator";
60 }
61 if (o2::GeometryManagerParam::Instance().useParallelWorld) {
62 LOG(fatal) << "G4.navmode=kVecGeom cannot be used with GeometryManagerParam.useParallelWorld: VecGeom "
63 "has no equivalent of the TGeo priority world";
64 }
65 auto* navMgr = TG4RootNavMgr::GetInstance();
66 if (navMgr == nullptr || navMgr->GetDetConstruction() == nullptr) {
67 LOG(fatal) << "G4.navmode=kVecGeom needs the Geant4 geometry built from TGeo by g4root, which is what "
68 "the geomRoot option provides; no TG4RootNavMgr was found";
69 }
70 auto* detConstruction = navMgr->GetDetConstruction();
71 if (!detConstruction->IsConstructed()) {
72 LOG(fatal) << "The Geant4 geometry has not been built yet; installVecGeomNavigator must be called after "
73 "the TGeant4 engine has been created";
74 }
75
76 if (!g4Params.vecgeomFlattenAssemblies && g4Params.vecgeomNavigator == o2::conf::EVecGeomNav::kRelocating) {
77 LOG(fatal) << "G4.vecgeomNavigator=kRelocating needs G4.vecgeomFlattenAssemblies=true: it enters a daughter by "
78 "locating inside it, which an assembly cannot answer";
79 }
80
81 TStopwatch timer;
82 timer.Start();
83 o2::base::GeometryManager::buildVecGeomGeometry(g4Params.vecgeomFlattenAssemblies, g4Params.vecgeomBooleanThreshold);
84 timer.Stop();
85 LOG(info) << "VecGeom geometry built in " << timer.RealTime() << " s";
86
87 timer.Start();
88 // Owned here for the lifetime of the process; the navigator keeps a reference to it.
89 static VecGeomG4Map map;
90 map.build(*detConstruction, g4Params.vecgeomFlattenAssemblies);
91 timer.Stop();
92 LOG(info) << "VecGeom to Geant4 map built in " << timer.RealTime() << " s";
93
94 if (!g4Params.vecgeomCheckVolumes.empty()) {
95 std::stringstream names(g4Params.vecgeomCheckVolumes);
96 std::string one;
97 while (std::getline(names, one, ',')) {
98 checkVecGeomVolume(one.c_str(), 20, 200);
99 }
100 }
101 if (g4Params.vecgeomCheckRays > 0) {
102 checkVecGeomRays(static_cast<std::size_t>(g4Params.vecgeomCheckRays));
103 }
104 if (g4Params.vecgeomCheckLocation > 0) {
105 checkVecGeomLocation(static_cast<std::size_t>(g4Params.vecgeomCheckLocation));
106 }
107
108 G4Navigator* navigator = nullptr;
109 if (g4Params.vecgeomNavigator == o2::conf::EVecGeomNav::kPropagated) {
110 LOG(info) << "VecGeom navigation: propagating navigator";
111 navigator = new VecGeomG4PropagatingNavigator(map, g4Params.vecgeomZeroSafety);
112 } else {
113 navigator = new VecGeomG4Navigator(map, g4Params.vecgeomPushDepth, g4Params.vecgeomZeroSafety);
114 }
115 navigator->SetWorldVolume(detConstruction->GetTopPV());
116
117 // Same sequence TG4RootNavMgr::SetNavigator uses, run here because by the time the engine
118 // exists the navigator manager considers itself connected and refuses to swap.
119 auto* trMgr = G4TransportationManager::GetTransportationManager();
120 trMgr->SetNavigatorForTracking(navigator);
121 auto* fieldMgr = trMgr->GetPropagatorInField()->GetCurrentFieldManager();
122 delete trMgr->GetPropagatorInField();
123 trMgr->SetPropagatorInField(new G4PropagatorInField(navigator, fieldMgr));
124 trMgr->ActivateNavigator(navigator);
125 if (auto* evtMgr = G4EventManager::GetEventManager()) {
126 evtMgr->GetTrackingManager()->GetSteppingManager()->SetNavigator(navigator);
127 }
128
129 LOG(info) << "VecGeom navigator registered with the Geant4 transportation manager";
130}
