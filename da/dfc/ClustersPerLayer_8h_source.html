45template <typename T>
46class ClustersPerLayer
47{
48 public:
50 void init(int nLr)
51 {
52 if (nLr < 1 || nLr > MaxClusLayers) {
53 LOGP(fatal, "Clusters container cannot be initialized for {} layers, must be within 1:{}", nLr, MaxClusLayers);
54 }
55 mClusters.clear();
56 mLrFirst.fill(0);
57 mNLayers = nLr;
58 }
59
61 void beginLayer(int lr)
62 {
63 if (lr < 0 || lr >= mNLayers) {
64 LOGP(fatal, "Clusters container was initialized for {} layers, cannot fill the layer {}", mNLayers, lr);
65 }
66 mLrFirst[lr] = int(mClusters.size());
67 }
68
70 void finalize()
71 {
72 for (int lr = mNLayers; lr <= MaxClusLayers; lr++) { // the slots above the filled ones are empty
73 mLrFirst[lr] = int(mClusters.size());
74 }
75 }
76
77 void clear()
78 {
79 mClusters.clear();
80 mLrFirst.fill(0);
81 mNLayers = 1;
82 }
83
85 auto& getClusters() { return mClusters; }
86 const auto& getClusters() const { return mClusters; }
87
88 auto getNLayers() const { return mNLayers; }
89 auto getFirstIndex(int lr) const { return mLrFirst[lr]; }
90 auto getNClusters(int lr) const { return mLrFirst[lr + 1] - mLrFirst[lr]; }
91 auto size() const { return mClusters.size(); }
92 bool empty() const { return mClusters.empty(); }
93
95 int flatIndex(int composedID) const
96 {
97 return mLrFirst[clusID2Layer(composedID)] + clusID2Index(composedID);
98 }
99 const T& operator[](int composedID) const { return mClusters[flatIndex(composedID)]; }
100
101 private:
102 std::vector<T> mClusters{};
103 std::array<int, MaxClusLayers + 1> mLrFirst{};
104 int mNLayers = 1;
105};
