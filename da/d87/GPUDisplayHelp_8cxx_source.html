30void GPUDisplay::PrintGLHelpText(float colorValue)
31{
32 const auto& lines = mFrontend->getHelpText();
33 const int32_t maxChars = mBackend->mFontCharWidth > 0.f ? (mBackend->mScreenWidth / mFrontend->getDisplayScale() - 80.f) / mBackend->mFontCharWidth : 0; // Characters fitting between the margins, 0 = unknown (no Freetype font), no wrapping
34 uint32_t row = 0;
35 auto print = [&](const char* s, uint32_t nCharacters, uint32_t indent) { GLPrintText(s, 40.f + indent * mBackend->mFontCharWidth, 35 + std::max(20, mDrawTextFontSize + 4) * (1 + row++), colorValue, colorValue, colorValue, mInfoHelpTimer.GetCurrentElapsedTime() >= 5 ? (6 - mInfoHelpTimer.GetCurrentElapsedTime()) : 1, nCharacters, false); };
36 for (const auto& line : lines) {
37 if (maxChars <= 0 || line.size() <= (size_t)maxChars) {
38 print(line.c_str(), 0, 0);
39 continue;
40 }
41 const size_t textStart = line.find(": ");
42 const size_t indent = textStart != std::string::npos && textStart + 2 < (size_t)maxChars / 2 ? textStart + 2 : 0; // Continuation lines aligned with the text after the keys, if that leaves enough space
43 size_t pos = 0; // Start of the part not yet printed
44 for (bool first = true;; first = false) { // Wrap at the last space that fits, or hard if there is none (also if it is only in the keys)
45 const size_t lineIndent = first ? 0 : indent, available = maxChars - lineIndent;
46 if (line.size() - pos <= available) {
47 print(line.c_str() + pos, line.size() - pos, lineIndent);
48 break;
49 }
50 size_t cut = line.rfind(' ', pos + available);
51 if (cut == std::string::npos || cut <= pos + (first ? indent : 0)) {
52 cut = pos + available;
53 }
54 print(line.c_str() + pos, cut - pos, lineIndent);
55 pos = line[cut] == ' ' ? cut + 1 : cut;
56 }
57 }
