71 genAndLoadRTC();
72 exit(0);
73 }
74
75 if (mMaster == nullptr) {
76 mMultiprocessorCount = std::max<int32_t>(1, mMaxHostThreads); // One device block runs as one TBB task

135int32_t GPUReconstructionCPUDevice::ExitDevice_Runtime()
136{
137 SynchronizeGPU();
138 mEventStorage.clear();
139
140 if (mMaster == nullptr) {
141#ifndef _WIN32
142 for (auto* lib : mInternals->rtcLibraries) {
143 dlclose(lib);
144 }
145#endif
146 mInternals->rtcLibraries.clear();
147 mInternals->rtcKernels.clear();
148 mInternals->arena.reset();
149 alignedDefaultBufferDeleter()(mInternals->deviceMemory);
150 alignedDefaultBufferDeleter()(mInternals->hostMemory);
151 alignedDefaultBufferDeleter()(mInternals->constantMemory);

167void GPUReconstructionCPUDeviceInternals::enqueue(int32_t stream, GPUReconstructionCPUDeviceTask&& task)
168{
169 auto& q = queues[stream];
170 bool start = false;
171 {
172 std::lock_guard lock(q.mutex);
173 q.tasks.emplace_back(std::move(task));
174 if (!q.active) {
175 q.active = start = true;
176 }
177 }
178 if (start) {
179 startQueue(stream);
180 }

188void GPUReconstructionCPUDeviceInternals::processQueue(int32_t stream)
189{
190 auto& q = queues[stream];
191 while (true) {
192 GPUReconstructionCPUDeviceTask* task;
193 {
194 std::lock_guard lock(q.mutex);
195 if (q.tasks.empty()) {
196 q.active = false;
197 q.cv.notify_all();
198 return;
199 }
200 task = &q.tasks.front(); // Only this TBB task removes elements, and push_back does not invalidate references into a deque
201 }
202 while (task->nextWait < task->waits.size()) {
203 auto [e, value] = task->waits[task->nextWait];
204 std::lock_guard lock(e->mutex);
205 if (e->nCompleted < value) { // Do not block the TBB thread, suspend the queue, completeEvent will start it again
206 e->waiting.push_back({value, this, stream});
207 return;
208 }
209 task->nextWait++;
210 }
211 if (!error) {
212 try {
213 task->work();
214 } catch (const std::exception& ex) {
215 GPUError("CPUDevice: Error in command queue %d: %s", stream, ex.what());
216 error = 1;
217 }
218 }
219 if (task->record) {
220 completeEvent(task->record, task->recordValue); // Also after errors, so that nothing waits forever
221 }
222 std::lock_guard lock(q.mutex);
223 q.tasks.pop_front();
224 }
225}

227void GPUReconstructionCPUDeviceInternals::completeEvent(GPUReconstructionCPUDeviceEvent* e, uint64_t value)
228{
229 std::vector<GPUReconstructionCPUDeviceEvent::waitingQueue> resume;
230 {
231 std::lock_guard lock(e->mutex);
232 e->nCompleted = std::max(e->nCompleted, value);
233 for (auto it = e->waiting.begin(); it != e->waiting.end();) {
234 if (it->value <= e->nCompleted) {
235 resume.emplace_back(*it);
236 it = e->waiting.erase(it);
237 } else {
238 it++;
239 }
240 }
241 e->cv.notify_all();
242 }
243 for (auto& w : resume) {
244 w.internals->startQueue(w.stream);
245 }
246}

248void GPUReconstructionCPUDevice::EnqueueTask(int32_t stream, deviceEvent* evList, int32_t nEvents, deviceEvent* ev, std::function<void()>&& work)
249{
250 if (stream < 0 || stream >= mNStreams) {
251 throw std::runtime_error("CPUDevice: invalid stream " + std::to_string(stream));
252 }
253 GPUReconstructionCPUDeviceTask task;
254 if (evList) { // Capture the event states at enqueue time, such that the queue waits for the markers recorded so far, like cudaStreamWaitEvent
255 for (int32_t k = 0; k < nEvents; k++) {
256 auto* e = getEvent(evList[k]);
257 std::lock_guard lock(e->mutex);
258 task.waits.emplace_back(e, e->nRecorded);
259 }
260 }
261 if (ev) {
262 task.record = getEvent(*ev);
263 std::lock_guard lock(task.record->mutex);
264 task.recordValue = ++task.record->nRecorded;
265 }
266 task.work = std::move(work);
267 mInternals->enqueue(stream, std::move(task));
268}

270void GPUReconstructionCPUDevice::SynchronizeStream(int32_t stream)
271{
272 auto& q = mInternals->queues[stream];

274 q.cv.wait(lock, [&q]() { return !q.active; });

284void GPUReconstructionCPUDevice::SynchronizeEvents(deviceEvent* evList, int32_t nEvents)
285{
286 for (int32_t i = 0; i < nEvents; i++) {
287 auto* e = getEvent(evList[i]);
288 std::unique_lock lock(e->mutex);
289 const uint64_t value = e->nRecorded;
290 e->cv.wait(lock, [e, value]() { return e->nCompleted >= value; });

342size_t GPUReconstructionCPUDevice::GPUMemCpy(void* dst, const void* src, size_t size, int32_t stream, int32_t toGPU, deviceEvent* ev, deviceEvent* evList, int32_t nEvents)
343{
344 if (GetProcessingSettings().debugLevel >= 3) {
345 stream = -1;
346 }
347 if (stream == -1) {
348 SynchronizeGPU();
349 if (size) {
350 memcpy(dst, src, size);
351 }
352 if (ev) {
353 auto* e = getEvent(*ev);
354 uint64_t value;
355 {
356 std::lock_guard lock(e->mutex);
357 value = ++e->nRecorded;
358 }
359 mInternals->completeEvent(e, value);
360 }
361 } else {
362 EnqueueTask(stream, evList, evList ? nEvents : 0, ev, [dst, src, size]() {
363 if (size) {
364 memcpy(dst, src, size);
365 }
366 });
367 }
368 if (GetProcessingSettings().serializeGPU & 2) {
369 GPUDebug(("GPUMemCpy " + std::to_string(toGPU)).c_str(), stream, true);
370 }
371 return size;
372}

374size_t GPUReconstructionCPUDevice::WriteToConstantMemory(size_t offset, const void* src, size_t size, int32_t stream, deviceEvent* ev)
375{
376 char* dst = (char*)mDeviceConstantMem + offset;
377 if (stream == -1) {
378 SynchronizeGPU();
379 memcpy(dst, src, size);
380 } else {
381 // The source may be a temporary, the CUDA backend stages the copy too, so we take a copy for the asynchronous transfer
382 auto buffer = std::make_shared<std::vector<char>>((const char*)src, (const char*)src + size);
383 EnqueueTask(stream, nullptr, 0, ev, [dst, buffer]() { memcpy(dst, buffer->data(), buffer->size()); });
384 }
385 if (GetProcessingSettings().serializeGPU & 2) {
386 GPUDebug("WriteToConstantMemory", stream, true);
387 }
388 return size;
389}

391template <class T, int32_t I, typename... Args>
392inline void GPUReconstructionCPUDevice::runKernelBackend(const krnlSetupTime& _xyz, const Args&... args)
393{
394 auto& x = _xyz.x;
395 auto& y = _xyz.y;
396 auto& z = _xyz.z;
397 if (x.nThreads != 1) {
398 throw std::runtime_error("CPUDevice backend supports only kernels with nThreads = 1");
399 }
400 const uint32_t nBlocks = x.nBlocks;
401 const int32_t iSector = y.index;
402 GPUConstantMem* constantMem = mDeviceConstantMem;
403 using rtcKernel = void (*)(GPUConstantMem*, int32_t, uint32_t, uint32_t, uint32_t, Args...);
404 rtcKernel rtc = mInternals->rtcKernels.size() ? reinterpret_cast<rtcKernel>(mInternals->rtcKernels[GetKernelNum<T, I>()]) : nullptr;
405
406 // The kernel arguments are captured by value, since the kernel runs asynchronously in the command queue
407 EnqueueTask(x.stream, z.evList, z.evList ? z.nEvents : 0, z.ev, [nBlocks, iSector, constantMem, rtc, ... args = args]() {
408 tbb::this_task_arena::isolate([&] { // Do not pick up tasks of other command queues while waiting inside parallel_for
409 tbb::parallel_for(tbb::blocked_range<uint32_t>(0, nBlocks, 1), [&](const tbb::blocked_range<uint32_t>& r) {
410 if (rtc) {
411 rtc(constantMem, iSector, nBlocks, r.begin(), r.end(), args...);
412 } else {
413 for (uint32_t iBlock = r.begin(); iBlock < r.end(); iBlock++) {
414 typename T::GPUSharedMemory smem;
415 T::template Thread<I>(nBlocks, 1, iBlock, 0, smem, T::Processor(*constantMem)[iSector], args...);
416 }
417 }
418 });
419 });
420 });
421}
