41// "DF_<n>/<tree>", with or without a leading slash. Empty strings if it is not that.
42std::pair<std::string, std::string> splitTreePath(std::string const& path)
43{
44 size_t start = path.starts_with("/") ? 1 : 0;
45 auto slash = path.find('/', start);
46 if (slash == std::string::npos) {
47 return {};
48 }
49 return {path.substr(start, slash - start), path.substr(slash + 1)};
50}

63std::shared_ptr<ReadPlan> NativeAODFileSystem::planFor(std::string const& path)
64{
65 auto [df, treeName] = splitTreePath(path);
66 if (df.empty()) {
67 return nullptr;
68 }
69 std::lock_guard lock{mMutex};
70 if (mFile->source().canMap()) {
71 releaseMappedDF(df);
72 return nullptr;
73 }
74 if (df != mPlanDF) {
75 auto previous = std::move(mPlanTrees);
76 mPlanTrees.clear();
77 mPlanOf.clear();
78 mPlanBytes = 0;
79 mPlanDF = df;
80 // Also when the directory is already known, e.g. the first DF, read at open.
81 adoptLookahead(df);
82 directory(df);
83 if (mAdoptedDF == df) {
84 mPlan = std::move(mAdoptedPlan);
85 for (auto& name : mAdoptedTrees) {
86 mPlanOf[name] = mPlan;
87 if (auto* handle = findTree(df, name)) {
88 mPlanBytes += ReadPlan::bytesOf(handle->tree);
89 }
90 }
91 mPlanTrees = std::move(mAdoptedTrees);
92 } else {
93 mPlan = std::make_shared<ReadPlan>(mFile->sourcePtr(), mOptions, mPool);
94 }
95 mAdoptedDF.clear();
96 auto trees = previous;
97 if (std::ranges::find(trees, treeName) == trees.end()) {
98 trees.push_back(treeName);
99 }
100 if (!previous.empty()) {
101 mFile->setTreesToPrefetch(trees);
102 }
103 for (auto& name : previous) {
104 addToPlan(name);
105 }
106 // Only once a whole DF has shown which trees the reader asks for.
107 if (!previous.empty() && mFile->source().isRemote() && mOptions.prefetchDepth > 0) {
108 startLookahead(df, std::move(trees));
109 }
110 }
111 addToPlan(treeName);
112 // Let go of the plan once its tree is handed out: the plan, and its buffers,
113 // then go away when the reads of its trees are done.
114 auto it = mPlanOf.find(treeName);
115 if (it == mPlanOf.end()) {
116 return nullptr;
117 }
118 auto plan = std::move(it->second);
119 mPlanOf.erase(it);
120 return plan;
121}

123void NativeAODFileSystem::releaseMappedDF(std::string const& df)
124{
125 if (df == mMappedDF) {
126 return;
127 }
128 // The pages of a DF which is done stay mapped and count in the RSS of the
129 // device until the kernel reclaims them: give them back. Rereading them is
130 // still correct, as they come back from the page cache.
131 uint64_t begin = UINT64_MAX;
132 uint64_t end = 0;
133 for (auto it = mTrees.lower_bound(mMappedDF + "/"); it != mTrees.end() && it->first.starts_with(mMappedDF + "/"); ++it) {
134 for (auto const& branch : it->second->tree.branches) {
135 for (auto const& basket : branch.baskets) {
136 if (!basket.embedded && basket.nbytes > 0) {
137 begin = std::min(begin, basket.seek);
138 end = std::max(end, basket.seek + basket.nbytes);
139 }
140 }
141 }
142 }
143 if (!mMappedDF.empty() && begin < end) {
144 mFile->source().dontNeed(begin, end - begin);
145 }
146 mMappedDF = df;
147}

149void NativeAODFileSystem::adoptLookahead(std::string const& df)
150{
151 if (!mLookahead.valid() || mLookaheadDF != df) {
152 return;
153 }
154 auto lookahead = mLookahead.get();
155 if (!lookahead) {
156 return;
157 }
158 mDirectories.emplace(df, std::move(lookahead->keys));
159 mAdoptedDF = df;
160 mAdoptedPlan = lookahead->plan;
161 mAdoptedTrees.clear();
162 for (auto& [name, tree] : lookahead->trees) {
163 // The plan already has the baskets of these trees.
164 mTrees.emplace(df + "/" + name, std::make_unique<NativeTreeHandle>(NativeTreeHandle{std::move(tree), mFile->sourcePtr(), mOptions}));
165 mAdoptedTrees.push_back(name);
166 }
167}

193void NativeAODFileSystem::startLookahead(std::string const& afterDF, std::vector<std::string> trees)
194{
195 if (mDataFrames.empty()) {
196 mDataFrames = mFile->dataFrames();
197 }
198 if (!mLookaheadFile) {
199 mLookaheadFile = std::make_shared<RootFile>(*mFile);
200 }
201 auto it = std::ranges::find(mDataFrames, afterDF);
202 if (it == mDataFrames.end() || ++it == mDataFrames.end()) {
203 return;
204 }
205 mLookaheadDF = *it;
206 // Errors are left to the normal path, which reports them when the DF is read.
207 mLookahead = std::async(std::launch::async, [file = mLookaheadFile, df = *it, trees = std::move(trees), options = mOptions, pool = mPool]() -> std::optional<Lookahead> {
208 try {
209 Lookahead lookahead{df, {}, {}, std::make_shared<ReadPlan>(file->sourcePtr(), options, pool)};
210 file->setTreesToPrefetch(trees);
211 lookahead.keys = file->keysOf(df);
212 // Only the trees which fit in the budget; the others are read when asked for.
213 uint64_t bytes = 0;
214 for (auto const& name : trees) {
215 for (auto const& key : lookahead.keys) {
216 if (key.name == name && key.className == "TTree") {
217 auto tree = file->readTree(key);
218 auto treeBytes = ReadPlan::bytesOf(tree);
219 if (options.bufferBudget > 0 && bytes > 0 && bytes + treeBytes > options.bufferBudget) {
220 lookahead.plan->fetch();
221 return lookahead;
222 }
223 bytes += treeBytes;
224 lookahead.trees.emplace_back(name, std::move(tree));
225 lookahead.plan->add(lookahead.trees.back().second);
226 }
227 }
228 }
229 lookahead.plan->fetch();
230 return lookahead;
231 } catch (std::exception const&) {
232 return std::nullopt;
233 }
234 });
235}

237void NativeAODFileSystem::addToPlan(std::string const& treeName)
238{
239 if (std::ranges::find(mPlanTrees, treeName) != mPlanTrees.end()) {
240 return;
241 }
242 NativeTreeHandle* handle = nullptr;
243 try {
244 handle = findTree(mPlanDF, treeName);
245 } catch (std::exception const&) {
246 // Reported when the tree itself is asked for.
247 }
248 if (handle) {
249 auto bytes = ReadPlan::bytesOf(handle->tree);
250 if (mOptions.bufferBudget > 0 && mPlanBytes > 0 && mPlanBytes + bytes > mOptions.bufferBudget) {
251 mPlan = std::make_shared<ReadPlan>(mFile->sourcePtr(), mOptions, mPool);
252 mPlanBytes = 0;
253 }
254 mPlan->add(handle->tree);
255 mPlanBytes += bytes;
256 mPlanOf[treeName] = mPlan;
257 mPlanTrees.push_back(treeName);
258 }

261NativeTreeHandle* NativeAODFileSystem::findTree(std::string const& df, std::string const& treeName)
262{
263 auto path = df + "/" + treeName;
264 if (auto it = mTrees.find(path); it != mTrees.end()) {
265 return it->second.get();
266 }
267 if (!mDirectories.contains(df)) {
268 bool exists = false;
269 for (auto& k : mFile->topKeys()) {
270 exists |= k.name == df;
271 }
272 if (!exists) {
273 return nullptr;
274 }
275 }
276 for (auto& key : directory(df)) {
277 if (key.name == treeName && key.className == "TTree") {
278 auto handle = std::make_unique<NativeTreeHandle>(NativeTreeHandle{mFile->readTree(key), mFile->sourcePtr(), mOptions});
279 return mTrees.emplace(path, std::move(handle)).first->second.get();
280 }
281 }
282 return nullptr;
283}

285std::shared_ptr<RootObjectHandler> NativeAODFileSystem::GetObjectHandler(arrow::dataset::FileSource source)
286{
287 for (auto& capability : mObjectFactory.capabilities) {
288 if (capability.name != "native-ttree") {
289 continue;
290 }
291 auto objectPath = capability.lfn2objectPath(source.path());
292 void* handle = capability.getHandle(shared_from_this(), objectPath);
293 if (!handle) {
294 continue;
295 }
296 return std::make_shared<RootObjectHandler>(handle, capability.factory().format());
297 }
298 throw runtime_error_f("Unable to get handler for object %s", source.path().c_str());
299}

302{
303 for (auto& capability : mObjectFactory.capabilities) {
304 if (capability.name == "native-ttree" && capability.getHandle(shared_from_this(), capability.lfn2objectPath(source.path()))) {
305 return true;
306 }
307 }
308 return false;

311arrow::Result<arrow::fs::FileInfo> NativeAODFileSystem::GetFileInfo(const std::string& path)
312{
313 arrow::fs::FileInfo result;
314 result.set_path(path);
315 result.set_type(arrow::fs::FileType::NotFound);
316 for (auto& k : mFile->topKeys()) {
317 if (k.name == path) {
318 result.set_type(k.className == "TTree" ? arrow::fs::FileType::File : arrow::fs::FileType::Directory);
319 return result;
320 }
321 }
322 if (getTreeHandle(path)) {
323 result.set_type(arrow::fs::FileType::File);
324 }
325 return result;

333std::shared_ptr<NativeAODInputFile> NativeAODInputFile::open(std::unique_ptr<ByteSource> source, RootObjectReadingFactory& factory, Options const& options, std::string const& name)
334{
335 auto result = std::shared_ptr<NativeAODInputFile>(new NativeAODInputFile);
336 result->mName = name;
337 result->mFile = std::make_shared<RootFile>(std::move(source));
338 result->mFilesystem = std::make_shared<NativeAODFileSystem>(result->mFile, options, factory);
339 result->mFilesystem->setName(result->mName);
340 result->mParentFiles = result->mFile->readStringMap("parentFiles");
341 result->mMetaData = result->mFile->readStringMap("metaData");
342 // Parse the trees of the first DF now: a layout or version we do not support
343 // is then found while the file can still be read with ROOT instead.
344 auto dfs = result->mFile->dataFrames();
345 if (!dfs.empty()) {
346 for (auto& key : result->mFile->keysOf(dfs.front())) {
347 if (key.className == "TTree") {
348 result->mFilesystem->getTreeHandle(dfs.front() + "/" + key.name);
349 }
350 }
351 }
352 return result;
353}
