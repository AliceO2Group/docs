103 void initDigitizerTask(framework::InitContext& ic) override
104 {
105 LOG(info) << "Initializing TPC digitization";
106
107 mLaneId = ic.services().get<const o2::framework::DeviceSpec>().rank;
108
109 mWithMCTruth = o2::conf::DigiParams::Instance().mctruth;
110 auto triggeredMode = ic.options().get<bool>("TPCtriggered");
111 mConditions.init(ic, mDistortionType);
112
113 mDigitizer.setContinuousReadout(!triggeredMode);
114 mDigitizer.setDistortionScaleType(mDistortionType);
115
116 if (ic.options().get<bool>("tpc-sector-seeding")) {
117 const auto seed = o2::conf::DigiParams::Instance().seed;
118 if (seed == 0) {
119 LOG(warning) << "TPC sector seeding with DigiParams.seed=0 is not reproducible";
120 }
121 // with seed 0, gRandom was seeded randomly in the base class
122 mDigitizer.setUseSectorSeeding(seed != 0 ? seed : gRandom->Integer(kMaxUInt));
123 LOG(info) << "TPC: using sector seeding";
124 }
125
126 // we send the GRP data once if the corresponding output channel is available
127 // and set the flag to false after
128 mWriteGRP = true;
129
130 // clean up (possibly) existing digit files
131 if (mInternalWriter) {
132 cleanDigitFile();
133 }
134 }

150 std::stringstream tmp;
151 tmp << "tpc_driftime_digits_lane" << mLaneId << ".root";

187 void run(framework::ProcessingContext& pc)
188 {
189 LOG(info) << "Processing TPC digitization";
190
191 mConditions.update(pc, mDigitizer, [this](float vdrift, float tdriftOffset) {
192 mDigitizer.setVDrift(vdrift);
193 mDigitizer.setTDriftOffset(tdriftOffset);
194 });
195
196 for (auto it = pc.inputs().begin(), end = pc.inputs().end(); it != end; ++it) {
197 for (auto const& inputref : it.parts()) {
198 if (inputref.spec->lifetime == o2::framework::Lifetime::Condition) { // process does not need conditions
199 continue;
200 }
201 process(pc, inputref);
202 if (mInternalWriter) {
203 mInternalROOTFlushTTree->SetEntries(mFlushCounter);
204 mInternalROOTFlushFile->Write("", TObject::kOverwrite);
205 mInternalROOTFlushFile->Close();
206 // delete mInternalROOTFlushTTree; --> automatically done by ->Close()
207 delete mInternalROOTFlushFile;
208 mInternalROOTFlushFile = nullptr;
209 }
210 // TODO: make generic reset method?
211 mFlushCounter = 0;
212 }
213 }

216 // process one sector
217 void process(framework::ProcessingContext& pc, framework::DataRef const& inputref)
218 {
219 // read collision context from input
220 auto context = pc.inputs().get<o2::steer::DigitizationContext*>(inputref);
221 context->initSimChains(o2::detectors::DetID::TPC, mSimChains);
222 auto& irecords = context->getEventRecords();
223 // A timeframe holds no collision at all whenever the interaction rate is low enough. Do not
224 // return here: the sector still has to produce its usual output, an empty one, so that the
225 // digit file has the same shape as an ordinary timeframe that happens to contain nothing.
226 // Returning also left the chunked writer without a file and a null tree to flush.
227 LOG(info) << "TPC: Processing " << irecords.size() << " collisions";
228 auto const* dh = DataRefUtils::getHeader<o2::header::DataHeader*>(inputref);
229
230 bool isContinuous = mDigitizer.isContinuousReadout();
231 // we publish the GRP data once if the output channel is there
232 if (mWriteGRP && pc.outputs().isAllowed({"TPC", "ROMode", 0})) {
233 auto roMode = isContinuous ? o2::parameters::GRPObject::CONTINUOUS : o2::parameters::GRPObject::PRESENT;
234 LOG(info) << "TPC: Sending ROMode= " << (mDigitizer.isContinuousReadout() ? "Continuous" : "Triggered")
235 << " to GRPUpdater from channel " << dh->subSpecification;
236 pc.outputs().snapshot(Output{"TPC", "ROMode", 0}, roMode);
237 }
238 mWriteGRP = false;
239
240 // extract which sector to treat
241 auto const* sectorHeader = DataRefUtils::getHeader<TPCSectorHeader*>(inputref);
242 if (sectorHeader == nullptr) {
243 LOG(error) << "TPC: Sector header missing, skipping processing";
244 return;
245 }
246 auto sector = sectorHeader->sector();
247 mSector = sector;
248 mListOfSectors.push_back(sector);
249 LOG(info) << "TPC: Processing sector " << sector;
250 // the active sectors need to be propagated
251 uint64_t activeSectors = 0;
252 activeSectors = sectorHeader->activeSectors;
253
254 // lambda that creates a DPL owned buffer to accumulate the digits (in shared memory)
255 auto makeDigitBuffer = [this, sector, &pc, activeSectors, &dh]() {
256 o2::tpc::TPCSectorHeader header{sector};
257 header.activeSectors = activeSectors;
258 if (mInternalWriter) {
259 using ContainerType = std::decay_t<decltype(pc.outputs().make<std::vector<o2::tpc::Digit>>(Output{"", "", 0}))>*;
260 return ContainerType(nullptr);
261 } else {
262 // default case
263 return &pc.outputs().make<std::vector<o2::tpc::Digit>>(Output{"TPC", "DIGITS", static_cast<SubSpecificationType>(dh->subSpecification), header});
264 }
265 };
266 // lambda that snapshots the common mode vector to be sent out; prepares and attaches header with sector information
267 auto snapshotCommonMode = [this, sector, &pc, activeSectors, &dh](std::vector<o2::tpc::CommonMode> const& commonMode) {
268 o2::tpc::TPCSectorHeader header{sector};
269 header.activeSectors = activeSectors;
270 if (!mInternalWriter) {
271 // note that snapshoting only works with non-const references (to be fixed?)
272 pc.outputs().snapshot(Output{"TPC", "COMMONMODE", static_cast<SubSpecificationType>(dh->subSpecification), header},
273 const_cast<std::vector<o2::tpc::CommonMode>&>(commonMode));
274 }
275 };
276 // lambda that snapshots labels to be sent out; prepares and attaches header with sector information
277 auto snapshotLabels = [this, &sector, &pc, activeSectors, &dh](o2::dataformats::MCTruthContainer<o2::MCCompLabel> const& labels) {
278 o2::tpc::TPCSectorHeader header{sector};
279 header.activeSectors = activeSectors;
280 if (mWithMCTruth) {
281 if (!mInternalWriter) {
282 auto& sharedlabels = pc.outputs().make<o2::dataformats::ConstMCTruthContainer<o2::MCCompLabel>>(Output{"TPC", "DIGITSMCTR", static_cast<SubSpecificationType>(dh->subSpecification), header});
283 labels.flatten_to(sharedlabels);
284 }
285 }
286 };
287 // lambda that snapshots digits grouping (triggers) to be sent out; prepares and attaches header with sector information
288 auto snapshotEvents = [this, sector, &pc, activeSectors, &dh](const std::vector<DigiGroupRef>& events) {
289 o2::tpc::TPCSectorHeader header{sector};
290 header.activeSectors = activeSectors;
291 if (!mInternalWriter) {
292 LOG(info) << "TPC: Send TRIGGERS for sector " << sector << " channel " << dh->subSpecification << " | size " << events.size();
293 pc.outputs().snapshot(Output{"TPC", "DIGTRIGGERS", static_cast<SubSpecificationType>(dh->subSpecification), header},
294 const_cast<std::vector<DigiGroupRef>&>(events));
295 }
296 };
297
298 auto digitsAccum = makeDigitBuffer(); // accumulator for digits
299 o2::dataformats::MCTruthContainer<o2::MCCompLabel> labelAccum; // timeframe accumulator for labels
300 std::vector<CommonMode> commonModeAccum;
301 std::vector<DigiGroupRef> eventAccum;
302
303 // this should not happen any more, legacy condition when the sector variable was used
304 // to transport control information
305 if (sector < 0) {
306 throw std::runtime_error("Legacy control information is not expected any more");
307 }
308
309 // the TPCSectorHeader now allows to transport information for more than one sector,
310 // e.g. for transporting clusters in one single data block. The digitization is however
311 // only on sector level
312 if (sector >= TPCSectorHeader::NSectors) {
313 throw std::runtime_error("Digitizer can only work on single sectors");
314 }
315
316 TStopwatch timer;
317 timer.Start();
318
319 auto onFlush = [this, digitsAccum, &labelAccum, &commonModeAccum]() {
320 mFlushCounter++;
321 if (mInternalWriter) {
322 // the natural place to write out this independent datachunk immediately ...
323 writeToROOTFile();
324 } else {
325 // ... or to accumulate and later forward to next DPL proc
326 std::copy(mFlushBuffers.digits.begin(), mFlushBuffers.digits.end(), std::back_inserter(*digitsAccum));
327 if (mWithMCTruth) {
328 labelAccum.mergeAtBack(mFlushBuffers.labels);
329 }
330 std::copy(mFlushBuffers.commonMode.begin(), mFlushBuffers.commonMode.end(), std::back_inserter(commonModeAccum));
331 }
332 };
333 eventAccum = digitizeSector(mDigitizer, *context, mSimChains, sector, dh->tfCounter, mFlushBuffers, onFlush);
334
335 if (!mInternalWriter) {
336 // send out to next stage
337 snapshotEvents(eventAccum);
338 // snapshotDigits(digitsAccum); --> done automatically
339 snapshotCommonMode(commonModeAccum);
340 snapshotLabels(labelAccum);
341 }
342
343 timer.Stop();
344 LOG(info) << "TPC: Digitization took " << timer.CpuTime() << "s";
345 }

347 private:
348 o2::tpc::Digitizer mDigitizer;
349 TPCDigitizerConditions mConditions;
350 std::vector<TChain*> mSimChains;
351 SectorFlushBuffers mFlushBuffers; // output of the latest flush
352 std::vector<int> mListOfSectors; // a list of sectors treated by this task
353 TFile* mInternalROOTFlushFile = nullptr;
354 TTree* mInternalROOTFlushTTree = nullptr;
355 size_t mFlushCounter = 0;
356 int mLaneId = 0; // the id of the current process within the parallel pipeline
357 int mSector = 0;
358 bool mWriteGRP = false;
359 bool mWithMCTruth = true;
360 bool mInternalWriter = false;
361 int mDistortionType = 0;
362};

364o2::framework::DataProcessorSpec getTPCDigitizerSpec(int channel, bool writeGRP, bool mctruth, bool internalwriter, int distortionType)
365{
366 // create the full data processor spec using
367 // a name identifier
368 // input description
369 // algorithmic description (here a lambda getting called once to setup the actual processing function)
370 // options that can be used for this processor (here: input file names where to take the hits)
371 std::stringstream id;
372 id << "TPCDigitizer" << channel;
373
374 std::vector<OutputSpec> outputs; // define channel by triple of (origin, type id of data to be sent on this channel, subspecification)
375
376 if (!internalwriter) {
377 outputs.emplace_back("TPC", "DIGITS", static_cast<SubSpecificationType>(channel), Lifetime::Timeframe);
378 outputs.emplace_back("TPC", "DIGTRIGGERS", static_cast<SubSpecificationType>(channel), Lifetime::Timeframe);
379 if (mctruth) {
380 outputs.emplace_back("TPC", "DIGITSMCTR", static_cast<SubSpecificationType>(channel), Lifetime::Timeframe);
381 }
382 outputs.emplace_back("TPC", "COMMONMODE", static_cast<SubSpecificationType>(channel), Lifetime::Timeframe);
383 }
384 if (writeGRP) {
385 outputs.emplace_back("TPC", "ROMode", 0, Lifetime::Timeframe);
386 LOG(debug) << "TPC: Channel " << channel << " will supply ROMode";
387 }
388
389 std::vector<InputSpec> inputs{InputSpec{"collisioncontext", "SIM", "COLLISIONCONTEXT", static_cast<SubSpecificationType>(channel), Lifetime::Timeframe}};
390 auto options = TPCDigitizerConditions::options();
391 options.push_back({"tpc-sector-seeding", VariantType::Bool, false, {"Seed the random numbers of each sector from DigiParams.seed and the sector, so that the digits do not depend on the lane assignment"}});
392 return DataProcessorSpec{
393 id.str().c_str(),
394 inputs,
395 outputs,
396 AlgorithmSpec{adaptFromTask<TPCDPLDigitizerTask>(internalwriter, distortionType)},
397 options};
398}

400o2::framework::WorkflowSpec getTPCDigitizerSpec(int nLanes, std::vector<int> const& sectors, bool mctruth, bool internalwriter, int distortionType)
401{
402 // channel parameter is deprecated in the TPCDigitizer processor, all descendants
403 // are initialized not to publish GRP mode, but the channel will be added to the first
404 // processor after the pipelines have been created. The processor will decide upon
405 // the index in the ParallelContext whether to publish
406 WorkflowSpec pipelineTemplate{getTPCDigitizerSpec(0, false, mctruth, internalwriter, distortionType)};
407 // override the predefined name, index will be added by parallelPipeline method
408 pipelineTemplate[0].name = "TPCDigitizer";
409 WorkflowSpec pipelines = parallelPipeline(
410 pipelineTemplate, nLanes, [size = sectors.size()]() { return size; }, [&sectors](size_t index) { return sectors[index]; });
411 // add the channel for the GRP information to the first processor
412 for (auto& spec : pipelines) {
413 TPCDigitizerConditions::addInputs(spec.inputs, distortionType); // the same CCDB request for each pipeline
414 }
415 pipelines[0].outputs.emplace_back("TPC", "ROMode", 0, Lifetime::Timeframe);
416 return pipelines;
417}
