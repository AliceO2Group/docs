53struct GPUReconstructionMetalInternals {
54 id<MTLDevice> device;
55
56 std::array<id<MTLCommandQueue>, constants::GPU_MAX_STREAMS> commandQueues;
57 // The buffer currently open for encoding, plus the ones already committed and
58 // not yet waited for. A Metal command buffer takes one commit and cannot be
59 // encoded into afterwards, so the two cannot be the same object.
60 std::array<id<MTLCommandBuffer>, constants::GPU_MAX_STREAMS> commandBuffers;
61 std::array<std::vector<id<MTLCommandBuffer>>, constants::GPU_MAX_STREAMS> submittedBuffers;
62 // Signalled by each committed buffer and waited for by the next one, see CommitStream
63 std::array<id<MTLSharedEvent>, constants::GPU_MAX_STREAMS> streamOrder;
64 std::array<uint64_t, constants::GPU_MAX_STREAMS> streamOrderValue;
65 // Orders the encoders within the open command buffer of a stream, which
66 // collects the stream's work until something has to wait for it.
67 std::array<id<MTLFence>, constants::GPU_MAX_STREAMS> streamFence;
68 std::array<bool, constants::GPU_MAX_STREAMS> streamHasWork;
69
70 std::vector<id<MTLFunction>> functions;
71 std::vector<id<MTLComputePipelineState>> pipelines;
72 std::vector<uint32_t> maxThreadsPerGroup;
73
74 id<MTLBuffer> mem_gpu;
75 id<MTLBuffer> mem_constant;
76 id<MTLBuffer> mem_host;
77
78 id<MTLLibrary> library;
79
80 // The value each event was last recorded with. An MTLSharedEvent only counts
81 // up, so recording signals the next value and waiting asks for the last one,
82 // which makes waiting for a never recorded event a no-op, as in CUDA.
83 std::unordered_map<void*, uint64_t> eventValues;
84 std::mutex eventMutex;
85};
86// An encoder of a stream starts after the previous one of the same stream and
87// leaves its work in the open command buffer.
88template <class E>
89inline void metalBeginEncoder(GPUReconstructionMetalInternals* internals, E encoder, int32_t stream)
90{
91 [encoder waitForFence:internals->streamFence[stream]];
92}
93template <class E>
94inline void metalEndEncoder(GPUReconstructionMetalInternals* internals, E encoder, int32_t stream)
95{
96 [encoder updateFence:internals->streamFence[stream]];
97 [encoder endEncoding];
98 internals->streamHasWork[stream] = true;
99}
