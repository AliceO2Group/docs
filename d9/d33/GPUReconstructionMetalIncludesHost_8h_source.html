114// An encoder of a stream starts after the previous one of the same stream and
115// leaves its work in the open command buffer.
116template <class E>
117inline void metalBeginEncoder(GPUReconstructionMetalInternals* internals, E encoder, int32_t stream)
118{
119 [encoder waitForFence:internals->streamFence[stream]];
120}
121// A failed command buffer does not signal its events, and the work waiting for
122// them would wait forever: signal them from the host, so that everything
123// completes and the failure is reported when the host synchronises.
124inline void metalSignalOnError(id<MTLCommandBuffer> cb, id<MTLSharedEvent> event, uint64_t value)
125{
126 [cb addCompletedHandler:^(id<MTLCommandBuffer> c) {
127 if (c.status == MTLCommandBufferStatusError && event.signaledValue < value) {
128 event.signaledValue = value;
129 }
130 }];
131}
132// Before a buffer of an idle stream is committed, wait until fewer than the
133// limit of streams have work in flight. This cannot deadlock: a buffer only waits
134// for events that buffers committed before it signal (RecordMarker commits the
135// signal at once), so the work in flight completes without the host's help.
136inline void metalLimitStreamsInFlight(GPUReconstructionMetalInternals* internals, id<MTLCommandBuffer> cb, int32_t stream)
137{
138 std::unique_lock<std::mutex> lock(internals->inFlightMutex);
139 if (internals->streamBuffersInFlight[stream]++ == 0) {
140 internals->inFlightDone.wait(lock, [internals] { return internals->streamsInFlight < internals->maxStreamsInFlight; });
141 internals->streamsInFlight++;
142 }
143 [cb addCompletedHandler:^(id<MTLCommandBuffer>) {
144 std::lock_guard<std::mutex> l(internals->inFlightMutex);
145 if (--internals->streamBuffersInFlight[stream] == 0) {
146 internals->streamsInFlight--;
147 internals->inFlightDone.notify_all();
148 }
149 }];
150}
151// In the time base of the command buffers' GPUStartTime and GPUEndTime
152inline double metalHostTime()
153{
154 static const double scale = [] { mach_timebase_info_data_t tb; mach_timebase_info(&tb); return 1e-9 * tb.numer / tb.denom; }();
155 return mach_absolute_time() * scale;
156}

165// Runs after each kernel in its encoder, so a kernel that ran while the pool
166// was not mapped for the GPU is counted and the run fails instead of going on
167// with the results it did not produce.
168inline void metalEncodeCanary(GPUReconstructionMetalInternals* internals, id<MTLComputeCommandEncoder> encoder, uint64_t word)
169{
170 [encoder setComputePipelineState:internals->canaryPipeline];
171 [encoder setBuffer:internals->canaryBad offset:0 atIndex:0];
172 [encoder setBytes:&word length:sizeof(word) atIndex:1];
173 [encoder dispatchThreadgroups:MTLSizeMake(1, 1, 1) threadsPerThreadgroup:MTLSizeMake(1, 1, 1)];
174}
175template <class E>
176inline void metalEndEncoder(GPUReconstructionMetalInternals* internals, E encoder, int32_t stream)
177{
178 [encoder updateFence:internals->streamFence[stream]];
179 [encoder endEncoding];
180 internals->streamHasWork[stream] = true;
181}
