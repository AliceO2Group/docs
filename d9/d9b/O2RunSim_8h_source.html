41class O2RunSim : public FairRunSim
42{
43 public:
44 O2RunSim(bool devicemode, bool evalmat) : FairRunSim(), mDeviceMode(devicemode), mEvalMat(evalmat) {}
45 ~O2RunSim() override = default;
46
47 void Init() final
48 {
49 LOG(info) << "O2RUNSIM SPECIFIC INIT CALLED";
50
51 fRootManager->InitSink();
52
53 // the geometry manager that FairGeoLoader used to create here (its ASCII media part is not used by ALICE)
54 new TGeoManager("FAIRGeom", "FAIR geometry");
55
56 if (mDeviceMode) {
57 fApp = new O2MCApplication("Fair", "The Fair VMC App", ListOfModules, MatFname);
58 } else {
59 if (!mEvalMat) {
60 fApp = new O2MCApplicationBase("Fair", "The Fair VMC App", ListOfModules, MatFname);
61 } else {
62 fApp = new O2MCApplicationEvalMat("Fair", "The Fair VMC App", ListOfModules, MatFname);
63 }
64 }
65
66 fApp->SetGenerator(fGen);
67
68 FairRunIdGenerator genid;
69 fRunId = genid.generateId();
70
71 fFileHeader->SetRunId(fRunId);
72
73 // Set global Parameter Info
74 if (fPythiaDecayer) {
75 fApp->SetPythiaDecayer(fPythiaDecayer);
76 if (fPythiaDecayerConfig) {
77 fApp->SetPythiaDecayerConfig(fPythiaDecayerConfig);
78 }
79 }
80 if (fUserDecay) {
81 fApp->SetUserDecay(fUserDecay);
82 if (fUserDecayConfig) {
83 fApp->SetUserDecayConfig(fUserDecayConfig);
84 }
85 }
86
87 if (fField) {
88 fField->Init();
89 }
90 fApp->SetField(fField);
91
92 fSimSetup();
93 fApp->InitMC("foo", "bar");
94 fRootManager->WriteFileHeader(fFileHeader);
95 }
96
97 void Run(int n = 0, int b = 0) final
98 {
99 FairRunSim::Run(n, b);
100 }
101
102 private:
103 bool mDeviceMode{false};
104 bool mEvalMat{false};
105
106 ClassDefOverride(O2RunSim, 0);
107};
