66 [[gnu::always_inline]] static bool directionLeaves(vecgeom::NavigationState const& exited, const V3& point,
67 const V3& dir)
68 {
69 if (exited.IsOutside() || exited.Top() == nullptr) {
70 return false;
71 }
72 vecgeom::Transformation3D m;
73 exited.TopMatrix(m);
74 V3 n;
75 exited.Top()->GetUnplacedVolume()->Normal(m.Transform(point), n);
76 if (!(n.Mag2() > 0.5)) {
77 return false; // no valid exit normal: Geant4 does not block either
78 }
79 return directionLeaves(m.InverseTransformDirection(n), dir);
80 }
