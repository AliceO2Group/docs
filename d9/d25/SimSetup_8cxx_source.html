26void execFromPlugin(const char* libname, const char* funcname)
27{
28 const std::string filename = std::string(libname) + DLLOADER_MAC_LINUX(".dylib", ".so");
29 auto libHandle = dlopen(filename.c_str(), RTLD_NOW);
30 if (!libHandle) {
31 const char* error = dlerror();
32 LOG(fatal) << "Cannot load " << filename << ": " << (error ? error : "unknown error");
33 // LOG(fatal) ends the process only when a fatal handler is installed
34 throw fair::FatalException("Cannot load " + filename);
35 }
36 dlerror(); // clear any pending error before dlsym
37 auto func = (setup_fnc)dlsym(libHandle, funcname);
38 if (!func) {
39 const char* error = dlerror();
40 LOG(fatal) << "Cannot find " << funcname << " in " << filename << ": " << (error ? error : "the symbol is null");
41 throw fair::FatalException("Cannot find " + std::string(funcname) + " in " + filename);
42 }
43 func();
44}

46void execSetupFromPlugin(const char* libname, const char* funcname)

52void SimSetup::setup(const char* engine)
53{
54 if (strcmp(engine, "TGeant3") == 0) {
55 execSetupFromPlugin("libO2G3Setup", "_ZN2o28g3config8G3ConfigEv");
56 } else if (strcmp(engine, "TGeant4") == 0) {
57 execSetupFromPlugin("libO2G4Setup", "_ZN2o28g4config8G4ConfigEv");
58 } else if (strcmp(engine, "TFluka") == 0) {
59 execSetupFromPlugin("libO2FLUKASetup", "_ZN2o211flukaconfig11FlukaConfigEv");
60 } else if (strcmp(engine, "MCReplay") == 0) {
61 execSetupFromPlugin("libO2MCReplaySetup", "_ZN2o214mcreplayconfig14MCReplayConfigEv");
62 } else if (strcmp(engine, "O2TrivialMCEngine") == 0) {
63 execSetupFromPlugin("libO2O2TrivialMCEngineSetup", "_ZN2o223o2trivialmcengineconfig23O2TrivialMCEngineConfigEv");
64 } else {
65 LOG(fatal) << "Unsupported engine " << engine;
66 }
67 o2::SetCuts();
68}
