54// a primary generator particle moving along z
55TParticle makeParticle(int pdg, double p, bool toBeDone, double vx = 0, double vy = 0, double vz = 0)
56{
57 bool ok = false;
58 double m = o2::O2DatabasePDG::Mass(pdg, ok);
59 double e = std::sqrt(p * p + m * m);
60 TParticle part(pdg, 1, -1, -1, -1, -1, 0., 0., p, e, vx, vy, vz, 0.);
61 o2::mcutils::MCGenHelper::encodeParticleStatusAndTracking(part, 1, 1, toBeDone);
62 part.SetUniqueID(kPPrimary);
63 return part;
64}

71bool toBeDone(TParticle const& p)

78{

82 o2::conf::ConfigurableParam::setValue("EventDecayer.config[2]", extraCfg);
83 o2::conf::ConfigurableParam::setValue("EventDecayer.g4ParticleTable", g4table);
84 o2::conf::ConfigurableParam::setValue("EventDecayer.seed", std::to_string(seed));

86 o2::conf::ConfigurableParam::setValue("EventDecayer.skipPdg", DEFAULTSKIP);

89// the output of decaying one Xic+ at (0.1,0.2,0.3) cm with p = 5 GeV
90std::vector<TParticle> decayOneXic(EventDecayerReport& report)
91{
92 setParams();
93 EventDecayer decayer;
94 BOOST_REQUIRE(decayer.init());
95 auto xic = makeParticle(4232, 5., true, 0.1, 0.2, 0.3);
96 // a distinctive generator status and a non-zero start time
97 o2::mcutils::MCGenHelper::encodeParticleStatusAndTracking(xic, 1, 85, true);
98 xic.SetProductionVertex(0.1, 0.2, 0.3, 1e-9);
99 std::vector<TParticle> particles{xic};
100 report = decayer.decay(particles);
101 return particles;
102}

104bool sameParticle(TParticle const& a, TParticle const& b)
105{
106 return a.GetPdgCode() == b.GetPdgCode() && a.Px() == b.Px() && a.Py() == b.Py() && a.Pz() == b.Pz() &&
107 a.Energy() == b.Energy() && a.Vx() == b.Vx() && a.Vy() == b.Vy() && a.Vz() == b.Vz() && a.T() == b.T() &&
108 a.GetStatusCode() == b.GetStatusCode() && a.TestBits(0xffffffff) == b.TestBits(0xffffffff) &&
109 a.GetFirstMother() == b.GetFirstMother() && a.GetFirstDaughter() == b.GetFirstDaughter() &&
110 a.GetLastDaughter() == b.GetLastDaughter() && a.GetUniqueID() == b.GetUniqueID();
111}

113// checks that a decay call left the particles as they were
114void checkUnchanged(std::vector<TParticle> const& particles, std::vector<TParticle> const& before)
115{
116 BOOST_REQUIRE_EQUAL(particles.size(), before.size());
117 for (size_t i = 0; i < before.size(); ++i) {
118 BOOST_CHECK(sameParticle(particles[i], before[i]));
119 }
120}

128// 20000 Xic+ at p = 3 GeV decayed by one decayer, shared by the statistics cases
129std::vector<TParticle> const& manyXic()
130{
131 static std::vector<TParticle> particles;
132 if (particles.empty()) {
133 setParams();
134 EventDecayer decayer;
135 BOOST_REQUIRE(decayer.init());
136 for (int i = 0; i < 20000; ++i) {
137 particles.push_back(makeParticle(4232, 3., true));
138 }
139 auto report = decayer.decay(particles);
140 BOOST_REQUIRE(report.decayed[4232] == 20000);
141 }
142 return particles;
143}

147BOOST_AUTO_TEST_CASE(xic_to_xistar_chain)
148{
149 EventDecayerReport report;
150 auto particles = decayOneXic(report);
151
152 auto const& xic = particles[0];
153 BOOST_CHECK(!toBeDone(xic));
154 BOOST_CHECK_EQUAL(hepmc(xic), 2);
155 BOOST_CHECK_EQUAL(o2::mcgenstatus::getGenStatusCode(xic.GetStatusCode()), 85);
156
157 int nXiStar = 0;
158 int iXiStar = -1;
159 int iPi = -1;
160 for (int i = 1; i < (int)particles.size(); ++i) {
161 if (particles[i].GetPdgCode() == 3324) {
162 ++nXiStar;
163 iXiStar = i;
164 }
165 if (particles[i].GetPdgCode() == 211 && particles[i].GetFirstMother() == 0) {
166 iPi = i;
167 }
168 }
169 BOOST_REQUIRE_EQUAL(nXiStar, 1);
170 BOOST_REQUIRE(iPi > 0);
171
172 // the daughter range of every decayed particle holds exactly its direct products
173 for (int m = 0; m < (int)particles.size(); ++m) {
174 auto const& mother = particles[m];
175 int nChildren = 0;
176 for (auto const& p : particles) {
177 nChildren += p.GetFirstMother() == m;
178 }
179 if (hepmc(mother) != 2) {
180 BOOST_CHECK_EQUAL(nChildren, 0);
181 continue;
182 }
183 BOOST_CHECK_EQUAL(mother.GetLastDaughter() - mother.GetFirstDaughter() + 1, nChildren);
184 for (int d = mother.GetFirstDaughter(); d <= mother.GetLastDaughter(); ++d) {
185 BOOST_CHECK_EQUAL(particles[d].GetFirstMother(), m);
186 }
187 }
188 BOOST_CHECK_EQUAL(xic.GetLastDaughter() - xic.GetFirstDaughter() + 1, 2);
189
190 // products carry the Pythia decay status 91 and are flagged as primaries
191 for (int i = 1; i < (int)particles.size(); ++i) {
192 BOOST_CHECK_EQUAL(o2::mcgenstatus::getGenStatusCode(particles[i].GetStatusCode()), 91);
193 BOOST_CHECK_EQUAL(particles[i].GetUniqueID(), kPPrimary);
194 }
195 auto const& xiStar = particles[iXiStar];
196 BOOST_CHECK_EQUAL(xiStar.GetFirstMother(), 0);
197 BOOST_CHECK(!toBeDone(xiStar));
198 BOOST_CHECK_EQUAL(hepmc(xiStar), 2);
199
200 std::vector<int> dPdg;
201 for (int d = xiStar.GetFirstDaughter(); d >= 0 && d <= xiStar.GetLastDaughter(); ++d) {
202 dPdg.push_back(particles[d].GetPdgCode());
203 BOOST_CHECK(toBeDone(particles[d]));
204 BOOST_CHECK_EQUAL(hepmc(particles[d]), 1);
205 BOOST_CHECK_EQUAL(particles[d].GetFirstMother(), iXiStar);
206 }
207 BOOST_REQUIRE_EQUAL(dPdg.size(), 2);
208 bool charged = (dPdg[0] == 3312 && dPdg[1] == 211) || (dPdg[0] == 211 && dPdg[1] == 3312);
209 bool neutral = (dPdg[0] == 3322 && dPdg[1] == 111) || (dPdg[0] == 111 && dPdg[1] == 3322);
210 BOOST_CHECK(charged || neutral);
211
212 int first = xic.GetFirstDaughter();
213 int last = xic.GetLastDaughter();
214 BOOST_CHECK(first <= iXiStar && iXiStar <= last);
215 BOOST_CHECK(first <= iPi && iPi <= last);
216
217 // all products start at the Xic decay point
218 double dx = xiStar.Vx() - 0.1;
219 double dy = xiStar.Vy() - 0.2;
220 double dz = xiStar.Vz() - 0.3;
221 BOOST_CHECK(std::sqrt(dx * dx + dy * dy + dz * dz) > 0.);
222 for (int i = 1; i < (int)particles.size(); ++i) {
223 BOOST_CHECK_CLOSE(particles[i].Vx(), xiStar.Vx(), 1e-6);
224 BOOST_CHECK_CLOSE(particles[i].Vy(), xiStar.Vy(), 1e-6);
225 BOOST_CHECK_CLOSE(particles[i].Vz(), xiStar.Vz(), 1e-6);
226 }
227
228 // products start at the Xic time plus its flight time, |d| / (beta c) with c in cm/ns and T in s
229 double flightTime = std::sqrt(dx * dx + dy * dy + dz * dz) / (xic.P() / xic.Energy() * 29.9792458) * 1e-9;
230 for (int i = 1; i < (int)particles.size(); ++i) {
231 BOOST_CHECK_CLOSE(particles[i].T(), xic.T() + flightTime, 1e-6);
232 }
233 BOOST_CHECK_EQUAL(report.decayed[4232], 1);
234}

251BOOST_AUTO_TEST_CASE(proper_decay_length)
252{
253 auto const& particles = manyXic();
254 double sum = 0.;
255 int n = 0;
256 for (int i = 0; i < 20000; ++i) {
257 auto const& mother = particles[i];
258 auto const& daughter = particles[mother.GetFirstDaughter()];
259 double dx = daughter.Vx() - mother.Vx();
260 double dy = daughter.Vy() - mother.Vy();
261 double dz = daughter.Vz() - mother.Vz();
262 sum += std::sqrt(dx * dx + dy * dy + dz * dz) * mother.GetCalcMass() / mother.P();
263 ++n;
264 }
265 BOOST_CHECK_CLOSE(sum / n, 0.01358, 2.);
266}

439BOOST_AUTO_TEST_CASE(units_mm_and_mmc)
440{
441 // the same decay in cm and s, and in mm and mm/c (the Generator default units)
442 const double mmc = 3.3356410e-12;
443 std::vector<TParticle> inCm;
444 std::vector<TParticle> inMm;
445 for (bool mm : {false, true}) {
446 setParams("", "", 4711);
447 EventDecayer decayer;
448 BOOST_REQUIRE(decayer.init());
449 auto xic = makeParticle(4232, 5., true);
450 if (mm) {
451 decayer.setUnits(0.1, mmc);
452 xic.SetProductionVertex(1., 2., 3., 1e-9 / mmc);
453 } else {
454 xic.SetProductionVertex(0.1, 0.2, 0.3, 1e-9);
455 }
456 auto& particles = mm ? inMm : inCm;
457 particles.push_back(xic);
458 decayer.decay(particles);
459 }
460 setParams();
461 BOOST_REQUIRE(inCm.size() > 1);
462 BOOST_REQUIRE_EQUAL(inCm.size(), inMm.size());
463 for (size_t i = 1; i < inCm.size(); ++i) {
464 BOOST_CHECK_CLOSE(inMm[i].Vz(), 10. * inCm[i].Vz(), 1e-6);
465 BOOST_CHECK_CLOSE(inMm[i].T(), inCm[i].T() / mmc, 1e-6);
466 }
467 // in mm and mm/c the flight time is the decay length over beta
468 auto const& xic = inMm[0];
469 auto const& d = inMm[xic.GetFirstDaughter()];
470 double length = std::sqrt(std::pow(d.Vx() - 1., 2) + std::pow(d.Vy() - 2., 2) + std::pow(d.Vz() - 3., 2));
471 BOOST_CHECK(length > 0.);
472 BOOST_CHECK_CLOSE(d.T() - xic.T(), length / (xic.P() / xic.Energy()), 1e-4);
473}

475BOOST_AUTO_TEST_CASE(seeding)
476{
477 // decays one Xic with the given seed and returns the seed in use and the decay position
478 auto run = [](int seed) {
479 setParams("", "", seed);
480 EventDecayer decayer;
481 BOOST_REQUIRE(decayer.init());
482 std::vector<TParticle> particles{makeParticle(4232, 5., true)};
483 decayer.decay(particles);
484 BOOST_REQUIRE(particles.size() > 1);
485 return std::make_pair(decayer.seed(), particles[1].Vz());
486 };
487 // a seed above the Pythia8 maximum is taken modulo 900000001
488 auto large = run(1000000123);
489 auto reduced = run(100000122);
490 BOOST_CHECK_EQUAL(large.first, 100000122);
491 BOOST_CHECK_EQUAL(large.second, reduced.second);
492 // seed 0 derives the seed from gRandom, differently from GeneratorPythia8 which takes it unchanged
493 gRandom->SetSeed(777);
494 auto fromGRandom = run(0);
495 BOOST_CHECK(fromGRandom.first != 777);
496 BOOST_CHECK(fromGRandom.first >= 0 && fromGRandom.first <= 900000000);
497 auto explicitSeed = run(fromGRandom.first);
498 BOOST_CHECK_EQUAL(fromGRandom.second, explicitSeed.second);
499 gRandom->SetSeed(777);
500 BOOST_CHECK_EQUAL(run(0).first, fromGRandom.first);
501 gRandom->SetSeed(778);
502 BOOST_CHECK(run(0).first != fromGRandom.first);
503 BOOST_CHECK(run(1).second != run(2).second);
504 setParams();
505}

507// quarks, gluon, diquarks and the Pythia8 pseudo-particles 81-99
508bool isPartonic(int pdg)
509{
510 int a = std::abs(pdg);
511 bool diquark = a > 1100 && a < 5600 && a / 1000 < 6 && (a / 10) % 10 == 0 && (a / 100) % 10 <= a / 1000 && (a / 100) % 10 > 0 && (a % 10) % 2 == 1;
512 return (a >= 1 && a <= 8) || a == 21 || (a >= 81 && a <= 99) || diquark;
513}
