92void ObjectReader::checkVersion(char const* className, int version)
93{
94 auto& table = supportedClassVersions();
95 auto it = table.find(className);
96 if (it == table.end() || std::find(it->second.begin(), it->second.end(), version) == it->second.end()) {
97 throw UnsupportedVersion(className, version, mC.file());
98 }
99}

101ObjectReader::Versioned ObjectReader::header(char const* className)
102{
103 auto start = mC.pos();
104 auto bcnt = mC.u32();
105 if (!(bcnt & kByteCountMask)) {
106 throw UnsupportedLayout(std::string("object of class ") + className + " without byte count");
107 }
108 size_t end = start + 4 + (bcnt & ~kByteCountMask);
109 if (end > mC.buffer().size()) {
110 mC.fail(std::string("byte count of ") + className + " past the end of the buffer");
111 }
112 auto version = mC.u16();
113 if (version & kStreamedMemberWise) {
114 throw UnsupportedLayout(std::string("memberwise streaming of ") + className);
115 }
116 checkVersion(className, version);
117 return {end, version};
118}

120void ObjectReader::tobject()
121{
122 auto version = mC.u16();
123 if (version & kStreamedMemberWise) {
124 // Written with a byte count: those were its high bytes. Skip the low ones.
125 mC.u16();
126 version = mC.u16();
127 }
128 checkVersion("TObject", version);
129 mC.u32(); // fUniqueID
130 auto bits = mC.u32();
131 if (bits & kIsReferenced) {
132 mC.u16(); // pidf
133 }
134}

146void ObjectReader::skipObject(char const* className, bool check)
147{
148 auto start = mC.pos();
149 auto bcnt = mC.u32();
150 if (!(bcnt & kByteCountMask)) {
151 throw UnsupportedLayout(std::string("object of class ") + className + " without byte count");
152 }
153 auto version = mC.u16();
154 if (check) {
155 checkVersion(className, version);
156 }
157 mC.seek(start + 4);
158 mC.skip(bcnt & ~kByteCountMask);
159}

179// ROOT's read_object_any scheme. Recorded offsets are buffer displacements
180// (relative to the key start) plus kMapOffset.
181Object ObjectReader::anyObject()
182{
183 auto beg = mC.displacement();
184 auto begPos = mC.pos();
185 auto bcnt = mC.u32();
186 if (!(bcnt & kByteCountMask) || bcnt == kNewClassTag) {
187 // No byte count: the word is the tag itself. Only references to objects
188 // already read are written like this; new objects without a byte count
189 // come from very old files.
190 auto tag = bcnt;
191 if (tag & kClassMask) {
192 throw UnsupportedLayout("new object without byte count");
193 }
194 if (tag == 0) {
195 return {};
196 }
197 if (tag == 1) {
198 throw UnsupportedLayout("reference to the parent object");
199 }
200 auto it = mObjects.find(tag);
201 if (it == mObjects.end()) {
202 mC.fail("dangling object reference " + std::to_string(tag));
203 }
204 return it->second;
205 }
206 size_t end = begPos + 4 + (bcnt & ~kByteCountMask);
207 if (end > mC.buffer().size()) {
208 mC.fail("object byte count past the end of the buffer");
209 }
210 auto tagAt = mC.displacement();
211 auto tag = mC.u32();
212 Object result;
213 if (!(tag & kClassMask)) {
214 throw UnsupportedLayout("object reference with byte count");
215 } else if (tag == kNewClassTag) {
216 auto className = mC.cstring();
217 mClasses[tagAt + kMapOffset] = className;
218 result = readByClass(className, beg + kMapOffset);
219 } else {
220 auto it = mClasses.find(tag & ~kClassMask);
221 if (it == mClasses.end()) {
222 mC.fail("dangling class reference " + std::to_string(tag & ~kClassMask));
223 }
224 result = readByClass(it->second, beg + kMapOffset);
225 }
226 mC.seek(end);
227 return result;
228}

230Object ObjectReader::readByClass(std::string const& className, int64_t mapAt)
231{
232 if (className == "TBranch") {
233 return tbranch(mapAt);
234 }
235 if (className.size() == 6 && className.starts_with("TLeaf") && std::string_view("OBSILFD").find(className[5]) != std::string_view::npos) {
236 return tleaf(className, mapAt);
237 }
238 if (className == "TBasket") {
239 return tbasket(mapAt);
240 }
241 if (className == "TObjString") {
242 return tobjstring(mapAt);
243 }
244 throw UnsupportedLayout("objects of class " + className);
245}

247std::shared_ptr<LeafInfo> ObjectReader::tleaf(std::string const& className, int64_t mapAt)
248{
249 // ROOT maps an object before streaming its members.
250 auto leaf = std::make_shared<LeafInfo>();
251 mObjects[mapAt] = leaf;
252 auto vx = header(className.c_str());
253 auto vl = header("TLeaf");
254 leaf->className = className;
255 leaf->name = tnamed();
256 leaf->len = mC.i32();
257 if (leaf->len <= 0) {
258 mC.fail("leaf " + leaf->name + " with length " + std::to_string(leaf->len));
259 }
260 mC.i32(); // fLenType
261 mC.i32(); // fOffset
262 mC.u8(); // fIsRange
263 leaf->isUnsigned = mC.u8() != 0;
264 auto count = anyObject();
265 if (auto countLeaf = as<LeafInfo>(count)) {
266 leaf->countLeaf = countLeaf->name;
267 } else if (!std::holds_alternative<std::monostate>(count)) {
268 throw UnsupportedLayout("fLeafCount of " + leaf->name + " is not a leaf");
269 }
270 mC.seek(vl.end);
271 // fMinimum and fMaximum, in the type of the leaf.
272 mC.seek(vx.end);
273 return leaf;
274}

276std::shared_ptr<EmbeddedBasket> ObjectReader::tbasket(int64_t mapAt)
277{
278 // TBasket has no versioned header of its own: it starts with its TKey.
279 auto basket = std::make_shared<EmbeddedBasket>();
280 mObjects[mapAt] = basket;
281 auto header = readBasketHeader(mC);
282 checkVersion("TBasket", header.version % 1000);
283 auto keylen = header.keylen;
284 auto nevbuf = header.nevbuf;
285 auto last = header.last;
286 int flag = header.flag;
287 if (nevbuf < 0 || last < keylen) {
288 mC.fail("inconsistent embedded basket header");
289 }
290 basket->nEntries = nevbuf;
291 if (flag == 99) {
292 // Header only, no data.
293 return basket;
294 }
295 bool displacements = flag > 40;
296 if (displacements) {
297 flag -= 40;
298 }
299 bool hasBuffer = flag > 10;
300 if (hasBuffer) {
301 flag -= 10;
302 }
303 if (flag == 1 && nevbuf > 0) {
304 // Entry offsets (and displacements), each a count followed by fNevBuf ints.
305 mC.skip(4 + 4 * (size_t)nevbuf);
306 if (displacements) {
307 mC.skip(4 + 4 * (size_t)nevbuf);
308 }
309 }
310 if (hasBuffer) {
311 // The buffer starts with a copy of the key, then the values.
312 mC.skip(keylen);
313 auto data = mC.bytes(last - keylen);
314 basket->data.assign(data.begin(), data.end());
315 }
316 return basket;
317}

330std::shared_ptr<BranchInfo> ObjectReader::tbranch(int64_t mapAt)
331{
332 auto b = std::make_shared<BranchInfo>();
333 mObjects[mapAt] = b;
334 auto v = header("TBranch");
335 b->name = tnamed();
336 skipObject("TAttFill", false);
337 mC.i32(); // fCompress
338 mC.i32(); // fBasketSize
339 mC.i32(); // fEntryOffsetLen
340 auto writeBasket = mC.i32();
341 mC.i64(); // fEntryNumber
342 skipObject("ROOT::TIOFeatures", false);
343 mC.i32(); // fOffset
344 auto maxBaskets = mC.i32();
345 mC.i32(); // fSplitLevel
346 b->entries = mC.i64();
347 mC.i64(); // fFirstEntry
348 b->totBytes = mC.i64();
349 b->zipBytes = mC.i64();
350 if (maxBaskets < 0 || writeBasket < 0 || b->entries < 0) {
351 mC.fail("inconsistent TBranch header for " + b->name);
352 }
353 // The three basket arrays take 20 bytes per basket: check before allocating them.
354 if ((size_t)maxBaskets > mC.remaining() / 20) {
355 mC.fail("fMaxBaskets " + std::to_string(maxBaskets) + " of branch " + b->name + " exceeds the buffer");
356 }
357
358 if (!objArray().empty()) {
359 throw UnsupportedLayout("branch " + b->name + " has sub-branches");
360 }
361 auto leaves = objArray();
362 if (leaves.size() != 1 || !as<LeafInfo>(leaves[0])) {
363 throw UnsupportedLayout("branch " + b->name + " does not have exactly one leaf");
364 }
365 auto baskets = objArray();
366
367 auto readArray = [this, maxBaskets](auto reader) {
368 mC.u8(); // speedbump
369 std::vector<int64_t> values(maxBaskets);
370 for (auto& value : values) {
371 value = reader();
372 }
373 return values;
374 };
375 auto basketBytes = readArray([this] { return (int64_t)mC.i32(); });
376 auto basketEntry = readArray([this] { return mC.i64(); });
377 auto basketSeek = readArray([this] { return mC.i64(); });
378 if (!mC.string().empty()) {
379 throw UnsupportedLayout("branch " + b->name + " stores its baskets in another file");
380 }
381 mC.seek(v.end);
382
383 auto leaf = as<LeafInfo>(leaves[0]);
384 b->leafClass = leaf->className[5];
385 b->isUnsigned = leaf->isUnsigned;
386 b->fixedLen = leaf->len;
387 b->countBranch = leaf->countLeaf;
388
389 // Normal baskets end at the first empty seek, or at fWriteBasket.
390 int nNormal = 0;
391 while (nNormal < maxBaskets && nNormal < writeBasket && basketSeek[nNormal] != 0) {
392 ++nNormal;
393 }
394 for (int i = 0; i < nNormal; ++i) {
395 if (basketBytes[i] <= 0 || basketSeek[i] < 0) {
396 mC.fail("invalid basket location in branch " + b->name);
397 }
398 // When fBasketEntry has no slot after the last basket, the entry count
399 // is only in the basket header. The container layer fills it in (-1).
400 bool hasNext = i + 1 < maxBaskets;
401 if (basketEntry[i] < 0 || (hasNext && basketEntry[i + 1] < basketEntry[i])) {
402 mC.fail("invalid basket entries in branch " + b->name);
403 }
404 int64_t n = hasNext ? basketEntry[i + 1] - basketEntry[i] : -1;
405 if (n > INT32_MAX) {
406 mC.fail("invalid basket entries in branch " + b->name);
407 }
408 b->baskets.push_back(BasketRef{(uint64_t)basketSeek[i], (uint32_t)basketBytes[i], basketEntry[i], (int32_t)n, false, {}});
409 }
410 // Embedded baskets exist only if the normal ones do not hold all the entries.
411 bool knownNormal = nNormal < maxBaskets;
412 int64_t normalEntries = nNormal == 0 ? 0 : (knownNormal ? basketEntry[nNormal] : -1);
413 if (normalEntries != b->entries) {
414 int64_t first = knownNormal ? normalEntries : 0;
415 for (auto& o : baskets) {
416 auto e = as<EmbeddedBasket>(o);
417 if (!e || (!knownNormal && e->nEntries == 0)) {
418 continue;
419 }
420 b->baskets.push_back(BasketRef{0, 0, first, e->nEntries, true, std::move(e->data)});
421 first += e->nEntries;
422 }
423 }
424 // The last normal basket holds what the others do not: computing it saves
425 // reading its header, a round trip for a remote file. The decoder checks it
426 // against the header anyway. If it does not add up, the container layer
427 // reads the header.
428 if (nNormal > 0 && b->baskets[nNormal - 1].nEntries < 0) {
429 int64_t rest = b->entries - b->baskets[nNormal - 1].firstEntry;
430 for (size_t i = nNormal; i < b->baskets.size(); ++i) {
431 rest -= b->baskets[i].nEntries;
432 }
433 if (rest >= 0 && rest <= INT32_MAX) {
434 b->baskets[nNormal - 1].nEntries = (int32_t)rest;
435 }
436 }
437 return b;
438}

440TreeInfo ObjectReader::tree()
441{
442 TreeInfo t;
443 auto v = header("TTree");
444 t.name = tnamed();
445 skipObject("TAttLine", false);
446 skipObject("TAttFill", false);
447 skipObject("TAttMarker", false);
448 t.entries = mC.i64();
449 t.totBytes = mC.i64();
450 t.zipBytes = mC.i64();
451 mC.skip(2 * 8); // fSavedBytes, fFlushedBytes
452 mC.f64(); // fWeight
453 mC.skip(4 * 4); // fTimerInterval, fScanField, fUpdate, fDefaultEntryOffsetLen
454 auto nClusterRange = mC.u32();
455 mC.skip(6 * 8); // fMaxEntries ... fEstimate
456 mC.u8(); // speedbump
457 mC.skip(8 * (size_t)nClusterRange);
458 mC.u8(); // speedbump
459 mC.skip(8 * (size_t)nClusterRange);
460 skipObject("ROOT::TIOFeatures", false);
461 for (auto& o : objArray()) {
462 auto b = as<BranchInfo>(o);
463 if (!b) {
464 throw UnsupportedLayout("tree " + t.name + " has a branch which is not a TBranch");
465 }
466 t.branches.push_back(std::move(*b));
467 }
468 // fLeaves, fAliases, fIndexValues, fIndex, fTreeIndex, fFriends, fUserInfo, fBranchRef
469 mC.seek(v.end);
470 return t;
471}

473std::map<std::string, std::string> ObjectReader::stringMap()
474{
475 std::map<std::string, std::string> result;
476 auto v = header("TMap");
477 tobject();
478 mC.string(); // fName
479 auto n = mC.i32();
480 for (int i = 0; i < n; ++i) {
481 auto key = as<std::string>(anyObject());
482 auto value = as<std::string>(anyObject());
483 if (!key || !value) {
484 throw UnsupportedLayout("TMap entry which is not a TObjString pair");
485 }
486 result[*key] = *value;
487 }
488 mC.seek(v.end);
489 return result;
490}
