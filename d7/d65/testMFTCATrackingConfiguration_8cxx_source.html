80{
81 const DetectorConfiguration layout{kMFTSurfaces};
82 for (const auto mode : {TrackingMode::Sync, TrackingMode::Async}) {
83 for (const uint32_t mask : {uint32_t{1} << 8, uint32_t{1} << 9, (uint32_t{1} << 8) | (uint32_t{1} << 9)}) {
84 ConfigurableParam::updateFromString("MFTCATrackerParam.startLayerMask[0]=" + std::to_string(mask));
85 BOOST_CHECK_EQUAL(MFTParam::Instance().startLayerMask[0], mask);
86 const auto topology = deriveTraversalTopology(layout, resolve(mode).front());
87 BOOST_REQUIRE(topology.ok());
88 LayerMask actual;
89 for (const auto path : topology.topology->roadStartPaths) {
90 const auto edge = topology.topology->paths[path.value()].second;
91 actual.set(topology.topology->edges[edge.value()].to.value());
92 }
93 BOOST_CHECK_EQUAL(actual.value(), mask);
94 }
95 }
96 ConfigurableParam::updateFromString("MFTCATrackerParam.startLayerMask[0]=1024");
97 BOOST_CHECK_THROW(resolve(TrackingMode::Async), std::invalid_argument);
98 ConfigurableParam::updateFromString("MFTCATrackerParam.startLayerMask[0]=0");
99 BOOST_CHECK_EQUAL(resolve(TrackingMode::Sync).front().StartLayerMask.count(), MFTNLayers);

101 BOOST_REQUIRE(dictionary);
