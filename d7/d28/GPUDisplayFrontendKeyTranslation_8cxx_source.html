20static constexpr uint64_t encodeSequence(const char* s) // As encoded by the console: first character in the lowest byte
21{
22 uint64_t retVal = 0;
23 for (uint32_t i = 0; i < 8 && s[i]; i++) {
24 retVal |= (uint64_t)(uint8_t)s[i] << (8 * i);
25 }
26 return retVal;
27}
