32 for (int i = 1; i < argc; ++i) {
33 if (std::strcmp(argv[i], "--threads") == 0 && i + 1 < argc) {
34 options.threads = std::stoi(argv[++i]);
35 } else if (std::strcmp(argv[i], "--pread") == 0) {
36 options.localIO = Options::LocalIO::PRead;
37 } else {
38 args.push_back(argv[i]);
39 }
40 }
41 if (args.size() != 3 || args[1].find('/') == std::string::npos) {
42 std::fprintf(stderr, "usage: %s <file> <DF_x/O2tree> <output.arrow> [--threads N] [--pread]\n", argv[0]);
43 return 2;
44 }
45 auto df = args[1].substr(0, args[1].find('/'));
46 auto treeName = args[1].substr(args[1].find('/') + 1);
47 try {
48 RootFile file{ByteSource::open(args[0], options)};
49 std::optional<TreeInfo> tree;
50 for (auto& key : file.keysOf(df)) {
51 if (key.name == treeName) {
52 tree = file.readTree(key);
53 }
54 }
55 if (!tree) {
56 throw FormatError("no tree " + treeName + " in " + df);
57 }
58 auto batch = readTree(file.source(), *tree, options).ValueOrDie();
59 auto out = arrow::io::FileOutputStream::Open(args[2]).ValueOrDie();
60 auto writer = arrow::ipc::MakeStreamWriter(out, batch->schema()).ValueOrDie();
61 auto status = writer->WriteRecordBatch(*batch);
62 if (status.ok()) {
63 status = writer->Close();
64 }
65 if (!status.ok()) {
66 std::fprintf(stderr, "%s: %s\n", argv[0], status.ToString().c_str());
67 return 1;
68 }
69 } catch (FormatError const& e) {
70 std::fprintf(stderr, "%s: %s\n", argv[0], e.what());
71 return 1;
72 }
