105 {
106 if (n > mBuf.size() - mPos) {
107 fail("read of " + std::to_string(n) + " bytes past the end of a " + std::to_string(mBuf.size()) + " byte buffer");

112 template <typename T>
113 T load()
114 {
115 T v;

144inline BasketHeader readBasketHeader(Cursor& c)
145{
146 auto start = c.pos();
147 BasketHeader h;
148 h.nbytes = c.u32();
149 bool big = c.i16() > 1000;
150 h.objlen = c.u32();
151 c.u32(); // fDatime
152 h.keylen = c.u16();
153 c.u16(); // fCycle
154 c.skip(big ? 16 : 8); // fSeekKey, fSeekPdir
155 c.string(); // fClassName
156 c.string(); // fName
157 c.string(); // fTitle
158 h.version = c.u16();
159 c.i32(); // fBufferSize
160 if (c.i32() < 0) {
161 c.u8(); // fIOBits
162 }
163 h.nevbuf = c.i32();
164 h.last = c.i32();
165 h.flag = c.u8();
166 if (c.pos() != start + h.keylen) {
167 c.fail("TBasket key of " + std::to_string(c.pos() - start) + " bytes, but fKeylen is " + std::to_string(h.keylen));
168 }
169 return h;
170}
