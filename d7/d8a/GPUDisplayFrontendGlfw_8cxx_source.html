146void GPUDisplayFrontendGlfw::GetKey(int32_t key, int32_t scancode, int32_t mods, int32_t& keyOut, int32_t& keyPressOut)
147{
148 int32_t specialKey = GetKey(key);
149 const char* str = glfwGetKeyName(key, scancode);
150 char localeKey = str ? str[0] : 0;
151 if ((mods & GLFW_MOD_SHIFT) && localeKey >= 'a' && localeKey <= 'z') {
152 localeKey += 'A' - 'a';
153 }
154 // GPUInfo("Key: key %d (%c) scancode %d -> %d (%c) special %d (%c)", key, (char)key, scancode, (int32_t)localeKey, localeKey, specialKey, (char)specialKey);
155
156 if (specialKey) {
157 keyOut = keyPressOut = specialKey;
158 } else {
159 keyOut = keyPressOut = (uint8_t)localeKey;
160 if (keyPressOut >= 'a' && keyPressOut <= 'z') {
161 keyPressOut += 'A' - 'a';
162 }
163 }
164}

168void GPUDisplayFrontendGlfw::key_callback(GLFWwindow* window, int32_t key, int32_t scancode, int32_t action, int32_t mods)
169{
170 int32_t handleKey = 0, keyPress = 0;
171 GetKey(key, scancode, mods, handleKey, keyPress);
172 if (handleKey < 32) {
173 if (action == GLFW_PRESS || action == GLFW_REPEAT) { // Repeat as the other frontends (e.g. backspace in the console), characters are repeated by char_callback
174 me->mKeys[keyPress] = keyPress == KEY_SHIFT || !me->consoleShown(); // Held keys only without console, SHIFT also with it (scrolling)
175 me->HandleKey(handleKey);
176 } else if (action == GLFW_RELEASE) {
177 me->mKeys[keyPress] = false;
178 }
179 } else if (handleKey < 256) {
180 if (action == GLFW_PRESS) {
181 me->mLastKeyDown = handleKey;
182 if (key == GLFW_KEY_GRAVE_ACCENT) { // Console key, its character (if not a dead key) follows in the same glfwPollEvents
183 me->mConsoleKeyPressed = true;
184 me->HandleKey('^');
185 }
186 } else if (action == GLFW_RELEASE) {
187 keyPress = (uint8_t)me->mGLFWKeyDownMap[handleKey];
188 me->mKeys[keyPress] = false;
189 }
190 }
191}

193void GPUDisplayFrontendGlfw::char_callback(GLFWwindow* window, uint32_t codepoint)
194{
195 // GPUInfo("Key (char callback): %d %c - key: %d", codepoint, (char)codepoint, (int32_t)me->mLastKeyDown);
196 if (me->mConsoleKeyPressed) { // Character of the console key, handled in key_callback
197 me->mConsoleKeyPressed = false;
198 return;
199 }
200 if (codepoint < 256) {
201 uint8_t keyPress = codepoint;
202 if (keyPress >= 'a' && keyPress <= 'z') {
203 keyPress += 'A' - 'a';
204 }
205 me->mGLFWKeyDownMap[me->mLastKeyDown] = keyPress;
206 me->mKeys[keyPress] = keyPress == KEY_SHIFT || !me->consoleShown(); // Held keys only without console, SHIFT also with it (scrolling)
207 me->HandleKey(codepoint);
208 }
209}

211void GPUDisplayFrontendGlfw::mouseButton_callback(GLFWwindow* window, int32_t button, int32_t action, int32_t mods)
212{

221 me->mMouseDnX = me->mMouseMvX;
222 me->mMouseDnY = me->mMouseMvY;

224 if (button == 0) {
225 me->mMouseDn = false;
226 } else if (button == 1) {
227 me->mMouseDnR = false;
228 } else if (button == GLFW_MOUSE_BUTTON_MIDDLE) {
229 me->mMouseDnM = false;

263{
264 me->setWindowState(glfwGetWindowMonitor(window) != nullptr, maximized);
265 if (glfwGetPlatform() == GLFW_PLATFORM_WAYLAND) { // Called before the resize. GLFW keeps the maximized size if the compositor leaves the size to the client when unmaximizing, so restore the last windowed size
266 if (maximized) {
267 glfwGetWindowSize(window, &me->mWindowedWidth, &me->mWindowedHeight);
268 } else {
269 me->mRestoreWindowedSize = true; // Must be applied after GLFW has processed the configure event, otherwise GLFW resizes to the pending maximized size again

274int32_t GPUDisplayFrontendGlfw::FrontendMain()
275{
276 me = this;
277
278 if (!glfwInit()) {
279 fprintf(stderr, "Error initializing glfw\n");
280 return (-1);
281 }
282 glfwSetErrorCallback(error_callback);
283
284 if (backend()->backendType() == GPUDisplayBackend::TYPE_VULKAN) {
285 glfwWindowHint(GLFW_CLIENT_API, GLFW_NO_API);
286 }
287 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
288 glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, GL_MIN_VERSION_MAJOR);
289 glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, GL_MIN_VERSION_MINOR);
290 glfwWindowHint(GLFW_OPENGL_FORWARD_COMPAT, 0);
291 glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
292 }
293 glfwWindowHint(GLFW_SCALE_TO_MONITOR, GLFW_TRUE); // Initial size scaled by the content scale on X11 / Win32
294 mWindow = glfwCreateWindow(INIT_WIDTH, INIT_HEIGHT, DISPLAY_WINDOW_NAME, nullptr, nullptr);
295 if (!mWindow) {
296 fprintf(stderr, "Error creating glfw window\n");
297 glfwTerminate();
298 return (-1);
299 }
300 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
301 glfwMakeContextCurrent(mWindow);
302 }
303
304 glfwSetKeyCallback(mWindow, key_callback);
305 glfwSetCharCallback(mWindow, char_callback);
306 glfwSetMouseButtonCallback(mWindow, mouseButton_callback);
307 glfwSetScrollCallback(mWindow, scroll_callback);
308 glfwSetCursorPosCallback(mWindow, cursorPos_callback);
309 glfwSetFramebufferSizeCallback(mWindow, resize_callback);
310 glfwSetWindowMaximizeCallback(mWindow, maximize_callback);
311 glfwSetWindowFocusCallback(mWindow, focus_callback);
312 float xscale, yscale;
313 glfwGetWindowContentScale(mWindow, &xscale, &yscale);
314 contentScale_callback(mWindow, xscale, yscale);
315 glfwSetWindowContentScaleCallback(mWindow, contentScale_callback);
316 const float limitScale = (glfwGetPlatform() == GLFW_PLATFORM_X11 || glfwGetPlatform() == GLFW_PLATFORM_WIN32) ? mScale : 1.f; // Window coordinates are pixels on X11 / Win32, logical on Wayland / macOS
317 glfwSetWindowSizeLimits(mWindow, MIN_WIDTH * limitScale, MIN_HEIGHT * limitScale, GLFW_DONT_CARE, GLFW_DONT_CARE);
318 int32_t fbWidth, fbHeight, windowWidth, windowHeight;
319 glfwGetFramebufferSize(mWindow, &fbWidth, &fbHeight);
320 glfwGetWindowSize(mWindow, &windowWidth, &windowHeight);
321 mPixelRatio = windowWidth ? (double)fbWidth / windowWidth : 1.; // Updated in resize_callback, which must not be called before InitDisplay
322 applyInitialWindowState();
323 SetVSync(vsyncEnabled());
324
325 if (mBackend->ExtInit()) {
326 fprintf(stderr, "Error initializing GL extension wrapper\n");
327 return (-1);
328 }
329
330 if (InitDisplay()) {
331 fprintf(stderr, "Error in GLFW display initialization\n");
332 return (1);
333 }
334
335 int32_t retVal = 0;
336 while (!glfwWindowShouldClose(mWindow)) {
337 if (HandleUpdatesAndDrawGLScene()) {
338 fprintf(stderr, "Error drawing GL scene\n");
339 retVal = 1;
340 break; // Cleanup below
341 }
342 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
343 glfwSwapBuffers(mWindow);
344 }
345 waitForFPSLimit();
346 glfwPollEvents();
347 mConsoleKeyPressed = false; // No character followed (dead key)
348 if (mRestoreWindowedSize) {
349 mRestoreWindowedSize = false;
350 if (mWindowedWidth > 0 && mWindowedHeight > 0 && !glfwGetWindowAttrib(mWindow, GLFW_MAXIMIZED)) {
351 glfwSetWindowSize(mWindow, mWindowedWidth, mWindowedHeight);
352 }
353 }
354 }
355
356 ExitDisplay();
357 {
358 std::lock_guard lock(mExitStateMutex);
359 setDisplayControl(2);
360 glfwDestroyWindow(mWindow);
361 glfwTerminate();
362 mDisplayRunning = false;
363 mDisplayStoppedCV.notify_all();
364 }
365
366 return retVal;
367}

406float GPUDisplayFrontendGlfw::getRefreshRate()
407{
408 GLFWmonitor* monitor = glfwGetWindowMonitor(mWindow) ? glfwGetWindowMonitor(mWindow) : glfwGetPrimaryMonitor(); // Monitor of a windowed window is not known, use the primary one
