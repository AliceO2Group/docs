53// splitmix64 finaliser, to derive well separated seeds from the base seed
54ULong64_t mixSeed(ULong64_t x)
55{
56 x += 0x9e3779b97f4a7c15ULL;
57 x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
58 x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
59 return x ^ (x >> 31);
60}
61
62// TRandom3 takes 32 bits of the seed and treats 0 as a request for a random seed
63UInt_t toTRandom3Seed(ULong64_t x)
64{
65 const UInt_t seed = static_cast<UInt_t>(x ^ (x >> 32));
66 return seed ? seed : 1;
67}

91void Digitizer::initSectorSeeding()
92{
93 if (mUseSectorSeeding && !mGEMAmplification) {
94 // the rings are filled once, in the construction order of the singletons
95 TRandom3 rng(toTRandom3Seed(mixSeed(mSeed)));
96 mGEMAmplification = std::make_unique<GEMAmplification>(&rng);
97 mElectronTransport = std::make_unique<ElectronTransport>(&rng);
98 mSAMPAProcessing = std::make_unique<SAMPAProcessing>(&rng);
99 mDigitContainer.setSAMPAProcessing(mSAMPAProcessing.get());
100 }

116void Digitizer::init()
117{
118 std::lock_guard<std::mutex> lock(calibrationMutex());
119 if (mUseSectorSeeding) {
120 initSectorSeeding();
121 const ULong64_t sectorSeed = mixSeed(mSeed ^ mixSeed(static_cast<ULong64_t>(mSector) + 1));
122 TRandom3 rng(toTRandom3Seed(mixSeed(sectorSeed ^ mixSeed(mSeedingTimeframe))));
123 mGEMAmplification->setRandomRingPositions(rng);
124 mElectronTransport->setRandomRingPositions(rng);
125 mSAMPAProcessing->setRandomRingPositions(rng);
126 }
127 auto& gemAmplification = this->gemAmplification();
128 gemAmplification.updateParameters();
129 auto& electronTransport = this->electronTransport();
130 electronTransport.updateParameters(mVDrift);
131 auto& sampaProcessing = this->sampaProcessing();
132 // the calibration update may reload FEE parameters from file, so SAMPAProcessing picks up its pointers afterwards
133 mDigitContainer.updateCalibrations();
134 sampaProcessing.updateParameters(mVDrift);
135}

137void Digitizer::process(const std::vector<o2::tpc::HitGroup>& hits,
138 const int eventID, const int sourceID)
139{
140 const Mapper& mapper = Mapper::instance();
141 auto& detParam = ParameterDetector::Instance();
142 auto& eleParam = ParameterElectronics::Instance();
143 auto& gemParam = ParameterGEM::Instance();
144
145 auto& gemAmplification = this->gemAmplification();
146 auto& electronTransport = this->electronTransport();
147 auto& sampaProcessing = this->sampaProcessing();
148
149 const int nShapedPoints = eleParam.NShapedPoints;
150 const auto amplificationMode = gemParam.AmplMode;
151 mSignalArray.resize(nShapedPoints);
152
154 mDigitContainer.reserve(sampaProcessing.getTimeBinFromTime(mEventTime - mOutputDigitTimeOffset));
155
157 float maxEleTime = (int(mDigitContainer.size()) - nShapedPoints) * eleParam.ZbinWidth;
158
159 for (auto& hitGroup : hits) {
160 const int MCTrackID = hitGroup.GetTrackID();
161 for (size_t hitindex = 0; hitindex < hitGroup.getSize(); ++hitindex) {
162 const auto& eh = hitGroup.getHit(hitindex);
163
164 GlobalPosition3D posEle(eh.GetX(), eh.GetY(), eh.GetZ());
165
166 // Distort the electron position in case space-charge distortions are used
167 if (mDistortionScaleType == 1) {
168 mSpaceCharge->distortElectron(posEle);
169 } else if (mDistortionScaleType == 2) {
170 mSpaceCharge->distortElectron(posEle, (mUseScaledDistortions ? nullptr : mSpaceChargeDer), mLumiScaleFactor);
171 }
172
175 if (electronTransport.isCompletelyOutOfSectorCoarseElectronDrift(posEle, mSector)) {
176 continue;
177 }
178
180 const int nPrimaryElectrons = static_cast<int>(eh.GetEnergyLoss());
181 const float hitTime = eh.GetTime() * 0.001;
182 float driftTime = 0.f;
183
185
187 for (int iEle = 0; iEle < nPrimaryElectrons; ++iEle) {
188
190 const GlobalPosition3D posEleDiff = electronTransport.getElectronDrift(posEle, driftTime);
191 const float eleTime = driftTime + hitTime;
192 if (eleTime >= maxEleTime) {
193 // LOG(warning) << "Skipping electron with driftTime " << driftTime << " from hit at time " << hitTime;
194 continue;
195 }
196 const float absoluteTime = eleTime + mTDriftOffset + (mEventTime - mOutputDigitTimeOffset);
197
200 if (!(absoluteTime >= 0 /* && absoluteTime <= timeframelength */)) {
201 continue;
202 }
203
205 if (electronTransport.isElectronAttachment(driftTime)) {
206 continue;
207 }
208
210 if (std::abs(posEleDiff.Z()) > detParam.TPClength) {
211 continue;
212 }
213
215 if (mapper.isOutOfSector(posEleDiff, mSector)) {
216 continue;
217 }
218
220 const DigitPos digiPadPos = mapper.findDigitPosFromGlobalPosition(posEleDiff, mSector);
221 if (!digiPadPos.isValid()) {
222 continue;
223 }
224
226 if (digiPadPos.getCRU().sector() != mSector) {
227 continue;
228 }
229
231 const int nElectronsGEM = gemAmplification.getStackAmplification(digiPadPos.getCRU(), digiPadPos.getPadPos(), amplificationMode);
232 if (nElectronsGEM == 0) {
233 continue;
234 }
235
236 const GlobalPadNumber globalPad = mapper.globalPadNumber(digiPadPos.getGlobalPadPos());
237 const float ADCsignal = sampaProcessing.getADCvalue(static_cast<float>(nElectronsGEM));
238 const MCCompLabel label(MCTrackID, eventID, sourceID, false);
239 sampaProcessing.getShapedSignal(ADCsignal, absoluteTime, mSignalArray);
240 for (float i = 0; i < nShapedPoints; ++i) {
241 const float time = absoluteTime + i * eleParam.ZbinWidth;
242 mDigitContainer.addDigit(label, digiPadPos.getCRU(), sampaProcessing.getTimeBinFromTime(time), globalPad,
243 mSignalArray[i]);
244 }
246 }
248 }
249 }

313void Digitizer::setStartTime(double time)
314{
315 // this is setting the first timebin index for the digit container
316 // note that negative times w.r.t start of timeframe/data-taking == mOutputDigitTimeOffset
317 // will yield the 0-th bin (due to casting logic in sampaProcessing)
318 SAMPAProcessing& sampaProcessing = this->sampaProcessing();
319 {
320 std::lock_guard<std::mutex> lock(calibrationMutex());
321 sampaProcessing.updateParameters(mVDrift);
322 }
323 const auto timediff = time - mOutputDigitTimeOffset;
324 const auto starttimebin = sampaProcessing.getTimeBinFromTime(timediff);
325 mDigitContainer.setStartTime(starttimebin);
326}

354void Digitizer::recalculateDistortions()
355{
356 if (!mSpaceCharge || !mSpaceChargeDer) {
357 LOGP(info, "Average or derivative distortions not set");
358 return;
359 }
360
361 // recalculate distortions only in case the inst lumi differs from the avg lumi
362 if (mSpaceCharge->getMeanLumi() != CorrMapParam::Instance().lumiInst) {
363 for (int iside = 0; iside < 2; ++iside) {
364 const o2::tpc::Side side = (iside == 0) ? Side::A : Side::C;
365 // this needs to be done only once
366 LOGP(info, "Calculating corrections for average distortions");
367 mSpaceCharge->calcGlobalCorrWithGlobalDistIterative(side, nullptr, 0);
368
369 LOGP(info, "Calculating corrections for derivative distortions");
370 mSpaceChargeDer->calcGlobalCorrWithGlobalDistIterative(side, nullptr, 0);
371
372 LOGP(info, "Calculating scaled distortions with scaling factor {}", mLumiScaleFactor);
373 mSpaceCharge->calcGlobalDistWithGlobalCorrIterativeLinearCartesian(side, mSpaceChargeDer, mLumiScaleFactor);
374 }
375 // set new lumi of avg map
376 mSpaceCharge->setMeanLumi(CorrMapParam::Instance().lumiInst);
377 } else {
378 LOGP(info, "Inst. lumi {} is same as mean lumi {}. Skip recalculation of distortions", CorrMapParam::Instance().lumiInst, mSpaceCharge->getMeanLumi());
379 }
380
381 mUseScaledDistortions = true;
