140void FairRunSim::Init()
141{
143 fRootManager->InitSink();
144
145 CheckFlukaExec();
146
147 // fOutFile=fRootManager->OpenOutFile(fOutname);
148 LOG(info) << "============== FairRunSim: Initialising simulation run ==============";
149
150 FairGeoLoader* loader = new FairGeoLoader(fLoaderName->Data(), "Geo Loader");
151 FairGeoInterface* GeoInterFace = loader->getGeoInterface();
152 GeoInterFace->SetNoOfSets(ListOfModules->GetEntries());
153 GeoInterFace->setMediaFile(MatFname.Data());
154 GeoInterFace->readMedia();
155
156 // gSystem->cd(flout.Data());
157
158 fApp = new FairMCApplication("Fair", "The Fair VMC App", ListOfModules, MatFname);
159 fApp->SetGenerator(fGen);
160
161 // Add a Generated run ID to the FairRunTimeDb
162 if (fRunId == 0) {
163 FairRunIdGenerator genid;
164 fRunId = genid.generateId();
165 }
166 fFileHeader->SetRunId(fRunId);
168 fApp->AddTask(fTask);
169 // The FairBaseParSet / FairGeoParSet runtime-database containers were filled
170 // here but never written out nor read back; the database has been removed.
171
172 // Set global Parameter Info
173
174 if (fPythiaDecayer) {
175 fApp->SetPythiaDecayer(fPythiaDecayer);
176 if (fPythiaDecayerConfig) {
177 fApp->SetPythiaDecayerConfig(fPythiaDecayerConfig);
178 }
179 }
180 if (fUserDecay) {
181 fApp->SetUserDecay(fUserDecay);
182 if (fUserDecayConfig) {
183 fApp->SetUserDecayConfig(fUserDecayConfig);
184 }
185 }
186 // on/off visualisation
187 if (fStoreTraj) {
188 LOG(info) << "Create visualisation manager ";
189 new FairTrajFilter();
190 }
191 if (fRadLength) {
192 fApp->SetRadiationLengthReg(fRadLength);
193 }
194 if (fRadMap) {
195 fApp->SetRadiationMapReg(fRadMap);
196 }
197 if (fRadGrid) {
198 fApp->AddMeshList(fMeshList);
199 }
200
201 if (fField) {
202 fField->Init();
203 }
204 fApp->SetField(fField);
205
207 SetMCConfig();
208 fRootManager->WriteFileHeader(fFileHeader);
209}

212void FairRunSim::CheckFlukaExec()
213{
215 TString work = getenv("VMCWORKDIR");
216 TString work_config = work + "/gconfig/";
217 work_config.ReplaceAll("//", "/");
218
219 TString config_dir = getenv("CONFIG_DIR");
220 config_dir.ReplaceAll("//", "/");
221 if (!config_dir.EndsWith("/")) {
222 config_dir += "/";
223 }
224
225 TString flout;
226 if (strcmp(GetName(), "TFluka") == 0) {
227 TString flexec = "run_fluka.sh";
228 if (TString(gSystem->FindFile(config_dir.Data(), flexec)) != TString("")) {
229 LOG(info) << "---User path for Configuration is used: " << config_dir.Data();
230 } else {
231 flexec = work_config + "run_fluka.sh";
232 }
233 gSystem->Exec(flexec.Data());
234 // More generic for different macro
235 // dir layout <D.B>
236 TString macrodir = getenv("PWD");
237 // cout << macrodir << endl;
238 flout = macrodir + "/fluka_out";
239 flout.ReplaceAll("//", "/");
240 gSystem->cd(flout.Data());
241 }
242}

264void FairRunSim::SetMaterials(const char* MatFileName)
265{

269 if (work.IsNull()) {

276 Mat += "/";
277 }
278 }
