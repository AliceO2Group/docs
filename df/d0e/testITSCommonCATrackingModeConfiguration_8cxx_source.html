141// process-local OnFatal handler converts it into a catchable exception so
142// this remains a normal, non-crashing ctest case. Each ITSMFT test source

144// file), so this handler cannot leak into unrelated test binaries.

194BOOST_AUTO_TEST_CASE(SharedSchemaKeepsDetectorArraySizesAndIndependentOverrides)
195{
196 const auto checkDictionary = []<typename Config>(const Config&) {
197 auto* dictionary = TClass::GetClass(typeid(Config));
198 BOOST_REQUIRE(dictionary);
199 for (const auto* name : {"addTimeError", "sysErr2Row", "sysErr2Col"}) {
200 auto* member = dictionary->GetDataMember(name);
201 BOOST_REQUIRE(member);
202 BOOST_CHECK_EQUAL(member->GetMaxIndex(0), Config::NLayers);
203 }
204 };
205 checkDictionary(ITSCommonCATrackerParam::Instance());
206 checkDictionary(MFTCATrackerParam::Instance());
207
208 using o2::conf::ConfigurableParam;
209 for (const auto detector : {o2::detectors::DetID::ITS, o2::detectors::DetID::MFT}) {
210 const auto other = detector == o2::detectors::DetID::ITS ? o2::detectors::DetID::MFT : o2::detectors::DetID::ITS;
211 const std::string key = detector == o2::detectors::DetID::ITS ? "ITSCommonCATrackerParam." : "MFTCATrackerParam.";
212 struct Restore {
213 std::string key;
214 ~Restore()
215 {
216 ConfigurableParam::updateFromString(key + "nIterations=-1;" + key + "startLayerMask[0]=0;" +
217 key + "minTrackLgtIter[0]=0;" + key + "maxHolesIter[0]=0;" +
218 key + "maxChi2NDF=-1;" + key + "addTimeError[0]=0;" +
219 key + "sysErr2Row[0]=0");
220 }
221 } restore{key};
222 const auto baseline = TrackingMode::getTrackingPlan(other, TrackingMode::Async);
223 ConfigurableParam::updateFromString(key + "nIterations=2;" + key + "startLayerMask[0]=64;" +
224 key + "minTrackLgtIter[0]=5;" + key + "maxHolesIter[0]=1;" +
225 key + "maxChi2NDF=12;" + key + "addTimeError[0]=3;" +
226 key + "sysErr2Row[0]=0.01");
227 const auto plan = TrackingMode::getTrackingPlan(detector, TrackingMode::Async);
228 BOOST_REQUIRE_EQUAL(plan.iterations.size(), 2);
229 BOOST_CHECK_EQUAL(plan.iterations[0].StartLayerMask.value(), 64);
230 BOOST_CHECK_EQUAL(plan.iterations[0].MinTrackLength, 5);
231 BOOST_CHECK_EQUAL(plan.iterations[0].MaxHoles, 1);
232 BOOST_CHECK_EQUAL(plan.iterations[0].MaxChi2NDF, 12.f);
233 BOOST_CHECK_EQUAL(plan.detector.AddTimeError[0], 3);
234 BOOST_CHECK_EQUAL(plan.detector.SystError2Row[0], 0.01f);
235 const auto unchanged = TrackingMode::getTrackingPlan(other, TrackingMode::Async);
236 BOOST_CHECK_EQUAL(unchanged.iterations.size(), baseline.iterations.size());
237 BOOST_CHECK_EQUAL(unchanged.iterations[0].MinTrackLength, baseline.iterations[0].MinTrackLength);
238 BOOST_CHECK_EQUAL(unchanged.detector.SystError2Row[0], baseline.detector.SystError2Row[0]);
239
240 ConfigurableParam::updateFromString(key + "minTrackLgtIter[0]=0;" + key + "nIterations=4");
241 BOOST_CHECK_THROW(TrackingMode::getTrackingPlan(detector, TrackingMode::Async), std::invalid_argument);
242 ConfigurableParam::updateFromString(key + "nIterations=-1;" + key + "startLayerMask[0]=1024");
243 BOOST_CHECK_THROW(TrackingMode::getTrackingPlan(detector, TrackingMode::Async), std::invalid_argument);
244 if (detector == o2::detectors::DetID::ITS) {
245 ConfigurableParam::updateFromString(key + "startLayerMask[0]=128");
246 BOOST_CHECK_THROW(TrackingMode::getTrackingPlan(detector, TrackingMode::Async), std::invalid_argument);
247 }
248 }
