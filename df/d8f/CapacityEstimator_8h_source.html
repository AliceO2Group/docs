115template <typename Emit>
116int runOnSlab(CapacityEstimator& estimator, const CapacityEstimator::KeyType key, const double scale, Emit&& emit, const size_t floorCapacity = 0)
117{
118 const auto toInt = [](const size_t v) { return static_cast<int>(std::min(v, static_cast<size_t>(std::numeric_limits<int>::max()))); };
119 const int initialCapacity = toInt(estimator.capacity(key, scale));
120 int capacity = std::max(initialCapacity, toInt(floorCapacity));
121 int emitted = 0;
122 bool overflowed = false;
123 bool needsRetry = false;
124 do {
125 const int attemptCapacity = capacity;
126 emitted = emit(attemptCapacity);
127 needsRetry = emitted > attemptCapacity;
128 overflowed |= needsRetry;
129 capacity = emitted;
130 } while (needsRetry);
131 estimator.update(key, scale, emitted, initialCapacity, overflowed, false);
132 return emitted;
133}
