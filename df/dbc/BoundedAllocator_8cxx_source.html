58void* BoundedMemoryResource::do_allocate(size_t bytes, size_t alignment)
59{
60 size_t newUsed{0};
61 size_t currentUsed{mUsedMemory.load(std::memory_order_relaxed)};
62 for (;;) {
63 const size_t limit = mMaxMemory.load(std::memory_order_relaxed);
64 // Test before adding, so even SIZE_MAX requests cannot wrap under budget.
65 if (currentUsed > limit || bytes > limit - currentUsed) {
66 const size_t attempted = bytes > std::numeric_limits<size_t>::max() - currentUsed ? std::numeric_limits<size_t>::max() : currentUsed + bytes;
67 mCountThrow.fetch_add(1, std::memory_order_relaxed);
68 throw MemoryLimitExceeded(attempted, currentUsed, limit);
69 }
70 newUsed = currentUsed + bytes;
71 if (mUsedMemory.compare_exchange_weak(currentUsed, newUsed, std::memory_order_acq_rel, std::memory_order_relaxed)) {
72 break;
73 }
74 }
75
76 void* p{nullptr};
77 try {
78 p = mUpstream->allocate(bytes, alignment);
79 } catch (...) {
80 mUsedMemory.fetch_sub(bytes, std::memory_order_relaxed);
81#ifdef BOUNDED_MR_STATS
82 mStats.upstreamFailures.fetch_add(1, std::memory_order_relaxed);
83#endif
84 throw;
85 }
86
87 mAllocations.fetch_add(1, std::memory_order_relaxed);
88 mAllocatedBytes.fetch_add(bytes, std::memory_order_relaxed);
89
90 size_t peak = mPeakUsedMemory.load(std::memory_order_relaxed);
91 while (newUsed > peak && !mPeakUsedMemory.compare_exchange_weak(peak, newUsed, std::memory_order_relaxed)) {
92 }
93
94#ifdef BOUNDED_MR_STATS
95 size_t statsPeak = mStats.peak.load(std::memory_order_relaxed);
96 while (newUsed > statsPeak && !mStats.peak.compare_exchange_weak(statsPeak, newUsed, std::memory_order_relaxed)) {
97 }
98 mStats.live.fetch_add(1, std::memory_order_relaxed);
99 mStats.nAlloc.fetch_add(1, std::memory_order_relaxed);
100 mStats.totalAlloc.fetch_add(bytes, std::memory_order_relaxed);
101
102 size_t maxAlignment = mStats.maxAlign.load(std::memory_order_relaxed);
103 while (alignment > maxAlignment && !mStats.maxAlign.compare_exchange_weak(maxAlignment, alignment, std::memory_order_relaxed)) {
104 }
105#endif
106 return p;
107}

110{

115 mStats.nFree.fetch_add(1, std::memory_order_relaxed);

145size_t BoundedMemoryResource::getPeakMemoryDelta() const noexcept
146{
147 const size_t peak = mPeakUsedMemory.load(std::memory_order_relaxed);
148 const size_t baseline = mPeakBaselineMemory.load(std::memory_order_relaxed);

166void BoundedMemoryResource::setMaxMemory(size_t max)
167{
168 size_t current = mMaxMemory.load(std::memory_order_relaxed);
169 if (max == current) {
170 return;
171 }
172 for (;;) {
173 const size_t used = mUsedMemory.load(std::memory_order_acquire);
174 if (used > max) {
175 mCountThrow.fetch_add(1, std::memory_order_relaxed);
176 throw MemoryLimitExceeded(0, used, max);
177 }
178 if (mMaxMemory.compare_exchange_weak(current, max, std::memory_order_release, std::memory_order_relaxed)) {
179 return;
180 }
181 if (current == max) {
182 return;
183 }
184 }
185}

187std::string BoundedMemoryResource::asString() const
188{
189 const auto throwCount = mCountThrow.load(std::memory_order_relaxed);
190 const auto used = static_cast<double>(mUsedMemory.load(std::memory_order_relaxed));
191 const auto peak = static_cast<double>(mPeakUsedMemory.load(std::memory_order_relaxed));
192 const auto peakDelta = static_cast<double>(getPeakMemoryDelta());
193 const auto maxMemory = mMaxMemory.load(std::memory_order_relaxed);
194 std::string result;
195 if (maxMemory == std::numeric_limits<size_t>::max()) {
196 result += std::format("maxthrow={} maxmem=unbounded used={:.2f} GB stagepeak={:.2f} GB stagealloc={:.2f} GB", throwCount, used / kGigabyte, peak / kGigabyte, peakDelta / kGigabyte);
197 } else {
198 result += std::format("maxthrow={} maxmem={:.2f} GB used={:.2f} GB ({:.2f}%) stagepeak={:.2f} GB stagealloc={:.2f} GB", throwCount, static_cast<double>(maxMemory) / kGigabyte, used / kGigabyte, 100.0 * used / static_cast<double>(maxMemory), peak / kGigabyte, peakDelta / kGigabyte);
199 }
200#ifdef BOUNDED_MR_STATS
201 result += std::format(" peak={:.2f} GB live={} nAlloc={} nFree={} totalAlloc={:.2f} GB totalFreed={:.2f} GB maxAlign={} upstreamFail={}",
202 static_cast<float>(mStats.peak.load(std::memory_order_relaxed)) / kGigabyte,
203 mStats.live.load(std::memory_order_relaxed),
204 mStats.nAlloc.load(std::memory_order_relaxed),
205 mStats.nFree.load(std::memory_order_relaxed),
206 static_cast<float>(mStats.totalAlloc.load(std::memory_order_relaxed)) / kGigabyte,
207 static_cast<float>(mStats.totalFreed.load(std::memory_order_relaxed)) / kGigabyte,
208 mStats.maxAlign.load(std::memory_order_relaxed),
209 mStats.upstreamFailures.load(std::memory_order_relaxed));
210#endif
211 return result;
212}
