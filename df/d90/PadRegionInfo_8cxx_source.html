35void PadRegionInfo::init()
36{
37
38 const float ks = mPadHeight / mPadWidth * tan(1.74532925199432948e-01); // tan(10deg)
39 // initialize number of pads per row
40 for (unsigned char irow = 0; irow < mNumberOfPadRows; ++irow) {
41 mPadsPerRow[irow] = 2 * std::floor(ks * (irow + mRowOffset) + mXhelper);
42 mNumberOfPads += mPadsPerRow[irow];
43 }
44}
