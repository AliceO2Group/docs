60 void FinishEvent() override
61 {
62 if (mIsEndOfWork) {
63 // the empty event that ends a worker thread's loop: nothing to send
64 for (auto det : listActiveDetectors) {
65 det->EndOfEvent();
66 }
67 fStack->Reset();
68 return;
69 }
70 // update the stack
71 fStack->FillTrackArray();
72 fStack->UpdateTrackIndex(fActiveDetectors);
73
74 finishEventCommon();
75
76 // detectors finalize their hits (e.g. sorting, summing duplicates) before these are sent
77 for (auto det : listActiveDetectors) {
78 det->FinishEvent();
79 }
80
81 // This special finish event version does not fill the output tree of FairRootManager
82 // but forwards the data to the HitMerger
83 SendData();
84
85 // call end of event on active detectors
86 for (auto det : listActiveDetectors) {
87 det->EndOfEvent();
88 }
89 fStack->Reset();
90 }
