66void* GPUDisplayFrontend::FrontendThreadWrapper(void* ptr)
67{
68 GPUDisplayFrontend* me = reinterpret_cast<GPUDisplayFrontend*>(ptr);
69 int32_t retVal;
70 try {
71 retVal = me->FrontendMain();
72 } catch (const std::exception& e) {
73 GPUError("Error in display frontend: %s", e.what());
74 retVal = -1;
75 }
76 if (retVal && me->mDisplay->mInitResult == 0) { // Failed before InitDisplay was called, report failure to StartDisplay(), which otherwise waits forever
77 me->InitDisplay(true);
78 }
79 return ((void*)(size_t)retVal);
80}

82void GPUDisplayFrontend::waitForFPSLimit()
83{
84 mNeedUpdate = 0;
85 const auto now = std::chrono::steady_clock::now();
86 const auto& cfg = mDisplay->cfgR();
87 if (mBackend->fpsLimitMode() != GPUDisplayBackend::FPS_LIMIT_RATE || cfg.fpsRate <= 0) {
88 mNextFrameTime = now;
89 return;
90 }
91 const auto period = std::chrono::duration_cast<std::chrono::steady_clock::duration>(std::chrono::duration<double>(1. / cfg.fpsRate)); // TODO: wake up on HasExternalGUISignals
92 mNextFrameTime += period;
93 if (mNextFrameTime < now - period) { // More than one frame behind (slow rendering, first frame), do not try to catch up
94 mNextFrameTime = now;
95 }
96 std::this_thread::sleep_until(mNextFrameTime);
97}

99void GPUDisplayFrontend::HandleExternalGUISignals()
100{
101 const GPUSettingsDisplayHeavy oldCfgH = mDisplay->mCfgH;
102 const GPUSettingsDisplayLight oldCfgL = mDisplay->mCfgL;
103 const GPUSettingsDisplayRenderer oldCfgR = mDisplay->mCfgR;
104 bool changes = HandleExternalGUIKeys();
105
106 const bool gui = isGUIRunning();
107 std::optional<GPUDisplayGUIChannel::Update> update;
108 if (gui) {
109 std::lock_guard lock(mGUIChannel->mutex);
110 update = std::exchange(mGUIChannel->update, std::nullopt);
111 mGUIChannel->displayProcessingCurrentUpdate = update && mGUIChannel->guiSleeping; // GUI waits for the reply when it wakes up
112 }
113 if (update) {
114 for (int32_t key : update->keys) {
115 mDisplay->HandleExternalGUIKey(key);
116 }
117 if (update->lineWidth) {
118 mDisplay->mCfgL.lineWidth = *update->lineWidth;
119 }
120 changes = true;
121 }
122
123 if (changes) {
124 mDisplay->checkAndApplyConfigChanges(oldCfgH, oldCfgL, oldCfgR);
125 }
126 if (gui) {
127 {
128 std::lock_guard lock(mGUIChannel->mutex);
129 mGUIChannel->state.cfg = {mDisplay->mCfgH, mDisplay->mCfgL, mDisplay->mCfgR};
130 if (update) {
131 mGUIChannel->state.appliedUpdate = update->id;
132 }
133 mGUIChannel->state.counter++;
134 mGUIChannel->displayProcessingCurrentUpdate = false;
135 }
136 mGUIChannel->stateUpdated.notify_all();
137 }
138}

186int32_t GPUDisplayFrontend::startGUI()
187{
188 int32_t retVal = 1;
189#ifdef GPUCA_BUILD_EVENT_DISPLAY_QT
190 if (!mGUI) {
191 mGUIChannel = std::make_unique<GPUDisplayGUIChannel>();
192 mGUI = std::make_unique<GPUDisplayGUIWrapper>(*mGUIChannel);
193 }
194 if (!mGUI->isRunning()) {
195 {
196 std::lock_guard lock(mGUIChannel->mutex);
197 mGUIChannel->update.reset();
198 mGUIChannel->state = {};
199 mGUIChannel->guiSleeping = mGUIChannel->displayProcessingCurrentUpdate = mGUIChannel->stop = false;
200 mGUIChannel->replyTimeout = std::chrono::milliseconds(mDisplay->cfg().guiReplyTimeout);
201 mGUIChannel->maxUpdatesAhead = std::max<uint32_t>(mDisplay->cfg().guiMaxUpdatesAhead, 1); // 0 would block all updates
202 mGUIChannel->windowName = DISPLAY_WINDOW_NAME;
203 }
204 mGUI->start();
205 } else {
206 mGUI->focus();
207 }
208#endif
209 return retVal;
210}

221GPUDisplayFrontend* GPUDisplayFrontend::getFrontend(const char* type)
222{ // clang-format off
223#if !defined(GPUCA_STANDALONE) && defined(GPUCA_BUILD_EVENT_DISPLAY_GLFW)
224 if (strcmp(type, "glfw") == 0 || strcmp(type, "auto") == 0) {
225 return new GPUDisplayFrontendGlfw;
226 } else
227#endif
228#ifdef _WIN32
229 if (strcmp(type, "windows") == 0 || strcmp(type, "auto") == 0) {
230 return new GPUDisplayFrontendWindows;
231 } else
232#elif defined(GPUCA_BUILD_EVENT_DISPLAY_X11)
233 if (strcmp(type, "x11") == 0 || strcmp(type, "auto") == 0) {
234 return new GPUDisplayFrontendX11;
235 } else
236#endif
237#if defined(GPUCA_STANDALONE) && defined(GPUCA_BUILD_EVENT_DISPLAY_GLFW)
238 if (strcmp(type, "glfw") == 0 || strcmp(type, "auto") == 0) {
239 return new GPUDisplayFrontendGlfw;
240 } else
241#endif
242#ifdef GPUCA_BUILD_EVENT_DISPLAY_WAYLAND
243 if (strcmp(type, "wayland") == 0 || (strcmp(type, "auto") == 0 && getenv("XDG_SESSION_TYPE") && strcmp(getenv("XDG_SESSION_TYPE"), "wayland") == 0)) {
244 return createGPUDisplayFrontendWayland();
245 } else
246#endif
247#ifdef GPUCA_BUILD_EVENT_DISPLAY_GLUT
248 if (strcmp(type, "glut") == 0 || strcmp(type, "auto") == 0) {
249 return new GPUDisplayFrontendGlut;
250 } else
251#endif
252 if (strcmp(type, "none") == 0) {
253 return new GPUDisplayFrontendNone;
254 } else {
255 GPUError("Requested frontend not available");
256 }
257 return nullptr;
258} // clang-format on

297int32_t GPUDisplayFrontend::StartDisplay()
298{
299#ifndef _WIN32
300 static pthread_t hThread;
301 if (pthread_create(&hThread, nullptr, FrontendThreadWrapper, this)) {
302 GPUError("Coult not Create frontend Thread...");
303 return (1);
304 }
305#else
306 HANDLE hThread;
307 if ((hThread = CreateThread(nullptr, nullptr, &OpenGLWrapper, this, nullptr, nullptr)) == nullptr) {
308 GPUError("Coult not Create GL Thread...");
309 return (1);
310 }
311#endif
312 return (0);
313}

315void GPUDisplayFrontend::StopDisplay()
316{
317 {
318 std::lock_guard lock(mExitStateMutex);
319 if (mDisplayRunning) {
320 mDisplayControl = 2;
321 StopDisplayA();
322 mDisplay->UnblockDisplay();
323 }
324 }
325 while (mDisplayRunning) {
326 std::this_thread::sleep_for(std::chrono::microseconds(10000)); // TODO: Signal by semaphore
327 }
328}
