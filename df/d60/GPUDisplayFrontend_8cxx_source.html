71void GPUDisplayFrontend::FrontendThreadWrapper()
72{
73 int32_t retVal;
74 try {
75 retVal = FrontendMain();
76 } catch (const std::exception& e) {
77 GPUError("Error in display frontend: %s", e.what());
78 retVal = -1;
79 }
80 std::unique_lock lock(mExitStateMutex);
81 mDisplayRunning = false; // Also after an error / exception in the main loop, StopDisplay waits for it
82 mDisplayStoppedCV.notify_all();
83 if (retVal) {
84 if (mDisplay->mInitResult == 0) { // Failed before InitDisplay was called, report failure to StartDisplay(), which otherwise waits forever
85 lock.unlock();
86 InitDisplay(true);
87 } else {
88 mDisplay->mInitResult = -1; // Error after the initialization: StopDisplay does not reset the state for a new start
89 }
90 }
91}

93void GPUDisplayFrontend::waitForFPSLimit()
94{
95 const auto now = std::chrono::steady_clock::now();
96 const auto& cfg = mDisplay->cfgR();
97 if (mBackend->fpsLimitMode() != GPUDisplayBackend::FPS_LIMIT_RATE || cfg.fpsRate <= 0) {
98 mNextFrameTime = now;
99 return;
100 }
101 const auto period = std::chrono::duration_cast<std::chrono::steady_clock::duration>(std::chrono::duration<double>(1. / cfg.fpsRate));
102 mNextFrameTime += period;
103 if (mNextFrameTime < now - period) { // More than one frame behind (slow rendering, first frame), do not try to catch up
104 mNextFrameTime = now;
105 }
106 std::unique_lock lock(mConfigUpdateMutex);
107 mConfigUpdateCV.wait_until(lock, mNextFrameTime, [this] { return mConfigUpdatePending; });
108}

119void GPUDisplayFrontend::HandleExternalGUISignals()
120{
121 {
122 std::lock_guard lock(mConfigUpdateMutex);
123 mConfigUpdatePending = false; // Before fetching the updates, so that updates sent in the meantime signal again
124 }
125 HandleExternalGUIKeys();
126
127 const bool gui = isGUIRunning();
128 std::optional<GPUDisplayGUIChannel::Update> update;
129 if (gui) {
130 std::lock_guard lock(mGUIChannel->mutex);
131 update = std::exchange(mGUIChannel->update, std::nullopt);
132 mGUIChannel->displayProcessingCurrentUpdate = update && mGUIChannel->guiSleeping; // GUI waits for the reply when it wakes up
133 }
134 if (update) {
135 for (int32_t key : update->keys) {
136 mDisplay->HandleExternalGUIKey(key);
137 }
138 applyGUIValues(*update);
139 if (update->guiScale) {
140 mDisplay->mCfgO.guiScale = *update->guiScale;
141 }
142 if (update->guiDarkMode) {
143 mDisplay->mCfgO.guiDarkMode = *update->guiDarkMode;
144 }
145 }
146
147 if (gui) {
148 GPUDisplayGUIChannel::Settings cfg{mDisplay->mCfgH, mDisplay->mCfgL, mDisplay->mCfgR, mDisplay->mCfgO, mDisplay->mCfgC};
149 {
150 std::lock_guard lock(mGUIChannel->mutex);
151 mGUIChannel->state.cfg = std::move(cfg);
152 mGUIChannel->state.infoStatus = mDisplay->mCfgO.printInfoTextGUI ? mDisplay->mGUIInfoStatus : "";
153 mGUIChannel->state.infoText = mDisplay->mCfgO.printInfoTextGUI ? mDisplay->mGUIInfoText : "";
154 mGUIChannel->state.infoTextCounter = mDisplay->mGUIInfoTextCounter;
155 if (mDisplay->mUpdateGUIOptions) {
156 mGUIChannel->state.optionsRevision++;
157 mDisplay->mUpdateGUIOptions = false;
158 }
159 if (update) {
160 mGUIChannel->state.appliedUpdate = update->id;
161 }
162 mGUIChannel->state.counter++;
163 mGUIChannel->displayProcessingCurrentUpdate = false;
164 }
165 mGUIChannel->stateUpdated.notify_all();
166 std::unique_lock lock(mGUIChannel->mutex);
167 if (mGUIChannel->optionsReadRevision != mGUIChannel->state.optionsRevision) { // The GUI calls the dynamic option lambdas (reading display data) only while we wait here, short timeout, retried in the next frame
168 mGUIChannel->displayWaitingForOptions = true;
169 mGUIChannel->optionsRead.wait_for(lock, std::chrono::milliseconds(20), [this] { return mGUIChannel->optionsReadRevision == mGUIChannel->state.optionsRevision || mGUIChannel->stop; });
170 mGUIChannel->displayWaitingForOptions = false;
171 }
172 }
173}

194int32_t GPUDisplayFrontend::HandleUpdatesAndDrawGLScene(bool suspended)
195{
196 HandleExternalGUISignals();
197 mDisplay->checkAndApplyConfigChanges();
198 const bool startupScript = !mStartupScriptDone && !mDisplay->cfg().runScript.empty();
199 const int32_t retVal = mDisplay->DrawGLScene(suspended || startupScript); // Before the startup script, the first frame only loads the event, it is not rendered
200 mDisplay->mOldCfg = {mDisplay->mCfgH, mDisplay->mCfgL, mDisplay->mCfgR}; // Also includes the changes during drawing (animation, scene reset, ...), which set their update flags themselves
201 if (startupScript) {
202 mStartupScriptDone = true;
203 mConsole->runScriptFile(mDisplay->cfg().runScript); // Settings it changes are applied with the next frame
204 } else {
205 mConsole->processScripts(true); // Waiting script lines, wait / benchmark
206 }
207 return retVal;
208}

233int32_t GPUDisplayFrontend::startGUI()
234{
235 int32_t retVal = 1;
236#ifdef GPUCA_BUILD_EVENT_DISPLAY_QT
237 if (!mGUI) {
238 mGUIChannel = std::make_unique<GPUDisplayGUIChannel>();
239 mGUIChannel->updatePosted = [this]() { signalConfigUpdate(); };
240 mGUI = std::make_unique<GPUDisplayGUIWrapper>(*mGUIChannel);
241 }
242 if (!mGUI->isRunning()) {
243 {
244 std::lock_guard lock(mGUIChannel->mutex);
245 mGUIChannel->update.reset();
246 mGUIChannel->state = {};
247 mGUIChannel->guiSleeping = mGUIChannel->displayProcessingCurrentUpdate = mGUIChannel->stop = mGUIChannel->displayWaitingForOptions = false;
248 mGUIChannel->optionsReadRevision = 0;
249 mDisplay->mUpdateGUIOptions = true;
250 mGUIChannel->replyTimeout = std::chrono::milliseconds(mDisplay->cfg().internal.guiReplyTimeout);
251 mGUIChannel->maxUpdatesAhead = std::max<uint32_t>(mDisplay->cfg().internal.guiMaxUpdatesAhead, 1); // 0 would block all updates
252 mGUIChannel->infoTextTime = std::chrono::milliseconds(std::lround((mDisplay->cfg().internal.setInfoTime + mDisplay->cfg().internal.setInfoTransition) * 1000.f));
253 mGUIChannel->windowName = DISPLAY_WINDOW_NAME;
254 setGUILayout(*mGUIChannel);
255 }
256 mGUI->start();
257 } else {
258 mGUI->focus();
259 }
260#endif
261 return retVal;
262}

362int32_t GPUDisplayFrontend::StartDisplay()
363{
364 try {
365 mFrontendThread = std::thread(&GPUDisplayFrontend::FrontendThreadWrapper, this);
366 } catch (const std::system_error& e) {
367 GPUError("Could not create frontend thread: %s", e.what());
368 return (1);
369 }
370 std::unique_lock lock(mExitStateMutex);
371 mStartDisplayDone = true;
372 mInitCV.notify_all();
373 mInitCV.wait(lock, [this] { return mDisplay->mInitResult != 0; }); // Both threads have finished their initialization
374 return (mDisplay->mInitResult != 1);
375}
376
377void GPUDisplayFrontend::StopDisplay()
378{
379 {
380 std::unique_lock lock(mExitStateMutex);
381 if (mDisplayRunning) {
382 mDisplay->mCfgC.displayControl.store(2);
383 StopDisplayA();
384 mDisplay->UnblockDisplay();
385 }
386 mDisplayStoppedCV.wait(lock, [this] { return !mDisplayRunning; });
387 }
388 if (mFrontendThread.joinable()) { // Also the wrapper has finished (it sets mInitResult = -1 for errors after the initialization), not holding mExitStateMutex, which the wrapper uses
389 mFrontendThread.join();
390 }
391 std::lock_guard lock(mExitStateMutex);
392 if (mDisplay->mInitResult == 1) { // For a new StartDisplay, not after a failed initialization or an error (undefined state)
393 mStartDisplayDone = mStartupScriptDone = false;
394 mDisplay->mInitResult = 0;
395 }
396}
