30static hmm_mat4 cameraViewMatrix(const float (&xyz)[3], const float (&angles)[3]) // Inverse of calcXYZ
31{
32 const float toDeg = 180.f / CAMath::Pi();
33 return HMM_Rotate(-angles[2] * toDeg, {{0, 0, 1}}) * HMM_Rotate(-angles[0] * toDeg, {{1, 0, 0}}) * HMM_Rotate(angles[1] * toDeg, {{0, 1, 0}}) * HMM_Translate({{-xyz[0], -xyz[1], -xyz[2]}});
34}

45void GPUDisplay::animationPointValues(const animationVector& p, int32_t mode, const float* last, float (&vals)[8])
46{
47 const float xyz[3] = {p.v[0], p.v[1], p.v[2]}, angles[3] = {p.v[3], p.v[4], p.v[5]};
48 if (mode & 4) // Spherical
49 {
50 float rxy = sqrtf(xyz[0] * xyz[0] + xyz[2] * xyz[2]);
51 float anglePhi = atan2f(xyz[0], xyz[2]);
52 float angleTheta = atan2f(xyz[1], rxy);
53 if (last) {
54 mAnimationCloseAngle(anglePhi, last[1]);
55 mAnimationCloseAngle(angleTheta, last[2]);
56 }
57 vals[0] = 0;
58 vals[1] = anglePhi;
59 vals[2] = angleTheta;
60 } else {
61 for (int32_t i = 0; i < 3; i++) {
62 vals[i] = xyz[i];
63 }
64 // Cartesian
65 }
66 vals[3] = sqrtf(xyz[0] * xyz[0] + xyz[1] * xyz[1] + xyz[2] * xyz[2]);
67 if (mode & 1) // Euler-angles
68 {
69 for (int32_t i = 0; i < 3; i++) {
70 float newangle = angles[i];
71 if (last) {
72 mAnimationCloseAngle(newangle, last[i + 4]);
73 }
74 vals[i + 4] = newangle;
75 }
76 vals[7] = 0;
77 } else { // Quaternions
78 float v[4];
79 const hmm_mat4 viewMatrix = cameraViewMatrix(xyz, angles);
80 createQuaternionFromMatrix(v, &viewMatrix.Elements[0][0]);
81 if (last) {
82 mAnimateCloseQuaternion(v, last[4], last[5], last[6], last[7]);
83 }
84 for (int32_t i = 0; i < 4; i++) {
85 vals[i + 4] = v[i];
86 }
87 }
88}

90static hmm_mat4 lookAtMatrix(const float (&eye)[3], const float (&target)[3]) // HMM_LookAt with y up, z up when looking along y (the cross product with y would be 0)
91{
92 const float d[3] = {target[0] - eye[0], target[1] - eye[1], target[2] - eye[2]};
93 const bool alongY = d[0] * d[0] + d[2] * d[2] <= 1e-10f * (d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
94 return HMM_LookAt({{eye[0], eye[1], eye[2]}}, {{target[0], target[1], target[2]}}, alongY ? hmm_vec3{{0, 0, 1}} : hmm_vec3{{0, 1, 0}});
95}

103bool GPUDisplay::lookAt(const float (&target)[3])
104{
105 if (target[0] == mXYZ[0] && target[1] == mXYZ[1] && target[2] == mXYZ[2]) {
106 return false;
107 }
108 *mViewMatrix = lookAtMatrix(mXYZ, target);
109 calcXYZ(&mViewMatrix->Elements[0][0]);
110 return true;

141void GPUDisplay::removeAnimationPoint(int32_t i)
142{
143 StopAnimation();
144 if (i < 0) {
145 i = mCfgC.animationPoints.selected >= 0 ? mCfgC.animationPoints.selected : (int32_t)mAnimationVectors.size() - 1;
146 }
147 if (i < 0 || i >= (int32_t)mAnimationVectors.size()) {
148 return;
149 }
150 auto& times = mCfgC.animationPoints.values;
151 mAnimationVectors.erase(mAnimationVectors.begin() + i);
152 times.erase(times.begin() + i);
153 if (i == 0 && times.size()) { // The first point stays at time 0
154 const float t0 = times[0];
155 for (float& t : times) {
156 t -= t0;
157 }
158 }
159 mCfgC.animationPoints.selected = -1;
160}

186void GPUDisplay::setAnimationPointsFromGUI(const GPUDisplayMultiSliderValue& points)
187{
188 if (mAnimate || points.values.size() != mAnimationVectors.size()) { // Not while the animation runs; the GUI may not yet know a changed number of points
189 return;
190 }
191 std::vector<uint32_t> order(points.values.size());
192 std::iota(order.begin(), order.end(), 0);
193 std::stable_sort(order.begin(), order.end(), [&](uint32_t a, uint32_t b) { return points.values[a] < points.values[b]; });
194 for (uint32_t i = 1; i < order.size(); i++) {
195 if (!(points.values[order[i]] > points.values[order[i - 1]])) { // Points must have different times, the GUI prevents this
196 return;
197 }
198 }
199 vecpod<animationVector> sorted(order.size());
200 vecpod<float> sortedTimes(order.size());
201 int32_t selected = -1;
202 for (uint32_t i = 0; i < order.size(); i++) {
203 sorted[i] = mAnimationVectors[order[i]];
204 sortedTimes[i] = points.values[order[i]] - points.values[order[0]]; // The first point at time 0
205 selected = (int32_t)order[i] == points.selected ? i : selected;
206 }
207 const bool selectionChanged = points.selected != mCfgC.animationPoints.selected;
208 mAnimationVectors = std::move(sorted);
209 mCfgC.animationPoints.values = std::move(sortedTimes);
210 if (selectionChanged) {
211 selectAnimationPoint(selected);
212 } else {
213 mCfgC.animationPoints.selected = selected;
214 }

237void GPUDisplay::startAnimation()
238{
239 mAnimationSplines.resize(ANIMATION_MODE_NAMES_SHORT.size());
240 vecpod<float> y[8];
241 for (auto& yi : y) {
242 yi.resize(mAnimationVectors.size());
243 }
244 for (uint32_t mode = 0; mode < mAnimationSplines.size(); mode++) { // Splines for all modes, points may have different modes
245 float vals[8], previous[8];
246 for (uint32_t k = 0; k < mAnimationVectors.size(); k++) {
247 animationPointValues(mAnimationVectors[k], mode, k ? previous : nullptr, vals); // Continuous to the previous point in time order
248 for (int32_t i = 0; i < 8; i++) {
249 y[i][k] = previous[i] = vals[i];
250 }
251 }
252 for (int32_t i = 0; i < 8; i++) {
253 mAnimationSplines[mode][i].create(mCfgC.animationPoints.values, y[i]);
254 }
255 }
256 mAnimationTimer.ResetStart();
257 mAnimationFrame = 0;
258 mAnimate = 1;
259 mCfgC.animationPoints.selected = -1; // The animation changes mCfgL, which would otherwise be stored in the selected point when it ends
260 mAnimationLastBase = 0;
261 mAnimationMixBase = -1;
262}

264int32_t GPUDisplay::animateCamera(bool suspended, float& animateTime, float& mixSlaveImage, hmm_mat4& nextViewMatrix)
265{
266 float time = animateTime;
267 if (mAnimate && time < 0) {
268 if (mCfgO.animateScreenshot) {
269 time = mAnimationFrame / 30.f;
270 } else {
271 time = mAnimationTimer.GetCurrentElapsedTime();
272 }
273
274 float maxTime = mCfgC.animationPoints.values.back();
275 mCfgC.animationPoints.progress = std::lround(std::clamp(time / maxTime, 0.f, 1.f) * 65535);
276 if (time >= maxTime) {
277 time = maxTime;
278 mAnimate = mCfgO.animateScreenshot = 0;
279 SetInfo("Animation finished. (%1.2f seconds, %d frames)", time, mAnimationFrame);
280 } else {
281 SetInfo("Running mAnimation: time %1.2f/%1.2f, frames %d", time, maxTime, mAnimationFrame);
282 }
283 mAnimationFrame++;
284 }
285 if (!mAnimate) {
286 return 0;
287 }
288 if (mCfgO.animationChangeConfig && animateTime < 0) {
289 int32_t base = 0;
290 const auto& times = mCfgC.animationPoints.values;
291 int32_t k = mAnimationVectors.size() - 1;
292 while (base < k && time > times[base]) {
293 base++;
294 }
295 if (base > mAnimationLastBase + 1) {
296 mAnimationLastBase = base - 1;
297 }
298
299 const bool mixing = base != mAnimationLastBase && times[mAnimationLastBase] != times[base] && memcmp(&mAnimationVectors[base].cfgL, &mAnimationVectors[mAnimationLastBase].cfgL, sizeof(mCfgL));
300 if ((mixing ? base : -1) != mAnimationMixBase) { // Blended configs changed: cached (Vulkan) command buffers must be re-recorded, otherwise they are kept while mixing
301 mAnimationMixBase = mixing ? base : -1;
302 mUpdateDrawCommands = true;
303 }
304 if (mixing) {
305 mixSlaveImage = 1.f - (time - times[mAnimationLastBase]) / (times[base] - times[mAnimationLastBase]);
306 if (mixSlaveImage > 0) {
307 mCfgL = mAnimationVectors[mAnimationLastBase].cfgL;
308 updateConfig();
309 DrawGLScene_internal(suspended, time, true);
310 }
311 }
312
313 if (memcmp(&mAnimationVectors[base].cfgL, &mCfgL, sizeof(mCfgL))) {
314 mCfgL = mAnimationVectors[base].cfgL;
315 updateConfig();
316 mUpdateDrawCommands |= !mixing; // mOldCfg is taken after drawing, so checkAndApplyConfigChanges does not see this change: cached (Vulkan) command buffers would keep the previous settings; while mixing, a change of mAnimationMixBase re-records them
317 }
318 }
319
320 const auto& times = mCfgC.animationPoints.values;
321 uint32_t k = 0; // Interval [times[k], times[k + 1]] of the time
322 while (k + 2 < times.size() && time > times[k + 1]) {
323 k++;
324 }
325 const int32_t maxMode = mAnimationSplines.size() - 1; // Modes from a loaded file are not checked
326 const int32_t modeA = std::clamp(mAnimationVectors[k].cfgL.animationMode, 0, maxMode), modeB = std::clamp(mAnimationVectors[k + 1].cfgL.animationMode, 0, maxMode);
327 auto cameraOfMode = [&](int32_t mode, hmm_mat4& viewMatrix) {
328 float vals[8];
329 for (int32_t i = 0; i < 8; i++) {
330 vals[i] = mAnimationSplines[mode][i].evaluate(time);
331 }
332 animationViewMatrix(vals, mode, viewMatrix);
333 };
334 if (modeA == modeB) {
335 cameraOfMode(modeA, nextViewMatrix);
336 } else { // Modes of the points differ: blend the cameras of both modes, position linearly, rotation as quaternion, with sin^2 for a smooth start and end
337 hmm_mat4 viewA = nextViewMatrix, viewB = nextViewMatrix;
338 cameraOfMode(modeA, viewA);
339 cameraOfMode(modeB, viewB);
340 const float s = sinf(std::clamp((time - times[k]) / (times[k + 1] - times[k]), 0.f, 1.f) * CAMath::Pi() / 2), w = s * s;
341 float qA[4], qB[4], vals[8];
342 createQuaternionFromMatrix(qA, &viewA.Elements[0][0]);
343 createQuaternionFromMatrix(qB, &viewB.Elements[0][0]);
344 const float sign = qA[0] * qB[0] + qA[1] * qB[1] + qA[2] * qB[2] + qA[3] * qB[3] < 0 ? -1.f : 1.f; // Shorter way
345 auto position = [](const hmm_mat4& m, int32_t i) { return -(m.Elements[i][0] * m.Elements[3][0] + m.Elements[i][1] * m.Elements[3][1] + m.Elements[i][2] * m.Elements[3][2]); }; // Camera position, as calcXYZ
346 for (int32_t i = 0; i < 3; i++) {
347 vals[i] = (1 - w) * position(viewA, i) + w * position(viewB, i);
348 }
349 vals[3] = 0;
350 for (int32_t i = 0; i < 4; i++) {
351 vals[4 + i] = (1 - w) * qA[i] + w * sign * qB[i]; // Normalized by animationViewMatrix
352 }
353 animationViewMatrix(vals, 0, nextViewMatrix); // Mode 0: Cartesian position, quaternion
354 }
355 return 1;
356}

358void GPUDisplay::animationViewMatrix(float (&vals)[8], int32_t mode, hmm_mat4& nextViewMatrix) const
359{
360 if (mode != 6) {
361 if (mode & 1) // Rotation from euler angles, same order as setCamera (inverse of calcXYZ)
362 {
363 nextViewMatrix = nextViewMatrix * HMM_Rotate(-vals[6] * 180.f / CAMath::Pi(), {{0, 0, 1}}) * HMM_Rotate(-vals[4] * 180.f / CAMath::Pi(), {{1, 0, 0}}) * HMM_Rotate(vals[5] * 180.f / CAMath::Pi(), {{0, 1, 0}});
364 } else { // Rotation from quaternion
365 const float mag = sqrtf(vals[4] * vals[4] + vals[5] * vals[5] + vals[6] * vals[6] + vals[7] * vals[7]);
366 if (mag < 0.0001f) {
367 vals[7] = 1;
368 } else {
369 for (int32_t i = 0; i < 4; i++) {
370 vals[4 + i] /= mag;
371 }
372 }
373
374 float xx = vals[4] * vals[4], xy = vals[4] * vals[5], xz = vals[4] * vals[6], xw = vals[4] * vals[7], yy = vals[5] * vals[5], yz = vals[5] * vals[6], yw = vals[5] * vals[7], zz = vals[6] * vals[6], zw = vals[6] * vals[7];
375 float mat[16] = {1 - 2 * (yy + zz), 2 * (xy - zw), 2 * (xz + yw), 0, 2 * (xy + zw), 1 - 2 * (xx + zz), 2 * (yz - xw), 0, 2 * (xz - yw), 2 * (yz + xw), 1 - 2 * (xx + yy), 0, 0, 0, 0, 1};
376 nextViewMatrix = nextViewMatrix * MY_HMM_FROM(mat);
377 }
378 }
379 if (mode & 4) // Compute cartesian translation from sperical coordinates (euler angles)
380 {
381 const float r = vals[3], phi = vals[1], theta = vals[2];
382 vals[2] = r * cosf(phi) * cosf(theta);
383 vals[0] = r * sinf(phi) * cosf(theta);
384 vals[1] = r * sinf(theta);
385 } else if (mode & 2) { // Scale cartesion translation to interpolated radius
386 float r = sqrtf(vals[0] * vals[0] + vals[1] * vals[1] + vals[2] * vals[2]);
387 if (fabsf(r) < 0.0001f) {
388 r = 1;
389 }
390 r = vals[3] / r;
391 for (int32_t i = 0; i < 3; i++) {
392 vals[i] *= r;
393 }
394 }
395 if (mode == 6) {
396 nextViewMatrix = lookAtMatrix({vals[0], vals[1], vals[2]}, {0.f, 0.f, 0.f});
397 } else {
398 nextViewMatrix = nextViewMatrix * HMM_Translate({{-vals[0], -vals[1], -vals[2]}});
399 }
