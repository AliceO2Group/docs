52struct TrackerACTSConfig {
53
54 TrackerACTSConfig()
55 {
56 std::ifstream input(configName);
57 if (!input) {
58 return;
59 }
60 LOG(info) << "Loading ACTS tracker configuration from " << configName;
61 nlohmann::json config;
62 try {
63 input >> config;
64 } catch (const nlohmann::json::parse_error& e) {
65 LOG(error) << "Failed to parse ACTS tracker configuration file " << configName << ": " << e.what();
66 return;
67 }
68 if (!config.is_object()) {
69 LOG(error) << "ACTS tracker configuration file " << configName << " must contain a JSON object";
70 return;
71 }
72
73 const auto load = [&config](const char* key, auto& value) {
74 if (!config.contains(key)) {
75 return;
76 }
77 try {
78 config.at(key).get_to(value);
79 } catch (const nlohmann::json::exception& e) {
80 LOG(error) << "Failed to load ACTS tracker configuration parameter " << key << ": " << e.what();
81 }
82 };
83
84 load("minPt", minPt);
85 load("maxImpactParameter", maxImpactParameter);
86 load("cotThetaMax", cotThetaMax);
87 load("deltaRMinBottom", deltaRMinBottom);
88 load("deltaRMaxBottom", deltaRMaxBottom);
89 load("deltaRMinTop", deltaRMinTop);
90 load("deltaRMaxTop", deltaRMaxTop);
91 load("zMin", zMin);
92 load("zMax", zMax);
93 load("collisionRegionMin", collisionRegionMin);
94 load("collisionRegionMax", collisionRegionMax);
95 load("maxSeedsPerMiddleSP", maxSeedsPerMiddleSP);
96 load("deltaPhiMax", deltaPhiMax);
97 load("writeDebugTrees", writeDebugTrees);
98 load("dumpTrackingGeometry", dumpTrackingGeometry);
99 load("debugMaxTimeFrames", debugMaxTimeFrames);
100 load("debugFileName", debugFileName);
101 load("geometryDumpFileName", geometryDumpFileName);
102 }
103
104 ~TrackerACTSConfig()
105 {
106 const nlohmann::json config = {
107 {"minPt", minPt},
108 {"maxImpactParameter", maxImpactParameter},
109 {"cotThetaMax", cotThetaMax},
110 {"deltaRMinBottom", deltaRMinBottom},
111 {"deltaRMaxBottom", deltaRMaxBottom},
112 {"deltaRMinTop", deltaRMinTop},
113 {"deltaRMaxTop", deltaRMaxTop},
114 {"zMin", zMin},
115 {"zMax", zMax},
116 {"collisionRegionMin", collisionRegionMin},
117 {"collisionRegionMax", collisionRegionMax},
118 {"maxSeedsPerMiddleSP", maxSeedsPerMiddleSP},
119 {"deltaPhiMax", deltaPhiMax},
120 {"writeDebugTrees", writeDebugTrees},
121 {"dumpTrackingGeometry", dumpTrackingGeometry},
122 {"debugMaxTimeFrames", debugMaxTimeFrames},
123 {"debugFileName", debugFileName},
124 {"geometryDumpFileName", geometryDumpFileName},
125 };
126 std::ofstream output(configName);
127 if (!output) {
128 LOG(error) << "Failed to open ACTS tracker configuration file " << configName;
129 return;
130 }
131
132 output << config.dump(2) << '\n';
133 if (!output) {
134 LOG(error) << "Failed to write ACTS tracker configuration file " << configName;
135 }
136 LOG(info) << "Saved ACTS tracker configuration to " << configName;
137 }
138
140 float minPt = 0.4f * Acts::UnitConstants::GeV;
142 float maxImpactParameter = 10.f * Acts::UnitConstants::mm;
144 float cotThetaMax = std::sinh(4.0f);
145
147 float deltaRMinBottom = 5.f * Acts::UnitConstants::mm;
148 float deltaRMaxBottom = 200.f * Acts::UnitConstants::mm;
150 float deltaRMinTop = 5.f * Acts::UnitConstants::mm;
151 float deltaRMaxTop = 200.f * Acts::UnitConstants::mm;
152
154 float zMin = -3000.f * Acts::UnitConstants::mm;
155 float zMax = 3000.f * Acts::UnitConstants::mm;
156
158 float collisionRegionMin = -150.f * Acts::UnitConstants::mm;
159 float collisionRegionMax = 150.f * Acts::UnitConstants::mm;
160
162 float maxSeedsPerMiddleSP = 2;
164 float deltaPhiMax = 0.1f;
165
167 bool writeDebugTrees = false;
169 bool dumpTrackingGeometry = false;
171 int debugMaxTimeFrames = 1;
173 std::string debugFileName = "TrackerACTS_debug.root";
175 std::string geometryDumpFileName = "TrackerACTS_geometry.json";
176
177 static constexpr const char* configName = "TrackerACTSConfig.json";
178};

343template <int nLayers>
344template <typename Func>
345float TrackerACTS<nLayers>::evaluateTask(Func&& task, std::string_view taskName)
346{
347 LOG(debug) << " + Starting " << taskName;
348 const auto start = std::chrono::high_resolution_clock::now();
349 task();
350 const auto end = std::chrono::high_resolution_clock::now();
351 std::chrono::duration<double, std::milli> diff{end - start};
352
353 LOG(debug) << " - " << taskName << " completed in: " << std::fixed << std::setprecision(2) << diff.count() << " ms";
354 return static_cast<float>(diff.count());
355}
