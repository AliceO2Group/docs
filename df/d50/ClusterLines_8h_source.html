49 GPUhdi() static float getDistance2FromPoint(const Line& line, const float point[3])
50 {
51 float delta[3], proj = 0.f;
52 for (int i = 0; i < 3; ++i) {
53 delta[i] = point[i] - line.originPoint[i];
54 proj += delta[i] * line.cosinesDirector[i];
55 }
56 float d2 = 0.f;
57 for (int i = 0; i < 3; ++i) {
58 const float residual = delta[i] - proj * line.cosinesDirector[i];
59 d2 += residual * residual;
60 }
61 return d2;
62 }

65 GPUhdi() static void getDCAComponents(const Line& line, const float point[3], float out[6])
66 {
67 float delta[3], proj = 0.f;
68 for (int i = 0; i < 3; ++i) {
69 delta[i] = line.originPoint[i] - point[i];
70 proj += delta[i] * line.cosinesDirector[i];
71 }
72 float r[3];
73 for (int i = 0; i < 3; ++i) {
74 r[i] = delta[i] - proj * line.cosinesDirector[i];
75 }
76 out[0] = r[0];
77 out[2] = r[1];
78 out[5] = r[2];
79#ifndef GPUCA_GPUCODE
80 out[1] = std::hypot(r[0], r[1]);
81 out[3] = std::hypot(r[0], r[2]);
82 out[4] = std::hypot(r[1], r[2]);
83#else
84 out[1] = o2::gpu::GPUCommonMath::Hypot(r[0], r[1]);
85 out[3] = o2::gpu::GPUCommonMath::Hypot(r[0], r[2]);
86 out[4] = o2::gpu::GPUCommonMath::Hypot(r[1], r[2]);
87#endif
88 }

90 GPUhdi() static bool getZAtBeam(const Line& line, const float beamX, const float beamY, float& z)
91 {
92 const float ux = line.cosinesDirector[0];
93 const float uy = line.cosinesDirector[1];
94 const float den = ux * ux + uy * uy;
95 if (den <= constants::Tolerance) {
96 return false;
97 }
98 const float dx = line.originPoint[0] - beamX;
99 const float dy = line.originPoint[1] - beamY;
100 const float s0 = -(dx * ux + dy * uy) / den;
101 z = line.originPoint[2] + s0 * line.cosinesDirector[2];
102 return true;
103 }

133 GPUhdi() void add(const Line& line)
134 {
135 const double d0 = line.cosinesDirector[0], d1 = line.cosinesDirector[1], d2 = line.cosinesDirector[2];
136 const double o0 = line.originPoint[0], o1 = line.originPoint[1], o2 = line.originPoint[2];
137 const double det = d0 * d0 + d1 * d1 + d2 * d2; // == 1 for a normalised director
138 if (det <= 0.) {
139 return;
140 }
141 if (mNContributors <= 0) {
142 mTime = line.mTime;
143 } else {
144 mTime += line.mTime;
145 }
146 mA[0] += (det - d0 * d0) / det;
147 mA[1] += (-d0 * d1) / det;
148 mA[2] += (-d0 * d2) / det;
149 mA[3] += (det - d1 * d1) / det;
150 mA[4] += (-d1 * d2) / det;
151 mA[5] += (det - d2 * d2) / det;
152 const double dDotO = d0 * o0 + d1 * o1 + d2 * o2;
153 mB[0] += (d0 * dDotO - det * o0) / det;
154 mB[1] += (d1 * dDotO - det * o1) / det;
155 mB[2] += (d2 * dDotO - det * o2) / det;
156 ++mNContributors;
157 }
158
160 GPUhdi() bool solve(float vertex[3]) const
161 {
162 const double a = mA[0], b = mA[1], c = mA[2], d = mA[3], e = mA[4], f = mA[5];
163 const double c00 = d * f - e * e;
164 const double c01 = c * e - b * f;
165 const double c02 = b * e - c * d;
166 const double c11 = a * f - c * c;
167 const double c12 = b * c - a * e;
168 const double c22 = a * d - b * b;
169 const double det = a * c00 + b * c01 + c * c02;
170 if (o2::gpu::GPUCommonMath::Abs(det) < 1.e-12) {
171 return false;
172 }
173 const double invDet = 1. / det;
174 const double x0 = (c00 * mB[0] + c01 * mB[1] + c02 * mB[2]) * invDet;
175 const double x1 = (c01 * mB[0] + c11 * mB[1] + c12 * mB[2]) * invDet;
176 const double x2 = (c02 * mB[0] + c12 * mB[1] + c22 * mB[2]) * invDet;
177 vertex[0] = static_cast<float>(-x0);
178 vertex[1] = static_cast<float>(-x1);
179 vertex[2] = static_cast<float>(-x2);
180 return true;
181 }

190 GPUhdi() void addResidual(const Line& line, const float vertex[3])
191 {
192 float dca[6];
193 Line::getDCAComponents(line, vertex, dca);
194 const float d2 = Line::getDistance2FromPoint(line, vertex);
195 ++mResidualCount;
196 const float inv = 1.f / static_cast<float>(mResidualCount);
197 for (int i = 0; i < 6; ++i) {
198 mRMS2[i] += (dca[i] - mRMS2[i]) * inv;
199 }
200 mAvgDistance2 += (d2 - mAvgDistance2) / static_cast<float>(mResidualCount);
201 }
