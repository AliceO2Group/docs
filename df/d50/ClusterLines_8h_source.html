52 GPUhdi() static float getDistance2FromPoint(const Line& line, const float point[3])
53 {
54 float delta[3], proj = 0.f;
55 for (int i = 0; i < 3; ++i) {
56 delta[i] = point[i] - line.originPoint[i];
57 proj += delta[i] * line.cosinesDirector[i];
58 }
59 float d2 = 0.f;
60 for (int i = 0; i < 3; ++i) {
61 const float residual = delta[i] - proj * line.cosinesDirector[i];
62 d2 += residual * residual;
63 }
64 return d2;
65 }

68 GPUhdi() static void getDCAComponents(const Line& line, const float point[3], float out[6])
69 {
70 float delta[3], proj = 0.f;
71 for (int i = 0; i < 3; ++i) {
72 delta[i] = line.originPoint[i] - point[i];
73 proj += delta[i] * line.cosinesDirector[i];
74 }
75 float r[3];
76 for (int i = 0; i < 3; ++i) {
77 r[i] = delta[i] - proj * line.cosinesDirector[i];
78 }
79 out[0] = r[0];
80 out[2] = r[1];
81 out[5] = r[2];
82#ifndef GPUCA_GPUCODE
83 out[1] = std::hypot(r[0], r[1]);
84 out[3] = std::hypot(r[0], r[2]);
85 out[4] = std::hypot(r[1], r[2]);
86#else
87 out[1] = o2::gpu::GPUCommonMath::Hypot(r[0], r[1]);
88 out[3] = o2::gpu::GPUCommonMath::Hypot(r[0], r[2]);
89 out[4] = o2::gpu::GPUCommonMath::Hypot(r[1], r[2]);
90#endif
91 }

93 GPUhdi() static bool getZAtBeam(const Line& line, const float beamX, const float beamY, float& z)
94 {
95 const float ux = line.cosinesDirector[0];
96 const float uy = line.cosinesDirector[1];
97 const float den = ux * ux + uy * uy;
98 if (den <= constants::Tolerance) {
99 return false;
100 }
101 const float dx = line.originPoint[0] - beamX;
102 const float dy = line.originPoint[1] - beamY;
103 const float s0 = -(dx * ux + dy * uy) / den;
104 z = line.originPoint[2] + s0 * line.cosinesDirector[2];
105 return true;
106 }

127GPUdi() bool getCellLineAtBeam(const CellSeed& cell, const float beamX, const float beamY, const float bz, const float curvatureScale, std::array<float, 3>& origin, std::array<float, 3>& direction)
128{
129 cell.getXYZGlo(origin);
130 if (!cell.getPxPyPzGlo(direction)) {
131 return false;
132 }
133 o2::track::TrackParametrization<float> par{cell};
134 par.setQ2Pt(par.getQ2Pt() * curvatureScale);
135 std::array<float, 3> dcaOrigin, dcaDirection;
136 if (par.propagateParamToDCA({beamX, beamY, 0.f}, bz)) {
137 par.getXYZGlo(dcaOrigin);
138 if (par.getPxPyPzGlo(dcaDirection)) {
139 origin = dcaOrigin;
140 direction = dcaDirection;
141 }
142 }
143 return true;

148class LineFit
149{
150 public:
152 GPUhdi() bool add(const Line& line)
153 {
154 const double d0 = line.cosinesDirector[0], d1 = line.cosinesDirector[1], d2 = line.cosinesDirector[2];
155 const double o0 = line.originPoint[0], o1 = line.originPoint[1], o2 = line.originPoint[2];
156 const double det = d0 * d0 + d1 * d1 + d2 * d2; // == 1 for a normalised director
157 if (det <= 0.) {
158 return false;
159 }
160 mA[0] += (det - d0 * d0) / det;
161 mA[1] += (-d0 * d1) / det;
162 mA[2] += (-d0 * d2) / det;
163 mA[3] += (det - d1 * d1) / det;
164 mA[4] += (-d1 * d2) / det;
165 mA[5] += (det - d2 * d2) / det;
166 const double dDotO = d0 * o0 + d1 * o1 + d2 * o2;
167 mB[0] += (d0 * dDotO - det * o0) / det;
168 mB[1] += (d1 * dDotO - det * o1) / det;
169 mB[2] += (d2 * dDotO - det * o2) / det;
170 return true;
171 }
172
174 GPUhdi() bool solve(float vertex[3]) const
175 {
176 const double c00 = mA[3] * mA[5] - mA[4] * mA[4];
177 const double c01 = mA[2] * mA[4] - mA[1] * mA[5];
178 const double c02 = mA[1] * mA[4] - mA[2] * mA[3];
179 const double c11 = mA[0] * mA[5] - mA[2] * mA[2];
180 const double c12 = mA[1] * mA[2] - mA[0] * mA[4];
181 const double c22 = mA[0] * mA[3] - mA[1] * mA[1];
182 const double det = mA[0] * c00 + mA[1] * c01 + mA[2] * c02;
183 if (o2::gpu::GPUCommonMath::Abs(det) < 1.e-12) {
184 return false;
185 }
186 const double invDet = 1. / det;
187 const double x0 = (c00 * mB[0] + c01 * mB[1] + c02 * mB[2]) * invDet;
188 const double x1 = (c01 * mB[0] + c11 * mB[1] + c12 * mB[2]) * invDet;
189 const double x2 = (c02 * mB[0] + c12 * mB[1] + c22 * mB[2]) * invDet;
190 vertex[0] = static_cast<float>(-x0);
191 vertex[1] = static_cast<float>(-x1);
192 vertex[2] = static_cast<float>(-x2);
193 return true;
194 }
195
197 GPUhdi() void computeClusterCentroid() { mIsValid = solve(mVertex); }

199 GPUhdi() bool isValid() const noexcept { return mIsValid; }
200 GPUhdi() const float* getVertex() const noexcept { return mVertex; }
201 GPUhdi() int getSize() const noexcept { return mNContributors; }
202 GPUhdi() int getNContributors() const noexcept { return mNContributors; }
203 GPUhdi() const TimeEstBC& getTimeStamp() const noexcept { return mTime; }
204 GPUhdi() float getR2() const noexcept { return (mVertex[0] * mVertex[0]) + (mVertex[1] * mVertex[1]); }
205 GPUhdi() float getR() const noexcept { return o2::gpu::GPUCommonMath::Sqrt(getR2()); }
206 GPUhdi() const float* getRMS2() const noexcept { return mRMS2; } // {XX, XY, YY, XZ, YZ, ZZ}
207 GPUhdi() float getAvgDistance2() const noexcept { return mAvgDistance2; }
208
209 protected:
210 double mA[6] = {0., 0., 0., 0., 0., 0.}; // AX=B, packed symmetric normal matrix
211 double mB[3] = {0., 0., 0.}; // AX=B, right-hand side
212 float mVertex[3] = {0.f, 0.f, 0.f}; // cluster centroid position
213 float mRMS2[6] = {0.f, 0.f, 0.f, 0.f, 0.f, 0.f}; // mean DCA components about the vertex
214 float mAvgDistance2 = 0.f; // mean squared distance about the vertex
215 int mNContributors = 0; // lines accumulated into the normal equations
216 bool mIsValid = false; // true if the linear system was solved successfully
217 TimeEstBC mTime; // intersection of the contributors' time intervals
218};

229 GPUhdi() void add(const Line& line)
230 {
231 if (!LineFit::add(line)) {
232 return;
233 }
234 if (mNContributors <= 0) {
235 mTime = line.mTime;
236 } else {
237 mTime += line.mTime;
238 }
239 ++mNContributors;
240 }
241
243 GPUhdi() void addResidual(const Line& line) { addResidual(line, mVertex); }
244
246 GPUhdi() void addResidual(const Line& line, const float vertex[3])
247 {
248 float dca[6];
249 Line::getDCAComponents(line, vertex, dca);
250 const float d2 = Line::getDistance2FromPoint(line, vertex);
251 ++mResidualCount;
252 const float inv = 1.f / static_cast<float>(mResidualCount);
253 for (int i = 0; i < 6; ++i) {
254 mRMS2[i] += (dca[i] - mRMS2[i]) * inv;
255 }
256 mAvgDistance2 += (d2 - mAvgDistance2) / static_cast<float>(mResidualCount);
257 }
258
259 bool operator==(const ClusterLines& rhs) const noexcept;
260
261 private:
262 int mResidualCount = 0; // lines folded into the running means
263};

280 template <int Width>
281 GPUdi() void reduce()
282 {
283 for (int i = 0; i < 6; ++i) {
284 mA[i] = warp_reduce_add<Width>(mA[i]);
285 }
286 for (int i = 0; i < 3; ++i) {
287 mB[i] = warp_reduce_add<Width>(mB[i]);
288 }
289 tLo = warp_reduce_max<Width>(tLo);
290 tHi = warp_reduce_min<Width>(tHi);
291 mNContributors = warp_reduce_add<Width>(mNContributors);
292 mTime = TimeEstBC{tLo, tHi - tLo};
293 }

295 GPUdi() void addResidual(const Line& l, const float vertex[3])
296 {
297 float dca[6];
298 Line::getDCAComponents(l, vertex, dca);
299 for (int i = 0; i < 6; ++i) {
300 rms2Sum[i] += dca[i];
301 }
302 dist2Sum += Line::getDistance2FromPoint(l, vertex);
303 }
304
306 template <int Width>
307 GPUdi() void reduceResiduals(const int n)
308 {
309 for (int i = 0; i < 6; ++i) {
310 mRMS2[i] = warp_reduce_add<Width>(rms2Sum[i]) / n;
311 }
312 mAvgDistance2 = warp_reduce_add<Width>(dist2Sum) / n;
313 }
314
315 double rms2Sum[6] = {0., 0., 0., 0., 0., 0.};
316 double dist2Sum = 0.;
