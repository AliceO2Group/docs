49{

145void GPUDisplayFrontendGlut::ResizeSceneWrapper(int32_t width, int32_t height)
146{
147 if (!mFullScreen) {
148 mWidth = width;
149 mHeight = height;
150 }
151 ResizeScene(width, height);
152#ifdef GPUCA_BUILD_EVENT_DISPLAY_X11
153 Display* display = glXGetCurrentDisplay(); // Maximizing / unmaximizing by the window manager changes the size, read the state as freeglut does not report it (fullscreen stays as requested, freeglut tracks it itself)
154 Atom type;
155 int32_t format;
156 unsigned long nItems, bytesAfter;
157 unsigned char* data = nullptr;
158 if (!display || XGetWindowProperty(display, glXGetCurrentDrawable(), XInternAtom(display, "_NET_WM_STATE", False), 0, 1024, False, XA_ATOM, &type, &format, &nItems, &bytesAfter, &data) != Success || data == nullptr) {
159 return;
160 }
161 const Atom maxHorz = XInternAtom(display, "_NET_WM_STATE_MAXIMIZED_HORZ", False), maxVert = XInternAtom(display, "_NET_WM_STATE_MAXIMIZED_VERT", False);
162 bool isMaxHorz = false, isMaxVert = false;
163 for (unsigned long i = 0; i < nItems; i++) {
164 isMaxHorz |= ((Atom*)data)[i] == maxHorz;
165 isMaxVert |= ((Atom*)data)[i] == maxVert;
166 }
167 XFree(data);
168 setWindowState(mFullScreen, isMaxHorz && isMaxVert);
169#endif
170}

172void GPUDisplayFrontendGlut::mouseFunc(int32_t button, int32_t state, int32_t x, int32_t y)
173{
174 if (button == 3) {
175 mMouseWheel += 100;
176 } else if (button == 4) {
177 mMouseWheel -= 100;
178 } else if (state == GLUT_DOWN) {
179 if (button == GLUT_LEFT_BUTTON) {
180 mMouseDn = true;
181 } else if (button == GLUT_RIGHT_BUTTON) {
182 mMouseDnR = true;
183 } else if (button == GLUT_MIDDLE_BUTTON) {
184 mMouseDnM = true;
185 }
186 mMouseDnX = x;
187 mMouseDnY = y;
188 } else if (state == GLUT_UP) {
189 if (button == GLUT_LEFT_BUTTON) {
190 mMouseDn = false;
191 } else if (button == GLUT_RIGHT_BUTTON) {
192 mMouseDnR = false;
193 } else if (button == GLUT_MIDDLE_BUTTON) {
194 mMouseDnM = false;
195 }
196 }

207int32_t GPUDisplayFrontendGlut::FrontendMain()
208{
209 if (backend()->backendType() != GPUDisplayBackend::TYPE_OPENGL) {
210 fprintf(stderr, "Only OpenGL backend supported\n");
211 return 1;
212 }
213 if (drawTextFontSize() == 0) {
214 drawTextFontSize() = 12;
215 }
216
217 int32_t nopts = 2;
218 char opt1[] = "progname";
219 char opt2[] = "-direct";
220 char* opts[] = {opt1, opt2};
221 glutInit(&nopts, opts);
222 glutInitContextVersion(GL_MIN_VERSION_MAJOR, GL_MIN_VERSION_MINOR);
223 glutInitContextProfile(GLUT_CORE_PROFILE);
224 glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB);
225 glutInitWindowSize(INIT_WIDTH, INIT_HEIGHT);
226 glutCreateWindow(DISPLAY_WINDOW_NAME);
227 glutSetOption(GLUT_ACTION_ON_WINDOW_CLOSE, GLUT_ACTION_GLUTMAINLOOP_RETURNS);
228
229 if (mBackend->ExtInit()) {
230 fprintf(stderr, "Error initializing GL extension wrapper\n");
231 return (-1);
232 }
233 auto mouseMoveFunc = [](int32_t x, int32_t y, void* p) { self(p)->mouseMoveFunc(x, y); };
234 glutDisplayFuncUcall([](void* p) { self(p)->displayFunc(); }, this); // Callbacks with this as user data, freeglut >= 3.2
235 glutIdleFuncUcall([](void* p) { self(p)->glutLoopFunc(); }, this);
236 glutReshapeFuncUcall([](int32_t width, int32_t height, void* p) { self(p)->ResizeSceneWrapper(width, height); }, this);
237 glutKeyboardFuncUcall([](uint8_t key, int32_t x, int32_t y, void* p) { self(p)->keyboardFunc(key, false, true); }, this);
238 glutKeyboardUpFuncUcall([](uint8_t key, int32_t x, int32_t y, void* p) { self(p)->keyboardFunc(key, false, false); }, this);
239 glutSpecialFuncUcall([](int32_t key, int32_t x, int32_t y, void* p) { self(p)->keyboardFunc(key, true, true); }, this);
240 glutSpecialUpFuncUcall([](int32_t key, int32_t x, int32_t y, void* p) { self(p)->keyboardFunc(key, true, false); }, this);
241 glutMouseFuncUcall([](int32_t button, int32_t state, int32_t x, int32_t y, void* p) { self(p)->mouseFunc(button, state, x, y); }, this);
242 glutMotionFuncUcall(mouseMoveFunc, this);
243 glutPassiveMotionFuncUcall(mouseMoveFunc, this);
244 glutMouseWheelFuncUcall([](int32_t button, int32_t dir, int32_t x, int32_t y, void* p) { self(p)->mMouseWheelFunc(button, dir, x, y); }, this);
245 applyInitialWindowState();
246 SetVSync(vsyncEnabled());
247
248 if (InitDisplay()) {
249 fprintf(stderr, "Error in OpenGL initialization\n");
250 return (1);
251 }
252 glutMainLoop();
253 {
254 std::lock_guard lock(mExitStateMutex);
255 setDisplayControl(2);
256 mDisplayRunning = false;
257 mDisplayStoppedCV.notify_all();
258 }
259
260 ExitDisplay();
261 return 0;
262}

279void GPUDisplayFrontendGlut::ToggleMaximized(bool set) // GLUT has no API for maximizing, request it from the window manager via GLX / X11 directly (no-op if freeglut does not use GLX)
280{
281#ifdef GPUCA_BUILD_EVENT_DISPLAY_X11
282 Display* display = glXGetCurrentDisplay();
283 if (!display) {
284 return;
285 }
286 XEvent xev;
287 memset(&xev, 0, sizeof(xev));
288 xev.type = ClientMessage;
289 xev.xclient.window = glXGetCurrentDrawable(); // freeglut renders directly to the X window
290 xev.xclient.message_type = XInternAtom(display, "_NET_WM_STATE", False);
291 xev.xclient.format = 32;
292 xev.xclient.data.l[0] = set ? 1 : 0; // _NET_WM_STATE_ADD / _NET_WM_STATE_REMOVE
293 xev.xclient.data.l[1] = XInternAtom(display, "_NET_WM_STATE_MAXIMIZED_HORZ", False);
294 xev.xclient.data.l[2] = XInternAtom(display, "_NET_WM_STATE_MAXIMIZED_VERT", False);
295 XSendEvent(display, DefaultRootWindow(display), False, SubstructureNotifyMask | SubstructureRedirectMask, &xev);
296 XFlush(display);
297#endif
298}
