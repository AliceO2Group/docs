74 static inline void qProcessSetting(qConfigSettings<T>& settings, qmin_t<T> minval)
75 {
76 static_assert(!std::is_same_v<T, bool>, "min option not supported for boolean settings");
77 settings.checkMin = true;
78 settings.min = minval.v;
79 }
80 static inline void qProcessSetting(qConfigSettings<T>& settings, qmax_t<T> maxval)
81 {
82 static_assert(!std::is_same_v<T, bool>, "max option not supported for boolean settings");
83 settings.checkMax = true;
84 settings.max = maxval.v;
85 }
86 static inline void qProcessSetting(qConfigSettings<T>& settings, qmessage_t msg) { settings.message = msg.v; }
87 template <typename S>
88 static inline void qProcessSetting(qConfigSettings<T>& settings, qset_t<S> set)
89 {
90 settings.doSet = true;
91 settings.set = set.v;
92 }
93 template <typename S>
94 static inline void qProcessSetting(qConfigSettings<T>& settings, qdef_t<S> set)
95 {
96 settings.doDefault = true;

100 static inline void qAddOptionSettings(qConfigSettings<T>& /*settings*/) {}
101 template <typename Arg1, typename... Args>
102 static inline void qAddOptionSettings(qConfigSettings<T>& settings, Arg1&& arg1, Args&&... args)
103 {
104 qProcessSetting(settings, arg1);
105 qAddOptionSettings(settings, args...);
