44bool hasO2GeometryObject(std::string_view fileName)
45{
46 TFile file(std::string(fileName).c_str());
47 if (file.IsZombie()) {
48 throw std::runtime_error("o2::acts::TrackingGeometryManager: cannot open '" + std::string(fileName) + "'");
49 }
50 return file.GetKey(std::string(o2::base::NameConf::CCDBOBJECT).c_str()) != nullptr ||

87void TrackingGeometryManager::build()
88{
89 if (isBuilt()) {
90 LOG(info) << "ACTS tracking geometry is already built, nothing to do";
91 return;
92 }
93 if (!o2::base::GeometryManager::isGeometryLoaded()) {
94 throw std::runtime_error(
95 "o2::acts::TrackingGeometryManager: no TGeo geometry is loaded. Declare "
96 "o2::base::GRPGeomRequest::Aligned in the workflow's GRPGeomRequest and call "
97 "GRPGeomHelper::instance().checkUpdates(pc) before asking for the ACTS geometry, or use "
98 "buildFromFile() outside a workflow.");
99 }
100 doBuild();
101}

103void TrackingGeometryManager::buildFromFile(std::string_view geomFileName)
104{
105 if (isBuilt()) {
106 LOG(info) << "ACTS tracking geometry is already built, nothing to do";
107 return;
108 }
109 if (o2::base::GeometryManager::isGeometryLoaded()) {
110 LOG(warn) << "A TGeo geometry is already loaded, ignoring the requested file and using it";
111 build();
112 return;
113 }
114
115 LOGP(info, "Loading TGeo geometry from '{}'", geomFileName);
116 if (hasO2GeometryObject(geomFileName)) {
117 o2::base::GeometryManager::loadGeometry(geomFileName);
118 } else {
119 // Not an o2-sim / CCDB geometry file. GeometryManager::loadGeometry() only
120 // accepts those two object names and aborts on anything else, so fall back to
121 // a plain import; this is the path for geometries exported from GDML, which
122 // the standalone actsO2 chain uses.
123 LOGP(warn, "'{}' holds no '{}' or '{}' object: importing it as a plain TGeo file",
124 geomFileName, o2::base::NameConf::CCDBOBJECT, o2::base::NameConf::GEOMOBJECTNAME_FAIR);
125 if (TGeoManager::Import(std::string(geomFileName).c_str()) == nullptr) {
126 throw std::runtime_error("o2::acts::TrackingGeometryManager: cannot import TGeo geometry from '" +
127 std::string(geomFileName) + "'");
128 }
129 }
130 build();
131}

133void TrackingGeometryManager::doBuild()
134{
135 if (mBuilder == nullptr) {
136 throw std::runtime_error(
137 "o2::acts::TrackingGeometryManager: no builder installed. Call setBuilder() with the "
138 "detector-specific ITrackingGeometryBuilder before requesting the geometry.");
139 }
140
141 LOGP(info, "Building the ACTS tracking geometry with the '{}' builder", mBuilder->getName());
142 auto output = mBuilder->build(*gGeoManager, mNominalContext);
143 if (output.geometry == nullptr) {
144 throw std::runtime_error(std::string("o2::acts::TrackingGeometryManager: builder '") +
145 mBuilder->getName() + "' returned no geometry");
146 }
147
148 mElementStore = std::move(output.elementStore);
149 mIndex = std::move(output.index);
150
151 if (!mMaterialMapFile.empty()) {
152 // Must happen while the geometry is still non-const: Gen3 construction takes
153 // no IMaterialDecorator, so material can only be injected afterwards.
154 applyMaterialMapImpl(*output.geometry);
155 }
156
157 mGeometry = std::const_pointer_cast<const Acts::TrackingGeometry>(output.geometry);
158
159 LOGP(info, "ACTS tracking geometry ready: {} mapped sensors, {} surfaces with material",
160 mIndex.size(), mNDecoratedSurfaces);
161}

163void TrackingGeometryManager::applyMaterialMapImpl(Acts::TrackingGeometry& geometry)
164{
165 if (!std::filesystem::exists(mMaterialMapFile)) {
166 throw std::runtime_error("o2::acts::TrackingGeometryManager: material map '" + mMaterialMapFile +
167 "' does not exist");
168 }
169
170 Acts::MaterialMapJsonConverter::Config converterConfig;
171 converterConfig.context = mNominalContext;
172 const Acts::JsonMaterialDecorator decorator(converterConfig, mMaterialMapFile, mLogLevel);
173
174 mNDecoratedSurfaces = 0;
175 geometry.apply([&](Acts::Surface& surface) {
176 decorator.decorate(surface);
177 if (surface.surfaceMaterial() != nullptr) {
178 ++mNDecoratedSurfaces;
179 }
180 });
181
182 if (mNDecoratedSurfaces == 0) {
183 // Material maps are keyed on Acts::GeometryIdentifier, which is assigned during
184 // construction. Any structural change to the builder shifts those ids and would
185 // otherwise silently yield a material-free geometry.
186 throw std::runtime_error(
187 "o2::acts::TrackingGeometryManager: material map '" + mMaterialMapFile +
188 "' matched no surface. It was produced for a different geometry structure and must be "
189 "regenerated for this one.");
190 }
191}

214const SurfaceIndexMap& TrackingGeometryManager::getIndex(const o2::detectors::DetMatrixCache& cache,
215 double toleranceCm,
216 SensorPathProvider pathProvider)
217{
218 if (!isBuilt()) {
219 build();
220 }
221 const int detID = cache.getDetID();
222 const auto it = mDetIndices.find(detID);
223 if (it != mDetIndices.end()) {
224 return it->second;
225 }
226 auto [inserted, ok] = mDetIndices.emplace(
227 detID, makeSurfaceIndex(*mGeometry, mNominalContext, cache, toleranceCm, std::move(pathProvider)));
228 return inserted->second;
229}
