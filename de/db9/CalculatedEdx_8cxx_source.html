46// dEdxSettings::sameRowClusterMethod == 1: one synthesized cluster summing a mergeable group's fragments'
47// qTot/qMax and charge-weighted pad/time
48o2::tpc::ClusterNative buildMergedClusterSum(const std::vector<o2::tpc::ClusterNative>& fragments)
49{
50 float weightedPadSum = 0.f;
51 float weightedTimeSum = 0.f;
52 float totalCharge = 0.f;
53 uint16_t maxCharge = 0;
54
55 const o2::tpc::ClusterNative& firstCluster = fragments[0];
56 o2::tpc::ClusterNative combinedCluster = firstCluster;
57
58 for (const auto& cl : fragments) {
59 const float clPad = cl.getPad();
60 const float clTime = cl.getTime();
61 const uint16_t clqTot = cl.getQtot();
62 const uint16_t clqMax = cl.qMax;
63
64 weightedPadSum += clPad * clqTot;
65 weightedTimeSum += clTime * clqTot;
66 totalCharge += clqTot;
67 maxCharge = std::max(maxCharge, clqMax);
68 }
69
70 if (totalCharge > o2::tpc::ClusterNative::maxRegularQtot) {
71 combinedCluster.setSaturatedQtot(static_cast<uint32_t>(totalCharge));
72 } else {
73 combinedCluster.qTotPacked = static_cast<uint16_t>(totalCharge);
74 }
75 combinedCluster.qMax = maxCharge;
76 combinedCluster.padPacked = static_cast<uint16_t>(weightedPadSum / totalCharge * o2::tpc::ClusterNative::scalePadPacked);
77 combinedCluster.timeFlagsPacked = (static_cast<uint32_t>(weightedTimeSum / totalCharge * o2::tpc::ClusterNative::scaleTimePacked) & 0xFFFFFF) | (firstCluster.timeFlagsPacked & 0xFF000000);
78 return combinedCluster;
79}

108void CalculatedEdx::setTPCCorrMap(const o2::gpu::TPCFastTransform& corrMap, const o2::gpu::TPCFastTransform* corrMapRef, float lumiScale)
109{
110 if (mRefit) {
111 // the existing refitter holds a pointer into the correction map buffer we are about to replace; drop it
112 // rather than leave it dangling. Normal per-event usage (loadCalibsFromCCDB()/loadCalibsFromLocalCCDBFolder()
113 // followed by setRefit()) re-creates it right after this call with the new map, so this is not an error.
114 LOGP(warning, "CalculatedEdx::setTPCCorrMap() called after setRefit(); invalidating the existing refitter, call setRefit() again before using it.");
115 mRefit.reset();
116 }
117 mRefitExt.reset();
118 mTPCCorrMapFull = TPCFastTransformHelperO2::instance()->create(0, corrMap.getCorrection());
119 if (corrMapRef && lumiScale >= 0) {
120 // linear lumi scaling as in the reconstruction (TPCScalerSpec): (mean - ref) * lumiScale + ref
121 auto& correction = mTPCCorrMapFull->getCorrection();
122 TPCFastSpaceChargeCorrectionHelper::instance()->addCorrections(correction, 1., {{&corrMapRef->getCorrection(), -1.}});
123 TPCFastSpaceChargeCorrectionHelper::instance()->addCorrections(correction, lumiScale, {{&corrMapRef->getCorrection(), 1.}});
124 }
125 rebuildTPCCorrMapPOD();
126}

166void CalculatedEdx::fillMissingClusters(int missingClusters[4], const float minChargeTot[4], const float minChargeMax[4], int method, std::array<std::vector<float>, 5>& chargeTotROC, std::array<std::vector<float>, 5>& chargeMaxROC)
167{
168 if (method != 0 && method != 1) {
169 LOGP(info, "Unrecognized subthreshold cluster treatment. Not adding virtual charges to the track!");
170 return;
171 }
172
173 for (int roc = 0; roc < 4; roc++) {
174 // minChargeTot/MaxROC[roc] is only ever updated from an accepted real cluster in that ROC
175 // a region this track never accepted a single cluster in leaves it at kNoValidCharge, so skip
176 // this ROC's fill entirely rather than injecting the sentinel as if it were a measured charge
177 if (minChargeTot[roc] >= kNoValidCharge || minChargeMax[roc] >= kNoValidCharge) {
178 continue;
179 }
180 const float chargeTot = (method == 1) ? minChargeTot[roc] / 2.f : minChargeTot[roc];
181 const float chargeMax = (method == 1) ? minChargeMax[roc] / 2.f : minChargeMax[roc];
182 for (int i = 0; i < missingClusters[roc]; i++) {
183
184 chargeTotROC[roc].emplace_back(chargeTot);
185 chargeTotROC[4].emplace_back(chargeTot);
186
187 chargeMaxROC[roc].emplace_back(chargeMax);
188 chargeMaxROC[4].emplace_back(chargeMax);
189 }
190 }

193void CalculatedEdx::handleSameRowClusters(o2::tpc::TrackTPC& track, std::vector<std::pair<unsigned char, unsigned char>>& rowOrder, std::map<std::pair<unsigned char, unsigned char>, std::vector<int>>& clustersByRow, std::set<std::pair<unsigned char, unsigned char>>& mergeableRows, std::map<int, std::tuple<unsigned char, unsigned char, unsigned int>>& clusterReferencesByIndex)
194{
195 // get number of clusters
196 const int nClusters = track.getNClusterReferences();
197
198 // group clusters by (sector, row)
199 for (int iCl = 0; iCl < nClusters; iCl++) {
200 unsigned char sectorIndex = 0;
201 unsigned char rowIndex = 0;
202 unsigned int clusterIndexNumb = 0;
203
204 track.getClusterReference(*mTPCTrackClIdxVecInput, iCl, sectorIndex, rowIndex, clusterIndexNumb);
205
206 const auto rowKey = std::make_pair(sectorIndex, rowIndex);
207 if (clustersByRow.find(rowKey) == clustersByRow.end()) {
208 rowOrder.emplace_back(rowKey);
209 }
210
211 // add the cluster index to the corresponding (sector, row) key in clustersByRow
212 clustersByRow[rowKey].emplace_back(iCl);
213
214 // store the reference data in clusterReferencesByIndex
215 clusterReferencesByIndex[iCl] = std::make_tuple(sectorIndex, rowIndex, clusterIndexNumb);
216 }
217
218 // flag groups of several clusters that ended up in the same (sector, row) as eligible for
219 // dEdxSettings::sameRowClusterMethod to merge but ONLY when they are close in pad and time
220 for (const auto& [rowKey, clusterIndices] : clustersByRow) {
221 if (clusterIndices.size() <= 1) {
222 continue;
223 }
224
225 // proximity gate: skip the merge if any pair in the group is farther apart than the pad/time window
226 float minPad = 1e9f, maxPad = -1e9f, minTime = 1e9f, maxTime = -1e9f;
227 for (int clusterIdx : clusterIndices) {
228 const o2::tpc::ClusterNative& cl = track.getCluster(*mTPCTrackClIdxVecInput, clusterIdx, *mClusterIndex);
229 minPad = std::min(minPad, cl.getPad());
230 maxPad = std::max(maxPad, cl.getPad());
231 minTime = std::min(minTime, cl.getTime());
232 maxTime = std::max(maxTime, cl.getTime());
233 }
234 if ((maxPad - minPad) > mSameRowMaxPadDiff || (maxTime - minTime) > mSameRowMaxTimeDiff) {
235 continue; // looper legs / distinct crossings -> keep as separate samples
236 }
237
238 mergeableRows.insert(rowKey);
239 }
240}

280void CalculatedEdx::gatherRowClusterData(o2::tpc::TrackTPC& track, std::vector<RowClusterData>& rowData, AverageOccupancy& averageOcc)
281{
282 rowData.clear();
283 averageOcc = AverageOccupancy{};
284 if (mDebug) {
285 ++mDebugTrackIndex;
286 }
287
288 const bool refitFailed = refitTrack(track);
289
290 // handle same (sector, row) clusters
291 std::vector<std::pair<unsigned char, unsigned char>> rowOrder;
292 std::map<std::pair<unsigned char, unsigned char>, std::vector<int>> clustersByRow;
293 std::set<std::pair<unsigned char, unsigned char>> mergeableRows;
294 std::map<int, std::tuple<unsigned char, unsigned char, unsigned int>> clusterReferencesByIndex;
295
296 handleSameRowClusters(track, rowOrder, clustersByRow, mergeableRows, clusterReferencesByIndex);
297
298 rowData.reserve(rowOrder.size());
299
300 // per-region occupancy, for the average occupancy output
301 std::array<std::vector<unsigned int>, 4> occupancyROC;
302
303 // for tracking missing clusters
304 unsigned char rowIndexOld = 255;
305 unsigned char sectorIndexOld = 255;
306
307 // loop over the clusters in the track's row-traversal order (rowOrder)
308 for (const auto& rowKey : rowOrder) {
309 const auto& clusterIndices = clustersByRow.at(rowKey);
310 const unsigned char rowIndex = rowKey.second;
311 const unsigned char sectorIndex = rowKey.first;
312
313 std::vector<o2::tpc::ClusterNative> fragmentClusters;
314 std::vector<bool> fragmentIsShared;
315 fragmentClusters.reserve(clusterIndices.size());
316 fragmentIsShared.reserve(clusterIndices.size());
317 for (const int clusterIdx : clusterIndices) {
318 fragmentClusters.emplace_back(track.getCluster(*mTPCTrackClIdxVecInput, clusterIdx, *mClusterIndex));
319 const auto& [fragSectorIndex, fragRowIndex, clusterIndexNumb] = clusterReferencesByIndex[clusterIdx];
320 const unsigned int absoluteIndex = mClusterIndex->clusterOffset[fragSectorIndex][fragRowIndex] + clusterIndexNumb;
321 fragmentIsShared.emplace_back(mRefit ? (mTPCRefitterShMap[absoluteIndex] & o2::gpu::GPUTPCGMMergedTrackHit::flagShared) : 0);
322 }
323
324 const bool mergeable = mergeableRows.count(rowKey) > 0;
325 gatherRowClusterDataForRow(track, fragmentClusters, fragmentIsShared, sectorIndex, rowIndex, mergeable, rowIndexOld, sectorIndexOld, occupancyROC, rowData, refitFailed);
326 rowData.back().inputClusterIndices = clusterIndices;
327 rowIndexOld = rowIndex;
328 sectorIndexOld = sectorIndex;
329 }
330
331 // calculate average cl occupancy for the track per TPC region; skip clusters where getOccupancy() had no data (sentinel -1)
332 double* const averageOccROC[4] = {&averageOcc.IROC, &averageOcc.OROC1, &averageOcc.OROC2, &averageOcc.OROC3};
333 for (int roc = 0; roc < 4; roc++) {
334 unsigned int sumOcc = 0;
335 size_t nValidOcc = 0;
336 for (const unsigned int occ : occupancyROC[roc]) {
337 if (occ != static_cast<unsigned int>(-1)) {
338 sumOcc += occ;
339 ++nValidOcc;
340 }
341 }
342 if (nValidOcc > 0) {
343 *averageOccROC[roc] = static_cast<double>(sumOcc) / nValidOcc;
344 }
345 }
346}

348bool CalculatedEdx::refitTrack(o2::tpc::TrackTPC& track)
349{
350 if (!mUseRefit) {
351 return false;
352 }
353 if (!mRefit) {
354 if (!mWarnedMissingRefit) {
355 LOGP(error, "CalculatedEdx: setRefit() is set but there is no refitter for the track's cluster references (setMembers() missing, or the refitter was dropped by setTPCCorrMap()/setTPCVDrift()), propagating the original tracks instead");
356 mWarnedMissingRefit = true;
357 }
358 return true;
359 }
360 // RefitTrackAsGPU() refits the whole track from its clusters (not just up to the current row) and may write into the track even if it fails, so keep the original state to fall back from
361 const o2::track::TrackParCov trackBeforeRefit = track;
362 // outward refit: the track ends at the outermost cluster, i.e. at the start of the cluster reference list that is traversed row by row
363 const bool failed = (mRefit->RefitTrackAsGPU(track, true, true) < 0) || std::isnan(track.getParam(1));
364 if (failed) {
365 static_cast<o2::track::TrackParCov&>(track) = trackBeforeRefit;
366 }
367 return failed;
368}

370bool CalculatedEdx::refitTrack(o2::tpc::TrackTPC& track, const std::vector<o2::tpc::ClusterNative>& clusters, const ClInfoVec& clusterInfos)
371{
372 if (!mUseRefit) {
373 return false;
374 }
375 const size_t nCl = clusters.size();
376 if (nCl == 0 || nCl != clusterInfos.size() || nCl > std::numeric_limits<uint16_t>::max()) {
377 return true;
378 }
379
380 if (!mExtClusterAccess) {
381 mExtClusterAccess = std::make_unique<o2::tpc::ClusterNativeAccess>();
382 }
383 auto& access = *mExtClusterAccess;
384 std::memset(access.nClusters, 0, sizeof(access.nClusters));
385 for (const auto& info : clusterInfos) {
386 if (info.sectorIndex >= constants::MAXSECTOR || info.rowIndex >= constants::MAXGLOBALPADROW) {
387 return true;
388 }
389 ++access.nClusters[info.sectorIndex][info.rowIndex];
390 }
391 const auto* refsBefore = mExtClusterRefs.data();
392 const auto* shMapBefore = mExtShMap.data();
393 mExtClusters.resize(nCl);
394 mExtClusterRefs.assign(nCl + (2 * nCl + sizeof(o2::tpc::TPCClRefElem) - 1) / sizeof(o2::tpc::TPCClRefElem), 0);
395 mExtShMap.assign(nCl, 0);
396 access.clustersLinear = mExtClusters.data();
397 access.clustersMCTruth = nullptr;
398 access.setOffsetPtrs();
399
400 std::map<std::pair<unsigned char, unsigned char>, unsigned int> nPlaced;
401 auto* sectorRowIndices = reinterpret_cast<uint8_t*>(mExtClusterRefs.data() + nCl);
402 for (size_t iCl = 0; iCl < nCl; ++iCl) {
403 const auto& info = clusterInfos[iCl];
404 const unsigned int indexInRow = nPlaced[{info.sectorIndex, info.rowIndex}]++;
405 const unsigned int linearIndex = access.clusterOffset[info.sectorIndex][info.rowIndex] + indexInRow;
406 mExtClusters[linearIndex] = clusters[iCl];
407 mExtShMap[linearIndex] = info.isShared ? o2::gpu::GPUTPCGMMergedTrackHit::flagShared : 0;
408 mExtClusterRefs[iCl] = indexInRow;
409 sectorRowIndices[iCl] = info.sectorIndex;
410 sectorRowIndices[iCl + nCl] = info.rowIndex;
411 }
412
413 // the refitter keeps raw pointers to the buffers, so recreate it when they moved
414 if (!mRefitExt || refsBefore != mExtClusterRefs.data() || shMapBefore != mExtShMap.data()) {
415 mRefitExt = std::make_unique<o2::gpu::GPUO2InterfaceRefit>(mExtClusterAccess.get(), mTPCCorrMap, mFieldNominalGPUBz, mExtClusterRefs.data(), mNHbfPerTf, mExtShMap.data(), mTPCRefitterOccMap.data(), mTPCRefitterOccMap.size());
416 }
417
418 const o2::track::TrackParCov trackBeforeRefit = track;
419 float chi2 = 0.f;
420 const o2::tpc::TrackTPCClusRef clusRef(0, static_cast<uint16_t>(nCl));
421 const bool failed = (mRefitExt->RefitTrackAsGPU(static_cast<o2::track::TrackParCov&>(track), clusRef, track.getTime0(), &chi2, true, true) < 0) || std::isnan(track.getParam(1));
422 if (failed) {
423 static_cast<o2::track::TrackParCov&>(track) = trackBeforeRefit;
424 } else {
425 track.setChi2(chi2);
426 }
427 return failed;
428}

430bool CalculatedEdx::propagateTrackToX(o2::track::TrackParCov& track, float xPosition, unsigned char sectorIndex) const
431{
432 const o2::track::TrackParCov trackBackup = track;
433 bool check = track.rotate(o2::math_utils::detail::sector2Angle<float>(sectorIndex));
434 if (check) {
435 check = o2::base::Propagator::Instance()->PropagateToXBxByBz(track, xPosition, 0.999f, 0.5f, o2::base::Propagator::MatCorrType::USEMatCorrLUT);
436 }
437 if (!check) {
438 track = trackBackup;
439 check = track.rotate(o2::math_utils::detail::sector2Angle<float>(sectorIndex));
440 if (check) {
441 check = o2::base::Propagator::Instance()->PropagateToXBxByBz(track, xPosition, 0.999f, 0.5f, o2::base::Propagator::MatCorrType::USEMatCorrNONE);
442 }
443 }
444 if (!check) {
445 track = trackBackup;
446 check = track.rotateParam(o2::math_utils::detail::sector2Angle<float>(sectorIndex));
447 if (check) {
448 check = track.propagateParamTo(xPosition, mFieldNominalGPUBz);
449 }
450 }
451 if (!check) {
452 track = trackBackup;
453 }
454 return check;
455}

457void CalculatedEdx::gatherRowClusterDataForRow(o2::tpc::TrackTPC& track, const std::vector<o2::tpc::ClusterNative>& fragmentClusters, const std::vector<bool>& fragmentIsShared, unsigned char sectorIndex, unsigned char rowIndex, bool mergeable, unsigned char rowIndexOld, unsigned char sectorIndexOld, std::array<std::vector<unsigned int>, 4>& occupancyROC, std::vector<RowClusterData>& rowData, bool refitFailed)
458{
459 RowClusterData row;
460 row.sectorIndex = sectorIndex;
461 row.rowIndex = rowIndex;
462 row.mergeable = mergeable;
463
464 // get region and stack
465 const int region = Mapper::REGION[rowIndex];
466 const CRU cru(Sector(sectorIndex), region);
467 const auto stack = cru.gemStack();
468 StackID stackID{sectorIndex, stack};
469 const int stackNumber = static_cast<int>(stack);
470
471 row.region = region;
472 row.stack = stack;
473 row.stackID = stackID;
474 row.stackNumber = stackNumber;
475
476 // per-fragment quantities: pad depends on the individual cluster's own pad, so threshold/gain/gainResidual/occupancy
477 // all looked up by pad, are computed once per fragment here
478 row.fragments.reserve(fragmentClusters.size());
479 for (size_t iFrag = 0; iFrag < fragmentClusters.size(); ++iFrag) {
480 const o2::tpc::ClusterNative& cl = fragmentClusters[iFrag];
481 RowFragment frag;
482 frag.cl = cl;
483 frag.isShared = fragmentIsShared[iFrag];
484 frag.pad = std::clamp(static_cast<unsigned int>(cl.getPad() + 0.5f), static_cast<unsigned int>(0), Mapper::PADSPERROW[region][Mapper::getLocalRowFromGlobalRow(rowIndex)] - 1); // the left side of the pad is defined at e.g. 3.5 and the right side at 4.5
485 frag.occupancy = getOccupancy(cl.getTime());
486 frag.threshold = mCalibCont.getZeroSupressionThreshold(sectorIndex, rowIndex, frag.pad);
487 frag.gain = mCalibCont.getGain(sectorIndex, rowIndex, frag.pad);
488 frag.gainResidual = mCalibCont.getResidualGain(sectorIndex, rowIndex, frag.pad);
489
490 row.fragments.emplace_back(std::move(frag));
491 }
492
493 // dEdxSettings::sameRowClusterMethod==1's merged sample
494 if (mergeable && fragmentClusters.size() > 1) {
495 row.mergedFragment.cl = buildMergedClusterSum(fragmentClusters);
496 row.mergedFragment.pad = std::clamp(static_cast<unsigned int>(row.mergedFragment.cl.getPad() + 0.5f), static_cast<unsigned int>(0), Mapper::PADSPERROW[region][Mapper::getLocalRowFromGlobalRow(rowIndex)] - 1);
497 row.mergedFragment.occupancy = getOccupancy(row.mergedFragment.cl.getTime());
498 row.mergedFragment.threshold = mCalibCont.getZeroSupressionThreshold(sectorIndex, rowIndex, row.mergedFragment.pad);
499 row.mergedFragment.gain = mCalibCont.getGain(sectorIndex, rowIndex, row.mergedFragment.pad);
500 row.mergedFragment.gainResidual = mCalibCont.getResidualGain(sectorIndex, rowIndex, row.mergedFragment.pad);
501 row.mergedFragment.isShared = row.fragments[0].isShared; // preserves the historical "first fragment's isShared" choice for a merged sample
502 }
503
504 // occupancy sample(s) for this row's average occupancy contribution
505 std::array<std::vector<unsigned int>, 4>::size_type occStackIdx;
506 if (stack == GEMstack::IROCgem) {
507 occStackIdx = 0;
508 } else if (stack == GEMstack::OROC1gem) {
509 occStackIdx = 1;
510 } else if (stack == GEMstack::OROC2gem) {
511 occStackIdx = 2;
512 } else {
513 occStackIdx = 3;
514 }
515 if (mergeable && fragmentClusters.size() > 1) {
516 occupancyROC[occStackIdx].emplace_back(row.mergedFragment.occupancy);
517 } else {
518 for (const auto& frag : row.fragments) {
519 occupancyROC[occStackIdx].emplace_back(frag.occupancy);
520 }
521 }
522
523 // get the x position of the track: the corrected cluster x (charge-weighted for several clusters in the row
524 float xPosition = Mapper::instance().getPadCentre(PadPos(rowIndex, 0)).X();
525 if (mPropagateToClusterX) {
526 float sumX = 0.f, sumQ = 0.f;
527 for (const auto& cl : fragmentClusters) {
528 float x, y, z;
529 mTPCCorrMap->Transform(sectorIndex, rowIndex, cl.getPad(), cl.getTime(), x, y, z, track.getTime0());
530 sumX += x * cl.getQtot();
531 sumQ += cl.getQtot();
532 }
533 if (sumQ > 0.f) {
534 xPosition = sumX / sumQ;
535 }
536 }
537 bool check = true;
538 if (mUseRefit || mPropagateTrack || !mPropagateParams) {
539 // mUseRefit: the track was refit once in gatherRowClusterData() and is only propagated from row to row here, if that refit failed, the propagation starts from the track's original state
540 // otherwise: propagate this track to the plane X=xk (cm) in the field "b" (kG)
541 check = propagateTrackToX(track, xPosition, sectorIndex);
542 } else {
543 // propagate the params of the track instead of full propagation; same rollback rationale as mPropagateTrack above
544 const o2::track::TrackParCov trackBackup = track;
545 check = track.rotateParam(o2::math_utils::detail::sector2Angle<float>(sectorIndex));
546 if (check) {
547 check = track.propagateParamTo(xPosition, mFieldNominalGPUBz);
548 }
549 if (!check) {
550 static_cast<o2::track::TrackParCov&>(track) = trackBackup;
551 }
552 }
553
554 row.propagationFailed = (!check || std::isnan(track.getParam(1)));
555 row.refitFellBack = mUseRefit && refitFailed && !row.propagationFailed;
556 ++mNRowsProcessed;
557 if (row.propagationFailed) {
558 ++mNPropagationFailed;
559 } else if (row.refitFellBack) {
560 ++mNRefitFallback;
561 }
562
563 // snapshot of the track state after refit/propagation to this row; reused by calculatedEdxFromRowData() for every settings entry
564 row.trackSnapshot = track;
565
566 // number of rows skipped between this row and the previous entry in rowOrder
567 row.sameSectorAsPrevRow = (sectorIndexOld == sectorIndex);
568 row.missingClusters = (rowIndexOld == 255) ? 0 : (std::abs(static_cast<int>(rowIndex) - static_cast<int>(rowIndexOld)) - 1);
569
570 // veto the gap as a subthreshold candidate if any of its missing row(s) would land on a dead channel or off the padrow edge
571 row.missingClusterGapDeadOrEdge = false;
572 if (row.missingClusters > 0 && row.missingClusters <= mMaxMissingCl) {
573 const o2::gpu::GPUTPCGeometry gpuGeom;
574 const RowClusterData& prevRow = rowData.back();
575 // bracket the gap by its lower/upper real row, independent of the rowOrder direction
576 const int rowLo = std::min<int>(rowIndex, rowIndexOld);
577 const int rowHi = std::max<int>(rowIndex, rowIndexOld);
578 // count every missing row in its own stack
579 for (int missingRow = rowLo + 1; missingRow < rowHi; ++missingRow) {
580 ++row.missingClustersStack[static_cast<int>(CRU(Sector(sectorIndex), Mapper::REGION[missingRow]).gemStack())];
581 }
582 // rotate the previous cluster into this row's sector frame
583 const float yPrevOwn = gpuGeom.LinearPad2Y(sectorIndexOld, rowIndexOld, prevRow.fragments[0].cl.getPad());
584 float yPrev = yPrevOwn;
585 if (sectorIndexOld != sectorIndex) {
586 if ((sectorIndexOld < Sector::MAXSECTOR / 2) != (sectorIndex < Sector::MAXSECTOR / 2)) {
587 row.missingClusterGapDeadOrEdge = true;
588 }
589 const float dAlpha = o2::math_utils::detail::sector2Angle<float>(sectorIndexOld) - o2::math_utils::detail::sector2Angle<float>(sectorIndex);
590 yPrev = gpuGeom.Row2X(rowIndexOld) * std::sin(dAlpha) + yPrevOwn * std::cos(dAlpha);
591 }
592 const float yCur = gpuGeom.LinearPad2Y(sectorIndex, rowIndex, row.fragments[0].cl.getPad());
593 const float yLo = (rowLo == static_cast<int>(rowIndexOld)) ? yPrev : yCur;
594 const float yHi = (rowHi == static_cast<int>(rowIndexOld)) ? yPrev : yCur;
595 for (int k = 1; k <= row.missingClusters && !row.missingClusterGapDeadOrEdge; ++k) {
596 const unsigned char missingRow = static_cast<unsigned char>(rowLo + k);
597 const float frac = static_cast<float>(k) / (row.missingClusters + 1);
598 const float missingPad = gpuGeom.LinearY2Pad(sectorIndex, missingRow, yLo + (yHi - yLo) * frac);
599 if (missingPad < 0.f || missingPad >= gpuGeom.NPads(missingRow)) {
600 row.missingClusterGapDeadOrEdge = true;
601 break;
602 }
603 const int missingRegion = Mapper::REGION[missingRow];
604 const unsigned char missingPadClamped = std::clamp(static_cast<unsigned int>(missingPad + 0.5f), static_cast<unsigned int>(0), Mapper::PADSPERROW[missingRegion][Mapper::getLocalRowFromGlobalRow(missingRow)] - 1);
605 if (mCalibCont.isDead(static_cast<unsigned int>(sectorIndex), static_cast<gpu::tpccf::Row>(missingRow), static_cast<gpu::tpccf::Pad>(missingPadClamped))) {
606 row.missingClusterGapDeadOrEdge = true;
607 break;
608 }
609 }
610 }
611
612 rowData.emplace_back(std::move(row));
613}

615void CalculatedEdx::gatherRowClusterData(o2::tpc::TrackTPC& track, const std::vector<o2::tpc::ClusterNative>& clusters, const ClInfoVec& clusterInfos, std::vector<RowClusterData>& rowData, AverageOccupancy& averageOcc)
616{
617 rowData.clear();
618 averageOcc = AverageOccupancy{};
619 if (mDebug) {
620 ++mDebugTrackIndex;
621 }
622
623 const bool refitFailed = refitTrack(track, clusters, clusterInfos);
624
625 // handle same (sector, row) clusters
626 std::vector<std::pair<unsigned char, unsigned char>> rowOrder;
627 std::map<std::pair<unsigned char, unsigned char>, std::vector<int>> clustersByRow;
628 std::set<std::pair<unsigned char, unsigned char>> mergeableRows;
629
630 handleSameRowClusters(clusters, clusterInfos, rowOrder, clustersByRow, mergeableRows);
631
632 rowData.reserve(rowOrder.size());
633
634 // per-region occupancy, for the average occupancy output
635 std::array<std::vector<unsigned int>, 4> occupancyROC;
636
637 // for tracking missing clusters
638 unsigned char rowIndexOld = 255;
639 unsigned char sectorIndexOld = 255;
640
641 // loop over the clusters in the rowOrder; the refit/propagation to a row is done exactly once here regardless of how many raw clusters the row has
642 for (const auto& rowKey : rowOrder) {
643 const auto& clusterIndices = clustersByRow.at(rowKey);
644 const unsigned char rowIndex = rowKey.second;
645 const unsigned char sectorIndex = rowKey.first;
646
647 std::vector<o2::tpc::ClusterNative> fragmentClusters;
648 std::vector<bool> fragmentIsShared;
649 fragmentClusters.reserve(clusterIndices.size());
650 fragmentIsShared.reserve(clusterIndices.size());
651 for (const int clusterIdx : clusterIndices) {
652 fragmentClusters.emplace_back(clusters[clusterIdx]);
653 // isShared cannot be looked up from mTPCRefitterShMap for externally supplied clusters, so it is taken from the info directly
654 fragmentIsShared.emplace_back(clusterInfos[clusterIdx].isShared);
655 }
656
657 const bool mergeable = mergeableRows.count(rowKey) > 0;
658 gatherRowClusterDataForRow(track, fragmentClusters, fragmentIsShared, sectorIndex, rowIndex, mergeable, rowIndexOld, sectorIndexOld, occupancyROC, rowData, refitFailed);
659 rowData.back().inputClusterIndices = clusterIndices; // positions in the externally supplied clusters vector grouped into this row
660 rowIndexOld = rowIndex;
661 sectorIndexOld = sectorIndex;
662 }
663
664 // calculate average cl occupancy for the track per TPC region; skip clusters where getOccupancy() had no data (sentinel -1)
665 double* const averageOccROC[4] = {&averageOcc.IROC, &averageOcc.OROC1, &averageOcc.OROC2, &averageOcc.OROC3};
666 for (int roc = 0; roc < 4; roc++) {
667 unsigned int sumOcc = 0;
668 size_t nValidOcc = 0;
669 for (const unsigned int occ : occupancyROC[roc]) {
670 if (occ != static_cast<unsigned int>(-1)) {
671 sumOcc += occ;
672 ++nValidOcc;
673 }
674 }
675 if (nValidOcc > 0) {
676 *averageOccROC[roc] = static_cast<double>(sumOcc) / nValidOcc;
677 }
678 }

681void CalculatedEdx::calculatedEdxFromRowData(const std::vector<RowClusterData>& rowData, const dEdxSettings& settings, size_t settingsIndex, float trackTime0, const o2::tpc::TrackTPC& trackOrig, const AverageOccupancy& averageOcc, dEdxInfo& output, const MCCompLabel* mcLabel)
682{
683 // NHits and NHitsSubthreshold values per region
684 int nClsROC[4] = {0, 0, 0, 0};
685 int nClsSubThreshROC[4] = {0, 0, 0, 0};
686
687 const unsigned short sameRowClusterMethod = (settings.sameRowClusterMethod <= 2) ? settings.sameRowClusterMethod : 0;
688 if (settings.sameRowClusterMethod > 2) {
689 LOGP(warning, "Unrecognized sameRowClusterMethod {} (expected 0, 1, or 2); treating same-row cluster groups as method 0 (do not merge)", settings.sameRowClusterMethod);
690 }
691
692 // corrected qTot and qMax values per region
693 const int nType = 5;
694 std::array<std::vector<float>, nType> chargeTotROC;
695 std::array<std::vector<float>, nType> chargeMaxROC;
696 for (int i = 0; i < nType; ++i) {
697 chargeTotROC[i].reserve(Mapper::PADROWS);
698 chargeMaxROC[i].reserve(Mapper::PADROWS);
699 }
700
701 // per-region (IROC, OROC1, OROC2, OROC3) running minimum charge among accepted clusters, used as the virtual charge for that region's subthreshold clusters below
702 float minChargeTotROC[4] = {kNoValidCharge, kNoValidCharge, kNoValidCharge, kNoValidCharge};
703 float minChargeMaxROC[4] = {kNoValidCharge, kNoValidCharge, kNoValidCharge, kNoValidCharge};
704
705 o2::utils::TreeStreamRedirector* debugStreamer = nullptr;
706 std::vector<unsigned int> occupancyVector;
707 if (mDebug) {
708 setStreamer(settings.debugRootFile.c_str());
709 debugStreamer = mStreamers.at(settings.debugRootFile).get();
710 occupancyVector.reserve(rowData.size());
711 }
712
713 // a gap is not filled as a subthreshold cluster when the row that closes it sits within the outermost min(nRows/2, mSubThreshEdgeRows) rows
714 const int edgeRowCut = std::min<int>(static_cast<int>(rowData.size()) / 2, mSubThreshEdgeRows);
715
716 for (size_t iRowData = 0; iRowData < rowData.size(); ++iRowData) {
717 const auto& row = rowData[iRowData];
718
719 // one effective sample per row for this settings entry, per dEdxSettings::sameRowClusterMethod; pointers into
720 // row.fragments/row.mergedFragment (both owned by rowData, alive for the whole calculatedEdxFromRowData() call)
721 std::vector<const RowFragment*> samples;
722
723 const bool doMerge = row.mergeable && row.fragments.size() > 1 && sameRowClusterMethod != 0;
724 if (!doMerge) {
725 samples.reserve(row.fragments.size());
726 for (const auto& frag : row.fragments) {
727 samples.push_back(&frag);
728 }
729 } else if (sameRowClusterMethod == 2) {
730 samples.push_back(&pickDominantFragment(row.fragments));
731 } else { // sameRowClusterMethod == 1: use the sum-merged sample gatherRowClusterDataForRow() already computed once for this row
732 samples.push_back(&row.mergedFragment);
733 }
734
735 // ExcludeSamePadRowCl: true whenever the row-group HAD >1 raw fragment, regardless of whether this settings entry actually merged them
736 const bool isCombined = row.fragments.size() > 1;
737
738 // find missing clusters
739 const int missingClusters = row.missingClusters;
740 const bool excludeSectorBoundary = ((settings.clusterMask & ClusterFlags::ExcludeSectorBoundaries) == ClusterFlags::ExcludeSectorBoundaries) && !row.sameSectorAsPrevRow;
741 if ((missingClusters > 0) && (missingClusters <= mMaxMissingCl) && !row.missingClusterGapDeadOrEdge && (static_cast<int>(iRowData) >= edgeRowCut) && !excludeSectorBoundary) {
742 for (int roc = 0; roc < 4; roc++) {
743 nClsSubThreshROC[roc] += row.missingClustersStack[roc];
744 nClsROC[roc] += row.missingClustersStack[roc];
745 }
746 }
747
748 for (const auto& sample : samples) {
749 if (mDebug) {
750 occupancyVector.emplace_back(sample->occupancy);
751 }
752
753 // get cluster values
754 float chargeTot = sample->cl.getQtot();
755 float chargeMax = sample->cl.getQmax();
756
757 // corrections
758 float effectiveLength = 1.0f;
759 float effectiveLengthTot = 1.0f;
760 float effectiveLengthMax = 1.0f;
761 float gain = 1.0f;
762 float gainResidual = 1.0f;
763 float corrTot = 1.0f;
764 float corrMax = 1.0f;
765 float scCorr = 1.0f;
766
767 int excludeCl = 0; // works as a bit mask
768 const uint8_t flagsCl = sample->cl.getFlags();
769 if (((settings.clusterMask & ClusterFlags::ExcludeSingleCl) == ClusterFlags::ExcludeSingleCl) && ((flagsCl & ClusterNative::flagSingle) == ClusterNative::flagSingle)) {
770 excludeCl += 0b001; // 1 for single cluster
771 }
772 if (((settings.clusterMask & ClusterFlags::ExcludeSplitPadCl) == ClusterFlags::ExcludeSplitPadCl) && ((flagsCl & ClusterNative::flagSplitPad) == ClusterNative::flagSplitPad)) {
773 excludeCl += 0b010; // 2 for split pad cluster
774 }
775 if (((settings.clusterMask & ClusterFlags::ExcludeSplitTimeCl) == ClusterFlags::ExcludeSplitTimeCl) && ((flagsCl & ClusterNative::flagSplitTime) == ClusterNative::flagSplitTime)) {
776 excludeCl += 0b0100; // 4 for split time cluster
777 }
778 if (((settings.clusterMask & ClusterFlags::ExcludeSplitCl) == ClusterFlags::ExcludeSplitCl) && (((flagsCl & ClusterNative::flagSplitPad) == ClusterNative::flagSplitPad) || ((flagsCl & ClusterNative::flagSplitTime) == ClusterNative::flagSplitTime))) {
779 excludeCl += 0b01000; // 8 for split cluster
780 }
781 if (((settings.clusterMask & ClusterFlags::ExcludeEdgeCl) == ClusterFlags::ExcludeEdgeCl) && ((flagsCl & ClusterNative::flagEdge) == ClusterNative::flagEdge)) {
782 excludeCl += 0b010000; // 16 for edge cluster
783 }
784 if (((settings.clusterMask & ClusterFlags::ExcludeSharedCl) == ClusterFlags::ExcludeSharedCl) && sample->isShared) {
785 excludeCl += 0b0100000; // 32 for shared cluster
786 }
787 if (((settings.clusterMask & ClusterFlags::ExcludeSamePadRowCl) == ClusterFlags::ExcludeSamePadRowCl) && isCombined) {
788 excludeCl += 0b01000000; // 64 for combined cluster
789 }
790 if ((settings.stackBoundaryMethod == 1 || settings.stackBoundaryMethod == 2) && isInStackBoundaries(row.stackNumber, row.rowIndex, settings.stackBoundaryMethod)) {
791 excludeCl += 0b010000000; // 128 for stack boundary cluster
792 }
793 if (row.propagationFailed) {
794 excludeCl += 0b01000000000; // 512 for failure of track propagation or refit
795 }
796
797 // get effective length
798 if ((settings.correctionMask & CorrectionFlags::TopologySimple) == CorrectionFlags::TopologySimple) {
799 effectiveLength = getTrackTopologyCorrection(row.trackSnapshot, row.region);
800 chargeTot /= effectiveLength;
801 chargeMax /= effectiveLength;
802 };
803
804 const bool gainFullApplied = (settings.correctionMask & CorrectionFlags::GainFull) == CorrectionFlags::GainFull;
805 float topoChargeTot = chargeTot;
806 float topoChargeMax = chargeMax;
807 if (gainFullApplied) {
808 gain = sample->gain;
809 chargeTot /= gain;
810 chargeMax /= gain;
811 } else {
812 topoChargeTot *= sample->gain;
813 topoChargeMax *= sample->gain;
814 }
815
816 // topology correction
817 if ((settings.correctionMask & CorrectionFlags::TopologyPol) == CorrectionFlags::TopologyPol) {
818 effectiveLengthTot = getTrackTopologyCorrectionPol(row.trackSnapshot, sample->cl, row.region, topoChargeTot, ChargeType::Tot, sample->threshold);
819 effectiveLengthMax = getTrackTopologyCorrectionPol(row.trackSnapshot, sample->cl, row.region, topoChargeMax, ChargeType::Max, sample->threshold);
820 chargeTot /= effectiveLengthTot;
821 chargeMax /= effectiveLengthMax;
822 };
823
824 // residual dE/dx correction on tgl and sector plane
825 if ((settings.correctionMask & CorrectionFlags::dEdxResidual) == CorrectionFlags::dEdxResidual) {
826 corrTot = mCalibCont.getResidualCorrection(row.stackID, ChargeType::Tot, row.trackSnapshot.getTgl(), row.trackSnapshot.getSnp());
827 corrMax = mCalibCont.getResidualCorrection(row.stackID, ChargeType::Max, row.trackSnapshot.getTgl(), row.trackSnapshot.getSnp());
828 if (corrTot > 0) {
829 chargeTot /= corrTot;
830 };
831 if (corrMax > 0) {
832 chargeMax /= corrMax;
833 };
834 };
835
836 // residual gain map
837 if ((settings.correctionMask & CorrectionFlags::GainResidual) == CorrectionFlags::GainResidual) {
838 gainResidual = sample->gainResidual;
839 chargeTot /= gainResidual;
840 chargeMax /= gainResidual;
841 };
842
843 // space-charge dEdx corrections
844 const float time = sample->cl.getTime() - trackTime0; // ToDo: get correct time from ITS-TPC track if possible
845 if ((settings.correctionMask & CorrectionFlags::dEdxSC) == CorrectionFlags::dEdxSC) {
846 scCorr = mSCdEdxCorrection.getCorrection(time, row.sectorIndex, row.rowIndex, sample->pad);
847 if (scCorr > 0) {
848 chargeTot /= scCorr;
849 };
850 if (scCorr > 0) {
851 chargeMax /= scCorr;
852 };
853 }
854
855 // for debugging
856 if (mDebug) {
857 const o2::gpu::GPUTPCGeometry gpuGeom;
858 const float localX = gpuGeom.Row2X(row.rowIndex);
859 const float localY = gpuGeom.LinearPad2Y(row.sectorIndex, row.rowIndex, sample->cl.getPad());
860 const LocalPosition2D l2D{localX, localY};
861 const auto g2D = Mapper::LocalToGlobal(l2D, Sector(row.sectorIndex));
862 const float globalX = g2D.x();
863 const float globalY = g2D.y();
864
865 // slice to the base parametrization (X, alpha, params, covariance) instead of the full TrackTPC, since only the parametrization changes cluster-to-cluster after refit/propagation
866 const o2::track::TrackParCov trackParam = row.trackSnapshot;
867
868 (*debugStreamer) << "dEdxDebugCl"
869 << "trackIndex=" << mDebugTrackIndex
870 << "trackParam=" << trackParam
871 << "cl=" << sample->cl
872 << "chargeTot=" << chargeTot
873 << "chargeMax=" << chargeMax
874 << "excludeCl=" << excludeCl
875 << "region=" << row.region
876 << "rowIndex=" << row.rowIndex
877 << "sectorIndex=" << row.sectorIndex
878 << "stack=" << row.stackNumber
879 << "localX=" << localX
880 << "localY=" << localY
881 << "globalX=" << globalX
882 << "globalY=" << globalY
883 << "isShared=" << sample->isShared
884 << "isCombined=" << isCombined
885 << "refitFellBack=" << row.refitFellBack
886 << "topologyCorr=" << effectiveLength
887 << "topologyCorrTot=" << effectiveLengthTot
888 << "topologyCorrMax=" << effectiveLengthMax
889 << "gain=" << gain
890 << "fullGain=" << sample->gain
891 << "gainResidual=" << gainResidual
892 << "residualCorrTot=" << corrTot
893 << "residualCorrMax=" << corrMax
894 << "scCorr=" << scCorr
895 << "occupancy=" << sample->occupancy
896 << "inputClusterIndices=" << row.inputClusterIndices
897 << "\n";
898 };
899
900 if (excludeCl != 0) {
901 continue;
902 }
903
904 // set the region's min charge, only from clusters actually included in the dEdx calculation,
905 // so excluded clusters (edge, failed propagation, ...) don't bias the virtual charge used for subthreshold filling
906 if (chargeTot < minChargeTotROC[row.stackNumber]) {
907 minChargeTotROC[row.stackNumber] = chargeTot;
908 };
909
910 if (chargeMax < minChargeMaxROC[row.stackNumber]) {
911 minChargeMaxROC[row.stackNumber] = chargeMax;
912 };
913
914 if (row.stack == GEMstack::IROCgem) {
915 chargeTotROC[0].emplace_back(chargeTot);
916 chargeMaxROC[0].emplace_back(chargeMax);
917 nClsROC[0]++;
918 } else if (row.stack == GEMstack::OROC1gem) {
919 chargeTotROC[1].emplace_back(chargeTot);
920 chargeMaxROC[1].emplace_back(chargeMax);
921 nClsROC[1]++;
922 } else if (row.stack == GEMstack::OROC2gem) {
923 chargeTotROC[2].emplace_back(chargeTot);
924 chargeMaxROC[2].emplace_back(chargeMax);
925 nClsROC[2]++;
926 } else if (row.stack == GEMstack::OROC3gem) {
927 chargeTotROC[3].emplace_back(chargeTot);
928 chargeMaxROC[3].emplace_back(chargeMax);
929 nClsROC[3]++;
930 };
931
932 chargeTotROC[4].emplace_back(chargeTot);
933 chargeMaxROC[4].emplace_back(chargeMax);
934 }
935 }
936
937 // fill subthreshold clusters if not excluded
938 if (((settings.clusterMask & ClusterFlags::ExcludeSubthresholdCl) == ClusterFlags::None)) {
939 float cappedMinChargeTotROC[4], cappedMinChargeMaxROC[4];
940 for (int roc = 0; roc < 4; roc++) {
941 cappedMinChargeTotROC[roc] = (minChargeTotROC[roc] >= kNoValidCharge) ? minChargeTotROC[roc] : std::min(minChargeTotROC[roc], settings.maxSubthresholdChargeTot);
942 cappedMinChargeMaxROC[roc] = (minChargeMaxROC[roc] >= kNoValidCharge) ? minChargeMaxROC[roc] : std::min(minChargeMaxROC[roc], settings.maxSubthresholdChargeMax);
943 // a ROC with no valid accepted-cluster charge at all makes fillMissingClusters() below skip it entirely
944 // (nothing is pushed into chargeTotROC/chargeMaxROC for it) -- so the gaps counted for this ROC earlier
945 // in the row loop must be un-counted here too, or NHits*/NHitsSubThreshold* and the
946 // mNSubThresholdFilledPerSettings diagnostic would report fills that never actually happened
947 if (minChargeTotROC[roc] >= kNoValidCharge || minChargeMaxROC[roc] >= kNoValidCharge) {
948 nClsROC[roc] -= nClsSubThreshROC[roc];
949 nClsSubThreshROC[roc] = 0;
950 }
951 }
952 fillMissingClusters(nClsSubThreshROC, cappedMinChargeTotROC, cappedMinChargeMaxROC, settings.subthresholdMethod, chargeTotROC, chargeMaxROC);
953 if (mNSubThresholdFilledPerSettings.size() <= settingsIndex) {
954 mNSubThresholdFilledPerSettings.resize(settingsIndex + 1, 0);
955 }
956 mNSubThresholdFilledPerSettings[settingsIndex] += nClsSubThreshROC[0] + nClsSubThreshROC[1] + nClsSubThreshROC[2] + nClsSubThreshROC[3];
957 }
958
959 // number of clusters
960 output.NHitsSubThresholdIROC = nClsROC[0];
961 output.NHitsSubThresholdOROC1 = nClsROC[1];
962 output.NHitsSubThresholdOROC2 = nClsROC[2];
963 output.NHitsSubThresholdOROC3 = nClsROC[3];
964
965 // the gaps found above are always treated as subthreshold clusters (except a ROC with no valid charge to fill
966 // them with at all, un-counted above so it isn't double-reported as both "hit" and "subthreshold hit")
967 output.NHitsIROC = nClsROC[0] - nClsSubThreshROC[0];
968 output.NHitsOROC1 = nClsROC[1] - nClsSubThreshROC[1];
969 output.NHitsOROC2 = nClsROC[2] - nClsSubThreshROC[2];
970 output.NHitsOROC3 = nClsROC[3] - nClsSubThreshROC[3];
971
972 // copy corrected cluster charges
973 auto chargeTotVector = mDebug ? chargeTotROC[4] : std::vector<float>();
974 auto chargeMaxVector = mDebug ? chargeMaxROC[4] : std::vector<float>();
975
976 // calculate dEdx
977 output.dEdxTotIROC = getTruncMean(chargeTotROC[0], settings.low, settings.high);
978 output.dEdxTotOROC1 = getTruncMean(chargeTotROC[1], settings.low, settings.high);

1113float CalculatedEdx::getTruncMean(std::vector<float>& charge, float low, float high) const
1114{
1115 // sort the charge vector
1116 std::sort(charge.begin(), charge.end());
1117
1118 // calculate truncated mean
1119 int nCl = 0;
1120 float sum = 0;
1121 size_t firstCl = charge.size() * low;
1122 size_t lastCl = charge.size() * high;
1123
1124 for (size_t iCl = firstCl; iCl < lastCl; ++iCl) {
1125 sum += charge[iCl];
1126 ++nCl;
1127 }
1128
1129 if (nCl > 0) {
1130 sum /= nCl;
1131 }
1132 // if nCl == 0 (charge was empty, or too few entries for low/high to select any index), sum stays 0
1133 return sum;
1134}

1136float CalculatedEdx::getTrackTopologyCorrection(const o2::tpc::TrackTPC& track, const unsigned int region) const
1137{
1138 const float padLength = Mapper::instance().getPadRegionInfo(region).getPadHeight();
1139 const float snp = track.getSnp();
1140 const float tgl = track.getTgl();
1141 const float snp2 = snp * snp;
1142 const float tgl2 = tgl * tgl;
1143 // calculate the trace length of the track over the pad
1144 const float effectiveLength = padLength * std::sqrt((1 + tgl2) / (1 - snp2));
1145 return effectiveLength;
1146}

1148float CalculatedEdx::getTrackTopologyCorrectionPol(const o2::tpc::TrackTPC& track, const o2::tpc::ClusterNative& cl, const unsigned int region, const float charge, ChargeType chargeType, const float threshold) const
1149{
1150 const float snp = std::abs(track.getSnp());
1151 const float tgl = track.getTgl();
1152 constexpr float maxSnp2 = 0.99f;
1153 float snp2 = snp * snp;
1154 if (snp2 > maxSnp2) {
1155 snp2 = maxSnp2;
1156 }
1157 const float tgl2 = tgl * tgl;
1158 const float sec2 = 1.f / (1.f - snp2);
1159 const float tanTheta = std::sqrt(tgl2 * sec2);
1160
1161 const float z = std::abs(track.getParam(1));
1162 const float padTmp = cl.getPad();
1163 const float absRelPad = std::abs(padTmp - int(padTmp + 0.5f));
1164 const float relTime = cl.getTime() - int(cl.getTime() + 0.5f);
1165
1166 const float effectiveLength = mCalibCont.getTopologyCorrection(region, chargeType, tanTheta, snp, z, absRelPad, relTime, threshold, charge);
1167 return effectiveLength;
1168}

1320float CalculatedEdx::getTPCScalerLumiScale(long timeStamp, float meanIDC) const
1321{
1322 // mean scaler of both sides (weighted over the ion drift time) over the map's mean IDC
1323 auto& cm = o2::ccdb::BasicCCDBManager::instance();
1324 const bool prevFatalWhenNull = cm.getFatalWhenNull();
1325 cm.setFatalWhenNull(false);
1326 auto* scalerTree = cm.getForTimeStamp<TTree>(o2::tpc::CDBTypeMap.at(o2::tpc::CDBType::CalScaler), timeStamp);
1327 auto* weights = cm.getForTimeStamp<o2::tpc::TPCScalerWeights>(o2::tpc::CDBTypeMap.at(o2::tpc::CDBType::CalScalerWeights), timeStamp);
1328 cm.setFatalWhenNull(prevFatalWhenNull);
1329 if (!scalerTree || meanIDC <= 0) {
1330 LOGP(warning, "no TPC scaler or mean IDC of the map ({}) at {}, the space-charge map is not scaled", meanIDC, timeStamp);
1331 return -1.f;
1332 }
1333 o2::tpc::TPCScaler scaler;
1334 scaler.setFromTree(*scalerTree);
1335 if (weights && weights->isValid()) {
1336 scaler.setScalerWeights(*weights);
1337 scaler.useWeights(true);
1338 const float integrationTimeMS = std::min(std::abs(weights->mFirstTimeStampMS), 500.f);
1339 if (integrationTimeMS > 0) {
1340 scaler.setIonDriftTimeMS(integrationTimeMS);
1341 }
1342 // the weights are only defined for a full window before the timestamp, otherwise use the unweighted mean of the closest window
1343 const double endMS = std::min(scaler.getEndTimeStampMS(o2::tpc::Side::A), scaler.getEndTimeStampMS(o2::tpc::Side::C));
1344 if (static_cast<double>(timeStamp) - scaler.getIonDriftTimeMS() < scaler.getStartTimeStampMS() || timeStamp > endMS) {
1345 LOGP(warning, "time stamp {} not covered by the TPC scaler window [{}, {}] ms, using the unweighted mean scaler", timeStamp, scaler.getStartTimeStampMS() + scaler.getIonDriftTimeMS(), endMS);
1346 scaler.useWeights(false);
1347 }
1348 }
1349 const float instIDC = (scaler.getMeanScaler(timeStamp, o2::tpc::Side::A) + scaler.getMeanScaler(timeStamp, o2::tpc::Side::C)) / 2;
1350 LOGP(info, "TPC scaler at {}: {}, map mean IDC: {}", timeStamp, instIDC, meanIDC);
1351 return instIDC > 0 ? instIDC / meanIDC : -1.f;
1352}
