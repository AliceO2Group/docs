24std::vector<uint8_t> sampleData(size_t n)
25{
26 // Compressible, but not trivially: a ramp with some noise.
27 std::vector<uint8_t> data(n);
28 std::mt19937 rng(42);
29 for (size_t i = 0; i < n; ++i) {
30 data[i] = (uint8_t)(i / 7 + (rng() % 4));
31 }
32 return data;
33}

72TEST_CASE("bounded decompression stops exactly at maxOut")
73{
74 auto data = sampleData(40000);
75 for (auto algo : allAlgos) {
76 for (size_t maxOut : {size_t(1), size_t(9999), size_t(10000), size_t(25001), size_t(39999)}) {
77 auto payload = rootCompress(algo, data, 10000);
78 // One guard byte after the bound must stay untouched.
79 std::vector<uint8_t> out(maxOut + 1, 0xAB);
80 REQUIRE(decompressBounded(payload, data.size(), out.data(), maxOut) == maxOut);
81 CHECK(std::equal(out.begin(), out.begin() + maxOut, data.begin()));
82 CHECK(out[maxOut] == 0xAB);

87TEST_CASE("prefix reads of large chunks stop exactly at the bound")
88{
89 // Several zstd blocks (at most 128 KiB each) per chunk, so that the bound
90 // falls before, at and after block boundaries, and in the last block.
91 auto data = sampleData(1 << 20);
92 constexpr size_t block = 128 * 1024;
93 for (auto algo : allAlgos) {
94 auto payload = rootCompress(algo, data, 600000);
95 for (size_t maxOut : {size_t(1), block - 1, block, block + 1, 2 * block + 17, size_t(599999), size_t(600000), size_t(600001), data.size() - 1}) {
96 INFO("algorithm " << (int)algo << ", bound " << maxOut);
97 std::vector<uint8_t> out(maxOut + 64, 0xAB);
98 PayloadDecompressor d{payload, (uint32_t)data.size(), "memory", 0};
99 d.readPrefix(out.data(), maxOut);
100 CHECK(std::equal(out.begin(), out.begin() + maxOut, data.begin()));
101 CHECK(std::all_of(out.begin() + maxOut, out.end(), [](uint8_t b) { return b == 0xAB; }));
102 }
103 }
104}

116TEST_CASE("incremental reads of odd sizes produce the same bytes")
117{
118 auto data = sampleData(70000);
119 for (auto algo : allAlgos) {
120 auto payload = rootCompress(algo, data, 16384);
121 PayloadDecompressor d{payload, (uint32_t)data.size(), "memory", 0};
122 std::vector<uint8_t> out(data.size());
123 size_t pos = 0;
124 for (size_t step = 1; pos < data.size(); step = step * 3 + 1) {
125 auto n = std::min(step, data.size() - pos);
126 d.read(out.data() + pos, n);
127 pos += n;
128 }
129 CHECK(out == data);
130 CHECK(d.produced() == data.size());
131 }

147TEST_CASE("corrupt payloads raise CorruptData")
148{
149 auto data = sampleData(30000);
150 for (auto algo : allAlgos) {
151 auto payload = rootCompress(algo, data);
152 std::vector<uint8_t> out(data.size());
153 // Flip bits in the middle of the compressed stream.
154 auto corrupt = payload;
155 for (size_t i = corrupt.size() / 3; i < corrupt.size() / 3 + 16; ++i) {
156 corrupt[i] ^= 0x5A;
157 }
158 // zlib and zstd may decode garbage of the right size without noticing; what
159 // matters is that nothing crashes and that detected errors are CorruptData.
160 try {
161 decompressBounded(corrupt, data.size(), out.data(), out.size());
162 } catch (CorruptData const&) {
163 }
164 // A wrong uncompressed size in the header is always detected.
165 auto wrongSize = payload;
166 wrongSize[6] ^= 0x01;
167 CHECK_THROWS_AS(decompressBounded(wrongSize, data.size(), out.data(), out.size()), CorruptData);
168 }
169}
