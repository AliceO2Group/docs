27// testing default binning and custom binning
28BOOST_AUTO_TEST_CASE(TrackResidualsBinning_test)
29{
30 TrackResiduals residUniform;
31 TrackResiduals residCustom;
32 residUniform.initBinning();
33
34 // fill uniform binning in Y/X manually
35 std::vector<float> binY2X;
36 for (int iBin = 0; iBin <= residUniform.getNY2XBins(); ++iBin) {
37 binY2X.push_back(-1. + 2. / residUniform.getNY2XBins() * iBin);
38 }
39 residCustom.setY2XBinning(binY2X);
40 // fill uniform binning in Z/X manually
41 std::vector<float> binZ2X;
42 for (int iBin = 0; iBin <= residUniform.getNZ2XBins(); ++iBin) {
43 binZ2X.push_back(1. / residUniform.getNZ2XBins() * iBin);
44 }
45 residCustom.setZ2XBinning(binZ2X);
46 residCustom.initBinning();
47
48 float x, p, z, xRef, pRef, zRef;
49 for (int ix = 0; ix < residUniform.getNXBins(); ++ix) {
50 for (int ip = 0; ip < residUniform.getNY2XBins(); ++ip) {
51 for (int iz = 0; iz < residUniform.getNZ2XBins(); ++iz) {
52 residUniform.getVoxelCoordinates(0, ix, ip, iz, xRef, pRef, zRef);
53 residCustom.getVoxelCoordinates(0, ix, ip, iz, x, p, z);
54 BOOST_CHECK_SMALL(x - xRef, 1e-6f);
55 BOOST_CHECK_SMALL(p - pRef, 1e-6f);
56 BOOST_CHECK_SMALL(z - zRef, 1e-6f);
57 }
58 }
59 }
