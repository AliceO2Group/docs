112void GPUDisplay::LoadGLAnimation()
113{
114 StopAnimation();
115 const char* file = mCfgO.glAnimationFile.c_str();
116 FILE* ftmp = fopen(file, "rb");
117 if (!ftmp) {
118 SetInfo("Error opening file %s", file);
119 return;
120 }
121 GPUSettingsDisplayLight cfg;
122 int32_t size = 0;
123 bool ok = fread(&cfg, sizeof(cfg), 1, ftmp) == 1 && fread(&size, sizeof(size), 1, ftmp) == 1 && size >= 0;
124 if (ok) { // Check the file size before allocating memory or changing anything
125 const long start = ftell(ftmp);
126 fseek(ftmp, 0, SEEK_END);
127 ok = (size_t)ftell(ftmp) == start + (size_t)size * (sizeof(float) + sizeof(animationVector));
128 fseek(ftmp, start, SEEK_SET);
129 }
130 if (ok) {
131 mCfgC.animationPoints.values.resize(size);
132 mAnimationVectors.resize(size);
133 ok = ok && fread(mCfgC.animationPoints.values.data(), sizeof(float), size, ftmp) == (size_t)size;
134 ok = ok && fread(mAnimationVectors.data(), sizeof(mAnimationVectors[0]), size, ftmp) == (size_t)size;
135 mCfgC.animationPoints.selected = -1;
136 if (ok) {
137 mCfgL = cfg;
138 updateConfig();
139 } else {
140 resetAnimation(); // Read error after the size check: no incomplete path
141 }
142 }
143 fclose(ftmp);
144 SetInfo(ok ? "Animation path loaded from file %s" : "Error reading animation path from file %s", file);
145}

155void GPUDisplay::DoScreenshot(const char* filename, std::vector<char>& pixels, float animateTime)
156{
157 size_t screenshot_x = mBackend->mScreenWidth * mCfgR.screenshotScaleFactor;
158 size_t screenshot_y = mBackend->mScreenHeight * mCfgR.screenshotScaleFactor;
159 size_t size = 4 * screenshot_x * screenshot_y;
160 if (size != pixels.size()) {
161 GPUError("Pixel array of incorrect size obtained");
162 filename = nullptr;
163 }
164
165 if (filename) {
166 FILE* fp = fopen(filename, "w+b");
167 if (fp == nullptr) {
168 GPUError("Error opening screenshot file %s", filename);
169 return;
170 }
171
172 BITMAPFILEHEADER bmpFH;
173 BITMAPINFOHEADER bmpIH;
174 memset(&bmpFH, 0, sizeof(bmpFH));
175 memset(&bmpIH, 0, sizeof(bmpIH));
176
177 bmpFH.bfType = 19778; //"BM"
178 bmpFH.bfSize = sizeof(bmpFH) + sizeof(bmpIH) + size;
179 bmpFH.bfOffBits = sizeof(bmpFH) + sizeof(bmpIH);
180
181 bmpIH.biSize = sizeof(bmpIH);
182 bmpIH.biWidth = screenshot_x;
183 bmpIH.biHeight = screenshot_y;
184 bmpIH.biPlanes = 1;
185 bmpIH.biBitCount = 32;
186 bmpIH.biCompression = BI_RGB;
187 bmpIH.biSizeImage = size;
188 bmpIH.biXPelsPerMeter = 5670;
189 bmpIH.biYPelsPerMeter = 5670;
190
191 fwrite(&bmpFH, 1, sizeof(bmpFH), fp);
192 fwrite(&bmpIH, 1, sizeof(bmpIH), fp);
193 fwrite(pixels.data(), 1, size, fp);
194 fclose(fp);
195 }
196}

198void GPUDisplay::showInfo(const char* info)
199{
200 if (mCfgO.printInfoTextGUI) {
201 mGUIInfoStatus = info;
202 }
203 if (!(mCfgO.printInfoText & 1)) {
204 return;
205 }
206 float colorValue = mCfgL.invertColors ? 0.f : 1.f;
207 GLPrintText(info, 40.f, 20.f + std::max(20, mDrawTextFontSize + 4), colorValue, colorValue, colorValue, 1);
208 if (mInfoText2Timer.IsRunning()) {
209 const float time = mInfoText2Timer.GetCurrentElapsedTime(), shown = mConfig.internal.setInfoTime, transition = mConfig.internal.setInfoTransition;
210 if (time >= shown + transition) {
211 mInfoText2Timer.Reset();
212 } else {
213 GLPrintText(mInfoText2, 40.f, 20.f, colorValue, colorValue, colorValue, time <= shown ? 1.f : 1.f - (time - shown) / transition); // Opaque, then fading out
214 }
215 }
216 if (mInfoHelpTimer.IsRunning() || mCfgO.printInfoTextAlways) {
217 if (mInfoHelpTimer.GetCurrentElapsedTime() >= 6) {
218 mInfoHelpTimer.Reset();
219 } else {
220 PrintGLHelpText(colorValue);
221 }
222 }
223}

225float GPUDisplay::consoleVerticalFraction()
226{
227 const bool running = mConsoleTimer.IsRunning();
228 if (!mCfgC.showConsole && !running) {
229 return 0.f;
230 }
231 const double time = mConsoleTimer.GetCurrentElapsedTime();
232 if (running && !mCfgC.showConsole && time >= mConfig.internal.consoleDelay) { // Keeps running while open, for the blinking cursor
233 mConsoleTimer.Stop();
234 }
235 const float progress = running && mConfig.internal.consoleDelay > 0.f ? std::min(1.f, (float)(time / mConfig.internal.consoleDelay)) : 1.f;
236 return mConfig.internal.consoleVertical * (mCfgC.showConsole ? progress : 1.f - progress);
237}
