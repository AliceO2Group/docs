39inline std::vector<uint8_t> compressChunk(Algo algo, std::span<const uint8_t> in)
40{
41 std::vector<uint8_t> out;
42 switch (algo) {
43 case Algo::ZSTD: {
44 out.resize(ZSTD_compressBound(in.size()));
45 out.resize(ZSTD_compress(out.data(), out.size(), in.data(), in.size(), 5));
46 } break;
47 case Algo::ZLIB: {
48 uLongf n = compressBound(in.size());
49 out.resize(n);
50 compress2(out.data(), &n, in.data(), in.size(), 6);
51 out.resize(n);
52 } break;
53 case Algo::LZMA: {
54 out.resize(lzma_stream_buffer_bound(in.size()));
55 size_t pos = 0;
56 lzma_easy_buffer_encode(6, LZMA_CHECK_CRC32, nullptr, in.data(), in.size(), out.data(), &pos, out.size());
57 out.resize(pos);
58 } break;
59 case Algo::LZ4: {
60 std::vector<uint8_t> block(LZ4_compressBound(in.size()));
61 block.resize(LZ4_compress_default((char const*)in.data(), (char*)block.data(), in.size(), block.size()));
62 auto checksum = xxh64(block);
63 for (int i = 7; i >= 0; --i) {
64 out.push_back((checksum >> (8 * i)) & 0xff);
65 }
66 out.insert(out.end(), block.begin(), block.end());
67 } break;
68 case Algo::None:
69 throw std::logic_error("no chunk for uncompressed data");
70 }
71 return out;
72}

76inline std::vector<uint8_t> rootCompress(Algo algo, std::span<const uint8_t> in, size_t chunkSize = 0xffffff)
77{
78 static const char* tags[] = {"ZS", "ZL", "XZ", "L4"};
79 std::vector<uint8_t> out;
80 for (size_t pos = 0; pos < in.size(); pos += chunkSize) {
81 auto piece = in.subspan(pos, std::min(chunkSize, in.size() - pos));
82 auto c = compressChunk(algo, piece);
83 out.push_back(tags[(int)algo][0]);
84 out.push_back(tags[(int)algo][1]);
85 out.push_back(algo == Algo::ZLIB ? 8 : 1);
86 for (auto v : {c.size(), piece.size()}) {
87 out.push_back(v & 0xff);
88 out.push_back((v >> 8) & 0xff);
89 out.push_back((v >> 16) & 0xff);
90 }
91 out.insert(out.end(), c.begin(), c.end());
92 }
93 return out;
94}

96struct BigEndianWriter {
97 std::vector<uint8_t> bytes;
98 template <typename T>
99 void put(T v)
100 {
101 for (int i = sizeof(T) - 1; i >= 0; --i) {
102 bytes.push_back((uint64_t(v) >> (8 * i)) & 0xff);
103 }
104 }
105 void string(std::string const& s)
106 {
107 bytes.push_back(s.size());
108 bytes.insert(bytes.end(), s.begin(), s.end());
109 }
110};

116inline std::vector<uint8_t> makeBasket(Algo algo, std::span<const uint8_t> values, int32_t nevbuf, size_t tailBytes = 0, size_t chunkSize = 0xffffff, bool ioBits = false)
117{
118 std::vector<uint8_t> raw(values.begin(), values.end());
119 // The tail is poisoned: a decoder that reads past the value region sees it.
120 raw.resize(values.size() + tailBytes, 0xEE);
121 std::vector<uint8_t> payload = algo == Algo::None ? raw : rootCompress(algo, raw, chunkSize);
122
123 BigEndianWriter h;
124 auto build = [&](uint16_t keylen) {
125 h.bytes.clear();
126 h.put<uint32_t>(keylen + payload.size()); // fNbytes
127 h.put<int16_t>(4); // key version
128 h.put<uint32_t>(raw.size()); // fObjlen
129 h.put<uint32_t>(0); // fDatime
130 h.put<uint16_t>(keylen);
131 h.put<uint16_t>(1); // fCycle
132 h.put<uint32_t>(0); // fSeekKey
133 h.put<uint32_t>(0); // fSeekPdir
134 h.string("TBasket"); // class, name, title
135 h.string("fX");
136 h.string("O2track");
137 h.put<uint16_t>(3); // fVersion
138 h.put<int32_t>(32000); // fBufferSize
139 h.put<int32_t>(ioBits ? -4 : 0); // fNevBufSize
140 if (ioBits) {
141 h.put<uint8_t>(0); // fIOBits
142 }
143 h.put<int32_t>(nevbuf); // fNevBuf
144 h.put<int32_t>(keylen + values.size()); // fLast
145 h.put<uint8_t>(tailBytes ? 1 : 2); // flag
146 };
147 build(0);
148 auto keylen = (uint16_t)h.bytes.size();
149 build(keylen);
150 h.bytes.insert(h.bytes.end(), payload.begin(), payload.end());
151 return h.bytes;
152}

155class MemoryByteSource : public ByteSource
156{
157 public:
158 explicit MemoryByteSource(std::vector<uint8_t> data, bool mappable = true) : mData(std::move(data)), mMappable(mappable) {}
159 uint64_t size() const override { return mData.size(); }
160 void readInto(uint64_t off, size_t len, void* dst) override
161 {
162 if (off > mData.size() || len > mData.size() - off) {
163 throw CorruptData("memory", off, "read past the end");
164 }
165 if (len == 0) {
166 return;
167 }
168 std::memcpy(dst, mData.data() + off, len);
169 ++mCalls;
170 mBytes += len;
171 }
172 void readvInto(std::span<ReadRequest> requests) override
173 {
174 for (auto& r : requests) {
175 readInto(r.offset, r.length, r.dst);
176 }
177 }
178 std::span<const uint8_t> tryMap(uint64_t off, size_t len) override
179 {
180 if (!mMappable) {
181 return {};
182 }
183 if (off > mData.size() || len > mData.size() - off) {
184 throw CorruptData("memory", off, "map past the end");
185 }
186 return std::span<const uint8_t>(mData).subspan(off, len);
187 }
188 uint64_t bytesRead() const override { return mBytes; }
189 uint64_t readCalls() const override { return mCalls; }
190 std::string url() const override { return "memory"; }
191
192 private:
193 std::vector<uint8_t> mData;
194 bool mMappable;
195 uint64_t mBytes = 0, mCalls = 0;
196};
