54void TPCScaler::dumpToFileSlices(const char* file, const char* name, int minutesPerObject, float marginMS, float marginCCDBMinutes)
55{
56 if (getNValues(o2::tpc::Side::A) != getNValues(o2::tpc::Side::C)) {
57 LOGP(error, "Number of points stored for A-side and C-side is different");
58 return;
59 }
60 const long marginCCDBMS = marginCCDBMinutes * 60 * 1000;
61 const float msPerObjectTmp = minutesPerObject * 60 * 1000;
62 const int valuesPerSlice = static_cast<int>(msPerObjectTmp / mIntegrationTimeMS);
63 const int marginPerSlice = static_cast<int>(marginMS / mIntegrationTimeMS);
64 int nSlices = getNValues(Side::A) / valuesPerSlice; // number of output objects
65
66 LOGP(info, "Producing {} objects with a CCDB margin of {} ms and {} margin per slice", nSlices, marginCCDBMS, marginPerSlice);
67 if (nSlices == 0) {
68 nSlices = 1;
69 }
70
71 for (int i = 0; i < nSlices; ++i) {
72 const int idxDataStart = (i == 0) ? 0 : (i * valuesPerSlice - marginPerSlice);
73 int idxDataEnd = (i == nSlices - 1) ? getNValues(Side::A) : ((i + 1) * valuesPerSlice + marginPerSlice);
74 if (idxDataEnd > getNValues(Side::A)) {
75 idxDataEnd = getNValues(Side::A);
76 }
77
78 TPCScaler scaler;
79 scaler.mIonDriftTimeMS = mIonDriftTimeMS;
80 scaler.mRun = mRun;
81 scaler.mTimeStampMS = (i == 0) ? mTimeStampMS : (mTimeStampMS + idxDataStart * static_cast<double>(mIntegrationTimeMS));
82 scaler.mFirstTFOrbit = mFirstTFOrbit + (scaler.mTimeStampMS - mTimeStampMS) / (o2::constants::lhc::LHCOrbitMUS * 0.001);
83 scaler.mIntegrationTimeMS = mIntegrationTimeMS;
84 scaler.mScalerA = std::vector<float>(mScalerA.begin() + idxDataStart, mScalerA.begin() + idxDataEnd);
85 scaler.mScalerC = std::vector<float>(mScalerC.begin() + idxDataStart, mScalerC.begin() + idxDataEnd);
86
87 const long timePerSliceMS = valuesPerSlice * mIntegrationTimeMS;
88 const long tsCCDBStart = mTimeStampMS + i * timePerSliceMS;
89 const long tsCCDBStartMargin = (i == 0) ? (tsCCDBStart - marginCCDBMS) : tsCCDBStart;
90 const long tsCCDBEnd = (i == nSlices - 1) ? (getEndTimeStampMS(o2::tpc::Side::A) + marginCCDBMS) : (tsCCDBStart + timePerSliceMS);
91 const std::string fileOut = fmt::format("{}_{}_{}_{}.root", file, i, tsCCDBStartMargin, tsCCDBEnd);
92 scaler.dumpToFile(fileOut.data(), name);
93 }
94}

116float TPCScaler::getMeanScaler(double timestamp, o2::tpc::Side side) const
117{
118 // index to data buffer
119 const int idxData = getDataIdx(timestamp);
120 const int nVals = getNValuesIonDriftTime();
121 const int nValues = getNValues(side);
122 if ((nVals == 0) || (nVals > nValues)) {
123 return -1;
124 LOGP(error, "Empty data provided {}", nValues);
125 }
126
127 // clamp indices to min and max
128 const int lastIdx = std::clamp(idxData, nVals, nValues);
129 const int firstIdx = (lastIdx == nValues) ? (nValues - nVals) : std::clamp(idxData - nVals, 0, nValues);
130
131 // sump up values from last ion drift time
132 float sum = 0;
133 float sumW = 0;
134 int nOutOfRange = 0;
135 const bool useWeights = mUseWeights && getScalerWeights().isValid();
136 for (int i = firstIdx; i < lastIdx; ++i) {
137 float weight = 1;
138 if (useWeights) {
139 const double relTSMS = mTimeStampMS + i * mIntegrationTimeMS - timestamp;
140 nOutOfRange += !getScalerWeights().isInRange(relTSMS);
141 weight = getScalerWeights().getWeight(relTSMS);
142 }
143 sum += getScalers(i, side) * weight;
144 sumW += weight;
145 }
146
147 // values outside the stored weights (e.g. timestamp within the first ion drift time of this object, where the clamped range extends
148 // beyond the timestamp) are used with weight 1: reported once per call and at most MaxOutOfRangeReports times per process
149 if (nOutOfRange > 0) {
150 constexpr int MaxOutOfRangeReports = 20;
151 static std::atomic<int> nReports{0};
152 const int report = nReports++;
153 if (report < MaxOutOfRangeReports) {
154 LOGP(error, "{} of {} scaler values for timestamp {} outside the weights (first timestamp {} ms, sampling {} ms): weight 1 used{}", nOutOfRange,
155 lastIdx - firstIdx, timestamp, getScalerWeights().mFirstTimeStampMS, getScalerWeights().mSamplingTimeMS,
156 (report + 1 == MaxOutOfRangeReports) ? ", further messages suppressed" : "");
157 }
158 }
159
160 if (sumW != 0) {
161 return (sum / sumW);
162 }
163 return 0;
164}

166bool TPCScalerWeights::isInRange(float deltaTime) const
167{
168 const int idx = (deltaTime - mFirstTimeStampMS) / mSamplingTimeMS;
169 return (idx >= 0) && (idx < static_cast<int>(mWeights.size()));
170}

172float TPCScalerWeights::getWeight(float deltaTime) const
173{
174 const float idxF = (deltaTime - mFirstTimeStampMS) / mSamplingTimeMS;
175 const int idx = idxF;
176 if (!isInRange(deltaTime)) {
177 // set weight 1 to in case it is out of range. This can only happen if the TPC scaler is not valid for given time (reported in getMeanScaler)
178 return 1;
179 }
180
181 if ((idxF == idx) || (idx == mWeights.size() - 1)) {
182 // no interpolation required
183 return mWeights[idx];
184 } else {
185 // interpolate scaler
186 const float y0 = mWeights[idx];
187 const float y1 = mWeights[idx + 1];
188 const float x0 = idx;
189 const float x1 = idx + 1;
190 const float x = idxF;
191 const float y = ((y0 * (x1 - x)) + y1 * (x - x0)) / (x1 - x0);
192 return y;
193 }
194}
