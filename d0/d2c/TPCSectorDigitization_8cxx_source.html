54std::vector<dataformats::RangeReference<int, int>> digitizeSector(Digitizer& digitizer, const steer::DigitizationContext& context,
55 const std::vector<TChain*>& simChains, int sector, ULong64_t timeframe,
56 SectorFlushBuffers& buffers, const std::function<void()>& onFlush)
57{
58 std::vector<dataformats::RangeReference<int, int>> events;
59 const bool isContinuous = digitizer.isContinuousReadout();
60 const auto& irecords = context.getEventRecords();
61 const auto& eventParts = context.getEventParts();
62
63 digitizer.setSector(sector);
64 digitizer.setSeedingTimeframe(timeframe);
65 digitizer.init();
66
67 size_t digitCounter = 0;
68 auto flush = [&](bool finalFlush = false) {
69 buffers.digits.clear();
70 buffers.labels.clear();
71 buffers.commonMode.clear();
72 digitizer.flush(buffers.digits, buffers.labels, buffers.commonMode, finalFlush);
73 LOG(info) << "TPC: Flushed " << buffers.digits.size() << " digits, " << buffers.labels.getNElements() << " labels and " << buffers.commonMode.size() << " common mode entries";
74 onFlush();
75 digitCounter += buffers.digits.size();
76 };
77
78 if (isContinuous) {
79 auto& hbfu = o2::raw::HBFUtils::Instance();
80 double time = hbfu.getFirstIRofTF(o2::InteractionRecord(0, hbfu.orbitFirstSampled)).bc2ns() / 1000.;
81 digitizer.setOutputDigitTimeOffset(time);
82 if (!irecords.empty()) {
83 digitizer.setStartTime(irecords[0].getTimeNS() / 1000.f);
84 }
85 }
86
87 // loop over all composite collisions given from context
88 // (aka loop over all the interaction records)
89 for (int collID = 0; collID < irecords.size(); ++collID) {
90 const double eventTime = irecords[collID].getTimeNS() / 1000.f;
91 LOG(info) << "TPC: Event time " << eventTime << " us";
92 digitizer.setEventTime(eventTime);
93 if (!isContinuous) {
94 digitizer.setStartTime(eventTime);
95 }
96 size_t startSize = digitCounter;
97
98 // for each collision, loop over the constituents event and source IDs
99 // (background signal merging is basically taking place here)
100 for (auto& part : eventParts[collID]) {
101 const int eventID = part.entryID;
102 const int sourceID = part.sourceID;
103
104 // get the hits for this event and this source
105 std::vector<o2::tpc::HitGroup> hitsLeft;
106 std::vector<o2::tpc::HitGroup> hitsRight;
107 context.retrieveHits(simChains, getBranchNameLeft(sector).c_str(), part.sourceID, part.entryID, &hitsLeft);
108 context.retrieveHits(simChains, getBranchNameRight(sector).c_str(), part.sourceID, part.entryID, &hitsRight);
109 LOG(debug) << "TPC: Found " << hitsLeft.size() << " hit groups left and " << hitsRight.size() << " hit groups right in collision " << collID << " eventID " << part.entryID;
110
111 digitizer.process(hitsLeft, eventID, sourceID);
112 digitizer.process(hitsRight, eventID, sourceID);
113
114 flush();
115
116 if (!isContinuous) {
117 events.emplace_back(startSize, buffers.digits.size());
118 }
119 }
120 }
121
122 // final flushing step; getting everything not yet written out
123 if (isContinuous) {
124 LOG(info) << "TPC: Final flush";
125 flush(true);
126 events.emplace_back(0, digitCounter); // all digits are grouped to 1 super-event pseudo-triggered mode
127 }
128 return events;
129}

144void TPCDigitizerConditions::addInputs(std::vector<framework::InputSpec>& inputs, int distortionType)
145{
146 using framework::Lifetime;
147 VDriftHelper::requestCCDBInputs(inputs);
148 inputs.emplace_back("gasparam", o2::header::gDataOriginTPC, "TPCGASPARAM", 0, Lifetime::Condition, framework::ccdbParamSpec(CDBTypeMap.at(CDBType::ParGas)));
149 inputs.emplace_back("gemparam", o2::header::gDataOriginTPC, "TPCGEMPARAM", 0, Lifetime::Condition, framework::ccdbParamSpec(CDBTypeMap.at(CDBType::ParGEM)));
150 if (distortionType) {
151 inputs.emplace_back("tpcdistortions", o2::header::gDataOriginTPC, "TPCDIST", 0, Lifetime::Condition, framework::ccdbParamSpec(CDBTypeMap.at(CDBType::DistortionMapMC), {}, 1)); // time-dependent
152 // load derivative map in case scaling was requested
153 if (distortionType == 2) {
154 inputs.emplace_back("tpcdistortionsderiv", o2::header::gDataOriginTPC, "TPCDISTDERIV", 0, Lifetime::Condition, framework::ccdbParamSpec(CDBTypeMap.at(CDBType::DistortionMapDerivMC), {}, 1)); // time-dependent
155 }
156 }

159void TPCDigitizerConditions::init(framework::InitContext& ic, int distortionType)
160{
161 mDistortionType = distortionType;
162 mRecalcDistortions = !(ic.options().get<bool>("do-not-recalculate-distortions"));
163 SpaceCharge<float>::setNThreads(ic.options().get<int>("n-threads-distortions"));
164 mUseCalibrationsFromCCDB = ic.options().get<bool>("TPCuseCCDB");
165 mMeanLumiDistortions = ic.options().get<float>("meanLumiDistortions");
166 mMeanLumiDistortionsDerivative = ic.options().get<float>("meanLumiDistortionsDerivative");
167 LOG(info) << "TPC calibrations from CCDB: " << mUseCalibrationsFromCCDB;
168}

170bool TPCDigitizerConditions::finaliseCCDB(framework::ConcreteDataMatcher& matcher, void* obj, Digitizer& digitizer)
171{
172 using framework::ConcreteDataMatcher;
173 using SC = SpaceCharge<float>;
174 if (mVDriftHelper.accountCCDBInputs(matcher, obj)) {
175 return true;
176 }
177 if (matcher == ConcreteDataMatcher(o2::header::gDataOriginTPC, "TPCDIST", 0)) {
178 LOGP(info, "Updating distortion map");
179 digitizer.setUseSCDistortions(static_cast<SC*>(obj));
180 if (mMeanLumiDistortions >= 0) {
181 digitizer.setMeanLumiDistortions(mMeanLumiDistortions);
182 }
183 return true;
184 }
185 if (matcher == ConcreteDataMatcher(o2::header::gDataOriginTPC, "TPCDISTDERIV", 0)) {
186 LOGP(info, "Updating reference distortion map");
187 digitizer.setSCDistortionsDerivative(static_cast<SC*>(obj));
188 if (mMeanLumiDistortionsDerivative >= 0) {
189 digitizer.setMeanLumiDistortionsDerivative(mMeanLumiDistortionsDerivative);
190 }
191 return true;
192 }
193 if (matcher == ConcreteDataMatcher(o2::header::gDataOriginTPC, "TPCGASPARAM", 0)) {
194 LOGP(info, "TPC gas param updated");
195 ParameterGas::Instance().printKeyValues(true, true);
196 return true;
197 }
198 if (matcher == ConcreteDataMatcher(o2::header::gDataOriginTPC, "TPCGEMPARAM", 0)) {
199 LOGP(info, "TPC GEM param updated");
200 ParameterGEM::Instance().printKeyValues(true, true);
201 return true;
202 }
203 return false;
204}

206void TPCDigitizerConditions::update(framework::ProcessingContext& pc, Digitizer& digitizer, const std::function<void(float, float)>& setVDrift)
207{
208 using SC = SpaceCharge<float>;
209 auto& cdb = CDBInterface::instance();
210 cdb.setUseDefaults(!mUseCalibrationsFromCCDB);
211 // whatever are global settings for CCDB usage, we have to extract the TPC vdrift from CCDB for anchored simulations
212 mVDriftHelper.extractCCDBInputs(pc);
213 pc.inputs().get<ParameterGas*>("gasparam");
214 pc.inputs().get<ParameterGEM*>("gemparam");
215 if (mDistortionType) {
216 pc.inputs().get<SC*>("tpcdistortions");
217 if (mDistortionType == 2) {
218 pc.inputs().get<SC*>("tpcdistortionsderiv");
219 digitizer.setLumiScaleFactor();
220 if (mRecalcDistortions) {
221 digitizer.recalculateDistortions();
222 }
223 }
224 }
225
226 if (mVDriftHelper.isUpdated()) {
227 const auto& vd = mVDriftHelper.getVDriftObject();
228 LOGP(info, "Updating TPC fast transform map with new VDrift factor of {} wrt reference {} and DriftTimeOffset correction {} wrt {} from source {}",
229 vd.corrFact, vd.refVDrift, vd.timeOffsetCorr, vd.refTimeOffset, mVDriftHelper.getSourceName());
230 setVDrift(vd.getVDrift(), vd.getTimeOffset());
231 mVDriftHelper.acknowledgeUpdate();
232 }
233
234 if (std::filesystem::exists("ThresholdMap.root")) {
235 LOG(info) << "TPC: Using zero suppression map from 'ThresholdMap.root'";
236 cdb.setThresholdMapFromFile("ThresholdMap.root");
237 }
238
239 if (std::filesystem::exists("GainMap.root")) {
240 LOG(info) << "TPC: Using gain map from 'GainMap.root'";
241 cdb.setGainMapFromFile("GainMap.root");
242 }
243}
