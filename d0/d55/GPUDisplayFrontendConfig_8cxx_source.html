123 void buildHelpText() // Keys left-aligned, the ':' aligned after the longest keys

135 void rebuildHelpText() // Keys of each entry from the current bindings, in the order of add_key_binding: [N/A] if unbound, [a|b] for several keys; entries without bindings (mouse) keep their text
136 {
137 for (auto& [keys, text, bindings] : helpTextEntries) {
138 if (bindings.empty()) {
139 continue;
140 }
141 keys.clear();
142 for (int32_t binding : bindings) {
143 std::string bound;
144 std::vector<int32_t> keyCodes = boundKeys(binding);
145 std::sort(keyCodes.begin(), keyCodes.end()); // By key code, independent of the order of binding
146 for (int32_t k : keyCodes) {
147 bound += (bound.empty() ? "" : "|") + getKeyName(k);
148 }
149 keys += (keys.empty() ? "[" : " [") + (bound.empty() ? "N/A" : bound) + "]";
150 }
151 }
152 buildHelpText();
153 }
154 std::vector<int32_t> boundKeys(int32_t binding) const // Keys bound to the action or key down mapping
155 {
156 if (isKeyMap(binding)) {
157 const int32_t key = currentMappedKeys[binding - KEY_MAP_INDEX];
158 return key < 0 ? std::vector<int32_t>() : std::vector<int32_t>{key};
159 }
160 return currentActionKeys[binding];
161 }
162 void setKeyBinding(uint32_t key, int32_t binding) // Binds the unbound key, also in the reverse lookups (mKeyDownMap is set by the caller)
163 {
164 configKeys[key] = binding;
165 if (isKeyMap(binding)) {
166 currentMappedKeys[binding - KEY_MAP_INDEX] = key;
167 } else {
168 currentActionKeys[binding].emplace_back(key);
169 }
170 helpTextUpToDate = false;
171 }
172 int32_t removeKeyBinding(uint32_t key) // Unbinds the key, also in the reverse lookups (mKeyDownMap is reset by the caller), returns the previous binding
173 {
174 const int32_t binding = configKeys[key];
175 if (isKeyMap(binding)) {
176 currentMappedKeys[binding - KEY_MAP_INDEX] = -1;
177 } else if (binding != KEY_UNBOUND) {
178 std::erase(currentActionKeys[binding], key);
179 }
180 configKeys[key] = KEY_UNBOUND;
181 helpTextUpToDate = false;
182 return binding;
183 }
184 void unbindAllKeys() // Unbinds all keys, also in the reverse lookups (mKeyDownMap is reset by the caller)
185 {
186 configKeys.fill(KEY_UNBOUND);
187 currentMappedKeys.assign(KEY_MAP_MAX, -1);
188 for (auto& keys : currentActionKeys) {
189 keys.clear();
190 }
191 helpTextUpToDate = false;
192 }
193 std::string bindingName(int32_t binding) const // Name of the action or key down mapping, empty if unbound
194 {
195 return isKeyMap(binding) ? keyMapNames[binding - KEY_MAP_INDEX] : (binding >= 0 ? keyActions[binding].actionName : "");
196 }
197 uint32_t keyRTTIIndex(uint32_t key) const // Option changed by the action bound to the key, INVALID_INDEX if none
198 {
199 const int32_t binding = configKeys[key];
200 return binding != KEY_UNBOUND && !isKeyMap(binding) ? keyActions[binding].rttiIndex : INVALID_INDEX;
201 }
202 std::string getKeyName(uint16_t key) const // Name of a special key (KEY_*), the character otherwise, the code if neither, SHIFT_ for shifted special keys
203 {
204 if (key >= 256) {
205 return "SHIFT_" + getKeyName(key - SPECIAL_KEYS_SHIFTED);
206 }
207 return key < keyNames.size() && !keyNames[key].empty() ? keyNames[key] : (key < 32 ? std::to_string(key) : std::string(1, (char)key));
208 }

217{
218 const auto& [oldCfgH, oldCfgL, oldCfgR] = mOldCfg;

220 if (mAnimate && (changedH || changedL || changedR || mCfgC.resetScene.load())) {

231 }

251 }

253 mUpdateTrackFilter = true;

255 mCfgO.rotateModel |= mCfgO.rotateAroundZ;
256 if (mCfgC.showConsole != mConsoleShown) {

276 updateAnimationPointsGUI();

301template <typename... Args>
302void GPUDisplayFrontend::add_key_binding(Args&&... args)
303{
304 constexpr size_t nKeys = (0 + ... + (std::is_same_v<std::remove_cvref_t<Args>, GPUDisplayConfiguration::keyLambdaEntry> || std::is_same_v<std::remove_cvref_t<Args>, GPUDisplayConfiguration::keyDownMapEntry>));
305 static_assert(nKeys >= 1 && (sizeof...(Args) == nKeys + 1 || sizeof...(Args) == nKeys + 2), "add_key_binding needs at least one key, the help text, and optionally the keys to show in the help text");
306 auto argTuple = std::forward_as_tuple(args...);
307 std::string keys;
308 std::vector<int32_t> bindings;
309 auto addKey = [&](auto& key) {
310 const uint16_t character = std::get<0>(key);
311 const std::string name = mConfiguration->getKeyName(character);
312 int32_t binding;
313 if (mConfiguration->configKeys[character] != GPUDisplayConfiguration::KEY_UNBOUND) {
314 throw std::runtime_error("Key [" + name + "] assigned twice in display frontend");
315 }
316 if constexpr (std::is_same_v<std::remove_cvref_t<decltype(key)>, GPUDisplayConfiguration::keyDownMapEntry>) {
317 const uint8_t id = std::get<1>(key);
318 if (mKeyDownMap[id]) {
319 throw std::runtime_error("Key down mapping " + mConfiguration->keyMapNames[id] + " assigned twice in display frontend");
320 }
321 mKeyDownMap[id] = character >= 'a' && character <= 'z' ? character + 'A' - 'a' : character; // mKeys stores letters in upper case
322 binding = GPUDisplayConfiguration::KEY_MAP_INDEX + id;
323 } else if constexpr (std::is_same_v<std::remove_cvref_t<decltype(key)>, GPUDisplayConfiguration::keyLambdaEntry>) {
324 const std::string& actionName = std::get<1>(key);
325 const auto& mapNames = mConfiguration->keyMapNames;
326 if (mConfiguration->keyActionsNames.contains(actionName) || std::find(mapNames.begin(), mapNames.end(), actionName) != mapNames.end()) {
327 throw std::runtime_error("Key action " + actionName + " defined twice in display frontend, or name of a key down mapping");
328 }
329 binding = mConfiguration->keyActions.size();
330 mConfiguration->keyActionsNames.emplace(actionName, binding);
331 mConfiguration->keyActions.push_back({actionName, std::get<2>(key), std::move(std::get<3>(key))});
332 mConfiguration->currentActionKeys.emplace_back();
333 } else {
334 static_assert(false, "invalid code path");
335 }
336 mConfiguration->setKeyBinding(character, binding);
337 bindings.emplace_back(binding);
338 keys += (keys.empty() ? "[" : " [") + name + "]";
339 };
340 [&]<size_t... I>(std::index_sequence<I...>) { (addKey(std::get<I>(argTuple)), ...); }(std::make_index_sequence<nKeys>());
341 if constexpr (sizeof...(Args) == nKeys + 2) {
342 keys = std::get<nKeys + 1>(argTuple);
343 }
344 mConfiguration->helpTextEntries.emplace_back(std::move(keys), std::get<nKeys>(argTuple), std::move(bindings));
345}

446void GPUDisplayFrontend::add_gui_binding(Args&&... args)
447{
448 constexpr size_t nElements = (0 + ... + std::is_same_v<std::remove_cvref_t<Args>, guiElement>);
449 constexpr size_t nHorizontal = (0 + ... + std::is_same_v<std::remove_cvref_t<Args>, bool>);
450 constexpr size_t nVertical = (0 + ... + std::is_same_v<std::remove_cvref_t<Args>, guiVertical>);

452 GPUDisplayGUIChannel::Frame frame;
453 bool horizontal = false; // Initially the flag of the frame, changed by guiVertical
454 auto getHorizontal = [&](auto& arg) {
455 if constexpr (std::is_same_v<std::remove_cvref_t<decltype(arg)>, bool>) {
456 horizontal = arg;
457 }
458 };
459 (getHorizontal(args), ...);
460 auto addElement = [&](auto& element) {
461 using Type = GPUDisplayGUIChannel::ElementType;
462 if (element.type.bitMask) { // The element sets / shows only these bits of the option
463 element.set = [this, set = element.set, get = element.get, mask = element.type.bitMask](const GPUDisplayGUIValue& v) {
464 const int64_t current = std::get<int64_t>(get({mDisplay->mCfgH, mDisplay->mCfgL, mDisplay->mCfgR, mDisplay->mCfgO, mDisplay->mCfgC}));
465 set(std::get<int64_t>(v) ? (current | mask) : (current & ~mask));
466 };
467 element.get = [get = element.get, mask = element.type.bitMask](const GPUDisplayGUIChannel::Settings& cfg) -> GPUDisplayGUIValue { return (int64_t)((std::get<int64_t>(get(cfg)) & mask) != 0); };
