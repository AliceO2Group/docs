145void GPUDisplayFrontendX11::GetKey(XEvent& event, int32_t& keyOut, int32_t& keyPressOut)
146{
147 char tmpString[9];
148 KeySym sym;
149 if (XLookupString(&event.xkey, tmpString, 8, &sym, nullptr) == 0) {
150 tmpString[0] = 0;
151 }
152 int32_t specialKey = GetKey(sym);
153 int32_t localeKey = (uint8_t)tmpString[0];
154 // GPUInfo("Key: keycode %d -> sym %d (%c) key %d (%c) special %d (%c)", (int32_t)event.xkey.keycode, (int32_t)sym, (char)sym, (int32_t)localeKey, (char)localeKey, (int32_t)specialKey, (char)specialKey);
155
156 if (specialKey) {
157 keyOut = keyPressOut = specialKey;
158 } else {
159 keyOut = keyPressOut = localeKey;
160 if (keyPressOut >= 'a' && keyPressOut <= 'z') {
161 keyPressOut += 'A' - 'a';
162 }
163 }

167{

181 glCallLists(strlen(s), GL_UNSIGNED_BYTE, (GLubyte*)s);

188int32_t GPUDisplayFrontendX11::FrontendMain()
189{
190 XSetWindowAttributes windowAttributes;
191 XVisualInfo* visualInfo = nullptr;
192 XEvent event;
193 Colormap colorMap;
194 GLXContext glxContext = nullptr;
195 int32_t errorBase;
196 int32_t eventBase;
197
198 // Open a connection to the X server
199 mXDisplay = XOpenDisplay(nullptr);
200
201 if (mXDisplay == nullptr) {
202 GPUError("could not open display");
203 return (-1);
204 }
205
206 XrmInitialize();
207 if (const char* resources = XResourceManagerString(mXDisplay)) {
208 XrmDatabase db = XrmGetStringDatabase(resources);
209 char* type = nullptr;
210 XrmValue value;
211 if (db && XrmGetResource(db, "Xft.dpi", "Xft.Dpi", &type, &value) && value.addr && atof(value.addr) > 0) {
212 mScale = atof(value.addr) / 96.;
213 }
214 if (db) {
215 XrmDestroyDatabase(db);
216 }
217 }
218
219 // Make sure OpenGL's GLX extension supported
220 if (!glXQueryExtension(mXDisplay, &errorBase, &eventBase)) {
221 GPUError("X server has no OpenGL GLX extension");
222 return (-1);
223 }
224
225 const char* glxExt = glXQueryExtensionsString(mXDisplay, DefaultScreen(mXDisplay));
226 if (strstr(glxExt, "GLX_EXT_swap_control") == nullptr) {
227 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
228 GPUError("GLX_EXT_swap_control not available, cannot change VSync");
229 }
230 vsync_supported = false;
231 } else {
232 vsync_supported = true;
233 }
234
235 // Require MSAA, double buffering, and Depth buffer
236 int32_t attribs[] = {GLX_X_RENDERABLE, True, GLX_DRAWABLE_TYPE, GLX_WINDOW_BIT, GLX_RENDER_TYPE, GLX_RGBA_BIT, GLX_X_VISUAL_TYPE, GLX_TRUE_COLOR, GLX_RED_SIZE, 8, GLX_GREEN_SIZE, 8, GLX_BLUE_SIZE, 8, GLX_ALPHA_SIZE, 8, GLX_DEPTH_SIZE, 24, GLX_STENCIL_SIZE, 8, GLX_DOUBLEBUFFER, True,
237 // GLX_SAMPLE_BUFFERS , 1, //Disable MSAA here, we do it by rendering to offscreenbuffer
238 // GLX_SAMPLES , MSAA_SAMPLES,
239 None};
240
241 GLXFBConfig fbconfig = nullptr;
242 int32_t fbcount;
243 GLXFBConfig* fbc = glXChooseFBConfig(mXDisplay, DefaultScreen(mXDisplay), attribs, &fbcount);
244 if (fbc == nullptr || fbcount == 0) {
245 GPUError("Failed to get MSAA GLXFBConfig");
246 return (-1);
247 }
248 fbconfig = fbc[0];
249 XFree(fbc);
250 visualInfo = glXGetVisualFromFBConfig(mXDisplay, fbconfig);
251
252 if (visualInfo == nullptr) {
253 GPUError("no RGB visual with depth buffer");
254 return (-1);
255 }
256
257 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
258 // Create an OpenGL rendering context
259 glXCreateContextAttribsARBProc glXCreateContextAttribsARB = (glXCreateContextAttribsARBProc)glXGetProcAddressARB((const GLubyte*)"glXCreateContextAttribsARB");
260 if (glXCreateContextAttribsARB) {
261 int32_t context_attribs[] = {
262 GLX_CONTEXT_MAJOR_VERSION_ARB, GL_MIN_VERSION_MAJOR,
263 GLX_CONTEXT_MINOR_VERSION_ARB, GL_MIN_VERSION_MINOR,
264 GLX_CONTEXT_PROFILE_MASK_ARB, mBackend->CoreProfile() ? GLX_CONTEXT_CORE_PROFILE_BIT_ARB : GLX_CONTEXT_COMPATIBILITY_PROFILE_BIT_ARB,
265 None};
266 glxContext = glXCreateContextAttribsARB(mXDisplay, fbconfig, nullptr, GL_TRUE, context_attribs);
267 } else {
268 glxContext = glXCreateContext(mXDisplay, visualInfo, nullptr, GL_TRUE);
269 }
270 if (glxContext == nullptr) {
271 GPUError("could not create rendering context");
272 return (-1);
273 }
274 }
275
276 Window win = RootWindow(mXDisplay, visualInfo->screen);
277 colorMap = XCreateColormap(mXDisplay, win, visualInfo->visual, AllocNone);
278 windowAttributes.colormap = colorMap;
279 windowAttributes.border_pixel = 0;
280 windowAttributes.event_mask = ExposureMask | VisibilityChangeMask | KeyPressMask | KeyReleaseMask | ButtonPressMask | ButtonReleaseMask | PointerMotionMask | StructureNotifyMask | SubstructureNotifyMask | FocusChangeMask | PropertyChangeMask;
281
282 // Create an X window with the selected visual
283 mWindow = XCreateWindow(mXDisplay, win, 50, 50, INIT_WIDTH * mScale, INIT_HEIGHT * mScale, // Position / Width and height of window
284 0, visualInfo->depth, InputOutput, visualInfo->visual, CWBorderPixel | CWColormap | CWEventMask, &windowAttributes);
285 XSetStandardProperties(mXDisplay, mWindow, DISPLAY_WINDOW_NAME, DISPLAY_WINDOW_NAME, None, nullptr, 0, nullptr);
286 XSizeHints sizeHints{};
287 sizeHints.flags = PMinSize;
288 sizeHints.min_width = MIN_WIDTH * mScale;
289 sizeHints.min_height = MIN_HEIGHT * mScale;
290 XSetWMNormalHints(mXDisplay, mWindow, &sizeHints);
291 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
292 glXMakeCurrent(mXDisplay, mWindow, glxContext);
293 }
294 XMapWindow(mXDisplay, mWindow);
295
296 XEvent mapEvent; // Maximize only after the window manager placed the window (on the active monitor), so that it restores to this position when un-maximized
297 XPeekIfEvent(mXDisplay, &mapEvent, [](Display*, XEvent* ev, XPointer window) -> Bool { return ev->type == MapNotify && ev->xmap.window == *(Window*)window; }, (XPointer)&mWindow);
298 applyInitialWindowState();
299
300 // Receive signal when window closed
301 Atom WM_DELETE_WINDOW = XInternAtom(mXDisplay, "WM_DELETE_WINDOW", False);
302 XSetWMProtocols(mXDisplay, mWindow, &WM_DELETE_WINDOW, 1);
303#ifndef GPUCA_DISPLAY_OPENGL_CORE
304 XFontStruct* font_info = nullptr;
305 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
306 // Prepare fonts
307 mFontBase = glGenLists(256);
308 if (!glIsList(mFontBase)) {
309 GPUError("Out of display lists.");
310 return (-1);
311 }
312 const char* f = "fixed";
313 font_info = XLoadQueryFont(mXDisplay, f);
314 if (!font_info) {
315 GPUError("XLoadQueryFont failed.");
316 return (-1);
317 } else {
318 int32_t first = font_info->min_char_or_byte2;
319 int32_t last = font_info->max_char_or_byte2;
320 glXUseXFont(font_info->fid, first, last - first + 1, mFontBase + first);
321 }
322 }
323 mCanDrawText = 1;
324 if (drawTextFontSize() == 0) {
325 drawTextFontSize() = 12;
326 }
327#endif
328
329 if (mBackend->ExtInit()) {
330 fprintf(stderr, "Error initializing backend\n");
331 return (-1);
332 }
333
334 XMapWindow(mXDisplay, mWindow);
335 XFlush(mXDisplay);
336
337 // Enable vsync
338 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL && vsync_supported) {
339 mGlXSwapIntervalEXT = (PFNGLXSWAPINTERVALEXTPROC)glXGetProcAddressARB((const GLubyte*)"glXSwapIntervalEXT");
340 if (mGlXSwapIntervalEXT == nullptr) {
341 GPUError("Cannot enable vsync");
342 return (-1);
343 }
344 mGlXSwapIntervalEXT(mXDisplay, glXGetCurrentDrawable(), vsyncEnabled());
345 }
346
347 if (InitDisplay()) {
348 return (1);
349 }
350
351 {
352 std::lock_guard lock(mExitStateMutex);
353 mDisplayRunning = true;
354 }
355
356 while (1) {
357 do {
358 if (mDisplayControl == 2) {
359 break;
360 }
361 if (!XPending(mXDisplay)) {
362 event.type = Expose;
363 } else {
364 XNextEvent(mXDisplay, &event);
365 }
366 switch (event.type) {
367 case ButtonPress: {
368 if (event.xbutton.button == 4) {
369 mMouseWheel += 100;
370 } else if (event.xbutton.button == 5) {
371 mMouseWheel -= 100;
372 } else {
373 if (event.xbutton.button == 1) {
374 mMouseDn = true;
375 } else if (event.xbutton.button == 2) {
376 mMouseDnM = true;
377 } else {
378 mMouseDnR = true;
379 }
380 mMouseDnX = event.xmotion.x;
381 mMouseDnY = event.xmotion.y;
382 }
383 break;
384 }
385
386 case ButtonRelease: {
387 if (event.xbutton.button != 4 && event.xbutton.button != 5) {
388 if (event.xbutton.button == 1) {
389 mMouseDn = false;
390 } else if (event.xbutton.button == 2) {
391 mMouseDnM = false;
392 } else {
393 mMouseDnR = false;
394 }
395 }
396 break;
397 }
398
399 case KeyPress: {
400 int32_t handleKey = 0, keyPress = 0;
401 GetKey(event, handleKey, keyPress);
402 mKeysShift[keyPress] = mKeys[KEY_SHIFT];
403 mKeys[keyPress] = true;
404 HandleKey(handleKey);
405 break;
406 }
407
408 case KeyRelease: {
409 int32_t handleKey = 0, keyPress = 0;
410 GetKey(event, handleKey, keyPress);
411 mKeys[keyPress] = false;
412 mKeysShift[keyPress] = false;
413 break;
414 }
415
416 case MotionNotify: {
417 mMouseMvX = event.xmotion.x;
418 mMouseMvY = event.xmotion.y;
419 break;
420 }
421
422 case FocusOut: {
423 focusLost();
424 break;
425 }
426
427 case PropertyNotify: {
428 if (event.xproperty.atom == XInternAtom(mXDisplay, "_NET_WM_STATE", False)) {
429 updateWindowState();
430 }
431 break;
432 }
433
434 case Expose: {
435 break;
436 }
437
438 case ConfigureNotify: {
439 ResizeScene(event.xconfigure.width, event.xconfigure.height);
440 break;
441 }
442
443 case ClientMessage: {
444 if (event.xclient.message_type == XInternAtom(mXDisplay, "_NET_WM_STATE", False)) {
445 XFlush(mXDisplay);
446 } else {
447 mDisplayControl = 2;
448 }
449 break;
450 }
451 }
452 } while (XPending(mXDisplay)); // Loop to compress events
453 if (mDisplayControl == 2) {
454 break;
455 }
456
457 HandleExternalGUISignals();
458 DrawGLScene();
459 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
460 glXSwapBuffers(mXDisplay, mWindow);
461 }
462 waitForFPSLimit();
463 }
464
465#ifndef GPUCA_DISPLAY_OPENGL_CORE
466 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
467 glDeleteLists(mFontBase, 256);
468 XUnloadFont(mXDisplay, font_info->fid);
469 }
470#endif
471 ExitDisplay();
472 if (backend()->backendType() == GPUDisplayBackend::TYPE_OPENGL) {
473 glXDestroyContext(mXDisplay, glxContext);
474 }
475 XFree(visualInfo);
476 XDestroyWindow(mXDisplay, mWindow);
477 XCloseDisplay(mXDisplay);
478
479 {
480 std::lock_guard lock(mExitStateMutex);
481 mDisplayRunning = false;
482 }
483
484 return (0);
485}

487void GPUDisplayFrontendX11::updateWindowState()
488{
489 Atom type;
490 int32_t format;
491 unsigned long nItems, bytesAfter;
492 unsigned char* data = nullptr;
493 if (XGetWindowProperty(mXDisplay, mWindow, XInternAtom(mXDisplay, "_NET_WM_STATE", False), 0, 1024, False, XA_ATOM, &type, &format, &nItems, &bytesAfter, &data) != Success || data == nullptr) {
494 return;
495 }
496 const Atom fullScreen = XInternAtom(mXDisplay, "_NET_WM_STATE_FULLSCREEN", False), maxHorz = XInternAtom(mXDisplay, "_NET_WM_STATE_MAXIMIZED_HORZ", False), maxVert = XInternAtom(mXDisplay, "_NET_WM_STATE_MAXIMIZED_VERT", False);
497 bool isFullScreen = false, isMaxHorz = false, isMaxVert = false;
498 for (unsigned long i = 0; i < nItems; i++) {
499 const Atom a = ((Atom*)data)[i];
500 isFullScreen |= a == fullScreen;
501 isMaxHorz |= a == maxHorz;
502 isMaxVert |= a == maxVert;
503 }
504 XFree(data);
505 setWindowState(isFullScreen, isMaxHorz && isMaxVert);
506}

508void GPUDisplayFrontendX11::SwitchFullscreen(bool set)
509{
510 XEvent xev;
511 memset(&xev, 0, sizeof(xev));
512 xev.type = ClientMessage;
513 xev.xclient.window = mWindow;
514 xev.xclient.message_type = XInternAtom(mXDisplay, "_NET_WM_STATE", False);
515 xev.xclient.format = 32;
516 xev.xclient.data.l[0] = 2; // _NET_WM_STATE_TOGGLE
517 xev.xclient.data.l[1] = XInternAtom(mXDisplay, "_NET_WM_STATE_FULLSCREEN", True);
518 xev.xclient.data.l[2] = 0;
519 XSendEvent(mXDisplay, DefaultRootWindow(mXDisplay), False, SubstructureNotifyMask, &xev);
520}
