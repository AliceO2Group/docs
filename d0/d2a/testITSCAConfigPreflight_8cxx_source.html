48BOOST_FIXTURE_TEST_CASE(LegacyNamespaceIsRejectedBeforeAnyUpdate, FatalToExceptionFixture)
49{
50 // Sentinel: if the rejection did not actually run before
51 // ConfigurableParam::updateFromString(), this legacy-namespace string
52 // would still throw from updateFromString() itself (unknown param), so
53 // this alone would not distinguish "preflight fired first" from "update
54 // itself fatal'd" -- the meaningful assertion is in the next test, which
55 // confirms the dedicated param was NOT mutated by the rejected string.
56 BOOST_CHECK_THROW(applyConfigKeyValuesOrFatal("ITSCATrackerParam.trackFollowerTop=1"), std::runtime_error);
57}

59BOOST_FIXTURE_TEST_CASE(RejectedStringNeverReachesConfigurableParamUpdate, FatalToExceptionFixture)
60{
61 // A malicious/confused string mixing a real dedicated-namespace override
62 // with an offending legacy one must not have its dedicated part applied

172BOOST_AUTO_TEST_CASE(DriverResolvesTruthContextIndependentlyOfMCLabels)
173{
174 using namespace o2::framework;
175 std::vector<ConfigParamSpec> specs{
176 {"configKeyValues", VariantType::String, "ITSCommonCATrackerParam.useDiamond=false;ITSVertexerParam.useTruthSeeding=false", {"parameters"}},
177 {"tracking-mode", VariantType::String, "async", {"mode"}},
178 {"vertex-source", VariantType::String, "truth", {"vertices"}},
179 {"truth-context", VariantType::String, "custom-context.root", {"context"}},
180 {"disable-mc", VariantType::Bool, true, {"MC labels"}},
181 {"disable-root-output", VariantType::Bool, false, {"output"}},
182 {"use-geom", VariantType::Bool, false, {"geometry"}},
183 {"use-full-geometry", VariantType::Bool, true, {"geometry alias"}}};
184 auto store = std::make_unique<ConfigParamStore>(specs, std::vector<std::unique_ptr<ParamRetriever>>{});
185 store->preload();
186 store->activate();
187 ConfigParamRegistry registry{std::move(store)};
188 ServiceRegistry services;
189 ConfigContext context{registry, ServiceRegistryRef{services}, 0, nullptr};
190 registry.override("configKeyValues", std::string{"ITSCommonCATrackerParam.trackingMode=0"});
191 BOOST_CHECK(readWorkflowOptions(context).mode == o2::itsmft::TrackingMode::Sync);
192 registry.override("configKeyValues", std::string{"ITSCommonCATrackerParam.trackingMode=-1"});
193 const auto resolved = readWorkflowOptions(context);
194 BOOST_CHECK(resolved.mode == o2::itsmft::TrackingMode::Async);
195 BOOST_CHECK(resolved.vertexSource == VertexSource::Truth);
196 BOOST_CHECK(!resolved.useMC);
197 BOOST_CHECK(resolved.useFullGeometry);
198 BOOST_CHECK_EQUAL(resolved.truthContext, "custom-context.root");
199 registry.override("truth-context", std::string{});
200 BOOST_CHECK_THROW(readWorkflowOptions(context), std::invalid_argument);
201 registry.override("vertex-source", std::string{"diamond"});
202 BOOST_CHECK(readWorkflowOptions(context).vertexSource == VertexSource::Diamond);
203 registry.override("configKeyValues", std::string{"ITSCommonCATrackerParam.nThreads=0"});
204 BOOST_CHECK_THROW(readWorkflowOptions(context), std::invalid_argument);
205 o2::conf::ConfigurableParam::setValue<int>("ITSCommonCATrackerParam", "nThreads", 1);
206}
