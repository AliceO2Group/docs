100inline void DigitTime::addDigit(const MCCompLabel& label, const CRU& cru, GlobalPadNumber globalPad, float signal)
101{
102 auto& paddigit = mGlobalPads[globalPad];
103 if (paddigit.getID() == -1) {
104 // this means we have a new digit
105 paddigit.setID(mDigitCounter++);
106 // could also register this pad in a vector of digits
107 }
108
109 // previous digit for CM and ToT calculation
110 paddigit.addDigit(label, signal, mLabels);
111 // mCommonMode[cru.gemStack()] += signal * 0.5; // TODO: Replace 0.5 by k-factor, take into account ion tail
112}

114inline void DigitTime::reset()

122inline float DigitTime::getCommonMode(const GEMstack& gemstack) const
123{
125 const Mapper& mapper = Mapper::instance();
126 const auto nPads = mapper.getNumberOfPads(gemstack);
127 return mCommonMode[gemstack] / static_cast<float>(nPads);
128}

130template <DigitzationMode MODE>
131inline void DigitTime::fillOutputContainer(std::vector<Digit>& output, dataformats::MCTruthContainer<MCCompLabel>& mcTruth,
132 std::vector<CommonMode>& commonModeOutput, const Sector& sector, TimeBin timeBin,
133 SAMPAProcessing& sampaProcessing, std::vector<std::pair<MCCompLabel, int>>& labelCollector,
134 PrevDigitInfoArray* prevTime, Streamer* debugStream, const CalPad* padParams[3],
135 const CalDet<bool>* deadMap)
136{
137 const auto& mapper = Mapper::instance();
138 const auto& eleParam = ParameterElectronics::Instance();
139
140 // at this point we only have the pure signals from tracks
141 // loop over all pads to calculated ion tail, common mode and ToT for saturated signals
142 for (size_t iPad = 0; iPad < mGlobalPads.size(); ++iPad) {
143 auto& digit = mGlobalPads[iPad];
144 if (prevTime) {
145 auto& prevDigit = (*prevTime)[iPad];
146 if (prevDigit.hasSignal()) {
147 digit.foldSignal(prevDigit, sector.getSector(), iPad, timeBin, debugStream, padParams);
148 }
149 prevDigit.signal = digit.getChargePad(); // to make hasSignal() check work in next time bin
150 }
151 const CRU cru = mapper.getCRU(sector, iPad);
152 const float cmKValue = (padParams[2]) ? padParams[2]->getValue(sector.getSector(), iPad) : 1.f;
153 mCommonMode[cru.gemStack()] += digit.getChargePad() * eleParam.commonModeCoupling * cmKValue; // TODO: Add stack-by-stack variation?
154 }
155
156 // fill common mode output container
157 for (size_t i = 0; i < mCommonMode.size(); ++i) {
158 const float cm = getCommonMode(GEMstack(i));
159 if (cm > 0.) {
160 commonModeOutput.push_back({cm, timeBin, static_cast<unsigned char>(i)});
161 }
162 }
163
164 for (size_t iPad = 0; iPad < mGlobalPads.size(); ++iPad) {
165 auto& digit = mGlobalPads[iPad];
166 if (eleParam.doNoiseEmptyPads || (digit.getChargePad() > 0.f)) {
167 PrevDigitInfo prevDigit;
168 if (prevTime) {
169 prevDigit = (*prevTime)[iPad];
170 }
171 const CRU cru = mapper.getCRU(sector, iPad);
172 digit.fillOutputContainer<MODE>(output, mcTruth, cru, timeBin, iPad, mLabels, sampaProcessing, labelCollector, getCommonMode(cru), prevDigit, debugStream, deadMap);
173 }
174 }
175}
