104// Position of beta*gamma in the table rows: lower row and fraction towards the next one
105inline void rowOf(double betaGamma, int& row, double& frac)
106{
107 const double x = (std::log10(std::max(betaGamma, 1e-6)) - kLogBetaGammaMin) / kLogBetaGammaStep;
108 if (x <= 0.) {
109 row = 0;
110 frac = 0.;
111 } else if (x >= kNBetaGamma - 1) {
112 row = kNBetaGamma - 2;
113 frac = 1.;
114 } else {
115 row = static_cast<int>(x);
116 frac = x - row;
117 }
118}
119
120// Log-log interpolation of a per-row table
121inline double interpolate(const double* table, int row, double frac)
122{
123 return std::exp((1. - frac) * std::log(table[row]) + frac * std::log(table[row + 1]));
124}

177double IonisationModel::sampleCollision(int row, double u) const
178{
179 // u is the fraction of collisions above the sampled energy; the tabulated fractions fall with energy
180 const float* f = mFractionAbove + row * mNEnergy;
181 int j = 1;
182 while (j < mNEnergy - 1 && f[j] > u) {
183 ++j;
184 }
185 const double t = (f[j - 1] - u) / std::max(f[j - 1] - f[j], 1e-12f);
186 return mEnergy[j - 1] * std::exp(t * std::log(mEnergy[j] / mEnergy[j - 1]));
187}

189double IonisationModel::stepEnergy(double length, double betaGamma, double z2, TRandom& rnd) const
190{
191 int row;
192 double frac;
193 rowOf(betaGamma, row, frac);
194 const int n = rnd.Poisson(length * z2 * interpolate(mCollisions, row, frac));
195 if (n == 0) {
196 return 0.;
197 }
198 // one of the two neighbouring spectra, with the interpolation weight as probability
199 if (rnd.Rndm() < frac) {
200 ++row;
201 }
202 double energy = 0.;
203 for (int i = 0; i < n; ++i) {
204 energy += sampleCollision(row, rnd.Rndm());
205 }
206 return energy;
207}
