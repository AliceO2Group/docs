50GPUDisplayGUIWrapper::GPUDisplayGUIWrapper(GPUDisplayGUIChannel& channel) : mChannel(channel)

71void GPUDisplayGUIWrapper::guiThread()
72{
73 mO->app = std::make_unique<QApplication>(mO->argc, mO->argv);
74 std::unique_lock lock(mO->mutex);
75 while (true) {
76 mO->signal.wait(lock, [this] { return mO->start || mO->terminate; });
77 if (mO->terminate) {
78 break;
79 }
80 mO->start = false;
81 mO->gui = std::make_unique<GPUDisplayGUI>(mChannel);
82 mO->gui->show();
83 mO->started = true;
84 mO->signal.notify_all();
85 lock.unlock();
86 mO->app->exec(); // Returns when the GUI window is closed, by the user or by stop()
87 lock.lock();
88 mO->gui.reset();
89 mO->started = false;
90 mO->signal.notify_all();
91 }
92 lock.unlock();
93 mO->app.reset();
94}

96int32_t GPUDisplayGUIWrapper::start()
97{
98 std::unique_lock lock(mO->mutex);
99 if (!mO->started) {
100 mO->start = true;
101 mO->signal.notify_all();
102 mO->signal.wait(lock, [this] { return mO->started.load(); });
103 }
104 return 0;
105}

107int32_t GPUDisplayGUIWrapper::stop()
108{
109 {
110 std::lock_guard lock(mChannel.mutex);
111 mChannel.stop = true;
112 }
113 mChannel.stateUpdated.notify_all(); // Wake up the GUI if it waits for a reply
114 std::unique_lock lock(mO->mutex);
115 mO->signal.wait(lock, [this] { return !mO->started; });
116 return 0;
117}
