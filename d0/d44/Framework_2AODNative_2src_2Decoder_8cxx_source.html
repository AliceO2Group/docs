32template <typename T>
33void byteswapAll(uint8_t* data, size_t count)
34{
35 for (size_t i = 0; i < count; ++i) {
36 T v;
37 std::memcpy(&v, data + i * sizeof(T), sizeof(T));
38 v = bswap(v);
39 std::memcpy(data + i * sizeof(T), &v, sizeof(T));
40 }
41}

44void byteswapInPlace(uint8_t* data, size_t count, size_t typeSize)
45{
46 if constexpr (std::endian::native == std::endian::big) {
47 return;
48 }
49 switch (typeSize) {
50 case 2:
51 byteswapAll<uint16_t>(data, count);
52 break;
53 case 4:
54 byteswapAll<uint32_t>(data, count);
55 break;
56 case 8:
57 byteswapAll<uint64_t>(data, count);
58 break;
59 }
60}

64struct BranchDecoder::BasketValues {
65 std::span<const uint8_t> plain;
66 std::optional<PayloadDecompressor> compressed;
67 size_t border = 0;
68 size_t consumed = 0;
69
70 void read(uint8_t* dst, size_t n)
71 {
72 if (compressed) {
73 compressed->read(dst, n);
74 } else {
75 std::memcpy(dst, plain.data() + consumed, n);
76 }
77 consumed += n;
78 }
79
81 void readAll(uint8_t* dst, size_t n)
82 {
83 if (compressed) {
84 compressed->readPrefix(dst, n);
85 consumed += n;
86 } else {
87 read(dst, n);
88 }
89 }
90};

97void BranchDecoder::checkBaskets(BranchInfo const& branch) const
98{
99 int64_t next = 0;
100 for (auto& b : branch.baskets) {
101 if (b.firstEntry != next || b.nEntries < 0) {
102 throw CorruptData(mUrl, b.seek, "baskets of branch " + branch.name + " do not cover its entries contiguously");
103 }
104 next += b.nEntries;
105 }
106 if (next != branch.entries) {
107 throw CorruptData(mUrl, 0, "baskets of branch " + branch.name + " hold " + std::to_string(next) + " entries, but the branch has " + std::to_string(branch.entries));
108 }
109}

111template <typename F>
112void BranchDecoder::forEachBasket(BranchInfo const& branch, bool parallel, F&& fn)
113{
114 checkBaskets(branch);
115 auto task = [this, &branch, &fn](size_t i) {
116 auto const& ref = branch.baskets[i];
117 BasketValues values;
118 std::vector<uint8_t> storage; // only when the source cannot be mapped
119 if (ref.embedded) {
120 values.plain = ref.embeddedPayload;
121 values.border = ref.embeddedPayload.size();
122 } else {
123 auto bytes = mSource.tryMap(ref.seek, ref.nbytes);
124 if (bytes.empty()) {
125 if (ref.seek > mSource.size() || ref.nbytes > mSource.size() - ref.seek) {
126 throw CorruptData(mUrl, ref.seek, "basket of branch " + branch.name + " past the end of the file");
127 }
128 storage.resize(ref.nbytes);
129 mSource.readInto(ref.seek, ref.nbytes, storage.data());
130 bytes = storage;
131 }
132 Cursor c{bytes, mUrl, ref.seek};
133 auto header = readBasketHeader(c);
134 auto nbytes = header.nbytes;
135 auto objlen = header.objlen;
136 auto keylen = header.keylen;
137 auto last = header.last;
138 if (nbytes != ref.nbytes || keylen > nbytes || header.nevbuf != ref.nEntries || last < keylen || (uint32_t)(last - keylen) > objlen) {
139 c.fail("basket header of branch " + branch.name + " does not match its TBranch");
140 }
141 values.border = last - keylen;
142 auto payload = bytes.subspan(keylen);
143 if (objlen != nbytes - keylen) {
144 values.compressed.emplace(payload, objlen, mUrl, ref.seek + keylen);
145 } else {
146 values.plain = payload;
147 }
148 }
149 fn(i, values);
150 };
151
152 auto n = branch.baskets.size();
153 int threads = parallel ? std::min<int>(mOptions.threads, n) : 1;
154 if (threads <= 1) {
155 for (size_t i = 0; i < n; ++i) {
156 task(i);
157 }
158 return;
159 }
160 // Baskets write disjoint slices of the destination, so no locking is needed.
161 std::atomic<size_t> next = 0;
162 std::exception_ptr error;
163 std::mutex errorMutex;
164 std::vector<std::thread> workers;
165 for (int t = 0; t < threads; ++t) {
166 workers.emplace_back([&] {
167 for (size_t i; (i = next.fetch_add(1)) < n;) {
168 try {
169 task(i);
170 } catch (...) {
171 std::lock_guard lock{errorMutex};
172 if (!error) {
173 error = std::current_exception();
174 }
175 next = n;
176 }
177 }
178 });
179 }
180 for (auto& w : workers) {
181 w.join();
182 }
183 if (error) {
184 std::rethrow_exception(error);
185 }
186}

198void BranchDecoder::decodeValues(BranchInfo const& branch, uint8_t* dst, size_t nbytes)
199{
200 auto elementSize = leafElementSize(branch.leafClass);
201 size_t rowBytes = elementSize * branch.fixedLen;
202 if (nbytes != branch.entries * rowBytes) {
203 throw FormatError("destination of " + std::to_string(nbytes) + " bytes for branch " + branch.name + " which needs " + std::to_string(branch.entries * rowBytes));
204 }
205 forEachBasket(branch, true, [&](size_t i, BasketValues& values) {
206 auto const& ref = branch.baskets[i];
207 size_t needed = ref.nEntries * rowBytes;
208 checkNeeded(mUrl, branch, ref, needed, values.border);
209 auto* out = dst + ref.firstEntry * rowBytes;
210 values.readAll(out, needed);
211 byteswapInPlace(out, needed / elementSize, elementSize);
212 });
213}

215void BranchDecoder::decodeVLAValues(BranchInfo const& branch, std::span<const int32_t> offsets, uint8_t* dst, size_t nbytes)
216{
217 auto elementSize = leafElementSize(branch.leafClass);
218 if (offsets.size() != (size_t)branch.entries + 1) {
219 throw FormatError("branch " + branch.name + " has " + std::to_string(branch.entries) + " entries, but " + std::to_string(offsets.size()) + " offsets were given");
220 }
221 if (nbytes != (size_t)offsets.back() * elementSize) {
222 throw FormatError("destination of " + std::to_string(nbytes) + " bytes for branch " + branch.name + " which needs " + std::to_string(offsets.back() * elementSize));
223 }
224 forEachBasket(branch, true, [&](size_t i, BasketValues& values) {
225 auto const& ref = branch.baskets[i];
226 auto begin = offsets[ref.firstEntry];
227 auto end = offsets[ref.firstEntry + ref.nEntries];
228 if (end < begin) {
229 throw CorruptData(mUrl, ref.seek, "invalid offset range for branch " + branch.name);
230 }
231 size_t needed = (size_t)(end - begin) * elementSize;
232 checkNeeded(mUrl, branch, ref, needed, values.border);
233 auto* out = dst + (size_t)begin * elementSize;
234 values.readAll(out, needed);
235 byteswapInPlace(out, needed / elementSize, elementSize);
236 });
237}

239void BranchDecoder::decodeBooleans(BranchInfo const& branch, uint8_t* bitmap, size_t nbytes)
240{
241 size_t total = branch.entries * branch.fixedLen;
242 if (nbytes * 8 < total) {
243 throw FormatError("bitmap of " + std::to_string(nbytes) + " bytes for " + std::to_string(total) + " booleans of branch " + branch.name);
244 }
245 std::memset(bitmap, 0, nbytes);
246 std::vector<uint8_t> window(std::min(kBooleanWindow, total));
247 // The value index is global across baskets, so a basket boundary can fall
248 // inside a byte of the bitmap: decode serially.
249 size_t index = 0;
250 forEachBasket(branch, false, [&](size_t i, BasketValues& values) {
251 auto const& ref = branch.baskets[i];
252 size_t remaining = (size_t)ref.nEntries * branch.fixedLen;
253 checkNeeded(mUrl, branch, ref, remaining, values.border);
254 while (remaining > 0) {
255 auto k = std::min(remaining, window.size());
256 values.read(window.data(), k);
257 for (size_t j = 0; j < k; ++j, ++index) {
258 bitmap[index / 8] |= (uint8_t)((window[j] != 0) << (index % 8));
259 }
260 remaining -= k;
261 }
262 });
263}

265std::vector<int32_t> BranchDecoder::decodeOffsets(BranchInfo const& sizeBranch)
266{
267 if (sizeBranch.leafClass != 'I' || sizeBranch.fixedLen != 1) {
268 throw UnsupportedLayout("size branch " + sizeBranch.name + " is not a scalar 32-bit integer");
269 }
270 // fEntries sizes the allocation: check it against the baskets first.
271 checkBaskets(sizeBranch);
272 // Decode the sizes after the leading 0, then prefix-sum in place.
273 std::vector<int32_t> offsets(sizeBranch.entries + 1);
274 decodeValues(sizeBranch, (uint8_t*)(offsets.data() + 1), sizeBranch.entries * sizeof(int32_t));
275 int64_t total = 0;
276 offsets[0] = 0;
277 for (int64_t i = 1; i <= sizeBranch.entries; ++i) {
278 total += (uint32_t)offsets[i];
279 // Arrow lists use 32 bit offsets, a larger total can only come from corrupted sizes.
280 if (total > INT32_MAX) {
281 throw CorruptData(mUrl, 0, "invalid sizes for branch " + sizeBranch.name + ": offsets overflow at entry " + std::to_string(i - 1));
282 }
283 offsets[i] = (int32_t)total;
284 }
285 return offsets;
286}
